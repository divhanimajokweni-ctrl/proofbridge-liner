### 7. Supporting APIs to add

Create these 6 endpoints to feed the terminals. They read from your existing services:

`app/api/metrics/gate-a/route.ts` through `gate-f/route.ts`. Example for Gate A:
import { NextResponse } from 'next/server'
export async function GET() {
  // Query Canton for last hour proposals
  const queue = await getCantonQueue(); // implement
  const avgMu = queue.reduce((s,p) => s + p.mu, 0) / queue.length;
  const rejectRate = queue.filter(p => p.mu < 0.60).length / queue.length;
  return NextResponse.json({
    avgMu, rejectRate, queue: queue.slice(0,20),
    chart: { labels: queue.map(p => p.ts), mu: queue.map(p => p.mu) }
  });
}
