- **B2B industrial targets** as 3D markers with status colours (contacted, proposal, pilot)

    class TrackField {
      constructor(max = 512) {
        const g = new THREE.BoxGeometry(1, 1, 1);
        g.translate(0, 0.5, 0);
        this.fill = new THREE.InstancedMesh(g, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0.2, depthWrite: false }), max);
        this.fill.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
        this.fill.frustumCulled = false;
        this.edges = new THREE.InstancedMesh(g, new THREE.MeshBasicMaterial({ wireframe: true, transparent: true, opacity: 0.6 }), max);
        this.edges.frustumCulled = false;
        this.fill.setColorAt(0, new THREE.Color(1,1,1));
        this.edges.instanceMatrix = this.fill.instanceMatrix;
        this.edges.instanceColor = this.fill.instanceColor;
        this.max = max;
        this.tracks = new Map();
        this.dummy = new THREE.Object3D();
        this.col = new THREE.Color();
        layers.agents.add(this.fill, this.edges);
      }
      ingest(t) {
        let r = this.tracks.get(t.id);
        if (!r) {
          r = {
            cur: new THREE.Vector3(...t.pos),
            tgt: new THREE.Vector3(...t.pos),
            cd: new THREE.Vector3(1, 1, 1),
            td: new THREE.Vector3(...t.dims),
            yaw: t.yaw || 0,
            tYaw: t.yaw || 0,
            vel: new THREE.Vector3(...t.vel),
            tVel: new THREE.Vector3(...t.vel),
            hasHdg: t.hasHdg || false,
            conf: t.conf || 0.8,
            cls: t.cls || 'unknown',
            seen: t.seen || performance.now(),
            born: performance.now()
          };
          this.tracks.set(t.id, r);
        } else {
          r.tgt.set(...t.pos);
          r.td.set(...t.dims);
          r.tYaw = t.yaw || 0;
          r.tVel.set(...t.vel);
          r.hasHdg = t.hasHdg || false;
          r.conf = t.conf || 0.8;
          r.cls = t.cls || 'unknown';
          r.seen = performance.now();
        }
      }
      update(dt, now) {
        const k = 1 - Math.exp(-dt * LERP_RATE);
        let i = 0;
        for (const [id, r] of this.tracks) {
          if (now - r.seen > TRACK_TTL) { this.tracks.delete(id); continue; }
          r.cur.lerp(r.tgt, k);
          r.cd.lerp(r.td, k);
          const dy = r.tYaw - r.yaw;
          r.yaw += Math.atan2(Math.sin(dy), Math.cos(dy)) * k;
          const s = Math.max(0.01, 1 - (now - r.seen) / TRACK_TTL);
          this.dummy.position.copy(r.cur);
          this.dummy.rotation.set(0, r.yaw, 0);
          this.dummy.scale.set(r.cd.x * s, r.cd.y * s, r.cd.z * s);
          this.dummy.updateMatrix();
          this.fill.setMatrixAt(i, this.dummy.matrix);
          this.col.setHex(CLASS_COLOR[r.cls] || CLASS_COLOR.unknown);
          this.fill.setColorAt(i, this.col);
          i++;
          if (i >= this.max) break;
        }
        this.fill.count = this.edges.count = i;
        this.fill.instanceMatrix.needsUpdate = true;
        if (this.fill.instanceColor) this.fill.instanceColor.needsUpdate = true;
      }
      getSnapshot() {
        const now = performance.now();
        const result = [];
        for (const [id, r] of this.tracks) {
          if (now - r.seen > TRACK_TTL) continue;
          result.push({
            id,
            pos: r.cur.clone(),
            cls: r.cls,
            conf: r.conf,
            dims: r.cd.clone(),
            yaw: r.yaw,
          });
        }
        return result;
      }
    }

    // ─── AGENTS SIMULATION ──────────────────────────────────────────
    let agents = [];
    let speedMultiplier = 1.0;
    let targetAgentCount = 14;

    function createAgents(count) {
      const newAgents = [];
      for (let i = 0; i < count; i++) {
        const angle = Math.random() * 2 * Math.PI;
        const dist = 200 + Math.random() * 300;
        const speed = (2 + Math.random() * 6) * speedMultiplier;
        const dir = Math.random() * 2 * Math.PI;
        const cls = ['person', 'vehicle', 'truck'][i % 3];
        newAgents.push({
          id: 'trk-' + i,
          cls: cls,
          conf: 0.7 + Math.random() * 0.25,
          pos: new THREE.Vector3(dist * Math.cos(angle), 0, dist * Math.sin(angle)),
          dir: dir,
          speed: speed,
          dims: cls === 'person' ? [0.5, 0.5, 1.7] : cls === 'truck' ? [8, 2.5, 3] : [4.5, 1.8, 1.5],
          yaw: 0,
          baseSpeed: speed / speedMultiplier
        });
      }
      return newAgents;
    }
    agents = createAgents(targetAgentCount);

    function updateAgentCount(newCount) {
      targetAgentCount = Math.max(1, newCount);
      if (targetAgentCount > agents.length) {
        const extra = targetAgentCount - agents.length;
        for (let i = 0; i < extra; i++) {
          const angle = Math.random() * 2 * Math.PI;
          const dist = 200 + Math.random() * 300;
          const speed = (2 + Math.random() * 6) * speedMultiplier;
          const dir = Math.random() * 2 * Math.PI;
          const cls = ['person', 'vehicle', 'truck'][(agents.length + i) % 3];
          agents.push({
            id: 'trk-' + (agents.length + i),
            cls: cls,
            conf: 0.7 + Math.random() * 0.25,
            pos: new THREE.Vector3(dist * Math.cos(angle), 0, dist * Math.sin(angle)),
            dir: dir,
            speed: speed,
            dims: cls === 'person' ? [0.5, 0.5, 1.7] : cls === 'truck' ? [8, 2.5, 3] : [4.5, 1.8, 1.5],
            yaw: 0,
            baseSpeed: speed / speedMultiplier
          });
        }
      } else if (targetAgentCount < agents.length) {
        agents = agents.slice(0, targetAgentCount);
      }
      document.getElementById('agent-count').textContent = targetAgentCount;
    }

    function updateSpeed(newSpeed) {
      speedMultiplier = parseFloat(newSpeed);

        a.speed = a.baseSpeed * speedMultiplier;
      }
    }

    function simulateTracks() {
      const now = performance.now();
      for (const a of agents) {
        a.pos.x += Math.cos(a.dir) * a.speed * 0.1;
        a.pos.z += Math.sin(a.dir) * a.speed * 0.1;
        if (Math.abs(a.pos.x) > 350) a.dir = Math.PI - a.dir;
        if (Math.abs(a.pos.z) > 350) a.dir = -a.dir;
        a.yaw = -a.dir;
        const track = {
          id: a.id,
          cls: a.cls,
          conf: a.conf,
          pos: [a.pos.x, 0, a.pos.z],
          dims: a.dims,
          yaw: a.yaw,
          vel: [Math.cos(a.dir) * a.speed, 0, Math.sin(a.dir) * a.speed],
          hasHdg: true,
          seen: now
        };
        field.ingest(track);
      }
    }
