import React, { useState, useEffect, useMemo, useRef } from 'react';
import { 
  Activity, ShieldCheck, ShieldAlert, Cpu, Eye, Navigation, 
  Map, BarChart2, Users, Truck, AlertTriangle, Layers, Maximize2, 
  ChevronRight, RefreshCw, Smartphone, TrendingUp, ShoppingBag,
  Camera, EyeOff, CheckCircle, Database
} from 'lucide-react';
import * as THREE from 'three';

// --- SYSTEM GEOLOCATION FALLBACK CONSTANTS ---
const GQEBERHA_LAT = -33.9608;
const GQEBERHA_LNG = 25.6022;

interface GISBenchProps {
  gpuTier?: number | null; // Passed from client-side detect-gpu hook (1-3)
}

export default function Geospatial3DGISBench({ gpuTier = 3 }: GISBenchProps) {
  // --- UI STATES ---
  const [isSidebarOpen, setIsSidebarOpen] = useState(false);
  const [isB2BOpen, setIsB2BOpen] = useState(false);
  const [isDemolished, setIsDemolished] = useState(false);
  const [gpsCoordinates, setGpsCoordinates] = useState({ lat: GQEBERHA_LAT, lng: GQEBERHA_LNG });
  const [gpsStatus, setGpsStatus] = useState<'STANDBY' | 'ACQUIRING' | 'LOCKED' | 'FAILED'>('STANDBY');
  const [selectedLayer, setSelectedLayer] = useState<'TERRAIN' | 'HEATMAP' | 'INFRASTRUCTURE'>('TERRAIN');
  const [activeVisualMode, setActiveVisualMode] = useState<'DEFAULT' | 'FLIR' | 'NVG'>('DEFAULT');

  // --- TELEMETRY / HOVER METRICS ---
  const [cursorCoordinates, setCursorCoordinates] = useState({ lat: GQEBERHA_LAT, lng: GQEBERHA_LNG, elev: 12.4 });
  const [activeLeakNode, setActiveLeakNode] = useState<string | null>(null);
  const [tankWaterLevel, setTankWaterLevel] = useState(8.5); // meters

  // --- DYNAMIC ANALYTICS SIMULATION STATE (UPDATES EVERY 2.5s) ---
  const [liveMetrics, setLiveMetrics] = useState({
    fps: 60,
    frameTimeMs: 16.6,
    apiCalls: 12,
    apiRequests: 148,
    peopleCountTotal: 1077,
    peopleStreet: 542,
    peopleSidewalk: 384,
    peopleAlley: 151,
    vehiclesTotal: 45,
    vehiclesTrucks: 8,
    vehiclesBuses: 4,
    vehiclesCars: 28,
    vehiclesVans: 5,
    marsoAnimals: 12,
    marsoVehicles: 31,
    factoryDefects: 3,
    factoryYieldRate: 99.4,
    retailCustomers: 188,
    retailFootTraffic: 420
  });

  // --- DOM REFS FOR THREE.JS AND LAYOUT ---
  const canvasContainerRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const rendererRef = useRef<THREE.WebGLRenderer | null>(null);

  // 🛡️ THE CRITICAL PERFORMANCE INVARIANT: Throttle heavy features on low-spec hardware (Tier < 2)
  const enableAnimations = useMemo(() => {
    if (typeof gpuTier === 'number' && gpuTier < 2) {
      console.warn("⚠️ [VVU PERF] Low-spec GPU detected (Tier < 2). Disabling post-processing and reducing WebGL terrain mesh density.");
      return false;
    }
    return true;
  }, [gpuTier]);

  // --- INITIAL GEOLOCATION RESOLUTION ---
  useEffect(() => {
    if (!navigator.geolocation) {
      setGpsStatus('FAILED');
      return;
    }
    setGpsStatus('ACQUIRING');
    navigator.geolocation.getCurrentPosition(
      (position) => {
        setGpsCoordinates({
          lat: position.coords.latitude,
          lng: position.coords.longitude
        });
        setGpsStatus('LOCKED');
      },
      (error) => {
        console.error("Geolocation capture error. Falling back to Gqeberha base coordinates:", error);
        setGpsStatus('FAILED');
      },
      { enableHighAccuracy: true, timeout: 5000 }
    );
  }, []);

  // --- DYNAMIC METRICS LOOP (Simulating real-time edge AI streams) ---
  useEffect(() => {
    const interval = setInterval(() => {
      setLiveMetrics(prev => {
        const fpsVariance = Math.floor(Math.random() * 5) - 2;
        const targetFps = enableAnimations ? 60 : 30;
        const currentFps = Math.max(targetFps + fpsVariance, 12);
        
        return {
          fps: currentFps,
          frameTimeMs: parseFloat((1000 / currentFps).toFixed(1)),
          apiCalls: prev.apiCalls + Math.floor(Math.random() * 3) - 1,
          apiRequests: prev.apiRequests + Math.floor(Math.random() * 5),
          peopleCountTotal: prev.peopleCountTotal + Math.floor(Math.random() * 11) - 5,
          peopleStreet: prev.peopleStreet + Math.floor(Math.random() * 5) - 2,
          peopleSidewalk: prev.peopleSidewalk + Math.floor(Math.random() * 4) - 2,
          peopleAlley: prev.peopleAlley + Math.floor(Math.random() * 2) - 1,
          vehiclesTotal: prev.vehiclesTotal + Math.floor(Math.random() * 3) - 1,
          vehiclesTrucks: prev.vehiclesTrucks,
          vehiclesBuses: prev.vehiclesBuses,
          vehiclesCars: prev.vehiclesCars + Math.floor(Math.random() * 2) - 1,
          vehiclesVans: prev.vehiclesVans,
          marsoAnimals: Math.max(0, prev.marsoAnimals + Math.floor(Math.random() * 3) - 1),
          marsoVehicles: prev.marsoVehicles + Math.floor(Math.random() * 2) - 1,
          factoryDefects: Math.max(0, prev.factoryDefects + (Math.random() > 0.85 ? 1 : -1)),
          factoryYieldRate: parseFloat((99.0 + Math.random() * 0.9).toFixed(2)),
          retailCustomers: prev.retailCustomers + Math.floor(Math.random() * 7) - 3,
          retailFootTraffic: prev.retailFootTraffic + Math.floor(Math.random() * 10) - 4
        };
      });
    }, 2500);

    return () => clearInterval(interval);
  }, [enableAnimations]);

  // --- REAL-TIME WATER LEVEL INTERACTION ---
  useEffect(() => {
    let animFrame: number;
    const animateTank = () => {
      setTankWaterLevel(prev => {
        if (activeLeakNode) {
          // Drops quickly during a simulated pipeline leak
          return Math.max(1.8, parseFloat((prev - 0.05).toFixed(3)));
        } else {
          // Recovers slowly up to normal capacity of 10.0m
          return Math.min(10.0, parseFloat((prev + 0.01).toFixed(3)));
        }
      });
      animFrame = requestAnimationFrame(animateTank);
    };
    animFrame = requestAnimationFrame(animateTank);
    return () => cancelAnimationFrame(animFrame);
  }, [activeLeakNode]);

  // --- MAIN THREE.JS SIMULATION IMPLEMENTATION ---
  useEffect(() => {
    if (!canvasRef.current || !canvasContainerRef.current) return;

    const width = canvasContainerRef.current.clientWidth;
    const height = canvasContainerRef.current.clientHeight;

    // 1. Initialise Scene, Camera and WebGL Renderer
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x020617); // Slate-950

    const camera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
    camera.position.set(35, 45, 50);

    const renderer = new THREE.WebGLRenderer({ 
      canvas: canvasRef.current, 
      antialias: enableAnimations,
      preserveDrawingBuffer: true // Required for clean high-res PNG snapshots
    });
    renderer.setSize(width, height);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    rendererRef.current = renderer;

    // Ambient and Directional Scene Lights
    const ambientLight = new THREE.AmbientLight(0x1e293b, 1.5);
    scene.add(ambientLight);

    const dirLight = new THREE.DirectionalLight(0x0ea5e9, 2.5);
    dirLight.position.set(10, 50, 10);
    scene.add(dirLight);

    // 2. stable Local Cartesian Coordinate transformation (ECEF relative to local origin)
    // Map bounds represents a stable 10km2 flat plane centered at Gqeberha coordinates
    const mapCenterLat = GQEBERHA_LAT;
    const mapCenterLng = GQEBERHA_LNG;

    const geodeticToLocalCartesian = (lat: number, lng: number) => {
      const R = 6378137; // Earth Radius in meters
      const dLat = (lat - mapCenterLat) * (Math.PI / 180);
      const dLng = (lng - mapCenterLng) * (Math.PI / 180);
      const x = R * dLng * Math.cos(mapCenterLat * Math.PI / 180);
      const z = R * dLat; // local elevation mapped along standard 3D space
      return { x, z };
    };

    const localCartesianToGeodetic = (x: number, z: number) => {
      const R = 6378137;
      const lat = mapCenterLat + (z / R) * (180 / Math.PI);
      const lng = mapCenterLng + (x / (R * Math.cos(mapCenterLat * Math.PI / 180))) * (180 / Math.PI);
      return { lat, lng };
    };

    // 3. Neon Grid Fragment Shader Material (Asynchronously Loaded Terrain Visual)
    const terrainSize = 80;
    const terrainSegments = enableAnimations ? 80 : 30; // Reduce vertices dynamically for Tier-1 GPUs
    const terrainGeo = new THREE.PlaneGeometry(terrainSize, terrainSize, terrainSegments, terrainSegments);

    // Elevate plane vertices to model realistic rolling coastal dunes
    const pos = terrainGeo.attributes.position;
    for (let i = 0; i < pos.count; i++) {
      const px = pos.getX(i);
      const py = pos.getY(i);
      // Realistic multi-frequency noise modeling Gqeberha dunes
      const elevation = Math.sin(px * 0.1) * Math.cos(py * 0.1) * 3 + Math.sin(px * 0.03) * 4;
      pos.setZ(i, elevation);
    }
    terrainGeo.computeVertexNormals();

    const customVertexShader = `
      varying vec2 vUv;
      varying float vElevation;
      void main() {
        vUv = uv;
        vElevation = position.z;
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
      }
    `;

    const customFragmentShader = `
      varying vec2 vUv;
      varying float vElevation;
      uniform float uTime;
      uniform vec3 uColor;
      void main() {
        // High-contrast Neon Grid aesthetic
        float gridX = abs(fract(vUv.x * 25.0 - 0.5) - 0.5) / fwidth(vUv.x * 25.0);
        float gridY = abs(fract(vUv.y * 25.0 - 0.5) - 0.5) / fwidth(vUv.y * 25.0);
        float grid = 1.0 - min(gridX, gridY);
        grid = clamp(grid, 0.0, 1.0);

        // Pulsing radar sweep scanline
        float scanline = sin(vUv.y * 50.0 - uTime * 4.0) * 0.5 + 0.5;
        vec3 finalColor = mix(vec3(0.01, 0.02, 0.08), uColor, grid * 0.4);
        finalColor += uColor * scanline * 0.1;
        
        // Color-scale based on height relief
        finalColor += vec3(0.0, 0.05, 0.1) * (vElevation + 7.0) * 0.1;
        
        gl_FragColor = vec4(finalColor, 1.0);
      }
    `;

    const terrainMaterial = new THREE.ShaderMaterial({
      vertexShader: customVertexShader,
      fragmentShader: customFragmentShader,
      uniforms: {
        uTime: { value: 0 },
        uColor: { value: new THREE.Color(0x0ea5e9) } // Cyan accent
      },
      wireframe: false
    });

    const terrainMesh = new THREE.Mesh(terrainGeo, terrainMaterial);
    terrainMesh.rotation.x = -Math.PI / 2;
    scene.add(terrainMesh);

    // 4. CityGML Utility Network ADE (Interactive Pipeline Structures)
    // Multi-material pipes and node tracking spheres
    const pipeGroup = new THREE.Group();
    scene.add(pipeGroup);

    // Dynamic pipe data containing positions, specifications, and materials
    const pipeSegments = [
      { id: 'PIPE-HDPE-01', material: 'HDPE', color: 0x10b981, start: [-25, -2], end: [-10, 1] },
      { id: 'PIPE-PVCO-02', material: 'PVC-O', color: 0x3b82f6, start: [-10, 1], end: [10, -3] },
      { id: 'PIPE-DI-03', material: 'Ductile Iron', color: 0x64748b, start: [10, -3], end: [25, 4] }
    ];

    const pipeMeshMap: { [key: string]: THREE.Mesh } = {};
    const nodeSpheres: THREE.Mesh[] = [];

    // Helper to evaluate absolute terrain elevation height at any local coordinate
    const getTerrainHeight = (lx: number, lz: number) => {
      // Linear approximation matching Plane dune height logic
      return Math.sin(lx * 0.1) * Math.cos(lz * 0.1) * 3 + Math.sin(lx * 0.03) * 4;
    };

    pipeSegments.forEach(pipe => {
      const hStart = getTerrainHeight(pipe.start[0], pipe.start[1]) - 1.5; // Placed at a burial depth of 1.5m
      const hEnd = getTerrainHeight(pipe.end[0], pipe.end[1]) - 1.5;

      const pStart = new THREE.Vector3(pipe.start[0], hStart, pipe.start[1]);
      const pEnd = new THREE.Vector3(pipe.end[0], hEnd, pipe.end[1]);

      const distance = pStart.distanceTo(pEnd);
      const pipeGeo = new THREE.CylinderGeometry(0.4, 0.4, distance, 12);
      
      const pipeMat = new THREE.MeshStandardMaterial({
        color: pipe.color,
        roughness: 0.2,
        metalness: 0.8,
        emissive: pipe.color,
        emissiveIntensity: 0.15
      });

      const mesh = new THREE.Mesh(pipeGeo, pipeMat);
      
      // Position and orient cylinder between points
      const position = pStart.clone().add(pEnd).multiplyScalar(0.5);
      mesh.position.copy(position);
      
      const direction = pEnd.clone().sub(pStart).normalize();
      const up = new THREE.Vector3(0, 1, 0);
      mesh.quaternion.setFromUnitVectors(up, direction);

      pipeGroup.add(mesh);
      pipeMeshMap[pipe.id] = mesh;

      // Create interactive, georeferenced Node Spheres at pipeline endpoints
      const nodeGeo = new THREE.SphereGeometry(0.7, 16, 16);
      const nodeMat = new THREE.MeshStandardMaterial({
        color: 0x3b82f6, // Vibrant Blue
        roughness: 0.1,
        emissive: 0x3b82f6,
        emissiveIntensity: 0.3
      });

      const startNode = new THREE.Mesh(nodeGeo, nodeMat.clone());
      startNode.position.copy(pStart);
      startNode.userData = { id: `${pipe.id}-START`, desc: `${pipe.material} Joint`, latLng: localCartesianToGeodetic(pStart.x, pStart.z) };
      pipeGroup.add(startNode);
      nodeSpheres.push(startNode);

      const endNode = new THREE.Mesh(nodeGeo, nodeMat.clone());
      endNode.position.copy(pEnd);
      endNode.userData = { id: `${pipe.id}-END`, desc: `${pipe.material} Joint`, latLng: localCartesianToGeodetic(pEnd.x, pEnd.z) };
      pipeGroup.add(endNode);
      nodeSpheres.push(endNode);
    });

    // 5. EPANET Storage Tank Model (Elevated topographic asset)
    const tankX = -18;
    const tankZ = -22;
    const tankBaseY = getTerrainHeight(tankX, tankZ) + 4; // Elevated on a structural framework

    const tankGroup = new THREE.Group();
    scene.add(tankGroup);

    // Structural support grid
    const trussGeo = new THREE.CylinderGeometry(0.1, 0.1, 8, 8);
    const trussMat = new THREE.MeshStandardMaterial({ color: 0x334155, metalness: 0.9 });
    for (let i = -1; i <= 1; i += 2) {
      for (let j = -1; j <= 1; j += 2) {
        const truss = new THREE.Mesh(trussGeo, trussMat);
        truss.position.set(tankX + i * 2, tankBaseY - 4, tankZ + j * 2);
        tankGroup.add(truss);
      }
    }

    // Storage Tank Reservoir Body
    const tankGeo = new THREE.CylinderGeometry(3.5, 3.5, 6, 24);
    const tankMat = new THREE.MeshStandardMaterial({
      color: 0x64748b, // Industrial Metallic Grey
      roughness: 0.3,
      metalness: 0.7,
      transparent: true,
      opacity: 0.85
    });
    const tankBody = new THREE.Mesh(tankGeo, tankMat);
    tankBody.position.set(tankX, tankBaseY + 3, tankZ);
    tankGroup.add(tankBody);

    // Dynamic, floating 3D water level disc representing current head capacity
    const waterGeo = new THREE.CylinderGeometry(3.4, 3.4, 0.1, 24);
    const waterMat = new THREE.MeshStandardMaterial({
      color: 0x0ea5e9, // Glowing Blue water
      roughness: 0.1,
      transparent: true,
      opacity: 0.9,
      emissive: 0x0ea5e9,
      emissiveIntensity: 0.4
    });
    const waterLevelMesh = new THREE.Mesh(waterGeo, waterMat);
    waterLevelMesh.position.set(tankX, tankBaseY, tankZ); // Height animated inside render loop
    tankGroup.add(waterLevelMesh);

    // 6. CesiumJS Style Dev Toggle Structure Group
    const buildGroup = new THREE.Group();
    scene.add(buildGroup);

    const mockBuildings: THREE.Mesh[] = [];
    const residentialColors = [0xd97706, 0xb45309]; // Warm amber/copper shades
    const industrialColors = [0x475569, 0x1e293b];  // Cool titanium/steel shades

    // Spawn 12 local structural models fanning around coordinates
    for (let i = 0; i < 12; i++) {
      const bx = (Math.sin(i * 0.5) * 22) + (Math.random() * 4 - 2);
      const bz = (Math.cos(i * 0.5) * 22) + (Math.random() * 4 - 2);
      const by = getTerrainHeight(bx, bz);

      const bWidth = 2 + Math.random() * 2;
      const bHeight = 4 + Math.random() * 8;
      const bGeo = new THREE.BoxGeometry(bWidth, bHeight, bWidth);
      
      const isInd = i % 3 === 0; // Classify 1 in 3 structures as industrial
      const color = isInd ? industrialColors[0] : residentialColors[0];
      
      const bMat = new THREE.MeshStandardMaterial({
        color: color,
        roughness: 0.4,
        metalness: isInd ? 0.9 : 0.1
      });

      const building = new THREE.Mesh(bGeo, bMat);
      building.position.set(bx, by + bHeight / 2, bz);
      building.userData = { isIndustrial: isInd, baseHeight: bHeight };
      buildGroup.add(building);
      mockBuildings.push(building);
    }

    // 7. Sentinel-2 Soil Moisture Anomaly (Dynamic vegetation/leak moisture circle)
    const leakDecalGeo = new THREE.RingGeometry(0.1, 6, 32);
    const leakDecalMat = new THREE.MeshBasicMaterial({
      color: 0x10b981, // Rich Moisture Green
      transparent: true,
      opacity: 0,
      side: THREE.DoubleSide
    });
    const leakDecal = new THREE.Mesh(leakDecalGeo, leakDecalMat);
    leakDecal.rotation.x = -Math.PI / 2;
    scene.add(leakDecal);

    // 8. Dynamic Camera Mouse Orbit controls (No external dependency required)
    let isDragging = false;
    let previousMousePosition = { x: 0, y: 0 };
    let cameraAngleX = 0.5;
    let cameraAngleY = 0.8;
    let cameraRadius = 65;

    const updateCameraPosition = () => {
      camera.position.x = cameraRadius * Math.sin(cameraAngleY) * Math.cos(cameraAngleX);
      camera.position.z = cameraRadius * Math.sin(cameraAngleY) * Math.sin(cameraAngleX);
      camera.position.y = cameraRadius * Math.cos(cameraAngleY) + 10;
      camera.lookAt(0, 0, 0);
    };
    updateCameraPosition();

    const handleMouseDown = () => { isDragging = true; };
    const handleMouseUp = () => { isDragging = false; };
    const handleMouseMove = (e: MouseEvent) => {
      const deltaMove = {
        x: e.offsetX - previousMousePosition.x,
        y: e.offsetY - previousMousePosition.y
      };

      if (isDragging) {
        cameraAngleX += deltaMove.x * 0.007;
        cameraAngleY = Math.max(0.1, Math.min(Math.PI / 2 - 0.05, cameraAngleY - deltaMove.y * 0.007));
        updateCameraPosition();
      }

      previousMousePosition = { x: e.offsetX, y: e.offsetY };

      // --- 9. INTERACTIVE RAYCASTER TELEMETRY OVERLAY ---
      const mouse = new THREE.Vector2(
        (e.offsetX / width) * 2 - 1,
        -(e.offsetY / height) * 2 + 1
      );

      const raycaster = new THREE.Raycaster();
      raycaster.setFromCamera(mouse, camera);

      // Project cursor onto terrain mesh
      const intersects = raycaster.intersectObject(terrainMesh);
      if (intersects.length > 0) {
        const point = intersects[0].point;
        const coords = localCartesianToGeodetic(point.x, point.z);
        setCursorCoordinates({
          lat: coords.lat,
          lng: coords.lng,
          elev: parseFloat((point.y + 12).toFixed(2)) // Map local space offset to elevation baseline
        });
      }
    };

    const handleWheel = (e: WheelEvent) => {
      cameraRadius = Math.max(20, Math.min(150, cameraRadius + e.deltaY * 0.05));
      updateCameraPosition();
    };

    const handleCanvasClick = (e: MouseEvent) => {
      const mouse = new THREE.Vector2(
        (e.offsetX / width) * 2 - 1,
        -(e.offsetY / height) * 2 + 1
      );

      const raycaster = new THREE.Raycaster();
      raycaster.setFromCamera(mouse, camera);

      // Cast rays across node endpoints
      const intersects = raycaster.intersectObjects(nodeSpheres);
      if (intersects.length > 0) {
        const node = intersects[0].object as THREE.Mesh;
        const nodeId = node.userData.id;
        
        if (activeLeakNode === nodeId) {
          // Deactivate leak if already running on node
          setActiveLeakNode(null);
          node.scale.set(1.0, 1.0, 1.0);
          (node.material as THREE.MeshStandardMaterial).color.setHex(0x3b82f6);
          (node.material as THREE.MeshStandardMaterial).emissive.setHex(0x3b82f6);
        } else {
          // Reset previous leak nodes
          nodeSpheres.forEach(s => {
            s.scale.set(1.0, 1.0, 1.0);
            (s.material as THREE.MeshStandardMaterial).color.setHex(0x3b82f6);
            (s.material as THREE.MeshStandardMaterial).emissive.setHex(0x3b82f6);
          });

          // Trigger Leak State on target Node
          setActiveLeakNode(nodeId);
          node.scale.set(1.6, 1.6, 1.6); // Scale up node physically
          (node.material as THREE.MeshStandardMaterial).color.setHex(0xef4444); // Glow Crimson red
          (node.material as THREE.MeshStandardMaterial).emissive.setHex(0xef4444);
          
          // Relocate Soil Moisture Decal onto leak epicenter
          leakDecal.position.copy(node.position);
          leakDecal.position.y += 0.15; // Raised slightly above physical terrain mesh
        }
      } else {
        // Clear leaks when clicking empty space
        const terrainIntersects = raycaster.intersectObject(terrainMesh);
        if (terrainIntersects.length > 0 && !intersects.length) {
          setActiveLeakNode(null);
          nodeSpheres.forEach(s => {
            s.scale.set(1.0, 1.0, 1.0);
            (s.material as THREE.MeshStandardMaterial).color.setHex(0x3b82f6);
            (s.material as THREE.MeshStandardMaterial).emissive.setHex(0x3b82f6);
          });
        }
      }
    };

    const canvas = canvasRef.current;
    canvas.addEventListener('mousedown', handleMouseDown);
    canvas.addEventListener('mouseup', handleMouseUp);
    canvas.addEventListener('mousemove', handleMouseMove);
    canvas.addEventListener('wheel', handleWheel);
    canvas.addEventListener('click', handleCanvasClick);

    // --- 10. ROBUST RESIZEOBSERVER INITIALIZATION GATES ---
    const resizeObserver = new ResizeObserver(entries => {
      for (let entry of entries) {
        const { width: w, height: h } = entry.contentRect;
        renderer.setSize(w, h);
        camera.aspect = w / h;
        camera.updateProjectionMatrix();
      }
    });
    resizeObserver.observe(canvasContainerRef.current);

    // 100ms Fallback sizing trigger
    const resizeTimeout = setTimeout(() => {
      if (canvasContainerRef.current) {
        const w = canvasContainerRef.current.clientWidth;
        const h = canvasContainerRef.current.clientHeight;
        renderer.setSize(w, h);
        camera.aspect = w / h;
        camera.updateProjectionMatrix();
      }
    }, 100);

    // --- 11. CENTRALIZED WEBGL RENDER LOOP ---
    const clock = new THREE.Clock();
    let animationFrameId: number;

    const render = () => {
      const elapsedTime = clock.getElapsedTime();

      // Update Custom fragment shader time uniforms
      terrainMaterial.uniforms.uTime.value = elapsedTime;

      // Animate fluid level within EPANET elevated reservoir
      const targetWaterY = tankBaseY + (tankWaterLevel / 10) * 5.8;
      waterLevelMesh.position.y = THREE.MathUtils.lerp(waterLevelMesh.position.y, targetWaterY, 0.1);

      // Handle CesiumJS Dev Toggle Style Swaps
      mockBuildings.forEach(building => {
        const scale = building.scale.y;
        if (isDemolished) {
          if (!building.userData.isIndustrial) {
            // Shrink/Hide residential structures
            building.scale.y = THREE.MathUtils.lerp(scale, 0, 0.1);
            building.position.y = THREE.MathUtils.lerp(building.position.y, -10, 0.1);
          } else {
            // Scale up & reveal Industrial Assets with glass-strip styling
            building.scale.y = THREE.MathUtils.lerp(scale, 1.4, 0.1);
            (building.material as THREE.MeshStandardMaterial).color.setHex(0x38bdf8); // Glowing Glass cyan
            (building.material as THREE.MeshStandardMaterial).metalness = 0.95;
          }
        } else {
          // Rebuild baseline urban distribution
