I’ve integrated the **3D GIS Bench** and the **Petal‑style 2D map** into a single dashboard where **both views show the same simulated traffic** – the map markers update in real time to match the 3D tracks.

Now you can **simulate traffic** (adjust speed, number of agents) and see it simultaneously in 3D and on the dark, Petal‑inspired map – perfect for visual validation.

---

## ✨ What’s new

- **Unified track data** – both views share the same `field.tracks` object.
- **Live map markers** – Leaflet markers update every second with the same positions as the 3D instances.
- **Traffic simulation controls** – a small panel (top‑left) lets you:
  - **Agent count** (slider) – add/remove vehicles/people on the fly.
  - **Speed multiplier** – make traffic faster or slower.
- **3D ↔ Map sync** – the map’s centre follows the 3D scene’s focus (optional, but you can enable it).
- **Full Petal aesthetic** – dark tiles, temperature, bottom bar, street labels.

    class TrackField {
      constructor(max = 256) {
        const g = new THREE.BoxGeometry(1, 1, 1);
        g.translate(0, 0.5, 0);
        this.fill = new THREE.InstancedMesh(g, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0.2, depthWrite: false }), max);

        this.fill.frustumCulled = false;

        if (!r) {
          r = {
            cur: new THREE.Vector3(...t.pos),
            tgt: new THREE.Vector3(...t.pos),
            cd: new THREE.Vector3(1, 1, 1),
            td: new THREE.Vector3(...t.dims),

            tYaw: t.yaw || 0,
            vel: new THREE.Vector3(...t.vel),
            tVel: new THREE.Vector3(...t.vel),
            hasHdg: t.hasHdg || false,
            conf: t.conf || 0.8,
            cls: t.cls || 'unknown',
            seen: t.seen || performance.now(),
            born: performance.now()
          };
          this.tracks.set(t.id, r);
        } else {
          r.tgt.set(...t.pos);
          r.td.set(...t.dims);
          r.tYaw = t.yaw || 0;
          r.tVel.set(...t.vel);
          r.hasHdg = t.hasHdg || false;
          r.conf = t.conf || 0.8;
          r.cls = t.cls || 'unknown';
          r.seen = performance.now();
        }
      }
      update(dt, now) {
        const k = 1 - Math.exp(-dt * LERP_RATE);
        let i = 0;
        for (const [id, r] of this.tracks) {
          if (now - r.seen > TRACK_TTL) { this.tracks.delete(id); continue; }
          r.cur.lerp(r.tgt, k);
          r.cd.lerp(r.td, k);
          const dy = r.tYaw - r.yaw;
          r.yaw += Math.atan2(Math.sin(dy), Math.cos(dy)) * k;
          const s = Math.max(0.01, 1 - (now - r.seen) / TRACK_TTL);
          this.dummy.position.copy(r.cur);
          this.dummy.rotation.set(0, r.yaw, 0);
          this.dummy.scale.set(r.cd.x * s, r.cd.y * s, r.cd.z * s);
          this.dummy.updateMatrix();
          this.fill.setMatrixAt(i, this.dummy.matrix);
          this.col.setHex(CLASS_COLOR[r.cls] || CLASS_COLOR.unknown);
          this.fill.setColorAt(i, this.col);
          i++;
          if (i >= this.max) break;
        }
        this.fill.count = this.edges.count = i;
        this.fill.instanceMatrix.needsUpdate = true;
        if (this.fill.instanceColor) this.fill.instanceColor.needsUpdate = true;

            pos: r.cur.clone(),
            cls: r.cls,
            conf: r.conf,
            dims: r.cd.clone(),
            yaw: r.yaw,
          });
        }
        return result;
      }

        // We'll store a baseSpeed property.
        if (!a.baseSpeed) a.baseSpeed = a.speed / (a._prevSpeedMul || 1);
        a._prevSpeedMul = speedMultiplier;
        a.speed = a.baseSpeed * speedMultiplier;
      }
    }

    // ─── SIMULATE TRACKS ─────────────────────────────────────────
    function simulateTracks() {
      const now = performance.now();
      for (const a of agents) {
        a.pos.x += Math.cos(a.dir) * a.speed * 0.1;
        a.pos.z += Math.sin(a.dir) * a.speed * 0.1;
        if (Math.abs(a.pos.x) > 900) a.dir = Math.PI - a.dir;
        if (Math.abs(a.pos.z) > 900) a.dir = -a.dir;
        a.yaw = -a.dir;
        const track = {
          id: a.id,
          cls: a.cls,
          conf: a.conf,
          pos: [a.pos.x, 0, a.pos.z],
          dims: a.dims,
          yaw: a.yaw,
          vel: [Math.cos(a.dir) * a.speed, 0, Math.sin(a.dir) * a.speed],
          hasHdg: true,
          seen: now
        };
        field.ingest(track);
      }
    }
    setInterval(simulateTracks, 100);

        const marker = L.circleMarker([lat, lon], {

## 🚦 How to use it

1. **Save** the file as `vvu-traffic-sim.html` and open in a browser.
2. **Toggle the map** – click the bottom‑right button.
3. **Adjust traffic** – use the sliders in the top‑left panel:
   - **Agents** – number of moving objects (people/vehicles/trucks).
   - **Speed** – global speed multiplier.
4. **Open analytics** – click the vertical tab on the right edge.
5. **Watch the sync** – the 2D map markers update every second to match the 3D tracks.

---

## 🧠 What makes this special

- **Shared state** – both views read from the same `TrackField` instance.
- **Performance‑aware** – map markers are batched, 3D uses instanced meshes.
- **Petal‑style map** – dark, clean, with street labels and temperature.
- **Full control** – you can simulate different traffic densities and speeds to test your visualisation.
