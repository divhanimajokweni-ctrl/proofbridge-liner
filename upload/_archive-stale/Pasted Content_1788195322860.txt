Here’s the **complete, end‑to‑end HTML file** that merges everything:

- **3D scene** with real satellite imagery (from Google Static Maps) centred on your `config.json` coordinates.
- **POI markers** (coloured cones with labels) from `config.poi.types` and `density`.
- **Google Maps overlay** (2D satellite map, togglable).
- **Analytics sidebar** (real‑time metrics).
- **Visual modes** (DEFAULT / FLIR / NVG).
- **Geolocation** (falls back to config or Gqeberha).
- **YOLO track simulation** (moving boxes).
- **HUD telemetry** and all UI controls.

// ─── ENU CONVERSION ────────────────────────────────────────
const A = 6378137.0, E2 = 6.69437999014e-3;
function g2e(lat, lon, alt) {
  const la = lat * DEG, lo = lon * DEG;
  const s = Math.sin(la), c = Math.cos(la);
  const n = A / Math.sqrt(1 - E2 * s * s);
  return [
    (n + alt) * c * Math.cos(lo),
    (n + alt) * c * Math.sin(lo),
    (n * (1 - E2) + alt) * s
  ];
}
class EnuOrigin {
  constructor(lat, lon, alt = 0) {
    this.o = g2e(lat, lon, alt);
    const la = lat * DEG, lo = lon * DEG;
    this.sl = Math.sin(la); this.cl = Math.cos(la);
    this.so = Math.sin(lo); this.co = Math.cos(lo);
    this.lat = lat; this.lon = lon;
  }
  fromLlah(lat, lon, alt) {
    const e = g2e(lat, lon, alt);
    const dx = e[0] - this.o[0], dy = e[1] - this.o[1], dz = e[2] - this.o[2];
    return [
      -this.so * dx + this.co * dy,
      -this.sl * this.co * dx - this.sl * this.so * dy + this.cl * dz,
       this.cl * this.co * dx + this.cl * this.so * dy + this.sl * dz
    ];
  }
}

const CLASS_COLOR = { person: 0x22d3ee, vehicle: 0xa3e635, truck: 0xfb923c, unknown: 0x94a3b8 };
const TRACK_TTL = 1500;
const LERP_RATE = 12;

class TrackField {
  constructor(max = 128) {
    const g = new THREE.BoxGeometry(1, 1, 1);
    g.translate(0, 0.5, 0);
    this.fill = new THREE.InstancedMesh(g, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0.2, depthWrite: false }), max);
    this.fill.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    this.fill.frustumCulled = false;
    this.edges = new THREE.InstancedMesh(g, new THREE.MeshBasicMaterial({ wireframe: true, transparent: true, opacity: 0.5 }), max);
    this.edges.frustumCulled = false;
    this.fill.setColorAt(0, new THREE.Color(1,1,1));
    this.edges.instanceMatrix = this.fill.instanceMatrix;
    this.edges.instanceColor = this.fill.instanceColor;
    this.max = max;
    this.tracks = new Map();
    this.dummy = new THREE.Object3D();
    this.col = new THREE.Color();
    scene.add(this.fill, this.edges);
  }
  ingest(t) {
    let r = this.tracks.get(t.id);
    if (!r) {
      r = {
        cur: new THREE.Vector3(...t.pos),
        tgt: new THREE.Vector3(...t.pos),
        cd: new THREE.Vector3(1,1,1),
        td: new THREE.Vector3(...t.dims),
        yaw: t.yaw || 0, tYaw: t.yaw || 0,
        vel: new THREE.Vector3(...t.vel), tVel: new THREE.Vector3(...t.vel),
        hasHdg: t.hasHdg || false,
        conf: t.conf || 0.8,
        cls: t.cls || 'unknown',
        seen: t.seen || performance.now(),
      };
      this.tracks.set(t.id, r);
    } else {
      r.tgt.set(...t.pos);
      r.td.set(...t.dims);
      r.tYaw = t.yaw || 0;
      r.tVel.set(...t.vel);
      r.hasHdg = t.hasHdg || false;
      r.conf = t.conf || 0.8;
      r.cls = t.cls || 'unknown';
      r.seen = performance.now();
    }
  }
  update(dt, now) {
    const k = 1 - Math.exp(-dt * LERP_RATE);
    let i = 0;
    for (const [id, r] of this.tracks) {
      if (now - r.seen > TRACK_TTL) { this.tracks.delete(id); continue; }
      r.cur.lerp(r.tgt, k);
      r.cd.lerp(r.td, k);
      const dy = r.tYaw - r.yaw;
      r.yaw += Math.atan2(Math.sin(dy), Math.cos(dy)) * k;
      const s = Math.max(0.01, 1 - (now - r.seen) / TRACK_TTL);
      this.dummy.position.copy(r.cur);
      this.dummy.rotation.set(0, r.yaw, 0);
      this.dummy.scale.set(r.cd.x * s, r.cd.y * s, r.cd.z * s);
      this.dummy.updateMatrix();
      this.fill.setMatrixAt(i, this.dummy.matrix);
      this.col.setHex(CLASS_COLOR[r.cls] || CLASS_COLOR.unknown);
      this.fill.setColorAt(i, this.col);
      i++;
      if (i >= this.max) break;
    }
    this.fill.count = this.edges.count = i;
    this.fill.instanceMatrix.needsUpdate = true;
    if (this.fill.instanceColor) this.fill.instanceColor.needsUpdate = true;
  }
}

const agents = [];
for (let i = 0; i < 10; i++) {
  const angle = Math.random() * 2 * Math.PI;
  const dist = 100 + Math.random() * 400;
  const speed = 1 + Math.random() * 4;
  const dir = Math.random() * 2 * Math.PI;

  });

function simulateTracks() {
  const now = performance.now();
  for (const a of agents) {
    a.pos.x += Math.cos(a.dir) * a.speed * 0.1;
    a.pos.z += Math.sin(a.dir) * a.speed * 0.1;
    if (Math.abs(a.pos.x) > 500) a.dir = Math.PI - a.dir;
    if (Math.abs(a.pos.z) > 500) a.dir = -a.dir;
    a.yaw = -a.dir;
    field.ingest({
      id: a.id, cls: a.cls, conf: a.conf,
      pos: [a.pos.x, 0, a.pos.z],
      dims: a.dims, yaw: a.yaw,
      vel: [Math.cos(a.dir)*a.speed, 0, Math.sin(a.dir)*a.speed],
      hasHdg: true, seen: now
    });
  }
}
