        // River valley depression
        const distToRiver = Math.abs(x - Math.sin(y * 0.03) * 30);
        if (distToRiver < 25) {

    function updateTerrainColors(geometry, leakCoord) {
      const colors = geometry.attributes.color;
      const pos = geometry.attributes.position;
      const count = pos.count;

      for (let i = 0; i < count; i++) {
        const x = pos.getX(i);
        const y = pos.getY(i);

        let r = 0.12, g = 0.16, b = 0.23; // Darker slate industrial grey

        const distToRiver = Math.abs(x - Math.sin(y * 0.03) * 30);
        if (distToRiver < 28) {
          r = 0.18; g = 0.22; b = 0.28;
        }

        if (leakCoord) {
          const distToLeak = Math.sqrt(Math.pow(x - leakCoord, 2) + Math.pow(y - (-leakCoord), 2));
          if (distToLeak < 22.0) {
            // Apply soil saturation color grading (simulate Sentinel-2 wet mud profile)
            const dampFactor = Math.pow(1.0 - (distToLeak / 22.0), 2);
            r += (0.01 - r) * dampFactor;
            g += (0.02 - g) * dampFactor;
            b += (0.04 - b) * dampFactor;
          }
        }

        colors.setXYZ(i, r, g, b);
      }
      colors.needsUpdate = true;
    }

        if (layout.futureOnly && !isFutureCity) return;
        if (!layout.futureOnly && layout.type === 'apartments' && isFutureCity) return;

        let bMesh;
        if (layout.type === 'tower' || layout.type === 'commercial') {
          const color = layout.type === 'tower' ? '#1e293b' : '#334155';
          bMesh = createWindowGrid(layout.size, layout.size, layout.size, color);
        } else if (layout.type === 'industrial_annex') {
          bMesh = createWindowGrid(layout.size, layout.size, layout.size, '#0f172a');
          const glowingDome = new THREE.Mesh(new THREE.SphereGeometry(8, 16, 16), new THREE.MeshBasicMaterial({ color: '#06b6d4', transparent: true, opacity: 0.4 }));
          glowingDome.position.set(0, layout.size / 2, 0);
          bMesh.add(glowingDome);
        } else {
          bMesh = new THREE.Mesh(new THREE.BoxGeometry(...layout.size), new THREE.MeshStandardMaterial({ color: '#475569', roughness: 0.7 }));
        }

        bMesh.position.set(layout.pos, (Array.isArray(layout.size) ? layout.size : layout.size) / 2 + 1.8, layout.pos);
        buildingsGroup.add(bMesh);
      });

      if (!isFutureCity) {
        const houseMat = new THREE.MeshStandardMaterial({ color: '#94a3b8', roughness: 0.8 });
        const roofMat = new THREE.MeshStandardMaterial({ color: '#991b1b', roughness: 0.5 });
        for (let i = 0; i < 22; i++) {
          const x = -15 - (i * 3) % 40;
          const z = -20 - (i * 7) % 60;
          const hGroup = new THREE.Group();
          const base = new THREE.Mesh(new THREE.BoxGeometry(4, 3, 4), houseMat);
          base.position.y = 1.5;
          const roof = new THREE.Mesh(new THREE.ConeGeometry(3.2, 2.5, 4), roofMat);
          roof.position.y = 3.5;
          roof.rotation.y = Math.PI / 4;
          hGroup.add(base, roof);
          hGroup.position.set(x, 1.8, z);
          buildingsGroup.add(hGroup);
        }
      }

      generateTrees();
    }

    function generateTrees() {
      while (treesGroup.children.length > 0) {
        treesGroup.remove(treesGroup.children);
      }
      const trunkMat = new THREE.MeshStandardMaterial({ color: '#78350f', roughness: 0.9 });
      const pineLeafMat = new THREE.MeshStandardMaterial({ color: '#14532d', roughness: 0.6 });
      const broadLeafMat = new THREE.MeshStandardMaterial({ color: '#065f46', roughness: 0.7 });

      const trunkGeo = new THREE.CylinderGeometry(0.3, 0.4, 2.5, 5);
      const pineGeo = new THREE.ConeGeometry(1.8, 4.5, 5);
      const deciduousGeo = new THREE.DodecahedronGeometry(2.0, 1);

      for (let i = 0; i < 80; i++) {
        const x = -90 + (i * 13) % 180;
        const z = -90 + (i * 7) % 180;
        const distToRiver = Math.abs(x - Math.sin(z * 0.03) * 30);
        if (distToRiver > 25 && Math.abs(x - (-60)) > 12 && Math.abs(z - 10) > 12) {
          const tGroup = new THREE.Group();
          const trunk = new THREE.Mesh(trunkGeo, trunkMat);
          trunk.position.y = 1.25;
          tGroup.add(trunk);

          if (i % 2 === 0) {
            const leaves = new THREE.Mesh(pineGeo, pineLeafMat);
            leaves.position.y = 4.0;
            tGroup.add(leaves);
          } else {
            const leaves = new THREE.Mesh(deciduousGeo, broadLeafMat);
            leaves.position.y = 3.8;
            leaves.scale.set(1.0, 1.3, 1.0);
            tGroup.add(leaves);
          }

          tGroup.scale.setScalar(0.8 + ((i * 3) % 5) * 0.1);
          tGroup.position.set(x, 1.0, z);
          treesGroup.add(tGroup);
        }
      }
    }

    function generateElevatedReservoir() {
      const tankNode = pipelineNetwork.nodes.find(n => n.isTank);

    function generateSubsurfaceNetwork() {
      const elementsToKeep = [];
      pipesGroup.children.forEach(child => {
        if (child.type === "Group") elementsToKeep.push(child);
      });
      while (pipesGroup.children.length > 0) {
        pipesGroup.remove(pipesGroup.children);
      }
      elementsToKeep.forEach(el => pipesGroup.add(el));

      particleSystems = [];
      nodeObjects = [];

      const materialColors = {
        'HDPE': '#10b981',        // Green
        'DUCTILE_IRON': '#4b5563', // Grey
        'PVC-O': '#64748b',        // Blue-Grey
        'STEEL': '#3b82f6'
      };

      pipelineNetwork.nodes.forEach(node => {
        if (node.isTank) return;
        const nodeMat = new THREE.MeshStandardMaterial({ color: materialColors[node.material] || '#f59e0b', roughness: 0.3, metalness: 0.8 });
        const sphere = new THREE.Mesh(new THREE.SphereGeometry(1.6, 16, 16), nodeMat);
        sphere.position.set(...node.coord);
        sphere.userData = { node_id: node.id, coord: node.coord, material: node.material };
        pipesGroup.add(sphere);
        nodeObjects.push(sphere);
      });

      pipelineNetwork.links.forEach(link => {
        const startNode = pipelineNetwork.nodes.find(n => n.id === link.from);
        const endNode = pipelineNetwork.nodes.find(n => n.id === link.to);

        if (startNode && endNode) {
          const p1 = new THREE.Vector3(...startNode.coord);
          const p2 = new THREE.Vector3(...endNode.coord);
          const tubeGeo = new THREE.TubeGeometry(new THREE.LineCurve3(p1, p2), 10, (link.diameter / 300.0) * 0.9, 8, false);
          const tube = new THREE.Mesh(tubeGeo, new THREE.MeshStandardMaterial({ color: materialColors[startNode.material] || '#38bdf8', transparent: true, opacity: 0.65, roughness: 0.2 }));
          pipesGroup.add(tube);
          createParticleFlowSystem(p1, p2);
        }
      });
    }

    function createParticleFlowSystem(start, end) {
      const pCount = 35;
      const geometry = new THREE.BufferGeometry();
      const positions = new Float32Array(pCount * 3);
      const states = []; 

      for (let i = 0; i < pCount; i++) {
        const progress = i / pCount;
        const pt = new THREE.Vector3().lerpVectors(start, end, progress);
        positions[i * 3] = pt.x;
        positions[i * 3 + 1] = pt.y;
        positions[i * 3 + 2] = pt.z;
        states.push({ progress: progress, start: start.clone(), end: end.clone() });
      }

      geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
      const points = new THREE.Points(geometry, new THREE.PointsMaterial({ color: '#0ea5e9', size: 0.8, transparent: true, opacity: 0.9, blending: THREE.AdditiveBlending, depthWrite: false }));
      particlesGroup.add(points);
      particleSystems.push({ points: points, states: states });
    }

    function animateParticles(delta) {
      particleSystems.forEach(sys => {
        const posAttr = sys.points.geometry.attributes.position;
        for (let i = 0; i < posAttr.count; i++) {
          const state = sys.states[i];
          state.progress += delta * (flowVelocity * 0.08);
          if (state.progress > 1.0) state.progress = 0.0;
          const currentPos = new THREE.Vector3().lerpVectors(state.start, state.end, state.progress);
          currentPos.x += Math.sin(Date.now() * 0.01 + i) * 0.05;
          currentPos.y += Math.cos(Date.now() * 0.012 + i) * 0.05;
          posAttr.setXYZ(i, currentPos.x, currentPos.y, currentPos.z);
        }
        posAttr.needsUpdate = true;
      });
    }

### Widescreen Optimization Breakdown

1. **Fixed-Height Container Flex-Grid:** This design avoids responsive layouts that reflow elements vertically or bundle controls into a collapsible tray. It is designed around a rigid, three-column desktop SCADA architecture that locks controls on the left, telemetry and public data on the right, and the 3D viewport in the center.
2. **True Resolution Initialization (`preserveDrawingBuffer`):** The 3D canvas is contained within a rigid CSS viewport block (`min-h-0 min-w-0`), completely isolating it from neighboring structural columns. This ensures that when your sales agents click the **"Capture Client Audit Snapshot"** button, the WebGL buffer draws the exact state of the terrain and subsurface pipeline network, exporting a pixel-perfect image alongside a structured cryptographic metadata file.
3. **No smartphone widgets:** No collapsible drawers, hamburger icons, or touch-swipe gestures. The controls are standard, high-density desktop inputs (sliders, checkboxes, and buttons) that give the operator immediate, precise control over all simulation parameters simultaneously.
