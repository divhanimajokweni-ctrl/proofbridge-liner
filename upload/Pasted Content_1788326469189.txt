async function executeScopedTransaction<T>(
  tenantId: string,
  transactionBlock: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await client.query(
      `SELECT set_config('vvu.current_tenant_id', $1, true)`,
      [tenantId]

    console.error(`❌ [VVU DATABASE RLS FAIL] Transaction rolled back for Tenant: ${tenantId}.`, error);
    throw error;
  } finally {
    client.release();
  }

LEDGER_ADDRESS = os.getenv("LEDGER_CONTRACT_ADDRESS")

    {"inputs": [{"name": "fileHash", "type": "string"}, {"name": "passed", "type": "bool"}], "name": "setVerificationResult", "outputs": [], "stateMutability": "nonpayable", "type": "function"}

def compute_sha256(filepath):
    sha256_hash = hashlib.sha256()
    with open(filepath, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()

def watch_step_files():

            print(f"✅ Registered {filename} -> {file_hash} (tx: {tx_hash.hex()})")
