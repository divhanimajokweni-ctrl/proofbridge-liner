    49→type DiffStatus = "added" | "removed" | "changed" | "unchanged";
    50→
    51→interface InvariantDiff {
    52→  name: string;
    53→  status: DiffStatus;
    54→  left?: InvariantNode;
    55→  right?: InvariantNode;
    56→  changes: { field: string; from: string; to: string }[];
    57→}
    58→
    59→interface PolicyDiffResult {
    60→  leftName: string;
    61→  rightName: string;
    62→  shardChanged: boolean;
    63→  shardFrom?: string;
    64→  shardTo?: string;
    65→  ancestryChanged: boolean;
    66→  shadowChanged: boolean;
    67→  repairChanged: boolean;
    68→  exportsAdded: string[];
    69→  exportsRemoved: string[];
    70→  invariants: InvariantDiff[];
    71→  added: number;
    72→  removed: number;
    73→  changed: number;
    74→  unchanged: number;
    75→}
    76→
    77→function diffPolicies(left: PolicyNode, right: PolicyNode): PolicyDiffResult {
    78→  const leftInv = new Map(left.invariants.map((i) => [i.name, i]));
    79→  const rightInv = new Map(right.invariants.map((i) => [i.name, i]));
    80→  const allNames = Array.from(new Set([...leftInv.keys(), ...rightInv.keys()]));
    81→
    82→  const invariants: InvariantDiff[] = allNames.map((name) => {
    83→    const l = leftInv.get(name);
    84→    const r = rightInv.get(name);
    85→    if (l && !r) {
    86→      return { name, status: "removed", left: l, changes: [] };
    87→    }
    88→    if (!l && r) {
    89→      return { name, status: "added", right: r, changes: [] };
    90→    }
    91→    if (l && r) {
    92→      const changes: { field: string; from: string; to: string }[] = [];
    93→      if (l.rawPredicate !== r.rawPredicate) {
    94→        changes.push({ field: "predicate", from: l.rawPredicate, to: r.rawPredicate });
    95→      }
    96→      if (l.severity !== r.severity) {
    97→        changes.push({ field: "severity", from: l.severity, to: r.severity });
    98→      }
    99→      if (l.soft !== r.soft) {
   100→        changes.push({ field: "soft", from: String(l.soft), to: String(r.soft) });
   101→      }
   102→      if (l.tolerance !== r.tolerance) {
   103→        changes.push({ field: "tolerance", from: String(l.tolerance), to: String(r.tolerance) });
   104→      }
   105→      return {
   106→        name,
   107→        status: changes.length ? "changed" : "unchanged",
   108→        left: l,
   109→        right: r,
   110→        changes,
   111→      };
   112→    }
   113→    return { name, status: "unchanged", changes: [] };
   114→  });
   115→
   116→  invariants.sort((a, b) => {
   117→    const order: Record<DiffStatus, number> = { changed: 0, added: 1, removed: 2, unchanged: 3 };
   118→    return order[a.status] - order[b.status];
   119→  });
   120→
   121→  const shardFrom = left.shard
   122→    ? `${left.shard.dimension ?? left.shard.key} · ${left.shard.strategy}`
   123→    : "—";
   124→  const shardTo = right.shard
   125→    ? `${right.shard.dimension ?? right.shard.key} · ${right.shard.strategy}`
   126→    : "—";
   127→
   128→  return {
   129→    leftName: left.name,
   130→    rightName: right.name,
   131→    shardChanged: shardFrom !== shardTo,
   132→    shardFrom,
   133→    shardTo,
   134→    ancestryChanged:
   135→      JSON.stringify(left.ancestry) !== JSON.stringify(right.ancestry),
   136→    shadowChanged:
   137→      JSON.stringify(left.shadowBridge) !== JSON.stringify(right.shadowBridge),
   138→    repairChanged:
   139→      JSON.stringify(left.onViolation) !== JSON.stringify(right.onViolation),
   140→    exportsAdded: right.exports.filter((e) => !left.exports.includes(e)),
   141→    exportsRemoved: left.exports.filter((e) => !right.exports.includes(e)),
   142→    invariants,
   143→    added: invariants.filter((i) => i.status === "added").length,
   144→    removed: invariants.filter((i) => i.status === "removed").length,
   145→    changed: invariants.filter((i) => i.status === "changed").length,
   146→    unchanged: invariants.filter((i) => i.status === "unchanged").length,
   147→  };
   148→}

   160→export function PolicyDiffSection() {

   166→
   167→  const result = useMemo<PolicyDiffResult | null>(() => {

   174→      return null;

   176→  }, [leftSource, rightSource]);
   177→

   179→    const s = SAMPLE_POLICIES.find((p) => p.filename === filename);
   180→    if (!s) return;
   181→    if (side === "left") {
   182→      setLeftSource(s.source);
   183→      setLeftFilename(s.filename);
   184→    } else {
   185→      setRightSource(s.source);
   186→      setRightFilename(s.filename);
   187→    }
   188→  };
   189→
   190→  const swap = () => {
   191→    setLeftSource(rightSource);
   192→    setLeftFilename(rightFilename);
   193→    setRightSource(leftSource);
   194→    setRightFilename(leftFilename);
   195→    toast({ title: "Policies swapped" });
   196→  };
   197→
   198→  const lineDiff = useMemo(() => {
   199→    if (!result) return [];
   200→    const leftLines = leftSource.split("\n");
   201→    const rightLines = rightSource.split("\n");
   202→    const maxLen = Math.max(leftLines.length, rightLines.length);
   203→    const lines: { leftNum: number; leftText: string; rightNum: number; rightText: string; type: "same" | "left-only" | "right-only" | "changed" }[] = [];

   205→    for (let i = 0; i < maxLen; i++) {
   206→      const ll = leftLines[i];
   207→      const rl = rightLines[i];

   209→        lines.push({ leftNum: 0, leftText: "", rightNum: i + 1, rightText: rl, type: "right-only" });

   211→        lines.push({ leftNum: i + 1, leftText: ll, rightNum: 0, rightText: "", type: "left-only" });
   212→      } else if (ll === rl) {
   213→        lines.push({ leftNum: i + 1, leftText: ll, rightNum: i + 1, rightText: rl, type: "same" });

   215→        lines.push({ leftNum: i + 1, leftText: ll ?? "", rightNum: i + 1, rightText: rl ?? "", type: "changed" });

   227→      { name: "Unchanged", value: result.unchanged, fill: "oklch(0.42 0.01 165 / 0.5)" },

   230→
   231→  return (
   232→    <motion.section
   233→      className="space-y-4"
   234→      variants={containerVariants}
   235→      initial="hidden"
   236→      animate="show"

   238→      {/* Header */}
   239→      <GradientBorderCard gradient="from-verified/40 via-repairing/20 to-violating/20" className="p-4">
   240→        <div className="bg-grid absolute inset-0 opacity-20 pointer-events-none" />
   241→        <div className="relative flex flex-wrap items-center gap-3">
   242→          <motion.div variants={cardVariants} className="flex h-10 w-10 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 shrink-0">
   243→            <GitCompare className="h-5 w-5 text-verified" />
   244→          </motion.div>
   245→          <motion.div variants={cardVariants} className="min-w-0 flex-1">
   246→            <h2 className="text-base font-semibold flex items-center gap-2">
   247→              Policy Diff
   248→              <Badge variant="outline" className="border-verified/30 bg-verified/10 text-verified text-[9px] font-mono">
   249→                .epd comparison
   250→              </Badge>
   251→            </h2>
   252→            <p className="text-xs text-muted-foreground mt-0.5">
   253→              Compare two `.epd` policies side-by-side — invariant additions, removals, predicate changes, and structural drift.
   254→            </p>
   255→          </motion.div>

   257→      </GradientBorderCard>

   285→            size="sm"

   287→            onClick={swap}

   310→                <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">

   326→                          fontSize: "11px",
   327→                        }}
   328→                      />
   329→                      <Bar dataKey="value" radius={[0, 4, 4, 0]}>

   331→                          <Cell key={`cell-${index}`} fill={entry.fill} />
   332→                        ))}
   333→                      </Bar>
   334→                    </BarChart>
   335→                  </ResponsiveContainer>
   336→                </div>
   337→              </div>

   339→
   340→            {/* Structural diff with drift visualization */}

   342→              <div className="relative space-y-2">

   348→                  </span>

   361→                  to={result.shardChanged ? "config B" : "config"}

   363→                  hideValues={!result.ancestryChanged}

   491→                        line.type === "changed" && "bg-repairing/5",
   492→                      )}
   493→                    >
   494→                      {/* Left side */}
   495→                      <div className="w-1/2 flex border-r border-border/20">
   496→                        <span className={cn(
   497→                          "select-none w-8 shrink-0 text-right pr-1.5 py-0.5 text-[9px]",
   498→                          line.type === "left-only" ? "text-violating/70 bg-violating/10" : line.type === "changed" ? "text-repairing/70 bg-repairing/10" : "text-muted-foreground/40",
   499→                        )}>
   500→                          {line.leftNum || ""}
   501→                        </span>
   502→                        <span className={cn(
   503→                          "flex-1 px-1.5 py-0.5 whitespace-pre overflow-hidden",
   504→                          line.type === "left-only" && "text-violating",
   505→                          line.type === "changed" && "text-violating/80",
   506→                        )}>
   507→                          {line.type === "left-only" && <Minus className="inline h-3 w-3 mr-0.5 text-violating" />}
   508→                          {line.type === "changed" && <Minus className="inline h-3 w-3 mr-0.5 text-violating/50" />}
   509→                          {line.leftText}
   510→                        </span>
   511→                      </div>
   512→                      {/* Right side */}
   513→                      <div className="w-1/2 flex">
   514→                        <span className={cn(
   515→                          "select-none w-8 shrink-0 text-right pr-1.5 py-0.5 text-[9px]",
   516→                          line.type === "right-only" ? "text-verified/70 bg-verified/10" : line.type === "changed" ? "text-repairing/70 bg-repairing/10" : "text-muted-foreground/40",
   517→                        )}>
   518→                          {line.rightNum || ""}
   519→                        </span>
   520→                        <span className={cn(
   521→                          "flex-1 px-1.5 py-0.5 whitespace-pre overflow-hidden",
   522→                          line.type === "right-only" && "text-verified",
   523→                          line.type === "changed" && "text-verified/80",
   524→                        )}>
   525→                          {line.type === "right-only" && <Plus className="inline h-3 w-3 mr-0.5 text-verified" />}
   526→                          {line.type === "changed" && <Plus className="inline h-3 w-3 mr-0.5 text-verified/50" />}
   527→                          {line.rightText}
   528→                        </span>
   529→                      </div>
   530→                    </motion.div>
   531→                  ))}
   532→                </div>
   533→              </div>
   534→            </GradientBorderCard>
   535→          </motion.div>
   536→        ) : (
   537→          <motion.div
   538→            key="diff-empty"
   539→            initial={{ opacity: 0 }}
   540→            animate={{ opacity: 1 }}
   541→            exit={{ opacity: 0 }}
   542→          >
   543→            <GradientBorderCard gradient="from-violating/30 to-violating/10" className="p-8 text-center">
   544→              <AlertTriangle className="mx-auto h-8 w-8 text-muted-foreground/40" />
   545→              <p className="mt-2 text-sm text-muted-foreground">
   546→                One or both policies failed to parse.
   547→              </p>
   548→              <p className="text-xs text-muted-foreground/70 mt-1">
   549→                Fix the `.epd` source in the editors above to see the diff.
   550→              </p>
   551→            </GradientBorderCard>
   552→          </motion.div>
   553→        )}
   554→      </AnimatePresence>
   555→    </motion.section>
   556→  );

   559→function DiffEditor({
   560→  side,
   561→  label,
   562→  filename,
   563→  source,
   564→  onSource,
   565→  onLoadSample,
   566→  accent,
   567→}: {
   568→  side: "left" | "right";
   569→  label: string;
   570→  filename: string;
   571→  source: string;
   572→  onSource: (s: string) => void;
   573→  onLoadSample: (filename: string) => void;
   574→  accent: "verified" | "violating";
   575→}) {
   576→  const lines = source.split("\n");
   577→  const borderColor = accent === "verified" ? "border-verified/30" : "border-violating/30";
   578→  const dotColor = accent === "verified" ? "bg-verified" : "bg-violating";
   579→  const gradient = accent === "verified" ? "from-verified/40 to-verified/10" : "from-violating/40 to-violating/10";
   580→
   581→  return (
   582→    <GradientBorderCard gradient={gradient} className={cn("p-0", borderColor)}>
   583→      <div className="flex items-center gap-2 border-b border-border/60 px-3 py-2 bg-background/40">
   584→        <span className={cn("h-2 w-2 rounded-full", dotColor)} />
   585→        <span className="text-xs font-medium">{label}</span>
   586→        <span className="font-mono text-[11px] text-muted-foreground truncate">{filename}</span>
   587→        <Select value={filename} onValueChange={onLoadSample}>
   588→          <SelectTrigger className="h-6 w-[150px] text-[10px] ml-auto">
   589→            <SelectValue />
   590→          </SelectTrigger>
   591→          <SelectContent>
   592→            {SAMPLE_POLICIES.map((s) => (
   593→              <SelectItem key={s.filename} value={s.filename}>
   594→                {s.filename}
   595→              </SelectItem>
   596→            ))}
   597→          </SelectContent>
   598→        </Select>
   599→      </div>
   600→      <div className="relative flex bg-background/30 h-[360px]">
   601→        <div
   602→          className="select-none overflow-hidden py-2 pl-2.5 pr-1.5 text-right font-mono text-[10px] text-muted-foreground/50 bg-background/20 shrink-0"
   603→          style={{ minWidth: 36 }}
   604→          aria-hidden
   605→        >
   606→          {lines.map((_, i) => (
   607→            <div key={i} style={{ height: "1.5em", lineHeight: "1.5em" }}>
   608→              {i + 1}
   609→            </div>
   610→          ))}
   611→        </div>
   612→        <Textarea
   613→          value={source}
   614→          onChange={(e) => onSource(e.target.value)}
   615→          spellCheck={false}
   616→          className="codeblock flex-1 h-full resize-none border-0 rounded-none bg-transparent focus-visible:ring-0 focus-visible:ring-offset-0 text-[11px]"
   617→        />
   618→      </div>
   619→    </GradientBorderCard>
   620→  );

   659→function StructDriftBar({ result }: { result: PolicyDiffResult }) {
   660→  const total = 4; // shard, ancestry, shadow, repair
   661→  const drifts = [
   662→    { label: "shard", changed: result.shardChanged },
   663→    { label: "ancestry", changed: result.ancestryChanged },
   664→    { label: "shadow", changed: result.shadowChanged },

   668→
   669→  return (
   670→    <div className="flex items-center gap-2 mb-1">
   671→      <span className="text-[10px] text-muted-foreground shrink-0">drift score</span>

   673→        {drifts.map((d, i) => (
   674→          <motion.div
   675→            key={d.label}
   676→            className={cn("h-full", d.changed ? "bg-repairing/70" : "bg-verified/30")}

   678→            animate={{ width: `${100 / total}%` }}
   679→            transition={{ delay: i * 0.08, type: "spring", stiffness: 300, damping: 25 }}
