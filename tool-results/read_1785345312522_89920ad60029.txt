    74→    74→class LedgerError(Exception):
    75→    75→    """Base exception for all ledger errors.
    76→    76→
    77→    77→    Attributes:
    78→    78→        code: Machine-readable error code (e.g. ``"LEDGER_UNKNOWN"``).
    79→    79→        detail: Optional dictionary with additional context.
    80→    80→    """
    81→    81→
    82→    82→    def __init__(
    83→    83→        self,
    84→    84→        message: str = "",
    85→    85→        *,
    86→    86→        code: str = "LEDGER_UNKNOWN",
    87→    87→        detail: dict[str, Any] | None = None,
    88→    88→    ) -> None:
    89→    89→        super().__init__(message)
    90→    90→        self.code: str = code
    91→    91→        self.detail: dict[str, Any] | None = detail
    92→    92→
    93→    93→    def __repr__(self) -> str:  # pragma: no cover
    94→    94→        return f"{type(self).__name__}(code={self.code!r}, message={str(self)!r})"
