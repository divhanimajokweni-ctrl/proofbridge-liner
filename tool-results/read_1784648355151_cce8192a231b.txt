    87→    87→function ScrollToTopButton() {
    88→    88→  const [visible, setVisible] = useState(false);
    89→    89→
    90→    90→  useEffect(() => {
    91→    91→    const onScroll = () => setVisible(window.scrollY > 300);

    93→    93→    return () => window.removeEventListener("scroll", onScroll);
    94→    94→  }, []);
    95→    95→

    97→    97→    window.scrollTo({ top: 0, behavior: "smooth" });

   101→   101→    <AnimatePresence>
   102→   102→      {visible && (
