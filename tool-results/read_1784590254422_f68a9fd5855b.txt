    86→/* ─── Color helpers ─── */
    87→function coverageColor(pct: number): string {
    88→  if (pct >= 80) return "oklch(0.78 0.16 160)"; // green
    89→  if (pct >= 60) return "oklch(0.70 0.13 120)"; // light green
    90→  if (pct >= 40) return "oklch(0.75 0.15 80)";  // amber
    91→  if (pct >= 20) return "oklch(0.72 0.13 50)";  // orange
    92→  return "oklch(0.65 0.2 25)";                   // red
    93→}
    94→
    95→function coverageTextClass(pct: number): string {
    96→  if (pct >= 80) return "text-verified";
    97→  if (pct >= 60) return "text-emerald-400";
    98→  if (pct >= 40) return "text-repairing";
    99→  if (pct >= 20) return "text-orange-400";
   100→  return "text-violating";
   101→}
   102→
   103→function coverageBgClass(pct: number): string {
   104→  if (pct >= 80) return "bg-verified/10 border-verified/30";
   105→  if (pct >= 60) return "bg-emerald-500/10 border-emerald-500/30";
   106→  if (pct >= 40) return "bg-repairing/10 border-repairing/30";
   107→  if (pct >= 20) return "bg-orange-500/10 border-orange-500/30";
   108→  return "bg-violating/10 border-violating/30";
   109→}
   110→
   111→function gapSeverityClass(severity: "critical" | "high" | "medium"): string {
   112→  if (severity === "critical") return "border-violating/40 bg-violating/10 text-violating";
   113→  if (severity === "high") return "border-repairing/40 bg-repairing/10 text-repairing";
   114→  return "border-quarantined/40 bg-quarantined/10 text-quarantined";
   115→}

   117→/* ─── Mock data generation ─── */
   118→function generateCoverageData(): CoverageData {
   119→  const shardNames = [
   120→    "us-east-1-alpha", "us-east-1-beta", "us-west-2-gamma",
   121→    "eu-west-1-delta", "eu-central-1-epsilon", "ap-south-1-zeta",
   122→    "ap-northeast-1-eta", "us-east-1-theta", "sa-east-1-iota",
   123→    "eu-north-1-kappa", "us-west-1-lambda", "ap-southeast-1-mu",
   124→  ];
   125→
   126→  const statuses: ShardStatus[] = ["healthy", "repairing", "violating"];
   127→
   128→  const shards: ShardCoverage[] = shardNames.map((name, i) => {
   129→    const coveragePct = Math.round(
   130→      i < 3 ? 85 + Math.random() * 15 :
   131→      i < 6 ? 50 + Math.random() * 30 :
   132→      i < 9 ? 20 + Math.random() * 30 :
   133→      Math.random() * 20
   134→    );
   135→    return {
   136→      id: `shard-${i}`,
   137→      name,
   138→      hash: `${name.slice(0, 3)}${Math.random().toString(36).slice(2, 8)}`,
   139→      invariantCount: Math.floor(8 + Math.random() * 50),
   140→      coveragePct: Math.min(100, coveragePct),
   141→      status: coveragePct >= 70 ? "healthy" : coveragePct >= 40 ? "repairing" : "violating",
   142→      lastChecked: new Date(Date.now() - Math.floor(Math.random() * 3600000)).toISOString(),
   143→      trend: coveragePct >= 70 ? (Math.random() > 0.3 ? "up" : "stable") :
   144→             coveragePct >= 40 ? (Math.random() > 0.5 ? "stable" : "down") :
   145→             (Math.random() > 0.4 ? "down" : "stable"),
   146→    };
   147→  });
   148→
   149→  const totalInvariants = shards.reduce((s, sh) => s + sh.invariantCount, 0);
   150→  const avgCoverage = Math.round(shards.reduce((s, sh) => s + sh.coveragePct, 0) / shards.length);
   151→  const uncoveredCount = shards.reduce((s, sh) => s + Math.round(sh.invariantCount * (1 - sh.coveragePct / 100)), 0);
   152→  const criticalGaps = shards.filter((sh) => sh.coveragePct === 0).reduce((s, sh) => s + sh.invariantCount, 0);
   153→
   154→  // Distribution buckets
   155→  const buckets = [
   156→    { label: "0-20%", min: 0, max: 20, fill: "oklch(0.65 0.2 25)" },
   157→    { label: "21-40%", min: 21, max: 40, fill: "oklch(0.72 0.13 50)" },
   158→    { label: "41-60%", min: 41, max: 60, fill: "oklch(0.75 0.15 80)" },
   159→    { label: "61-80%", min: 61, max: 80, fill: "oklch(0.70 0.13 120)" },
   160→    { label: "81-100%", min: 81, max: 100, fill: "oklch(0.78 0.16 160)" },
   161→  ];
   162→
   163→  const distribution = buckets.map((b) => ({
   164→    bucket: b.label,
   165→    count: shards.filter((sh) => sh.coveragePct >= b.min && sh.coveragePct <= b.max)
   166→      .reduce((s, sh) => s + sh.invariantCount, 0),
   167→    fill: b.fill,
   168→  }));
   169→
   170→  // Coverage gaps
   171→  const gaps: CoverageGap[] = [];
   172→  for (const shard of shards) {
   173→    if (shard.coveragePct < 40) {
   174→      const uncoveredCountForShard = Math.ceil(shard.invariantCount * (1 - shard.coveragePct / 100));
   175→      const count = Math.min(uncoveredCountForShard, 3);
   176→      for (let j = 0; j < count; j++) {
   177→        gaps.push({
   178→          id: `gap-${shard.id}-${j}`,
   179→          invariantName: `inv_${shard.name.replace(/-/g, "_")}_${["bounds_check", "state_integrity", "prop_constraint", "drift_limit", "hash_valid"][j % 5]}`,
   180→          shardName: shard.name,
   181→          shardHash: shard.hash,
   182→          coveragePct: Math.max(0, shard.coveragePct - Math.floor(Math.random() * 20)),
   183→          severity: shard.coveragePct < 10 ? "critical" : shard.coveragePct < 25 ? "high" : "medium",
   184→        });
   185→      }
   186→    }
   187→  }
   188→
   189→  return {
   190→    shards,
   191→    totalInvariants,
   192→    avgCoverage,
   193→    uncoveredCount,
   194→    criticalGaps,
   195→    distribution,
   196→    generatedAt: Date.now(),
   197→  };
   198→}

   200→/* ─── Custom Treemap Content ─── */
   201→function CoverageTreemapContent(props: {
   202→  x: number; y: number; width: number; height: number;
   203→  name: string; size: number; coverage: number; depth: number;
   204→}) {
   205→  const { x, y, width, height, name, size, coverage, depth } = props;
   206→  if (depth !== 1 || width < 20 || height < 16) return null;
   207→
   208→  const fillColor = coverageColor(coverage);
   209→  const opacity = 0.15 + (coverage / 100) * 0.35;
   210→
   211→  return (
   212→    <g>
   213→      <rect
   214→        x={x} y={y} width={width} height={height}
   215→        fill={fillColor} opacity={opacity} rx={3}
   216→        stroke={fillColor} strokeWidth={1} strokeOpacity={0.5}
   217→      />
   218→      {width > 50 && height > 22 && (
   219→        <text x={x + 6} y={y + 13} fill="var(--foreground)" fontSize={8} fontFamily="var(--font-geist-mono)" opacity={0.9}>
   220→          {name.length > 14 ? name.slice(0, 12) + "…" : name}
   221→        </text>
   222→      )}
   223→      {width > 50 && height > 36 && (
   224→        <text x={x + 6} y={y + 25} fill={fillColor} fontSize={9} fontFamily="var(--font-geist-mono)" fontWeight={600}>
   225→          {coverage}% · {size}
   226→        </text>
   227→      )}
   228→    </g>
   229→  );
   230→}

   251→/* ─── Main Component ─── */
   252→export function CoverageTreemapSection() {
   253→  const [data, setData] = useState<CoverageData | null>(null);
   254→  const [loading, setLoading] = useState(true);
   255→  const [autoRefresh, setAutoRefresh] = useState(false);
   256→  const [sortKey, setSortKey] = useState<SortKey>("coveragePct");
   257→  const [sortDir, setSortDir] = useState<SortDir>("asc");
   258→  const [lastRefresh, setLastRefresh] = useState<number>(Date.now());
   259→
   260→  const load = useCallback(() => {
   261→    const d = generateCoverageData();
   262→    setData(d);
   263→    setLastRefresh(Date.now());
   264→    setLoading(false);
   265→  }, []);
   266→
   267→  useEffect(() => { load(); }, [load]);
   268→
   269→  useEffect(() => {
   270→    if (!autoRefresh) return;
   271→    const t = setInterval(load, 30000);
   272→    return () => clearInterval(t);
   273→  }, [autoRefresh, load]);
   274→
   275→  /* ─── Derived data ─── */
   276→  const treemapData = useMemo(() => {
   277→    if (!data) return [];
   278→    return data.shards.map((sh) => ({
   279→      name: sh.name,
   280→      size: sh.invariantCount,
   281→      coverage: sh.coveragePct,
   282→    }));
   283→  }, [data]);
   284→
   285→  const sortedShards = useMemo(() => {
   286→    if (!data) return [];
   287→    const arr = [...data.shards];
   288→    return arr.sort((a, b) => {
   289→      let cmp = 0;
   290→      if (sortKey === "name") cmp = a.name.localeCompare(b.name);
   291→      else if (sortKey === "invariantCount") cmp = a.invariantCount - b.invariantCount;
   292→      else if (sortKey === "coveragePct") cmp = a.coveragePct - b.coveragePct;
   293→      else if (sortKey === "lastChecked") cmp = new Date(a.lastChecked).getTime() - new Date(b.lastChecked).getTime();
   294→      return sortDir === "asc" ? cmp : -cmp;
   295→    }).slice(0, 10);
   296→  }, [data, sortKey, sortDir]);
   297→
   298→  const coverageGaps = useMemo(() => {
   299→    if (!data) return [];
   300→    const gaps: CoverageGap[] = [];
   301→    for (const shard of data.shards) {
   302→      if (shard.coveragePct < 40) {
   303→        const uncovered = Math.ceil(shard.invariantCount * (1 - shard.coveragePct / 100));
   304→        const count = Math.min(uncovered, 3);
   305→        for (let j = 0; j < count; j++) {
   306→          gaps.push({
   307→            id: `gap-${shard.id}-${j}`,
   308→            invariantName: `inv_${shard.name.replace(/-/g, "_")}_${["bounds_check", "state_integrity", "prop_constraint", "drift_limit", "hash_valid"][j % 5]}`,
   309→            shardName: shard.name,
   310→            shardHash: shard.hash,
   311→            coveragePct: Math.max(0, shard.coveragePct - Math.floor(j * 8)),
   312→            severity: shard.coveragePct < 10 ? "critical" : shard.coveragePct < 25 ? "high" : "medium",
   313→          });
   314→        }
   315→      }
   316→    }
   317→    return gaps;
   318→  }, [data]);
   319→
   320→  const trendIcon = useMemo(() => {
   321→    if (!data) return Minus;
   322→    return data.avgCoverage >= 70 ? TrendingUp : data.avgCoverage >= 40 ? Minus : TrendingDown;
   323→  }, [data]);
   324→
   325→  const handleSort = useCallback((key: SortKey) => {
   326→    setSortKey((prev) => {
   327→      if (prev === key) {
   328→        setSortDir((d) => d === "asc" ? "desc" : "asc");
   329→      } else {
   330→        setSortDir("asc");
   331→      }
   332→      return key;
   333→    });
   334→  }, []);
   335→
   336→  const fmtTime = (iso: string) => {
   337→    const diff = Date.now() - new Date(iso).getTime();
   338→    const s = Math.floor(diff / 1000);
   339→    if (s < 60) return `${s}s ago`;
   340→    const m = Math.floor(s / 60);
   341→    if (m < 60) return `${m}m ago`;
   342→    const h = Math.floor(m / 60);
   343→    return `${h}h ago`;
   344→  };
   345→
   346→  if (loading) {
   347→    return (
   348→      <div className="flex items-center justify-center py-20">
   349→        <div className="flex items-center gap-3 text-muted-foreground">
   350→          <div className="h-5 w-5 animate-spin rounded-full border-2 border-muted-foreground/30 border-t-verified" />
   351→          <span className="text-sm">Loading coverage data…</span>
   352→        </div>
   353→      </div>
   354→    );
   355→  }
   356→
   357→  if (!data) return null;
   358→
   359→  const TrendIcon = trendIcon;
   360→
   361→  return (
   362→    <motion.div
   363→      className="space-y-5"
   364→      initial="hidden"
   365→      animate="visible"
   366→      variants={containerVariants}
   367→    >
   368→      {/* ── Header Card ── */}
   369→      <motion.div variants={cardVariants}>
   370→        <Card className="relative overflow-hidden bg-card/60 backdrop-blur border-border/60 p-5">
   371→          <TopAccentBar color="oklch(0.78 0.16 160)" />
   372→          <GridOverlay opacity="opacity-10" />
   373→          <div className="relative flex flex-col sm:flex-row items-start sm:items-center gap-3">
   374→            <SectionHeader
   375→              icon={GitGraph}
   376→              title="Coverage Treemap"
   377→              subtitle="Invariant coverage visualization across shards & policies"
   378→              iconClass="border-verified/30 bg-verified/10 text-verified"
   379→            />
   380→            <div className="sm:ml-auto flex flex-wrap items-center gap-2">
   381→              <Button
   382→                variant="outline"
   383→                size="sm"
   384→                className="h-7 gap-1.5 border-border/60 bg-muted/30 text-xs"
   385→                onClick={load}
   386→              >
   387→                <RefreshCw className={cn("h-3 w-3", loading && "animate-spin")} />
   388→                Refresh
   389→              </Button>
   390→              <Button
   391→                variant="outline"
   392→                size="sm"
   393→                className="h-7 gap-1.5 border-verified/30 bg-verified/10 text-verified text-xs hover:bg-verified/20"
   394→              >
   395→                <Download className="h-3 w-3" />
   396→                Export Report
   397→              </Button>
   398→              <div
   399→                className={cn(
   400→                  "flex items-center gap-2 rounded-full border px-3 py-1 text-[10px] cursor-pointer select-none transition-colors",
   401→                  autoRefresh
   402→                    ? "border-verified/30 bg-verified/10 text-verified"
   403→                    : "border-border/60 bg-muted/30 text-muted-foreground"

   405→                onClick={() => setAutoRefresh((v) => !v)}

   407→                <RefreshCw className={cn("h-3 w-3", autoRefresh && "animate-spin")} />

   419→
   420→      {/* ── 2. KPI Strip ── */}

   427→          <Card className="relative overflow-hidden bg-card/60 backdrop-blur border-border/60 p-4 h-full">
