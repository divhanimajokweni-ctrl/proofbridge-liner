    31→/* ─── Helpers ─── */
    32→function fmtTime(iso: string): string {
    33→  const d = Date.now() - new Date(iso).getTime();
    34→  if (d < 60000) return `${Math.floor(d / 1000)}s ago`;
    35→  if (d < 3600000) return `${Math.floor(d / 60000)}m ago`;
    36→  if (d < 86400000) return `${Math.floor(d / 3600000)}h ago`;
    37→  return `${Math.floor(d / 86400000)}d ago`;
    38→}
    39→
    40→function fmtMs(ms: number): string {
    41→  if (ms < 1) return `${(ms * 1000).toFixed(0)}μs`;
    42→  if (ms < 1000) return `${ms.toFixed(1)}ms`;
    43→  return `${(ms / 1000).toFixed(2)}s`;
    44→}
