    59→    59→function fmtRelative(iso: string | null | undefined): string {
    60→    60→  if (!iso) return "—";
    61→    61→  const t = new Date(iso).getTime();
    62→    62→  const diff = Date.now() - t;
    63→    63→  if (diff < 0) return "just now";
    64→    64→  const s = Math.floor(diff / 1000);
    65→    65→  if (s < 60) return `${s}s ago`;
    66→    66→  const m = Math.floor(s / 60);
    67→    67→  if (m < 60) return `${m}m ago`;
    68→    68→  const h = Math.floor(m / 60);
    69→    69→  if (h < 24) return `${h}h ago`;
    70→    70→  return `${Math.floor(h / 24)}d ago`;
    71→    71→}

   339→   339→export function MmrProofsSection() {
   340→   340→  const { toast } = useToast();
   341→   341→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   342→   342→  const [proofs, setProofs] = useState<ProofRow[]>([]);
   343→   343→  const [loading, setLoading] = useState(true);
   344→   344→  const [filterPolicyId, setFilterPolicyId] = useState<string>("all");
   345→   345→  const [mintPolicyId, setMintPolicyId] = useState<string>("");
   346→   346→  const [minting, setMinting] = useState(false);
   347→   347→
   348→   348→  const policyMap = useMemo(() => {
   349→   349→    const m: Record<string, PolicyRow> = {};
   350→   350→    for (const p of policies) m[p.id] = p;
   351→   351→    return m;
   352→   352→  }, [policies]);
   353→   353→
   354→   354→  // Load policies once.
   355→   355→  useEffect(() => {
   356→   356→    let cancelled = false;
   357→   357→    fetch("/api/policies")
   358→   358→      .then((r) => r.json())
   359→   359→      .then((d: { policies: PolicyRow[] }) => {
   360→   360→        if (cancelled) return;
   361→   361→        const ps = d.policies ?? [];
   362→   362→        setPolicies(ps);
   363→   363→        if (ps.length > 0) setMintPolicyId(ps[0].id);
   364→   364→      })
   365→   365→      .catch(() => {});
   366→   366→    return () => {
   367→   367→      cancelled = true;
   368→   368→    };
   369→   369→  }, []);
   370→   370→
   371→   371→  const loadProofs = useCallback(async () => {
   372→   372→    try {
   373→   373→      const url =
   374→   374→        filterPolicyId && filterPolicyId !== "all"
   375→   375→          ? `/api/proofs?policyId=${encodeURIComponent(filterPolicyId)}`
   376→   376→          : "/api/proofs";
   377→   377→      const r = await fetch(url);
   378→   378→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   379→   379→      const d: { proofs: ProofRow[] } = await r.json();
   380→   380→      setProofs(d.proofs ?? []);
   381→   381→    } catch (e) {
   382→   382→      toast({
   383→   383→        title: "Failed to load proofs",
   384→   384→        description: e instanceof Error ? e.message : "Unknown error",
   385→   385→        variant: "destructive",
   386→   386→      });
   387→   387→    } finally {
   388→   388→      setLoading(false);
   389→   389→    }
   390→   390→  }, [filterPolicyId, toast]);
   391→   391→
   392→   392→  useEffect(() => {
   393→   393→    setLoading(true);
   394→   394→    loadProofs();
   395→   395→    const t = setInterval(loadProofs, POLL_MS);
   396→   396→    return () => clearInterval(t);
   397→   397→  }, [loadProofs]);
   398→   398→
   399→   399→  const mintProof = useCallback(async () => {
   400→   400→    if (!mintPolicyId) {
   401→   401→      toast({ title: "Select a policy first", variant: "destructive" });
   402→   402→      return;
   403→   403→    }
   404→   404→    setMinting(true);
   405→   405→    try {
   406→   406→      const r = await fetch("/api/proofs", {
   407→   407→        method: "POST",
   408→   408→        headers: { "Content-Type": "application/json" },
   409→   409→        body: JSON.stringify({ policyId: mintPolicyId }),
   410→   410→      });
   411→   411→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   412→   412→      const d = await r.json();
   413→   413→      const created: ProofRow = {
   414→   414→        ...d.proof,
   415→   415→        verified: true,
   416→   416→        policy: d.proof.policy ?? {
   417→   417→          name: policyMap[mintPolicyId]?.name ?? "unknown",
   418→   418→          zkEnabled: policyMap[mintPolicyId]?.zkEnabled ?? false,
   419→   419→          proofKind: policyMap[mintPolicyId]?.proofKind ?? "mmr",
   420→   420→        },
   421→   421→      };
   422→   422→      setProofs((prev) => [created, ...prev]);
   423→   423→      toast({
   424→   424→        title: "MMR proof minted",
   425→   425→        description: `root ${created.mmrRoot} · ${created.proofPath.length} sibling(s)`,
   426→   426→      });
   427→   427→    } catch (e) {
   428→   428→      toast({
   429→   429→        title: "Mint failed",
   430→   430→        description: e instanceof Error ? e.message : "Unknown error",
   431→   431→        variant: "destructive",
   432→   432→      });
   433→   433→    } finally {
   434→   434→      setMinting(false);
   435→   435→    }
   436→   436→  }, [mintPolicyId, policyMap, toast]);
   437→   437→
   438→   438→  // Summary metrics.
   439→   439→  const total = proofs.length;
   440→   440→  const zkCount = proofs.filter((p) => p.zkProof).length;
   441→   441→  const anchoredCount = proofs.filter((p) => p.anchored).length;
   442→   442→  const anchoredRate = total > 0 ? Math.round((anchoredCount / total) * 100) : 0;
   443→   443→  const allVerified = total > 0 && proofs.every((p) => p.verified);
   444→   444→  const healthy = total > 0 && allVerified;
   445→   445→
   446→   446→  return (
   447→   447→    <TooltipProvider delayDuration={150}>
   448→   448→      <section className="space-y-4">
   449→   449→        {/* Header */}
