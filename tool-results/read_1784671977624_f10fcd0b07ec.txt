    20→function generateSparkline(base: number, variance: number, length = 12): number[] {
    21→  let seed = base * 7 + 31;
    22→  const rand = () => { seed = (seed * 1103515245 + 12345) & 0x7fffffff; return seed / 0x7fffffff; };
    23→  return Array.from({ length }, () => Math.max(0, Math.round((base + (rand() - 0.5) * 2 * variance) * 10) / 10));
    24→}
