    92→function FederationHealthGauge({ channels }: { channels: FedChannel[] }) {
    93→  const verified = channels.filter((c) => c.status === "verified").length;
    94→  const negotiating = channels.filter((c) => c.status === "negotiating").length;
    95→  const drift = channels.filter((c) => c.status === "drift").length;
    96→  const data = [{ name: "verified", value: verified, color: "oklch(0.78 0.16 160)" }, { name: "negotiating", value: negotiating, color: "oklch(0.80 0.15 80)" }, { name: "drift", value: drift, color: "oklch(0.64 0.21 25)" }].filter((d) => d.value > 0);
    97→  const total = verified + negotiating + drift;
    98→  const healthPct = total > 0 ? Math.round((verified / total) * 100) : 0;
    99→  if (!data.length) return <div className="flex items-center justify-center h-[120px] text-xs text-muted-foreground italic">No channels</div>;
   100→  return (
   101→    <div className="relative h-[120px]">
   102→      <ResponsiveContainer width="100%" height="100%">
   103→        <PieChart><Pie data={data} cx="50%" cy="50%" innerRadius={35} outerRadius={48} paddingAngle={2} dataKey="value" strokeWidth={0}>
   104→          {data.map((_, i) => <Cell key={i} fill={data[i].color} fillOpacity={0.8} />)}
   105→        </Pie></PieChart>
   106→      </ResponsiveContainer>
   107→      <div className="absolute inset-0 flex items-center justify-center">
   108→        <div className="text-center">
   109→          <motion.div className={cn("text-lg font-bold font-mono", healthPct >= 80 ? "text-verified" : healthPct >= 50 ? "text-repairing" : "text-violating")}
   110→            initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }} transition={{ type: "spring", stiffness: 300 }}>{healthPct}%</motion.div>
   111→          <div className="text-[8px] uppercase tracking-wide text-muted-foreground">health</div>
   112→        </div>
   113→      </div>
   114→    </div>
   115→  );
   116→}

   193→  const [activeChannelIdx, setActiveChannelIdx] = useState<number | null>(null);
   194→
   195→  const load = useCallback(() => {
   196→    fetch("/api/federation").then((r) => r.json()).then((d) => { setData(d); setLoading(false); }).catch(() => setLoading(false));
   197→  }, []);
   198→
   199→  useEffect(() => { load(); const t = setInterval(load, 12000); return () => clearInterval(t); }, [load]);
   200→

   202→    setReconciling(true);
