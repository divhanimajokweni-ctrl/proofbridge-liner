   392→export function TemplateLibrarySection({ onCreate }: { onCreate?: (source: string, filename: string) => void }) {
   393→  const { toast } = useToast();
   394→  const [selected, setSelected] = useState<Template | null>(null);
   395→  const [editedSource, setEditedSource] = useState("");
   396→  const [policyName, setPolicyName] = useState("");
   397→  const [validation, setValidation] = useState<{ ok: boolean; invCount: number; errors: number } | null>(null);
   398→  const [copied, setCopied] = useState(false);
   399→  const [searchQuery, setSearchQuery] = useState("");
   400→  const [activeCategory, setActiveCategory] = useState("all");
   401→  const [deploying, setDeploying] = useState(false);
   402→
   403→  const openTemplate = (t: Template) => {
   404→    setSelected(t);
   405→    setEditedSource(t.source);
   406→    const m = t.source.match(/policy "([^"]+)"/);
   407→    setPolicyName(m ? m[1] : t.name);
   408→    setValidation(null);
   409→  };
   410→
   411→  const validate = () => {
   412→    const result = validateEpd(editedSource);
   413→    setValidation({
   414→      ok: result.ok,
   415→      invCount: result.invariantCount,
   416→      errors: result.diagnostics.filter((d) => d.level === "error").length,
   417→    });
   418→    if (result.ok) {
   419→      toast({ title: "Template valid", description: `${result.invariantCount} invariants compiled` });
   420→    } else {
   421→      toast({ title: "Validation failed", description: `${result.diagnostics.filter((d) => d.level === "error").length} error(s)`, variant: "destructive" });
   422→    }
   423→  };
   424→
   425→  const create = async () => {
   426→    const result = validateEpd(editedSource);
   427→    if (!result.ok) {
   428→      toast({ title: "Cannot create — fix errors first", variant: "destructive" });
   429→      return;
   430→    }
   431→    setDeploying(true);
   432→    const filename = `${policyName.replace(/[^a-zA-Z0-9]+/g, "-").toLowerCase()}.epd`;
   433→    try {
   434→      const res = await fetch("/api/policies", {
   435→        method: "POST",
   436→        headers: { "Content-Type": "application/json" },
   437→        body: JSON.stringify({ source: editedSource, filename }),
   438→      });
   439→      const d = await res.json();
   440→      if (!res.ok) {
   441→        toast({ title: "Create failed", description: d.error ?? "invalid", variant: "destructive" });
   442→      } else {
   443→        toast({ title: "Policy created", description: `${d.policy.name} saved & compiled` });
   444→        setSelected(null);
   445→        onCreate?.(editedSource, filename);
   446→      }
   447→    } catch (e) {
   448→      toast({ title: "Create failed", description: String(e), variant: "destructive" });
   449→    } finally {
   450→      setDeploying(false);
   451→    }
   452→  };
   453→
   454→  const copy = async () => {
   455→    try {
   456→      await navigator.clipboard.writeText(editedSource);
   457→      setCopied(true);
   458→      setTimeout(() => setCopied(false), 1400);
   459→      toast({ title: "Source copied" });
   460→    } catch {
   461→      toast({ title: "Copy failed", variant: "destructive" });
   462→    }
   463→  };
   464→
   465→  const filteredTemplates = useMemo(() => {
   466→    return TEMPLATES.filter((t) => {
   467→      const matchesCategory = activeCategory === "all" || t.category === activeCategory;
   468→      const matchesSearch = !searchQuery ||
   469→        t.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   470→        t.domain.toLowerCase().includes(searchQuery.toLowerCase()) ||
   471→        t.description.toLowerCase().includes(searchQuery.toLowerCase());
   472→      return matchesCategory && matchesSearch;
   473→    });
   474→  }, [activeCategory, searchQuery]);
   475→
   476→  const totalInvariants = TEMPLATES.reduce((sum, t) => sum + t.invariants, 0);
   477→  const domains = new Set(TEMPLATES.map((t) => t.domain)).size;
   478→
   479→  return (
   480→    <section className="space-y-4">
   481→      {/* Header */}
   482→      <motion.div
   483→        initial={{ opacity: 0, y: -8 }}
   484→        animate={{ opacity: 1, y: 0 }}
   485→        transition={{ duration: 0.4 }}
   486→      >
   487→        <Card className="bg-card/60 backdrop-blur border-border/60 p-4 relative overflow-hidden">
   488→          <div className="absolute top-0 left-0 right-0 h-[2px] bg-gradient-to-r from-verified/0 via-verified/50 to-verified/0" />
   489→          <div className="bg-grid absolute inset-0 opacity-20" />
   490→          <div className="relative flex flex-wrap items-center gap-3">
   491→            <div className="flex h-10 w-10 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 shrink-0">
   492→              <Library className="h-5 w-5 text-verified" />
   493→            </div>
   494→            <div className="min-w-0 flex-1">
   495→              <h2 className="text-base font-semibold flex items-center gap-2">
   496→                Policy Template Library
   497→                <Badge variant="outline" className="border-verified/30 bg-verified/10 text-verified text-[9px] font-mono">
   498→                  {TEMPLATES.length} templates
   499→                </Badge>
   500→              </h2>
   501→              <p className="text-xs text-muted-foreground mt-0.5">
   502→                Start from a domain template — edit, validate, and compile a new `.epd` policy in one click.
   503→              </p>
   504→            </div>
   505→            {/* Stats */}
   506→            <div className="flex items-center gap-3 shrink-0">
   507→              <div className="text-center">
   508→                <div className="font-mono text-lg font-bold text-verified">{totalInvariants}</div>
   509→                <div className="text-[9px] uppercase tracking-wide text-muted-foreground">invariants</div>
   510→              </div>
   511→              <Separator orientation="vertical" className="h-8 bg-border/40" />
   512→              <div className="text-center">
   513→                <div className="font-mono text-lg font-bold text-foreground">{domains}</div>
   514→                <div className="text-[9px] uppercase tracking-wide text-muted-foreground">domains</div>
   515→              </div>
   516→            </div>
   517→          </div>
   518→        </Card>
   519→      </motion.div>
   520→
   521→      {/* Search + Category filter */}
   522→      <motion.div
   523→        initial={{ opacity: 0, y: 8 }}
   524→        animate={{ opacity: 1, y: 0 }}
   525→        transition={{ duration: 0.4, delay: 0.05 }}
   526→        className="flex flex-wrap items-center gap-2"
   527→      >
   528→        {/* Search */}
   529→        <div className="relative flex-1 min-w-[200px] max-w-sm">
   530→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   531→          <Input
   532→            value={searchQuery}
   533→            onChange={(e) => setSearchQuery(e.target.value)}
   534→            placeholder="Search templates…"
   535→            className="h-8 pl-8 text-xs bg-card/60 border-border/60"
   536→          />
   537→        </div>
   538→
   539→        {/* Category filter buttons */}
   540→        <div className="flex items-center gap-1">
   541→          {CATEGORIES.map((cat) => {
   542→            const Icon = cat.icon;
   543→            const isActive = activeCategory === cat.id;
   544→            return (
   545→              <motion.button
   546→                key={cat.id}
   547→                whileTap={{ scale: 0.95 }}
   548→                onClick={() => setActiveCategory(cat.id)}
   549→                className={cn(
   550→                  "inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs font-medium transition-colors",
   551→                  isActive
   552→                    ? "border-verified/40 bg-verified/10 text-verified"
   553→                    : "border-border/60 bg-muted/30 text-muted-foreground hover:text-foreground hover:border-border",
   554→                )}
   555→              >
   556→                <Icon className="h-3 w-3" />
   557→                <span className="hidden sm:inline">{cat.label}</span>
   558→              </motion.button>
   559→            );
   560→          })}
   561→        </div>
   562→      </motion.div>
   563→
   564→      {/* Template grid */}
   565→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   566→        <AnimatePresence mode="popLayout">
   567→          {filteredTemplates.map((t, i) => {
   568→            const Icon = t.icon;
   569→            const domainColor = DOMAIN_COLORS[t.domain] ?? "border-border/60";
   570→            const iconBgColor = DOMAIN_ICON_COLORS[t.domain] ?? "bg-verified/10 border-verified/20 text-verified group-hover:bg-verified/20";
   571→            return (
   572→              <motion.div
   573→                key={t.id}
   574→                initial={{ opacity: 0, y: 16, scale: 0.97 }}
   575→                animate={{ opacity: 1, y: 0, scale: 1 }}
   576→                exit={{ opacity: 0, scale: 0.95 }}
   577→                transition={{ duration: 0.3, delay: i * 0.05 }}
   578→                layout
   579→              >
   580→                <Card
   581→                  className="bg-card/60 backdrop-blur border-border/60 p-4 relative overflow-hidden hover:border-verified/40 transition-all duration-200 cursor-pointer group"
   582→                  onClick={() => openTemplate(t)}
   583→                >
   584→                  {/* Gradient border top accent */}
   585→                  <div className={cn(
   586→                    "absolute top-0 left-0 right-0 h-[2px]",
   587→                    t.domain === "iot" ? "bg-gradient-to-r from-teal-500/0 via-teal-500/50 to-teal-500/0" :
   588→                    t.domain === "datacenter" ? "bg-gradient-to-r from-verified/0 via-verified/50 to-verified/0" :
   589→                    t.domain === "healthcare" ? "bg-gradient-to-r from-rose-500/0 via-rose-500/50 to-rose-500/0" :
   590→                    t.domain === "energy" ? "bg-gradient-to-r from-amber-500/0 via-amber-500/50 to-amber-500/0" :
   591→                    "bg-gradient-to-r from-verified/0 via-verified/30 to-verified/0",
   592→                  )} />
   593→                  <div className="bg-grid-fine absolute inset-0 opacity-20" />
   594→                  <div className="relative">
   595→                    <div className="flex items-start gap-3">
