    83→function synthesizeCircuit(policy: PolicyNode): Circuit {
    84→  const gates: CircuitGate[] = [];
    85→  const wires: CircuitWire[] = [];
    86→
    87→  const fields = new Set<string>();
    88→  for (const inv of policy.invariants) {
    89→    if (inv.predicate) {
    90→      extractIdents(inv.predicate, fields);
    91→    }
    92→  }
    93→  const privateInputs = Array.from(fields);
    94→  const publicInputs = ["mmr_root", "policy_hash", "nonce"];
    95→
    96→  const inputX = 40;
    97→  const constraintX = 220;
    98→  const aggregateX = 420;
    99→  const outputX = 560;
   100→
   101→  privateInputs.forEach((f, i) => {
   102→    const y = 50 + i * 42;
   103→    gates.push({
   104→      id: `in-${f}`,
   105→      type: "input",
   106→      label: f,
   107→      sublabel: "private",
   108→      x: inputX,
   109→      y,
   110→    });
   111→    policy.invariants.forEach((inv) => {
   112→      const invFields = new Set<string>();
   113→      if (inv.predicate) extractIdents(inv.predicate, invFields);
   114→      if (invFields.has(f)) {
   115→        wires.push({ from: `in-${f}`, to: `c-${inv.name}`, kind: "private" });
   116→      }
   117→    });
   118→  });
   119→
   120→  policy.invariants.forEach((inv, j) => {
   121→    const y = 50 + j * 52;
   122→    gates.push({
   123→      id: `c-${inv.name}`,
   124→      type: "constraint",
   125→      label: inv.name,
   126→      sublabel: inv.severity + (inv.soft ? " · soft" : ""),
   127→      x: constraintX,
   128→      y,
   129→      invariant: inv.name,
   130→    });
   131→    wires.push({ from: `c-${inv.name}`, to: `agg-all`, kind: "private" });
   132→  });
   133→
   134→  gates.push({
   135→    id: "agg-all",
   136→    type: "aggregate",
   137→    label: "ALL(invariants)",
   138→    sublabel: "conjunction",
   139→    x: aggregateX,
   140→    y: 50 + (policy.invariants.length - 1) * 26,
   141→  });
   142→
   143→  gates.push({
   144→    id: "out-pass",
   145→    type: "output",
   146→    label: "pass",
   147→    sublabel: "public",
   148→    x: outputX,
   149→    y: 50 + (policy.invariants.length - 1) * 26 - 20,
   150→  });
   151→  gates.push({
   152→    id: "out-root",
   153→    type: "output",
   154→    label: "new_mmr_root",
   155→    sublabel: "public",
   156→    x: outputX,
   157→    y: 50 + (policy.invariants.length - 1) * 26 + 20,
   158→  });
   159→  wires.push({ from: "agg-all", to: "out-pass", kind: "public" });
   160→  wires.push({ from: "agg-all", to: "out-root", kind: "public" });
   161→
   162→  publicInputs.forEach((p, i) => {
   163→    gates.push({
   164→      id: `pub-${p}`,
   165→      type: "input",
   166→      label: p,
   167→      sublabel: "public",
   168→      x: inputX,
   169→      y: 50 + (privateInputs.length + i) * 42 + 20,
   170→    });
   171→    wires.push({ from: `pub-${p}`, to: "agg-all", kind: "public" });
   172→  });
   173→
   174→  const constraintCount = policy.invariants.length + privateInputs.length * 2 + 4;
   175→  const estimatedRows = constraintCount * 8 + privateInputs.length * 3;
   176→  const provingKeySize = `${(estimatedRows * 0.12).toFixed(1)} MB`;
   177→  const verificationKeySize = `${(constraintCount * 0.04 + 0.8).toFixed(1)} KB`;
   178→
   179→  return {
   180→    gates,
   181→    wires,
   182→    publicInputs,
   183→    privateInputs,
   184→    constraintCount,
   185→    estimatedRows,
   186→    provingKeySize,
   187→    verificationKeySize,
   188→  };
   189→}
   190→
   191→function extractIdents(expr: unknown, out: Set<string>) {
   192→  if (!expr || typeof expr !== "object") return;
   193→  const e = expr as Record<string, unknown>;
   194→  if (e.kind === "ident") {
   195→    out.add(e.name as string);
   196→  } else if (e.kind === "unary") {
   197→    extractIdents(e.operand, out);
   198→  } else if (e.kind === "binary" || e.kind === "logic" || e.kind === "compare") {
   199→    extractIdents(e.left, out);
   200→    extractIdents(e.right, out);
   201→  } else if (e.kind === "in") {
   202→    extractIdents(e.value, out);
   203→    extractIdents((e.range as unknown[])[0], out);
   204→    extractIdents((e.range as unknown[])[1], out);
   205→  } else if (e.kind === "call") {
   206→    (e.args as unknown[]).forEach((a) => extractIdents(a, out));
   207→  }
   208→}

   210→function generateProofHistory(circuit: Circuit): ProofHistoryEntry[] {
   211→  const entries: ProofHistoryEntry[] = [];
   212→  const now = Date.now();
   213→  for (let i = 0; i < 8; i++) {
   214→    const status = i === 2 ? "failed" : i === 5 ? "pending" : "verified";
   215→    entries.push({
   216→      id: `proof-${i + 1}`,
   217→      timestamp: new Date(now - (8 - i) * 180000).toISOString(),
   218→      status,
   219→      provingTimeMs: Math.round(800 + Math.random() * 2400),
   220→      constraintCount: circuit.constraintCount + Math.round(Math.random() * 4 - 2),
   221→      shardKey: `shard-${String.fromCharCode(65 + (i % 3))}`,
   222→    });
   223→  }
   224→  return entries;
   225→}

   628→function bezierPath(x1: number, y1: number, x2: number, y2: number): string {
   629→  const cx1 = x1 + (x2 - x1) * 0.4;
   630→  const cy1 = y1;
   631→  const cx2 = x2 - (x2 - x1) * 0.4;
   632→  const cy2 = y2;
   633→  return `M ${x1},${y1} C ${cx1},${cy1} ${cx2},${cy2} ${x2},${y2}`;
   634→}

   636→function GateShape({ gate, revealPrivate, isHovered, onHover }: { gate: CircuitGate; revealPrivate: boolean; isHovered: boolean; onHover: (id: string | null) => void }) {
   637→  const w = 90;
   638→  const h = 30;
   639→  const x = gate.x;
   640→  const y = gate.y - h / 2;
   641→  const isPrivate = gate.sublabel === "private";
   642→  const hidden = isPrivate && !revealPrivate;
   643→
   644→  const config: Record<
   645→    CircuitGate["type"],
   646→    { fill: string; stroke: string; text: string }
   647→  > = {
   648→    input: { fill: "oklch(0.22 0.014 168)", stroke: "oklch(0.45 0.01 165)", text: "oklch(0.85 0.01 165)" },
   649→    constraint: { fill: "oklch(0.25 0.03 80 / 0.6)", stroke: "oklch(0.80 0.15 80 / 0.7)", text: "oklch(0.90 0.10 80)" },
   650→    aggregate: { fill: "oklch(0.25 0.03 40 / 0.6)", stroke: "oklch(0.70 0.13 40 / 0.8)", text: "oklch(0.90 0.10 40)" },
   651→    output: { fill: "oklch(0.22 0.05 160 / 0.7)", stroke: "oklch(0.78 0.16 160)", text: "oklch(0.90 0.12 160)" },
   652→  };
   653→  const c = config[gate.type];
   654→
   655→  return (
   656→    <g
   657→      onMouseEnter={() => onHover(gate.id)}
   658→      onMouseLeave={() => onHover(null)}
   659→      style={{ cursor: "pointer" }}
   660→    >
   661→      <rect
   662→        x={x}
   663→        y={y}
   664→        width={w}
   665→        height={h}
   666→        rx={4}
   667→        fill={hidden ? "oklch(0.20 0.01 165)" : c.fill}
   668→        stroke={hidden ? "oklch(0.35 0.02 165)" : c.stroke}
   669→        strokeWidth={isHovered ? 2 : 1.2}
   670→        strokeDasharray={hidden ? "3 2" : "0"}
   671→        filter={isHovered ? "url(#gate-glow)" : "url(#gate-shadow)"}
   672→      />
   673→      {/* Animated pulse on hover */}
   674→      {isHovered && (
   675→        <rect
   676→          x={x - 2}
   677→          y={y - 2}
   678→          width={w + 4}
   679→          height={h + 4}
   680→          rx={5}
   681→          fill="none"
   682→          stroke={c.stroke}
   683→          strokeWidth={0.5}
   684→          opacity={0.5}
   685→        >
   686→          <animate attributeName="opacity" values="0.5;0.1;0.5" dur="1.5s" repeatCount="indefinite" />
   687→        </rect>
   688→      )}
   689→      <text
   690→        x={x + w / 2}
   691→        y={y + 13}
   692→        textAnchor="middle"
   693→        fontSize={9.5}
   694→        fontFamily="var(--font-geist-mono), monospace"
   695→        fill={hidden ? "oklch(0.50 0.02 165)" : c.text}
   696→        fontWeight={gate.type === "output" || gate.type === "aggregate" ? 600 : 400}
   697→      >
   698→        {hidden ? "•••••" : gate.label.length > 14 ? gate.label.slice(0, 12) + "…" : gate.label}
   699→      </text>
   700→      <text
   701→        x={x + w / 2}
   702→        y={y + 24}
   703→        textAnchor="middle"
   704→        fontSize={7}
   705→        fill="oklch(0.55 0.01 165)"
   706→      >
   707→        {gate.sublabel}
   708→      </text>
   709→    </g>
   710→  );
