    99→def derive_intent(text):
   100→    """
   101→    Auto-derive design intent from document title/description.
   102→    Scans for theme keywords and returns the best-matching intent.
   103→    Falls back to 'neutral' if no keywords match.
   104→    
   105→    Usage:
   106→        python3 design_engine.py derive "Social Media Operations Monthly Report"  → energy
   107→        python3 design_engine.py derive "2025 Annual Sustainability Report"  → nature
   108→    """
   109→    text_lower = text.lower()
   110→    scores = {}
   111→    for keyword, intent in THEME_KEYWORDS.items():
   112→        if keyword.lower() in text_lower:
   113→            scores[intent] = scores.get(intent, 0) + 1
   114→    if not scores:
   115→        return "neutral"
   116→    # When tied, prefer specific intents over 'neutral' (which is a generic fallback)
   117→    max_score = max(scores.values())
   118→    top_intents = [k for k, v in scores.items() if v == max_score]
   119→    if len(top_intents) > 1 and "neutral" in top_intents:
   120→        top_intents.remove("neutral")
   121→    return top_intents[0]

   124→# Used as fallback when LLM doesn't specify color_harmony
   125→INTENT_HARMONY_MAP = {

   128→    "energy":     "triadic",               # vibrant, multi-directional

   141→def _hsl_to_hex(h, s, l):
   142→    """Convert HSL (h: 0-360, s: 0-1, l: 0-1) to hex string."""
   143→    r, g, b = colorsys.hls_to_rgb(h / 360.0, l, s)
   144→    return f"#{int(r*255):02x}{int(g*255):02x}{int(b*255):02x}"
   145→
   146→
   147→# Muddy / ugly hue zones that produce unattractive accent colors at medium saturation.
   148→# These hue ranges tend to look "dirty" or "sickly" — yellow-green, brown-orange, etc.
   149→# When an accent lands here, we nudge it to the nearest attractive neighbor.
   150→_UGLY_HUE_ZONES = [
   151→    # (start, end, nudge_low, nudge_high) — hues that look muddy
   152→    # at S 0.4-0.7, L 0.35-0.55. When accent lands here, redirect.
   153→    (28, 105, 15, 120),  # Dirty yellow/brown/olive/yellow-green zone
   154→                          # nudge_low=15 (warm red-orange), nudge_high=120 (true green)
   155→]
   156→
   157→
   158→def _sanitize_accent_hue(hue):
   159→    """Nudge accent hue away from muddy/ugly zones toward attractive neighbors."""
   160→    hue = hue % 360
   161→    for start, end, nudge_low, nudge_high in _UGLY_HUE_ZONES:
   162→        if start <= hue <= end:
   163→            mid = (start + end) / 2
   164→            return nudge_low if hue < mid else nudge_high
   165→    return hue
   166→
   167→def _hex_to_rgb(hex_str):
   168→    """Convert hex to 'r,g,b' string for rgba() usage."""
   169→    hex_str = hex_str.lstrip('#')
   170→    return f"{int(hex_str[0:2], 16)},{int(hex_str[2:4], 16)},{int(hex_str[4:6], 16)}"
   171→
   172→def _relative_luminance(hex_str):
   173→    """WCAG 2.1 relative luminance from hex color."""
   174→    hex_str = hex_str.lstrip('#')
   175→    channels = []
   176→    for i in (0, 2, 4):
   177→        c = int(hex_str[i:i+2], 16) / 255.0
   178→        channels.append(c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4)
   179→    return 0.2126 * channels[0] + 0.7152 * channels[1] + 0.0722 * channels[2]
   180→
   181→def _contrast_ratio(hex1, hex2):
   182→    """WCAG contrast ratio between two hex colors."""
   183→    l1, l2 = _relative_luminance(hex1), _relative_luminance(hex2)
   184→    lighter, darker = max(l1, l2), min(l1, l2)
   185→    return (lighter + 0.05) / (darker + 0.05)

   187→def generate_color_palette(intent="neutral", mode="minimal", harmony=None, seed=None):
   188→    """
   189→    Generative Color Harmony Engine — geometric accent computation + 5 aesthetic modes.
   190→    
   191→    LLM is FORBIDDEN from specifying HEX/RGB. It only selects:
   192→      - intent   → base hue (from INTENT_HUES)
   193→      - mode     → S/L physical boundaries (minimal/dark/pastel/jewel/light)
   194→      - harmony  → accent hue geometry (auto-recommended from intent if omitted)
   195→    
   196→    Returns dict with:
   197→      - bg:      60% — dominant ground
   198→      - mid:     30% — structural shade
   199→      - accent:  10% — geometric harmony emphasis
   200→      - text:    primary text color
   201→      - muted:   secondary/caption text
   202→      - surface: card/container background (translucent)
   203→    """
   204→    if seed is not None:
   205→        random.seed(seed)
   206→
   207→    # Auto-recommend harmony from intent if not specified
   208→    if harmony is None or harmony == "auto":
   209→        harmony = INTENT_HARMONY_MAP.get(intent, "split_complementary")
   210→
   211→    base_hue = INTENT_HUES.get(intent, random.randint(0, 359))
   212→
   213→    # 1. Geometric Accent Hue Derivation
   214→    if harmony == "complementary":
   215→        accent_hue = (base_hue + 180) % 360
   216→    elif harmony == "split_complementary":
   217→        accent_hue = (base_hue + random.choice([150, 210])) % 360
   218→    elif harmony == "triadic":
   219→        accent_hue = (base_hue + random.choice([120, 240])) % 360
   220→    elif harmony == "analogous":
   221→        accent_hue = (base_hue + random.choice([30, -30, 45, -45])) % 360
   222→    else:  # monochrome
   223→        accent_hue = base_hue
   224→
   225→    # Sanitize accent hue — avoid muddy yellow-green and brown-orange zones
   226→    accent_hue = _sanitize_accent_hue(accent_hue)
   227→
   228→    # 2. Five Aesthetic Modes — hard-lock S and L boundaries
   229→    if mode == "minimal":
   230→        # DEFAULT for 50%+ of documents. Editorial paper tone + high-purity accent.
   231→        # bg has visible tint (S 0.08-0.15, L 0.92-0.96) — not dead white.
   232→        is_warm = random.choice([True, False])
   233→        paper_hue = random.randint(35, 45) if is_warm else random.randint(200, 215)
   234→        bg      = _hsl_to_hex(paper_hue,   random.uniform(0.08, 0.15), random.uniform(0.92, 0.96))
   235→        mid     = _hsl_to_hex(paper_hue,   random.uniform(0.10, 0.18), random.uniform(0.85, 0.90))
   236→        accent  = _hsl_to_hex(accent_hue,  random.uniform(0.55, 0.72), random.uniform(0.42, 0.52))
   237→        text    = _hsl_to_hex(paper_hue,   0.05,                       random.uniform(0.10, 0.15))
   238→        muted   = _hsl_to_hex(paper_hue,   0.05,                       random.uniform(0.45, 0.55))
   239→        surface = f"rgba(0,0,0,{random.uniform(0.01, 0.03):.2f})"
   240→
   241→    elif mode == "dark":
   242→        # Cyber/tech: ultra-low saturation, ultra-low lightness
   243→        bg      = _hsl_to_hex(base_hue,    random.uniform(0.04, 0.10), random.uniform(0.04, 0.08))
   244→        mid     = _hsl_to_hex(base_hue,    random.uniform(0.08, 0.15), random.uniform(0.12, 0.20))
   245→        accent  = _hsl_to_hex(accent_hue,  random.uniform(0.45, 0.60), random.uniform(0.52, 0.62))
   246→        text    = _hsl_to_hex(base_hue,    0.05,                       random.uniform(0.88, 0.95))
   247→        muted   = _hsl_to_hex(base_hue,    0.05,                       random.uniform(0.45, 0.55))
   248→        surface = f"rgba(255,255,255,{random.uniform(0.03, 0.06):.2f})"
   249→
   250→    elif mode == "pastel":
   251→        # Morandi/macaron: medium-low saturation, high lightness
   252→        # Accent tightened: S capped at 0.50 to avoid clashing with tinted bg
   253→        bg      = _hsl_to_hex(base_hue,    random.uniform(0.15, 0.35), random.uniform(0.85, 0.92))
   254→        mid     = _hsl_to_hex(base_hue,    random.uniform(0.20, 0.40), random.uniform(0.75, 0.82))
   255→        accent  = _hsl_to_hex(accent_hue,  random.uniform(0.38, 0.50), random.uniform(0.38, 0.48))
   256→        text    = _hsl_to_hex(base_hue,    0.15,                       random.uniform(0.15, 0.25))
   257→        muted   = _hsl_to_hex(base_hue,    0.20,                       random.uniform(0.45, 0.55))
   258→        surface = f"rgba(255,255,255,{random.uniform(0.30, 0.50):.2f})"
   259→
   260→    elif mode == "jewel":
   261→        # Gem/luxury: medium-high saturation bg, accent must NOT compete — lower S, higher L
   262→        bg      = _hsl_to_hex(base_hue,    random.uniform(0.40, 0.60), random.uniform(0.15, 0.25))
   263→        mid     = _hsl_to_hex(base_hue,    random.uniform(0.40, 0.60), random.uniform(0.25, 0.35))
   264→        accent  = _hsl_to_hex(accent_hue,  random.uniform(0.30, 0.50), random.uniform(0.65, 0.80))
   265→        text    = _hsl_to_hex(base_hue,    0.10,                       random.uniform(0.90, 0.96))
   266→        muted   = _hsl_to_hex(base_hue,    0.20,                       random.uniform(0.60, 0.70))
   267→        surface = f"rgba(0,0,0,{random.uniform(0.20, 0.40):.2f})"
   268→
   269→    else:
   270→        # light — noticeably tinted, not dead white (S 0.10-0.20, L 0.91-0.95)
   271→        bg      = _hsl_to_hex(base_hue,    random.uniform(0.10, 0.20), random.uniform(0.91, 0.95))
   272→        mid     = _hsl_to_hex(base_hue,    random.uniform(0.15, 0.25), random.uniform(0.84, 0.90))
   273→        accent  = _hsl_to_hex(accent_hue,  random.uniform(0.35, 0.45), random.uniform(0.35, 0.45))
   274→        text    = _hsl_to_hex(base_hue,    0.08,                       random.uniform(0.08, 0.15))
   275→        muted   = _hsl_to_hex(base_hue,    0.05,                       random.uniform(0.45, 0.55))
   276→        surface = f"rgba(0,0,0,{random.uniform(0.02, 0.05):.2f})"
   277→
   278→    # 3. WCAG Contrast Safety Net — ensure text:bg ratio ≥ 4.5:1
   279→    text_cr = _contrast_ratio(text, bg)
   280→    if text_cr < 4.5:
   281→        # Push text darker (light modes) or lighter (dark modes)
   282→        r, g, b = (int(bg.lstrip('#')[i:i+2], 16) / 255.0 for i in (0, 2, 4))
   283→        _, bg_l, _ = colorsys.rgb_to_hls(r, g, b)
   284→        if bg_l > 0.5:
   285→            text = _hsl_to_hex(base_hue, 0.08, 0.08)  # force near-black
   286→        else:
   287→            text = _hsl_to_hex(base_hue, 0.05, 0.95)  # force near-white
   288→
   289→    # 4. Accent-on-bg visibility check — ensure accent stands out (ratio ≥ 3:1)
   290→    accent_cr = _contrast_ratio(accent, bg)
   291→    if accent_cr < 3.0:
   292→        # Nudge accent lightness away from bg
   293→        r, g, b = (int(bg.lstrip('#')[i:i+2], 16) / 255.0 for i in (0, 2, 4))
   294→        _, bg_l, _ = colorsys.rgb_to_hls(r, g, b)
   295→        target_l = 0.35 if bg_l > 0.5 else 0.70
   296→        r2, g2, b2 = (int(accent.lstrip('#')[i:i+2], 16) / 255.0 for i in (0, 2, 4))
   297→        _, _, accent_s = colorsys.rgb_to_hls(r2, g2, b2)
   298→        accent = _hsl_to_hex(accent_hue, accent_s, target_l)
   299→
   300→    return {
   301→        "bg": bg, "bg_rgb": _hex_to_rgb(bg),
   302→        "mid": mid, "mid_rgb": _hex_to_rgb(mid),
   303→        "accent": accent, "accent_rgb": _hex_to_rgb(accent),
   304→        "text": text, "muted": muted,
   305→        "surface": surface,
   306→        "meta": {
   307→            "intent": intent, "mode": mode, "harmony": harmony,
   308→            "base_hue": base_hue, "accent_hue": accent_hue,
   309→            "contrast": {
   310→                "text_on_bg": round(_contrast_ratio(text, bg), 2),
   311→                "accent_on_bg": round(_contrast_ratio(accent, bg), 2),
   312→            }
   313→        }
   314→    }

   337→# 1b. CASCADE PALETTE — Role-Based Unified Color System
   338→# ═══════════════════════════════════════════════════════════════════════
   339→#
   340→# Iron Law: Area ∝ 1/Saturation.
   341→# The larger the colored area, the LOWER its saturation must be.
   342→#
   343→# Role tiers by area usage:
   344→#   XL (>50% of page): page_bg, section_bg   → S ≤ 0.08, near-white/near-black
   345→#   L  (20-50%):       card_bg, table_stripe  → S ≤ 0.15
   346→#   M  (5-20%):        header_fill, sidebar   → S ≤ 0.30
   347→#   S  (1-5%):         border, divider, icon   → S ≤ 0.50
   348→#   XS (<1%):          accent_dot, badge, tag  → S up to 0.75 (the only place high-sat lives)
   349→#
   350→# Every color is derived from one base hue. No orphan colors.
   351→
   352→# Tier saturation caps — enforced at generation AND audit time
   353→CASCADE_TIER_CAPS = {
   354→    "xl": 0.08,   # Page background, section background
   355→    "l":  0.15,   # Card background, table stripe
   356→    "m":  0.30,   # Header fill, sidebar, cover decorative blocks
   357→    "s":  0.50,   # Borders, dividers, icons, chart grid lines
   358→    "xs": 0.75,   # Accent dot, badge, tag, data point highlight
   359→}

   386→    if harmony == "complementary":
   387→        accent_hue = (base_hue + 180) % 360
   388→    elif harmony == "split_complementary":
   389→        accent_hue = (base_hue + random.choice([150, 210])) % 360
   390→    elif harmony == "triadic":
   391→        accent_hue = (base_hue + random.choice([120, 240])) % 360
   392→    elif harmony == "analogous":
   393→        accent_hue = (base_hue + random.choice([30, -30, 45, -45])) % 360
   394→    else:  # monochrome
   395→        accent_hue = base_hue
   396→
   397→    # Sanitize accent hue — avoid muddy zones
   398→    accent_hue = _sanitize_accent_hue(accent_hue)

   682→def audit_cascade_palette(roles, semantic):
   683→    """Audit cascade palette for tier saturation violations and WCAG contrast."""
   684→    violations = []
   685→    for name, role in {**roles, **semantic}.items():
   686→        tier = role["tier"]
   687→        if tier in CASCADE_TIER_CAPS:
   688→            cap = CASCADE_TIER_CAPS[tier]
   689→            s = role["hsl"][1]
   690→            if s > cap:
   691→                violations.append(f"{name}: S={s:.3f} exceeds tier '{tier}' cap {cap}")
   692→
   693→    # WCAG checks
   694→    bg_hex = roles["page_bg"]["hex"]
   695→    text_hex = roles["text_primary"]["hex"]
   696→    cr = _contrast_ratio(text_hex, bg_hex)
   697→    if cr < 4.5:
   698→        violations.append(f"text_primary:page_bg contrast {cr:.2f} < 4.5:1 (WCAG AA)")
   699→
   700→    accent_hex = roles["accent"]["hex"]
   701→    cr_a = _contrast_ratio(accent_hex, bg_hex)
   702→    if cr_a < 3.0:
   703→        violations.append(f"accent:page_bg contrast {cr_a:.2f} < 3.0:1 (accent invisible)")
   704→
   705→    # Header fill should be readable with white text on it
   706→    hf_hex = roles["header_fill"]["hex"]
   707→    cr_hf = _contrast_ratio("#ffffff", hf_hex)
   708→    if cr_hf < 3.0:
   709→        violations.append(f"white on header_fill contrast {cr_hf:.2f} < 3.0:1 (header text unreadable)")
   710→
   711→    return violations

   715→# 2. GENERATIVE SVG — Algorithmic Art Backgrounds
   716→# ═══════════════════════════════════════════════════════════════════════
   717→#
   718→# Two primary modes:
   719→#   - "flow": 3-5 large-radius bézier curves at ultra-low opacity (0.05)
   720→#             Creates organic, breathing atmospheric depth
   721→#   - "grid": 1px reference grid or noise texture
   722→#             Creates structured, architectural underlying rhythm
   723→#
   724→# All SVG is inline-ready (no external files needed).

   726→def _svg_open(w, h):
   727→    return f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {w} {h}" width="{w}" height="{h}">'
   728→
   729→def _random_bezier_path(w, h):
   730→    """Generate a single flowing bézier curve across the canvas."""
   731→    # Start from a random edge point
   732→    x0 = random.uniform(-w * 0.2, w * 0.3)
   733→    y0 = random.uniform(0, h)
   734→    # End at the opposite side
   735→    x3 = random.uniform(w * 0.7, w * 1.2)
   736→    y3 = random.uniform(0, h)
   737→    # Control points — large sweeps for organic feel
   738→    cx1 = random.uniform(w * 0.1, w * 0.5)
   739→    cy1 = random.uniform(-h * 0.3, h * 1.3)
   740→    cx2 = random.uniform(w * 0.5, w * 0.9)
   741→    cy2 = random.uniform(-h * 0.3, h * 1.3)
   742→    return f"M{x0:.0f},{y0:.0f} C{cx1:.0f},{cy1:.0f} {cx2:.0f},{cy2:.0f} {x3:.0f},{y3:.0f}"
   743→
   744→def generate_flow_svg(w, h, color="#8a8a8a", curves=4, stroke_width=80, opacity=0.05):
   745→    """
   746→    Flow mode: ultra-wide, ultra-faint bézier curves.
   747→    Creates atmospheric depth without competing with content.
   748→    """
   749→    random.seed(42)
   750→    svg = _svg_open(w, h)
   751→    svg += '\n  <defs>'
   752→    svg += f'\n    <linearGradient id="fg" x1="0" y1="0" x2="1" y2="1">'
   753→    svg += f'\n      <stop offset="0%" stop-color="{color}" stop-opacity="{opacity}"/>'
   754→    svg += f'\n      <stop offset="50%" stop-color="{color}" stop-opacity="{opacity * 1.5:.3f}"/>'
   755→    svg += f'\n      <stop offset="100%" stop-color="{color}" stop-opacity="{opacity * 0.5:.3f}"/>'
   756→    svg += '\n    </linearGradient>'
   757→    svg += '\n  </defs>'
   758→    
   759→    for i in range(curves):
   760→        path = _random_bezier_path(w, h)
   761→        sw = stroke_width + random.uniform(-20, 30)
   762→        svg += f'\n  <path d="{path}" fill="none" stroke="url(#fg)" '
   763→        svg += f'stroke-width="{sw:.0f}" stroke-linecap="round" opacity="{opacity + i * 0.01:.3f}"/>'
   764→    
   765→    svg += '\n</svg>'
   766→    return svg

   768→def generate_grid_svg(w, h, color="#888888", spacing=60, line_width=0.5, opacity=0.04):
   769→    """
   770→    Grid mode: architectural reference grid.
   771→    Ultra-faint 1px lines creating underlying structure.
   772→    """
   773→    svg = _svg_open(w, h)
   774→    svg += f'\n  <g opacity="{opacity}">'
   775→    # Vertical lines
   776→    for x in range(0, int(w) + 1, spacing):
   777→        svg += f'\n    <line x1="{x}" y1="0" x2="{x}" y2="{h}" stroke="{color}" stroke-width="{line_width}"/>'
   778→    # Horizontal lines
   779→    for y in range(0, int(h) + 1, spacing):
   780→        svg += f'\n    <line x1="0" y1="{y}" x2="{w}" y2="{y}" stroke="{color}" stroke-width="{line_width}"/>'
   781→    svg += '\n  </g>'
   782→    svg += '\n</svg>'
   783→    return svg

   785→def generate_noise_svg(w, h, frequency=0.8, octaves=4, opacity=0.035):
   786→    """
   787→    Noise mode: feTurbulence grain texture.
   788→    Adds tactile paper-like quality.
   789→    """
   790→    svg = _svg_open(w, h)
   791→    svg += f"""
   792→  <defs>
   793→    <filter id="grain" x="0" y="0" width="100%" height="100%">
   794→      <feTurbulence type="fractalNoise" baseFrequency="{frequency}" 
   795→                    numOctaves="{octaves}" stitchTiles="stitch" result="noise"/>
   796→      <feColorMatrix type="saturate" values="0" in="noise" result="grey"/>
   797→    </filter>
   798→  </defs>
   799→  <rect width="{w}" height="{h}" filter="url(#grain)" opacity="{opacity}"/>"""
   800→    svg += '\n</svg>'
   801→    return svg

   803→def _generate_data_driven_svg(data_points, w=720, h=960, color="#8a8a8a"):
   804→    """
   805→    Content-Aware SVG: Transform business data arrays into Bézier background curves.
   806→    The background subtly echoes the data's shape — a cross-modal design metaphor.
   807→    """
   808→    if not data_points or len(data_points) < 2:
   809→        return generate_flow_svg(w, h, color)
   810→
   811→    n = len(data_points)
   812→    min_v = min(data_points)
   813→    max_v = max(data_points)
   814→    rng = max_v - min_v if max_v != min_v else 1.0
   815→
   816→    # Normalize data to canvas coordinates
   817→    # X: evenly spaced across width; Y: mapped to 20%-80% of height (inverted for SVG)
   818→    points = []
   819→    for i, v in enumerate(data_points):
   820→        x = (i / (n - 1)) * w
   821→        y_norm = (v - min_v) / rng  # 0..1
   822→        y = h * 0.8 - y_norm * (h * 0.6)  # Map to 20%-80% height band
   823→        points.append((x, y))
   824→
   825→    # Build smooth Bézier path through all points (Catmull-Rom → cubic Bézier)
   826→    def catmull_to_bezier(p0, p1, p2, p3, tension=0.5):
   827→        """Convert 4 Catmull-Rom points to cubic Bézier control points for segment p1→p2."""
   828→        cp1x = p1[0] + (p2[0] - p0[0]) / (6 * tension)
   829→        cp1y = p1[1] + (p2[1] - p0[1]) / (6 * tension)
   830→        cp2x = p2[0] - (p3[0] - p1[0]) / (6 * tension)
   831→        cp2y = p2[1] - (p3[1] - p1[1]) / (6 * tension)
   832→        return (cp1x, cp1y), (cp2x, cp2y)
   833→
   834→    svg_paths = ""
   835→    # Generate 3 layers at different offsets for depth
   836→    for layer_idx, (opacity, y_offset, thickness) in enumerate([
   837→        (0.08, 0, 3), (0.05, 60, 2), (0.03, -40, 1.5)
   838→    ]):
   839→        pts = [(px, py + y_offset) for px, py in points]
   840→        # Pad start/end for Catmull-Rom
   841→        padded = [pts[0]] + pts + [pts[-1]]
   842→        d = f"M {padded[1][0]:.1f},{padded[1][1]:.1f} "
   843→        for i in range(1, len(padded) - 2):
   844→            cp1, cp2 = catmull_to_bezier(padded[i-1], padded[i], padded[i+1], padded[i+2])
   845→            d += f"C {cp1[0]:.1f},{cp1[1]:.1f} {cp2[0]:.1f},{cp2[1]:.1f} {padded[i+1][0]:.1f},{padded[i+1][1]:.1f} "
   846→
   847→        # Main stroke
   848→        svg_paths += f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{thickness}" opacity="{opacity}" />\n'
   849→        # Filled area below the curve
   850→        fill_d = d + f"L {w},{h} L 0,{h} Z"
   851→        svg_paths += f'<path d="{fill_d}" fill="{color}" opacity="{opacity * 0.4}" />\n'
   852→
   853→    return f'<svg viewBox="0 0 {w} {h}" xmlns="http://www.w3.org/2000/svg" style="width:100%;height:100%">{svg_paths}</svg>'

   856→def generate_supergraphic_svg(w, h, color="#8a8a8a", seed=42):
   857→    """
   858→    Supergraphic mode: oversized geometric shapes (circles, rectangles, polygons)
   859→    cropped by the canvas edge, rendered at 3-5% opacity.
   860→    Creates "blueprint of a larger world" feeling — McKinsey / Pentagram style.
   861→    Shapes deliberately overflow the viewport so only partial arcs/edges are visible.
   862→    """
   863→    random.seed(seed)
   864→    svg = _svg_open(w, h)
   865→    # Layer 1: Giant concentric circles, center placed off-canvas
   866→    cx = random.uniform(w * 0.6, w * 1.3)  # right-biased, partially off-canvas
   867→    cy = random.uniform(-h * 0.2, h * 0.4)  # top-biased
   868→    for i in range(4):
   869→        r = w * (0.6 + i * 0.35)  # radii from 60% to 165% of width — always overflowing
   870→        opacity = 0.04 - i * 0.008  # outer rings fainter
   871→        svg += f'\n  <circle cx="{cx:.0f}" cy="{cy:.0f}" r="{r:.0f}" '
   872→        svg += f'fill="none" stroke="{color}" stroke-width="1.5" opacity="{max(opacity, 0.015):.3f}"/>'
   873→
   874→    # Layer 2: Oversized rotated rectangles, clipped by viewport
   875→    for _ in range(2):
   876→        rx = random.uniform(-w * 0.3, w * 0.5)
   877→        ry = random.uniform(h * 0.3, h * 1.2)
   878→        rw = random.uniform(w * 0.8, w * 1.6)
   879→        rh = random.uniform(h * 0.5, h * 1.2)
   880→        angle = random.uniform(-15, 25)
   881→        svg += f'\n  <rect x="{rx:.0f}" y="{ry:.0f}" width="{rw:.0f}" height="{rh:.0f}" '
   882→        svg += f'fill="none" stroke="{color}" stroke-width="1" opacity="0.025" '
   883→        svg += f'transform="rotate({angle:.1f} {rx + rw/2:.0f} {ry + rh/2:.0f})"/>'
   884→
   885→    # Layer 3: A single massive polygon (pentagon/hexagon) barely visible
   886→    sides = random.choice([5, 6, 8])
   887→    pcx = random.uniform(-w * 0.1, w * 0.4)
   888→    pcy = random.uniform(h * 0.5, h * 1.1)
   889→    pr = w * random.uniform(0.9, 1.4)
   890→    angle_offset = random.uniform(0, 360 / sides)
   891→    points = []
   892→    for i in range(sides):
   893→        a = math.radians(angle_offset + i * 360 / sides)
   894→        px = pcx + pr * math.cos(a)
   895→        py = pcy + pr * math.sin(a)
   896→        points.append(f"{px:.0f},{py:.0f}")
   897→    svg += f'\n  <polygon points="{" ".join(points)}" '
   898→    svg += f'fill="none" stroke="{color}" stroke-width="1" opacity="0.02"/>'
   899→
   900→    svg += '\n</svg>'
   901→    return svg

   904→def generate_ordered_texture_svg(w, h, color="#8a8a8a", seed=42):
   905→    """
   906→    Ordered Texture mode: precision dot-matrix, coordinate grids, and contour lines
   907→    placed in specific corners/edges of the canvas (not full coverage).
   908→    Creates "engineered precision" feeling — ideal for tech, finance, data reports.
   909→    """
   910→    random.seed(seed)
   911→    svg = _svg_open(w, h)
   912→
   913→    # Region 1: Dot matrix in top-right corner (10x10 grid of small circles)
   914→    dot_cols, dot_rows = 12, 10
   915→    dot_spacing = 18
   916→    dot_r = 2
   917→    dot_origin_x = w - dot_cols * dot_spacing - 40  # right-aligned with margin
   918→    dot_origin_y = 30  # top margin
   919→    svg += f'\n  <g opacity="0.08">'
   920→    for row in range(dot_rows):
   921→        for col in range(dot_cols):
   922→            dx = dot_origin_x + col * dot_spacing
   923→            dy = dot_origin_y + row * dot_spacing
   924→            svg += f'\n    <circle cx="{dx}" cy="{dy}" r="{dot_r}" fill="{color}"/>'
   925→    svg += '\n  </g>'
   926→
   927→    # Region 2: Coordinate grid lines in bottom-left quadrant (blueprint style)
   928→    grid_x0, grid_y0 = 20, h * 0.7
   929→    grid_x1, grid_y1 = w * 0.45, h - 20
   930→    grid_spacing = 30
   931→    svg += f'\n  <g opacity="0.05" stroke="{color}" stroke-width="0.5" stroke-dasharray="4,6">'
   932→    # Vertical lines
   933→    x = grid_x0
   934→    while x <= grid_x1:
   935→        svg += f'\n    <line x1="{x:.0f}" y1="{grid_y0:.0f}" x2="{x:.0f}" y2="{grid_y1:.0f}"/>'
   936→        x += grid_spacing
   937→    # Horizontal lines
   938→    y = grid_y0
   939→    while y <= grid_y1:
   940→        svg += f'\n    <line x1="{grid_x0:.0f}" y1="{y:.0f}" x2="{grid_x1:.0f}" y2="{y:.0f}"/>'
   941→        y += grid_spacing
   942→    svg += '\n  </g>'
   943→
   944→    # Region 3: Tick marks along the grid (ruler effect)
   945→    svg += f'\n  <g opacity="0.06" stroke="{color}" stroke-width="0.8">'
   946→    x = grid_x0
   947→    while x <= grid_x1:
   948→        svg += f'\n    <line x1="{x:.0f}" y1="{grid_y1:.0f}" x2="{x:.0f}" y2="{grid_y1 + 5:.0f}"/>'
   949→        x += grid_spacing
   950→    svg += '\n  </g>'
   951→
   952→    # Region 4: Flowing contour lines (Bézier) across mid-right area
   953→    svg += f'\n  <g opacity="0.04" fill="none" stroke="{color}" stroke-width="1">'
   954→    for i in range(5):
   955→        y_base = h * 0.35 + i * 35
   956→        x_start = w * 0.55
   957→        x_end = w + 20  # overflow right edge
   958→        cx1 = x_start + (x_end - x_start) * 0.3 + random.uniform(-30, 30)
   959→        cy1 = y_base + random.uniform(-40, 40)
   960→        cx2 = x_start + (x_end - x_start) * 0.7 + random.uniform(-30, 30)
   961→        cy2 = y_base + random.uniform(-40, 40)
   962→        svg += f'\n    <path d="M{x_start:.0f},{y_base:.0f} C{cx1:.0f},{cy1:.0f} {cx2:.0f},{cy2:.0f} {x_end:.0f},{y_base + random.uniform(-20, 20):.0f}"/>'
   963→    svg += '\n  </g>'
   964→
   965→    # Region 5: Cross-hair markers at 2-3 strategic points
   966→    for _ in range(3):
   967→        mx = random.uniform(w * 0.15, w * 0.85)
   968→        my = random.uniform(h * 0.15, h * 0.85)
   969→        size = 8
   970→        svg += f'\n  <g opacity="0.06" stroke="{color}" stroke-width="0.8">'
   971→        svg += f'\n    <line x1="{mx - size}" y1="{my}" x2="{mx + size}" y2="{my}"/>'
   972→        svg += f'\n    <line x1="{mx}" y1="{my - size}" x2="{mx}" y2="{my + size}"/>'
   973→        svg += f'\n    <circle cx="{mx}" cy="{my}" r="{size * 0.6:.0f}" fill="none"/>'
   974→        svg += '\n  </g>'
   975→
   976→    svg += '\n</svg>'
   977→    return svg

   998→def generate_continuous_flow_svg(w, h, total_pages, color="#8a8a8a", curves=4, stroke_width=80, opacity=0.05):
   999→    """
  1000→    Continuous Flow mode: generates ONE large SVG spanning all pages.
  1001→    Returns a list of per-page SVG strings, each using viewBox to slice the master.
  1002→    
  1003→    The bezier curves are constrained to have anchor points every ~480px vertically,
  1004→    ensuring visible content on every page.
  1005→    """
  1006→    total_h = h * total_pages
  1007→    random.seed(42)
  1008→    
  1009→    # Build the master path data
  1010→    paths_data = []
  1011→    for _ in range(curves):
  1012→        # Generate anchor points — one every 480px ensures ~2 per page
  1013→        num_anchors = max(3, int(total_h / 480))
  1014→        anchors = []
  1015→        for i in range(num_anchors):
  1016→            ax = random.uniform(w * 0.1, w * 0.9)
  1017→            ay = (total_h / (num_anchors - 1)) * i
  1018→            anchors.append((ax, ay))
  1019→        
  1020→        # Build cubic bezier path through anchors
  1021→        path = f"M{anchors[0][0]:.0f},{anchors[0][1]:.0f}"
  1022→        for i in range(1, len(anchors)):
  1023→            prev = anchors[i - 1]
  1024→            curr = anchors[i]
  1025→            # Control points: spread horizontally for organic feel
  1026→            cx1 = random.uniform(w * 0.0, w * 1.0)
  1027→            cy1 = prev[1] + (curr[1] - prev[1]) * 0.33 + random.uniform(-100, 100)
  1028→            cx2 = random.uniform(w * 0.0, w * 1.0)
  1029→            cy2 = prev[1] + (curr[1] - prev[1]) * 0.66 + random.uniform(-100, 100)
  1030→            path += f" C{cx1:.0f},{cy1:.0f} {cx2:.0f},{cy2:.0f} {curr[0]:.0f},{curr[1]:.0f}"
  1031→        
  1032→        sw = stroke_width + random.uniform(-20, 30)
  1033→        paths_data.append((path, sw))
  1034→    
  1035→    # Generate per-page SVG slices
  1036→    page_svgs = []
  1037→    gradient_def = f'''<defs>
  1038→    <linearGradient id="fg" x1="0" y1="0" x2="1" y2="1">
  1039→      <stop offset="0%" stop-color="{color}" stop-opacity="{opacity}"/>
  1040→      <stop offset="50%" stop-color="{color}" stop-opacity="{opacity * 1.5:.3f}"/>
  1041→      <stop offset="100%" stop-color="{color}" stop-opacity="{opacity * 0.5:.3f}"/>
  1042→    </linearGradient>
  1043→  </defs>'''
  1044→    
  1045→    for page_idx in range(total_pages):
  1046→        vy = page_idx * h
  1047→        svg = f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 {vy} {w} {h}" width="{w}" height="{h}">'
  1048→        svg += f'\n  {gradient_def}'
  1049→        for i, (path, sw) in enumerate(paths_data):
  1050→            svg += f'\n  <path d="{path}" fill="none" stroke="url(#fg)" '
  1051→            svg += f'stroke-width="{sw:.0f}" stroke-linecap="round" opacity="{opacity + i * 0.01:.3f}"/>'
  1052→        svg += '\n</svg>'
  1053→        page_svgs.append(svg)
  1054→    
  1055→    return page_svgs

  1132→# ═══════════════════════════════════════════════════════════════════════
  1133→# 3. LAYOUT CALCULATOR — Deliberate Spatial Tension
  1134→# ═══════════════════════════════════════════════════════════════════════
  1135→#
  1136→# Returns absolute coordinates for elements with intentional:
  1137→#   - Offset: elements not perfectly centered (art tension)
  1138→#   - Overlap: controlled z-index collisions
  1139→#   - Breathing margin: 15% minimum on all edges
  1140→
  1141→BREATHING_MARGIN = 0.12  # 12% of canvas on each edge (was 15%, too tight for content-dense pages)
  1142→
  1143→def calculate_layout(elements, w=720, h=960, style="offset"):
  1144→    """
  1145→    Calculate positioned layout for named elements.
  1146→    
  1147→    Args:
  1148→        elements: list of element names (e.g. ["hero", "body", "meta", "footer"])
  1149→        style: "offset" (deliberate asymmetry), "centered" (formal), "stacked" (vertical flow)
  1150→    
  1151→    Returns:
  1152→        dict mapping element names to {x, y, w, h, rotation} in pixels
  1153→    """
  1154→    # Safe area (15% breathing margin on all sides)
  1155→    safe_x = w * BREATHING_MARGIN
  1156→    safe_y = h * BREATHING_MARGIN
  1157→    safe_w = w * (1 - 2 * BREATHING_MARGIN)
  1158→    safe_h = h * (1 - 2 * BREATHING_MARGIN)
  1159→    
  1160→    layout = {}
  1161→    
  1162→    if style == "offset":
  1163→        # Asymmetric placement — elements shift left/right of center
  1164→        regions = _divide_vertical(safe_x, safe_y, safe_w, safe_h, len(elements))
  1165→        for i, name in enumerate(elements):
  1166→            rx, ry, rw, rh = regions[i]
  1167→            # Apply deliberate offset: odd elements shift left, even shift right
  1168→            offset_x = rw * 0.08 * (-1 if i % 2 == 0 else 1)
  1169→            layout[name] = {
  1170→                "x": round(rx + offset_x, 1),
  1171→                "y": round(ry, 1),
  1172→                "w": round(rw * 0.85, 1),  # Don't fill the full width
  1173→                "h": round(rh * 0.85, 1),
  1174→                "rotation": round(random.uniform(-1.5, 1.5), 2) if name != "body" else 0,
  1175→            }
  1176→    
  1177→    elif style == "centered":
  1178→        # Formal centered — golden ratio vertical split
  1179→        regions = _divide_vertical(safe_x, safe_y, safe_w, safe_h, len(elements))
  1180→        for i, name in enumerate(elements):
  1181→            rx, ry, rw, rh = regions[i]
  1182→            layout[name] = {
  1183→                "x": round(rx + (rw * 0.075), 1),  # Slight horizontal centering margin
  1184→                "y": round(ry, 1),
  1185→                "w": round(rw * 0.85, 1),
  1186→                "h": round(rh * 0.9, 1),
  1187→                "rotation": 0,
  1188→            }
  1189→    
  1190→    elif style == "overlap":
  1191→        # Controlled overlaps — elements bleed into each other's space
  1192→        regions = _divide_vertical(safe_x, safe_y, safe_w, safe_h, len(elements))
  1193→        for i, name in enumerate(elements):
  1194→            rx, ry, rw, rh = regions[i]
  1195→            overlap_y = rh * 0.15 if i > 0 else 0  # Pull up into previous region
  1196→            layout[name] = {
  1197→                "x": round(rx, 1),
  1198→                "y": round(ry - overlap_y, 1),
  1199→                "w": round(rw, 1),
  1200→                "h": round(rh + overlap_y * 0.5, 1),
  1201→                "rotation": 0,
  1202→                "z_index": len(elements) - i,  # Later elements on top
  1203→            }
  1204→    
  1205→    return layout
  1206→
  1207→def _divide_vertical(x, y, w, h, n):
  1208→    """Divide a rectangle into n vertical bands with golden-ratio-inspired proportions."""
  1209→    if n <= 0:
  1210→        return []
  1211→    if n == 1:
  1212→        return [(x, y, w, h)]
  1213→    
  1214→    # Weighted distribution: first element (hero) gets more space
  1215→    weights = [1.618 if i == 0 else 1.0 for i in range(n)]
  1216→    total = sum(weights)
  1217→    
  1218→    regions = []
  1219→    current_y = y
  1220→    for i in range(n):
  1221→        region_h = h * weights[i] / total
  1222→        regions.append((x, current_y, w, region_h))
  1223→        current_y += region_h
  1224→    
  1225→    return regions
  1226→

  1228→# ═══════════════════════════════════════════════════════════════════════
  1229→# VALIDATION — Post-generation color audit
  1230→# ═══════════════════════════════════════════════════════════════════════
  1231→
  1232→def audit_palette(palette):
  1233→    """
  1234→    Strict audit: mode-specific S/L bounds + WCAG contrast checks.
  1235→    Returns list of violations (empty = clean).
  1236→    """
  1237→    violations = []
  1238→    mode = palette.get("meta", {}).get("mode", "minimal")
  1239→    
  1240→    for key in ["bg", "mid", "accent", "text"]:
  1241→        hex_val = palette.get(key, "")
  1242→        if not hex_val.startswith("#"):
  1243→            continue
  1244→        r, g, b = (int(hex_val[i:i+2], 16) / 255.0 for i in (1, 3, 5))
  1245→        h, l, s = colorsys.rgb_to_hls(r, g, b)
  1246→        
  1247→        # Tight bg/mid saturation limits per mode
  1248→        if key in ("bg", "mid"):
  1249→            limits = {"minimal": 0.20, "dark": 0.16, "pastel": 0.42, "jewel": 0.62, "light": 0.28}
  1250→            cap = limits.get(mode, 0.15)
  1251→            if s > cap:
  1252→                violations.append(f"{key}: S={s:.3f} > {cap} in {mode} mode")
  1253→
  1254→        # Tight accent saturation — pastel/jewel reined in
  1255→        if key == "accent":
  1256→            accent_caps = {"minimal": 0.78, "dark": 0.62, "pastel": 0.52, "jewel": 0.52, "light": 0.48}
  1257→            cap = accent_caps.get(mode, 0.60)
  1258→            if s > cap:
  1259→                violations.append(f"accent: S={s:.3f} > {cap} in {mode} mode")
  1260→
  1261→        # Lightness guardrails
  1262→        if key == "bg":
  1263→            if mode == "dark" and l > 0.10:
  1264→                violations.append(f"bg L={l:.3f} > 0.10 in dark (muddy middle)")
  1265→            elif mode == "minimal" and l < 0.90:
  1266→                violations.append(f"bg L={l:.3f} < 0.90 in minimal (too dark for paper)")
  1267→            elif mode == "light" and l < 0.88:
  1268→                violations.append(f"bg L={l:.3f} < 0.88 in light (too dark)")
  1269→            elif mode == "jewel" and l > 0.28:
  1270→                violations.append(f"bg L={l:.3f} > 0.28 in jewel (not deep enough)")
  1271→            elif mode == "pastel" and l < 0.83:
  1272→                violations.append(f"bg L={l:.3f} < 0.83 in pastel (too dark for Morandi)")
  1273→    
  1274→    # WCAG contrast checks
  1275→    bg_hex = palette.get("bg", "")
  1276→    text_hex = palette.get("text", "")
  1277→    accent_hex = palette.get("accent", "")
  1278→    if bg_hex.startswith("#") and text_hex.startswith("#"):
  1279→        cr = _contrast_ratio(text_hex, bg_hex)
  1280→        if cr < 4.5:
  1281→            violations.append(f"text:bg contrast {cr:.2f} < 4.5:1 (WCAG AA fail)")
  1282→    if bg_hex.startswith("#") and accent_hex.startswith("#"):
  1283→        cr = _contrast_ratio(accent_hex, bg_hex)
  1284→        if cr < 2.5:
  1285→            violations.append(f"accent:bg contrast {cr:.2f} < 2.5:1 (accent invisible)")
  1286→    
  1287→    return violations
