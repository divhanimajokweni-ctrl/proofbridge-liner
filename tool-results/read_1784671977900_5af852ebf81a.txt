    84→function ThemeToggleBtn() {
    85→  const { resolvedTheme, setTheme } = useTheme();
    86→  const [mounted, setMounted] = useState(false);
    87→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    88→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    89→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
    90→  return (
    91→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
    92→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
    93→    </button>
    94→  );
    95→}

    97→function ScrollToTopButton() {
    98→  const [visible, setVisible] = useState(false);
    99→
   100→  useEffect(() => {
   101→    const onScroll = () => setVisible(window.scrollY > 300);
   102→    window.addEventListener("scroll", onScroll, { passive: true });
   103→    return () => window.removeEventListener("scroll", onScroll);
   104→  }, []);
   105→
   106→  const scrollToTop = useCallback(() => {
   107→    window.scrollTo({ top: 0, behavior: "smooth" });
   108→  }, []);
   109→
   110→  return (
   111→    <AnimatePresence>
   112→      {visible && (
   113→        <motion.button
   114→          type="button"
   115→          onClick={scrollToTop}
   116→          initial={{ opacity: 0, scale: 0.8 }}
   117→          animate={{ opacity: 1, scale: 1 }}
   118→          exit={{ opacity: 0, scale: 0.8 }}
   119→          transition={{ duration: 0.2 }}
   120→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   121→          title="Scroll to top"
   122→        >
   123→          <ArrowUp className="h-4 w-4" />
   124→        </motion.button>
   125→      )}
   126→    </AnimatePresence>
   127→  );
   128→}

   432→/* ─── Quick Stats Bar ─── */
   433→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   434→  const meta = SECTION_META[sectionId];
   435→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   436→  const [isRefreshing, setIsRefreshing] = useState(false);
   437→
   438→  // Auto-refresh countdown (10s cycle, matches health fetch)
   439→  useEffect(() => {
   440→    const t = setInterval(() => {
   441→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   442→    }, 1000);
   443→    return () => clearInterval(t);
   444→  }, []);
   445→
   446→  const handleRefresh = useCallback(() => {
   447→    setIsRefreshing(true);
   448→    fetch("/api/stats").then(() => {
   449→      setIsRefreshing(false);
   450→    }).catch(() => {
   451→      setIsRefreshing(false);
   452→    });
   453→  }, []);
   454→
   455→  return (
   456→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   457→      {meta.stats.map((stat, i) => (
   458→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   459→          {stat}
   460→        </span>
   461→      ))}
   462→      <div className="ml-auto flex items-center gap-2">
   463→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   464→          <Timer className="h-2.5 w-2.5" />
   465→          {refreshCountdown}s
   466→        </span>
   467→        <button
   468→          type="button"
   469→          onClick={handleRefresh}
   470→          disabled={isRefreshing}
   471→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   472→          title="Refresh section data"
   473→        >
   474→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   475→          refresh
   476→        </button>
   477→      </div>
   478→    </div>
   479→  );
   480→}
