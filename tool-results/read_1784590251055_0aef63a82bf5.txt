    49→/** Policy health indicator showing pass/fail ratio */
    50→function PolicyHealthGauge({ evaluations }: { evaluations: { passed: boolean }[] | null }) {
    51→  if (!evaluations || evaluations.length === 0) {
    52→    return (
    53→      <div className="flex items-center gap-2">
    54→        <Shield className="h-4 w-4 text-muted-foreground" />
    55→        <span className="text-xs text-muted-foreground">No evaluations</span>
    56→      </div>
    57→    );
    58→  }
    59→  const passed = evaluations.filter((e) => e.passed).length;
    60→  const total = evaluations.length;
    61→  const ratio = Math.round((passed / total) * 100);
    62→  const status = ratio === 100 ? "verified" : ratio >= 60 ? "repairing" : "violating";
    63→
    64→  const colorMap: Record<string, string> = {
    65→    verified: "text-verified",
    66→    repairing: "text-repairing",
    67→    violating: "text-violating",
    68→  };
    69→
    70→  return (
    71→    <div className="space-y-1.5">
    72→      <div className="flex items-center justify-between">
    73→        <div className="flex items-center gap-2">
    74→          <Shield className={cn("h-4 w-4", colorMap[status])} />
    75→          <span className="text-xs font-medium">Policy Health</span>
    76→        </div>
    77→        <span className={cn("text-sm font-mono font-bold tabular-nums", colorMap[status])}>
    78→          {ratio}%
    79→        </span>
    80→      </div>
    81→      <div className="relative h-2 w-full overflow-hidden rounded-full bg-muted/40">
    82→        <motion.div
    83→          className={cn(
    84→            "h-full rounded-full",
    85→            status === "verified" ? "bg-verified" : status === "repairing" ? "bg-repairing" : "bg-violating",
    86→          )}
    87→          initial={{ width: 0 }}
    88→          animate={{ width: `${ratio}%` }}
    89→          transition={{ type: "spring", stiffness: 120, damping: 20, delay: 0.2 }}
    90→        />
    91→      </div>
    92→      <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
    93→        <span className="flex items-center gap-1">
    94→          <CheckCircle2 className="h-2.5 w-2.5 text-verified" /> {passed} pass
    95→        </span>
    96→        <span className="flex items-center gap-1">
    97→          <XCircle className="h-2.5 w-2.5 text-violating" /> {total - passed} fail
    98→        </span>
    99→      </div>
   100→    </div>
   101→  );
   102→}

   104→/** Live validation indicator */
   105→function ValidationIndicator({ validating, result }: { validating: boolean; result: ValidationResult | null }) {
   106→  return (
   107→    <AnimatePresence mode="wait">
   108→      {validating ? (
   109→        <motion.div
   110→          key="loading"
   111→          initial={{ opacity: 0, scale: 0.8 }}
   112→          animate={{ opacity: 1, scale: 1 }}
   113→          exit={{ opacity: 0, scale: 0.8 }}
   114→          className="flex items-center gap-1.5"
   115→        >
   116→          <Loader2 className="h-4 w-4 text-repairing animate-spin" />
   117→          <span className="text-xs text-repairing font-medium">Validating…</span>
   118→        </motion.div>
   119→      ) : result ? (
   120→        <motion.div
   121→          key={result.ok ? "ok" : "err"}
   122→          initial={{ opacity: 0, scale: 0.8 }}
   123→          animate={{ opacity: 1, scale: 1 }}
   124→          exit={{ opacity: 0, scale: 0.8 }}
   125→          className="flex items-center gap-1.5"
   126→        >
   127→          {result.ok ? (
   128→            <motion.div
   129→              initial={{ scale: 0 }}
   130→              animate={{ scale: 1 }}
   131→              transition={{ type: "spring", stiffness: 400, damping: 15 }}
   132→            >
   133→              <CheckCircle2 className="h-4 w-4 text-verified" />
   134→            </motion.div>
   135→          ) : (
   136→            <motion.div
   137→              initial={{ scale: 0 }}
   138→              animate={{ scale: 1 }}
   139→              transition={{ type: "spring", stiffness: 400, damping: 15 }}
   140→            >
   141→              <XCircle className="h-4 w-4 text-violating" />
   142→            </motion.div>
   143→          )}
   144→          <span className={cn("text-xs font-medium", result.ok ? "text-verified" : "text-violating")}>
   145→            {result.ok ? "Valid" : "Invalid"}
   146→          </span>
   147→        </motion.div>
   148→      ) : (
   149→        <motion.div
   150→          key="idle"
   151→          initial={{ opacity: 0 }}
   152→          animate={{ opacity: 1 }}
   153→          exit={{ opacity: 0 }}
   154→          className="flex items-center gap-1.5"
   155→        >
   156→          <Activity className="h-4 w-4 text-muted-foreground" />
   157→          <span className="text-xs text-muted-foreground">Idle</span>
   158→        </motion.div>
   159→      )}
   160→    </AnimatePresence>
   161→  );
   162→}

   530→                        </motion.div>
   531→                      );
   532→                    })}
   533→                  </div>
   534→                ) : (
   535→                  <p className="text-xs text-muted-foreground">
   536→                    Press <span className="font-mono text-verified">Validate</span> to parse & semantically check the policy.

   652→          </motion.div>
   653→        </div>
   654→      </motion.div>
   655→    </TooltipProvider>
   656→  );
   657→}
   658→
   659→function formatAst(p: PolicyNode): string {
   660→  const lines: string[] = [];
   661→  lines.push(`Policy "${p.name}"`);
   662→  if (p.domain) lines.push(`  domain: ${p.domain}`);
   663→  if (p.version) lines.push(`  version: ${p.version}`);
   664→  if (p.shard) {
   665→    lines.push(
   666→      `  shard by ${p.shard.dimension ?? "?"} {`,
   667→    );
   668→    lines.push(`    key: "${p.shard.key}"`);
   669→    lines.push(`    strategy: ${p.shard.strategy}`);
   670→    if (p.shard.count) lines.push(`    count: ${p.shard.count}`);
   671→    if (p.shard.replication) lines.push(`    replication: ${p.shard.replication}`);
   672→    lines.push(`  }`);
   673→  }
   674→  lines.push(`  invariants (${p.invariants.length}):`);
   675→  for (const inv of p.invariants) {
   676→    lines.push(
   677→      `    ${inv.soft ? "soft " : ""}${inv.name} [${inv.severity}]`,
   678→    );
   679→    lines.push(`      predicate: ${inv.rawPredicate}`);
   680→    if (inv.message) lines.push(`      message: "${inv.message}"`);
   681→  }
   682→  if (p.expectMerge) {
   683→    lines.push(`  expect merge {`);
   684→    for (const pres of p.expectMerge.preserves) lines.push(`    preserves ${pres}`);
   685→    if (p.expectMerge.localityPreserving !== undefined)
   686→      lines.push(`    locality_preserving: ${p.expectMerge.localityPreserving}`);
   687→    for (const req of p.expectMerge.requires) lines.push(`    requires ${req}`);
   688→    if (p.expectMerge.maxDivergence !== undefined)
   689→      lines.push(`    max_divergence: ${p.expectMerge.maxDivergence}`);
   690→    lines.push(`  }`);
   691→  }
   692→  if (p.onViolation) {
   693→    lines.push(`  on_violation {`);
   694→    lines.push(`    strategy: ${p.onViolation.strategy}`);
   695→    lines.push(`    objective: ${p.onViolation.objective}`);
   696→    if (p.onViolation.maxIters) lines.push(`    max_iters: ${p.onViolation.maxIters}`);
   697→    lines.push(`  }`);
   698→  }
   699→  if (p.ancestry) {
   700→    lines.push(`  ancestry {`);
   701→    lines.push(`    proof: ${p.ancestry.proof}`);
   702→    lines.push(`    zk: ${p.ancestry.zk}`);
   703→    lines.push(`    gossip: ${p.ancestry.gossip}`);
   704→    lines.push(`    anchor: ${p.ancestry.anchor}`);
   705→    lines.push(`  }`);
   706→  }
   707→  if (p.shadowBridge) {
   708→    lines.push(`  shadow_bridge {`);
   709→    lines.push(`    enabled: ${p.shadowBridge.enabled}`);
   710→    if (p.shadowBridge.takeoverLatencyMs !== undefined)
   711→      lines.push(`    takeover_latency_ms: ${p.shadowBridge.takeoverLatencyMs}`);
   712→    if (p.shadowBridge.whatifBranching !== undefined)
   713→      lines.push(`    whatif_branching: ${p.shadowBridge.whatifBranching}`);
   714→    lines.push(`  }`);
   715→  }
   716→  if (p.exports.length) lines.push(`  exports: ${p.exports.join(", ")}`);
   717→  return lines.join("\n");
   718→}
