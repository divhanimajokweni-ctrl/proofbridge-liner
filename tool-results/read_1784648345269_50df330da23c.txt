    74→function ThemeToggleBtn() {
    75→  const { resolvedTheme, setTheme } = useTheme();
    76→  const [mounted, setMounted] = useState(false);
    77→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    78→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    79→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
    80→  return (
    81→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
    82→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
    83→    </button>
    84→  );
    85→}

    87→function ScrollToTopButton() {
    88→  const [visible, setVisible] = useState(false);
    89→
    90→  useEffect(() => {
    91→    const onScroll = () => setVisible(window.scrollY > 300);
    92→    window.addEventListener("scroll", onScroll, { passive: true });
    93→    return () => window.removeEventListener("scroll", onScroll);
    94→  }, []);
    95→
    96→  const scrollToTop = useCallback(() => {
    97→    window.scrollTo({ top: 0, behavior: "smooth" });
    98→  }, []);
    99→
   100→  return (
   101→    <AnimatePresence>
   102→      {visible && (
   103→        <motion.button
   104→          type="button"
   105→          onClick={scrollToTop}
   106→          initial={{ opacity: 0, scale: 0.8 }}
   107→          animate={{ opacity: 1, scale: 1 }}
   108→          exit={{ opacity: 0, scale: 0.8 }}
   109→          transition={{ duration: 0.2 }}
   110→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   111→          title="Scroll to top"
   112→        >
   113→          <ArrowUp className="h-4 w-4" />
   114→        </motion.button>
   115→      )}
   116→    </AnimatePresence>

   409→/* ─── Quick Stats Bar ─── */
   410→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   411→  const meta = SECTION_META[sectionId];
   412→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   413→  const [isRefreshing, setIsRefreshing] = useState(false);
   414→
   415→  // Auto-refresh countdown (10s cycle, matches health fetch)
   416→  useEffect(() => {
   417→    const t = setInterval(() => {
   418→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   419→    }, 1000);
   420→    return () => clearInterval(t);
   421→  }, []);
   422→
   423→  const handleRefresh = useCallback(() => {
   424→    setIsRefreshing(true);
   425→    fetch("/api/stats").then(() => {
   426→      setIsRefreshing(false);
   427→    }).catch(() => {
   428→      setIsRefreshing(false);
   429→    });
   430→  }, []);
   431→
   432→  return (
   433→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   434→      {meta.stats.map((stat, i) => (
   435→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   436→          {stat}
   437→        </span>
   438→      ))}
   439→      <div className="ml-auto flex items-center gap-2">
   440→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   441→          <Timer className="h-2.5 w-2.5" />
   442→          {refreshCountdown}s
   443→        </span>
   444→        <button
   445→          type="button"
   446→          onClick={handleRefresh}
   447→          disabled={isRefreshing}
   448→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   449→          title="Refresh section data"
   450→        >
   451→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   452→          refresh
   453→        </button>
   454→      </div>
   455→    </div>
   456→  );
   457→}
