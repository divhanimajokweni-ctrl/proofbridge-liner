     1→# Bridging the Gap: Aligning the Epistemic Runtime with the Hydro-Bayesian Research Architecture
     2→
     3→---
     4→
     5→## Executive Summary
     6→
     7→The Epistemic Runtime and the Hydro-Bayesian Kernel represent two distinct but complementary systems. The **Hydro-Bayesian Kernel (HBK)** is the scientific model under investigation—a computational engine integrating EPANET hydraulic simulation with Bayesian filtering. The **Epistemic Runtime** is the execution and governance substrate—a deterministic, auditable execution environment that ensures every simulation trial is reproducible, verifiable, and cryptographically provable.
     8→
     9→This document proposes a **principled integration architecture** that bridges these systems without compromising the separation between scientific research and commercial infrastructure. The integration enables the HBK to leverage the Epistemic Runtime's reproducibility guarantees while preserving the HBK's status as a publishable, open research output.

    87→### 2.2 The Integration Interface
    88→
    89→The HBK should interact with the Epistemic Runtime through a **well-defined, minimal API** that captures execution metadata without exposing the runtime's internals.
    90→
    91→```typescript
    92→// packages/hbk/src/interfaces/runtime-adapter.ts
    93→
    94→/**
    95→ * Minimal interface between HBK and Epistemic Runtime.
    96→ * This is the ONLY integration point.
    97→ */
    98→export interface RuntimeAdapter {
    99→  /**
   100→   * Register a simulation trial with the runtime
   101→   * Returns a proof record that can be verified later
   102→   */
   103→  registerTrial(params: TrialParams): Promise<TrialProof>;
   104→
   105→  /**
   106→   * Record a simulation step's outputs
   107→   * Chain creates an immutable proof log
   108→   */
   109→  recordStep(params: StepParams): Promise<StepReceipt>;
   110→
   111→  /**
   112→   * Retrieve a previously recorded trial
   113→   * Allows verification and replay
   114→   */
   115→  getTrial(trialId: string): Promise<ArchivedTrial>;
   116→}
   117→
   118→export interface TrialParams {
   119→  trialId: string;
   120→  timestamp: string;
   121→  environment: {
   122→    epanetVersion: string;
   123→    networkHash: string;
   124→    leakParameters: LeakParameters;
   125→  };
   126→  algorithm: {
   127→    type: 'PF' | 'EnKF';
   128→    populationSize: number;
   129→    priors: any;
   130→    prngSeed: string;
   131→  };
   132→  evidence: {
   133→    observationCount: number;
   134→    durationSeconds: number;
   135→  };
   136→}
   137→
   138→export interface StepParams {
   139→  trialId: string;
   140→  stepIndex: number;
   141→  state: StateSnapshot;
   142→  evidence: EvidenceSnapshot;
   143→  posterior: PosteriorDistribution;
   144→  metrics: StepMetrics;
   145→}
   146→
   147→export interface TrialProof {
   148→  trialId: string;
   149→  proofHash: string;
   150→  proofPath: string;
   151→  chainIndex: number;
   152→}
   153→
   154→export interface ArchivedTrial {
   155→  trialId: string;
   156→  proof: TrialProof;
   157→  steps: StepReceipt[];
   158→  summary: TrialSummary;
   159→}
   160→```

   256→### 4. Concrete Integration: HBK Executing via Runtime
   257→
   258→#### 4.1 HBK Implementation with Runtime Integration
   259→
   260→```typescript
   261→// packages/hbk/src/simulation/hbk-runner.ts
   262→
   263→import { RuntimeAdapter, TrialParams, StepParams } from '../adapters/runtime-adapter';
   264→import { EpanetSimulator } from './epanet-simulator';
   265→import { ParticleFilter } from '../inference/particle-filter';
   266→import { EnsembleKalmanFilter } from '../inference/ensemble-kalman-filter';
   267→import { ScenarioConfig, ScenarioResult } from '../scenarios/types';
   268→
   269→export class HBKRuntimeRunner {
   270→  private runtime: RuntimeAdapter;
   271→
   272→  constructor(runtimeAdapter: RuntimeAdapter) {
   273→    this.runtime = runtimeAdapter;
   274→  }
   275→
   276→  /**
   277→   * Run a single trial through the Epistemic Runtime
   278→   * All execution metadata is automatically captured and proven
   279→   */
   280→  async runTrial(
   281→    scenario: ScenarioConfig,
   282→    algorithmType: 'PF' | 'EnKF',
   283→    prngSeed: string
   284→  ): Promise<ScenarioResult> {
   285→    const trialId = this.generateTrialId(scenario, algorithmType);
   286→
   287→    // Step 1: Register trial with runtime
   288→    const params: TrialParams = {
   289→      trialId,
   290→      timestamp: new Date().toISOString(),
   291→      environment: {
   292→        epanetVersion: this.getEpanetVersion(),
   293→        networkHash: scenario.networkHash,
   294→        leakParameters: scenario.leakParameters,
   295→      },
   296→      algorithm: {
   297→        type: algorithmType,
   298→        populationSize: algorithmType === 'PF' ? 750 : 64,
   299→        priors: scenario.priors,
   300→        prngSeed,
   301→      },
   302→      evidence: {
   303→        observationCount: scenario.durationSeconds / 5, // 0.2 Hz sampling
   304→        durationSeconds: scenario.durationSeconds,
   305→      },
   306→    };
   307→
   308→    // This registers the trial and returns a proof record
   309→    const proof = await this.runtime.registerTrial(params);
   310→
   311→    // Step 2: Initialize simulation and filter
   312→    const simulator = new EpanetSimulator(scenario.network);
   313→    const filter = algorithmType === 'PF'
   314→      ? new ParticleFilter(params.algorithm.populationSize, scenario.priors, prngSeed)
   315→      : new EnsembleKalmanFilter(params.algorithm.populationSize, scenario.priors, prngSeed);
   316→
   317→    // Step 3: Run simulation loop with step recording
   318→    let currentState = await simulator.initialize();
   319→    let stepIndex = 0;
   320→
   321→    while (!simulator.isComplete()) {
   322→      // Get evidence at this timestep
   323→      const evidence = await simulator.getEvidence();
   324→
   325→      // Run filter update
   326→      const posterior = filter.update(currentState, evidence);
   327→
   328→      // Record step with runtime
   329→      const stepParams: StepParams = {
   330→        trialId,
   331→        stepIndex,
   332→        state: currentState,
   333→        evidence,
   334→        posterior,
   335→        metrics: {
   336→          rmse: this.calculateRMSE(posterior, scenario.trueLeakLocation),
   337→          brierScore: this.calculateBrierScore(posterior, scenario.trueLeakLocation),
   338→          falsePositiveRate: this.calculateFPR(posterior, scenario.trueLeakLocation),
   339→          latencyMs: stepMetrics.latency,
   340→          convergenceTime: this.calculateConvergenceTime(posterior),
   341→        },
   342→      };
   343→
   344→      // Record step → immutable proof log entry
   345→      await this.runtime.recordStep(stepParams);
   346→
   347→      // Advance simulation
   348→      currentState = await simulator.step();
   349→      stepIndex++;
   350→    }
   351→
   352→    // Step 4: Return result (which is also archived)
   353→    return {
   354→      trialId,
   355→      proof,
   356→      algorithm: algorithmType,
   357→      finalMetrics: await this.getFinalMetrics(trialId),
   358→      convergencePath: await this.getConvergencePath(trialId),
   359→    };
   360→  }
   361→}
   362→```

   364→#### 4.2 Runtime Adapter Implementation
   365→
   366→```typescript
   367→// packages/hbk/src/adapters/runtime-adapter.ts
   368→
   369→import { RuntimeAdapter, TrialParams, StepParams } from '../interfaces/runtime-adapter';
   370→import { createHash } from 'crypto';
   371→
   372→/**
   373→ * Production implementation: Connects to the Epistemic Runtime
   374→ */
   375→export class EpistemicRuntimeAdapter implements RuntimeAdapter {
   376→  private runtimeEndpoint: string;
   377→
   378→  constructor(endpoint: string = 'http://localhost:3001/api/runtime') {
   379→    this.runtimeEndpoint = endpoint;
   380→  }
   381→
   382→  async registerTrial(params: TrialParams): Promise<TrialProof> {
   383→    const response = await fetch(`${this.runtimeEndpoint}/trials`, {
   384→      method: 'POST',
   385→      headers: { 'Content-Type': 'application/json' },
   386→      body: JSON.stringify(params),
   387→    });
   388→
   389→    if (!response.ok) {
   390→      throw new Error(`Runtime trial registration failed: ${response.status}`);
   391→    }
   392→
   393→    return response.json();
   394→  }
   395→
   396→  async recordStep(params: StepParams): Promise<StepReceipt> {
   397→    const response = await fetch(`${this.runtimeEndpoint}/trials/${params.trialId}/steps`, {
   398→      method: 'POST',
   399→      headers: { 'Content-Type': 'application/json' },
   400→      body: JSON.stringify(params),
   401→    });
   402→
   403→    if (!response.ok) {
   404→      throw new Error(`Runtime step recording failed: ${response.status}`);
   405→    }
   406→
   407→    return response.json();
   408→  }
   409→
   410→  async getTrial(trialId: string): Promise<ArchivedTrial> {
   411→    const response = await fetch(`${this.runtimeEndpoint}/trials/${trialId}`);
   412→    if (!response.ok) {
   413→      throw new Error(`Trial retrieval failed: ${response.status}`);
   414→    }
   415→    return response.json();
   416→  }
   417→}
   418→
   419→/**
   420→ * Mock implementation: For development and CI (no runtime required)
   421→ */
   422→export class MockRuntimeAdapter implements RuntimeAdapter {
   423→  private trials: Map<string, { params: TrialParams; steps: StepParams[] }> = new Map();
   424→
   425→  async registerTrial(params: TrialParams): Promise<TrialProof> {
   426→    const proof: TrialProof = {
   427→      trialId: params.trialId,
   428→      proofHash: createHash('sha256')
   429→        .update(JSON.stringify(params))
   430→        .digest('hex'),
   431→      proofPath: `mock/proof/${params.trialId}`,
   432→      chainIndex: this.trials.size + 1,
   433→    };
   434→
   435→    this.trials.set(params.trialId, { params, steps: [] });
   436→    return proof;
   437→  }
   438→
   439→  async recordStep(params: StepParams): Promise<StepReceipt> {
   440→    const trial = this.trials.get(params.trialId);
   441→    if (!trial) {
   442→      throw new Error(`Trial ${params.trialId} not found`);
   443→    }
   444→
   445→    trial.steps.push(params);
   446→
   447→    return {
   448→      trialId: params.trialId,
   449→      stepIndex: params.stepIndex,
   450→      receiptHash: createHash('sha256')
   451→        .update(JSON.stringify(params))
   452→        .digest('hex'),
   453→      timestamp: new Date().toISOString(),
   454→    };
   455→  }
   456→
   457→  async getTrial(trialId: string): Promise<ArchivedTrial> {
   458→    const trial = this.trials.get(trialId);
   459→    if (!trial) {
   460→      throw new Error(`Trial ${trialId} not found`);
   461→    }
   462→
   463→    return {
   464→      trialId,
   465→      proof: {
   466→        trialId,
   467→        proofHash: 'mock-hash',
   468→        proofPath: 'mock/path',
   469→        chainIndex: 0,
   470→      },
   471→      steps: trial.steps.map((s) => ({
   472→        trialId: s.trialId,
   473→        stepIndex: s.stepIndex,
   474→        receiptHash: 'mock-hash',
   475→        timestamp: new Date().toISOString(),
   476→      })),
   477→      summary: {
   478→        totalSteps: trial.steps.length,
   479→        durationSeconds: trial.params.evidence.durationSeconds,
   480→        finalRMSE: 0,
   481→      },
   482→    };
   483→  }
   484→}

   505→### 5.1 Verifying a Trial's Integrity
   506→
   507→```typescript
   508→// scripts/verify-proofs.ts
   509→
   510→import { EpistemicRuntimeAdapter } from '../packages/hbk/src/adapters/runtime-adapter';
   511→
   512→async function verifyTrial(trialId: string) {
   513→  const runtime = new EpistemicRuntimeAdapter();
   514→  
   515→  // 1. Retrieve the archived trial from runtime
   516→  const trial = await runtime.getTrial(trialId);
   517→  
   518→  // 2. Verify the proof chain
   519→  const isChainValid = await verifyProofChain(trial);
   520→  
   521→  // 3. Re-run the simulation from the manifest
   522→  // (This is the reproducibility guarantee)
   523→  const rerunResult = await rerunFromManifest(trial);
   524→  
   525→  // 4. Compare original metrics vs rerun
   526→  const matches = compareResults(trial.summary, rerunResult);
   527→  
   528→  console.log(`
   529→    Verification results for trial ${trialId}:
   530→    - Proof chain: ${isChainValid ? '✅ VALID' : '❌ INVALID'}
   531→    - Reproducibility: ${matches ? '✅ PASS' : '❌ FAIL'}
   532→    
   533→    Original RMSE: ${trial.summary.finalRMSE}
   534→    Re-run RMSE:   ${rerunResult.finalRMSE}
   535→  `);
   536→}

   857→
   858→#### 3.1.1 Fault-Tolerant Execution Engine
   859→
   860→**Current State**: `evaluate()` is a pure function—it either succeeds or throws.
   861→
   862→**Gap**: A hardware fault, network timeout, or transient error causes the entire transition to fail.
   863→
   864→**Solution**: Implement **circuit breaker patterns** at the runtime level.
   865→
   866→```typescript
   867→// packages/core/runtime/engine-fault-tolerant.ts
   868→
   869→export interface FaultTolerantRuntime<State, Observation> {
   870→  /**
   871→   * Evaluate with automatic retry and graceful degradation
   872→   * - Retries on transient failures (network, timeouts)
   873→   * - Falls back to safe state on persistent failures
   874→   * - Logs every failure for audit
   875→   */
   876→  evaluateWithResilience(
   877→    state: State,
   878→    observation: Observation,
   879→    ruleSet: CompiledRuleSet,
   880→    epoch: ConstitutionalEpoch,
   881→    options: ResilienceOptions
   882→  ): Promise<Verdict<State>>;
   883→}
   884→
   885→export interface ResilienceOptions {
   886→  maxRetries: number;          // Default: 3
   887→  retryBackoff: number;        // Exponential backoff factor
   888→  fallbackMode: 'safe' | 'maintain' | 'report';
   889→  timeoutMs: number;           // Per-evaluation timeout
   890→  circuitBreakerThreshold: number; // Failures before circuit opens
   891→}
   892→
   893→/**
   894→ * Implementation: Wraps evaluate() with resilience
   895→ */
   896→export class FaultTolerantRuntime<State, Observation>
   897→  implements FaultTolerantRuntime<State, Observation>
   898→{
   899→  private runtime: StateslessRuntime<State, Observation>;
   900→  private circuitState: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED';
   901→  private consecutiveFailures = 0;
   902→
   903→  async evaluateWithResilience(
   904→    state: State,
   905→    observation: Observation,
   906→    ruleSet: CompiledRuleSet,
   907→    epoch: ConstitutionalEpoch,
   908→    options: ResilienceOptions = { maxRetries: 3, retryBackoff: 2, fallbackMode: 'safe', timeoutMs: 5000, circuitBreakerThreshold: 5 }
   909→  ): Promise<Verdict<State>> {
   910→    
   911→    // Check circuit breaker
   912→    if (this.circuitState === 'OPEN') {
   913→      return this.fallbackVerdict(state, 'Circuit breaker open - system degraded');
   914→    }
   915→
   916→    let lastError: Error | null = null;
   917→    let attempt = 0;
   918→
   919→    while (attempt < options.maxRetries) {
   920→      try {
   921→        const result = await this.evaluateWithTimeout(
   922→          state, observation, ruleSet, epoch, options.timeoutMs
   923→        );
   924→
   925→        // Success: reset circuit
   926→        this.consecutiveFailures = 0;
   927→        this.circuitState = 'CLOSED';
   928→        return result;
   929→
   930→      } catch (error) {
   931→        lastError = error as Error;
   932→        attempt++;
   933→        
   934→        // Log failure for audit
   935→        await this.logFailure({
   936→          timestamp: new Date().toISOString(),
   937→          attempt,
   938→          error: lastError.message,
   939→          stateHash: hashObject(state),
   940→          observationHash: hashObject(observation),
   941→        });
   942→
   943→        // Check if we should open circuit
   944→        this.consecutiveFailures++;
   945→        if (this.consecutiveFailures >= options.circuitBreakerThreshold) {
   946→          this.circuitState = 'OPEN';
   947→          console.error(`[Runtime] Circuit breaker OPEN after ${this.consecutiveFailures} consecutive failures`);
   948→          break;
   949→        }
   950→
   951→        // Exponential backoff
   952→        const delay = options.retryBackoff ** attempt * 100;
   953→        await this.sleep(delay);
   954→      }
   955→    }
   956→
   957→    // All retries failed
   958→    return this.fallbackVerdict(state, `All retries failed: ${lastError?.message}`);
   959→  }
   960→
   961→  private async evaluateWithTimeout(
   962→    state: State,
   963→    observation: Observation,
   964→    ruleSet: CompiledRuleSet,
   965→    epoch: ConstitutionalEpoch,
   966→    timeoutMs: number
   967→  ): Promise<Verdict<State>> {
   968→    return new Promise((resolve, reject) => {
   969→      const timeout = setTimeout(() => {
   970→        reject(new Error(`Evaluation timeout after ${timeoutMs}ms`));
   971→      }, timeoutMs);
   972→
   973→      try {
   974→        const result = this.runtime.evaluate(state, observation, ruleSet, epoch, '', null);
   975→        clearTimeout(timeout);
   976→        resolve(result);
   977→      } catch (error) {
   978→        clearTimeout(timeout);
   979→        reject(error);
   980→      }
   981→    });
   982→  }
   983→
   984→  private fallbackVerdict(state: State, reason: string): Verdict<State> {
   985→    return {
   986→      accepted: false,
   987→      candidateState: state,
   988→      proof: {
   989→        epochId: 'fallback',
   990→        observationHash: 'fallback',
   991→        inferenceHash: '',
   992→        previousProofHash: null,
   993→        signature: '',
   994→        timestamp: new Date().toISOString(),
   995→        divergenceSignals: [reason],
   996→      },
   997→      divergenceSignals: [reason],
   998→    };
   999→  }
  1000→
  1001→  private sleep(ms: number): Promise<void> {
  1002→    return new Promise(resolve => setTimeout(resolve, ms));
  1003→  }
  1004→
  1005→  private async logFailure(failure: any): Promise<void> {
  1006→    // Append to immutable failure log
  1007→    // In production: write to distributed log
  1008→    console.warn('[Runtime] Evaluation failure:', failure);
  1009→  }
  1010→}
  1011→```

  1015→#### 3.1.2 State Recovery & Replay Protection
  1016→
  1017→**Current State**: Projector folds proof log to reconstruct state. No protection against replay attacks.
  1018→
  1019→**Gap**: An attacker could replay old proof logs to roll back state.
  1020→
  1021→**Solution**: Add **chain height checks** and **Merkle root verification**.
  1022→
  1023→```typescript
  1024→// packages/core/runtime/projector-replay-protected.ts
  1025→
  1026→export interface ReplayProtectedProjector<State> extends Projector<State> {
  1027→  /**
  1028→   * Fold with replay protection
  1029→   * - Verifies chain height monotonicity
  1030→   * - Checks Merkle root integrity
  1031→   * - Prevents rollback attacks
  1032→   */
  1033→  foldWithProtection(
  1034→    genesis: Genesis,
  1035→    proofLog: readonly ProofRecord[],
  1036→    activeEpoch: ConstitutionalEpoch,
  1037→    stateTransformers: Record<string, StateTransformer<State>>,
  1038→    lastKnownHeight?: number
  1039→  ): State;
  1040→}
  1041→
  1042→export class ReplayProtectedProjector<State>
  1043→  extends Projector<State>
  1044→  implements ReplayProtectedProjector<State>
  1045→{
  1046→  private merkleRootCache = new Map<string, string>();
  1047→
  1048→  foldWithProtection(
  1049→    genesis: Genesis,
  1050→    proofLog: readonly ProofRecord[],
  1051→    activeEpoch: ConstitutionalEpoch,
  1052→    stateTransformers: Record<string, StateTransformer<State>>,
  1053→    lastKnownHeight?: number
  1054→  ): State {
  1055→    
  1056→    // 1. Verify chain height (prevent rollback)
  1057→    if (lastKnownHeight !== undefined && proofLog.length < lastKnownHeight) {
  1058→      throw new Error(
  1059→        `Replay attack detected: chain height decreased from ${lastKnownHeight} to ${proofLog.length}`
  1060→      );
  1061→    }
  1062→
  1063→    // 2. Verify each proof's epoch validity
  1064→    const invalidEpochs: string[] = [];
  1065→    for (const proof of proofLog) {
  1066→      if (!this.isEpochValid(proof.epochId, activeEpoch)) {
  1067→        invalidEpochs.push(proof.epochId);
  1068→      }
  1069→    }
  1070→    if (invalidEpochs.length > 0) {
  1071→      throw new Error(
  1072→        `Invalid epochs in proof log: ${invalidEpochs.join(', ')}`
  1073→      );
  1074→    }
  1075→
  1076→    // 3. Compute Merkle root and verify against Genesis
  1077→    const computedRoot = this.computeMerkleRoot(proofLog);
  1078→    if (genesis.rootHash && computedRoot !== genesis.rootHash) {
  1079→      throw new Error(
  1080→        `Merkle root mismatch: expected ${genesis.rootHash}, got ${computedRoot}`
  1081→      );
  1082→    }
  1083→
  1084→    // 4. Proceed with normal fold
  1085→    return super.fold(genesis, proofLog, activeEpoch, stateTransformers);
  1086→  }
  1087→
  1088→  private computeMerkleRoot(proofLog: readonly ProofRecord[]): string {
  1089→    if (proofLog.length === 0) return 'empty';
  1090→    
  1091→    const hashes = proofLog.map(p => p.inferenceHash);
  1092→    return merkleRoot(hashes);
  1093→  }
  1094→}
  1095→```

  1099→#### 3.1.3 Graceful Degradation Mode
  1100→
  1101→**Current State**: Runtime either succeeds or rejects. No partial operation.
  1102→
  1103→**Gap**: In critical systems, a total failure may be worse than degraded operation.
  1104→
  1105→**Solution**: Implement **degradation states** with clear safety boundaries.
  1106→
  1107→```typescript
  1108→// packages/core/runtime/degradation.ts
  1109→
  1110→export type DegradationLevel = 'nominal' | 'degraded' | 'minimal' | 'failsafe';
  1111→
  1112→export interface DegradationState {
  1113→  level: DegradationLevel;
  1114→  reason: string;
  1115→  timestamp: string;
  1116→  capabilities: {
  1117→    evaluation: boolean;
  1118→    proofGeneration: boolean;
  1119→    stateProjection: boolean;
  1120→    governance: boolean;
  1121→  };
  1122→  constraints: {
  1123→    maxObservationRate?: number; // Observations per second
  1124→    maxStateSize?: number;       // Bytes
  1125→    allowedOperators?: string[]; // Limited rule set
  1126→  };
  1127→}
  1128→
  1129→export class DegradationManager {
  1130→  private currentState: DegradationState = {
  1131→    level: 'nominal',
  1132→    reason: 'System operating normally',
  1133→    timestamp: new Date().toISOString(),
  1134→    capabilities: {
  1135→      evaluation: true,
  1136→      proofGeneration: true,
  1137→      stateProjection: true,
  1138→      governance: true,
  1139→    },
  1140→    constraints: {},
  1141→  };
  1142→
  1143→  /**
  1144→   * Detect degradation triggers
  1145→   */
  1146→  public assessState(health: RuntimeHealth): DegradationLevel {
  1147→    // Check for resource exhaustion
  1148→    if (health.memoryUsage > 0.9) return 'minimal';
  1149→    if (health.memoryUsage > 0.75) return 'degraded';
  1150→    
  1151→    // Check for performance degradation
  1152→    if (health.averageLatency > 10000) return 'minimal';
  1153→    if (health.averageLatency > 5000) return 'degraded';
  1154→    
  1155→    // Check for error rate
  1156→    if (health.errorRate > 0.2) return 'failsafe';
  1157→    if (health.errorRate > 0.05) return 'minimal';
  1158→    
  1159→    return 'nominal';
  1160→  }
  1161→
  1162→  /**
  1163→   * Apply degradation constraints
  1164→   */
  1165→  public applyDegradation(level: DegradationLevel): DegradationState {
  1166→    switch (level) {
  1167→      case 'nominal':
  1168→        return {
  1169→          level: 'nominal',
  1170→          reason: 'Full capability',
  1171→          timestamp: new Date().toISOString(),
  1172→          capabilities: { evaluation: true, proofGeneration: true, stateProjection: true, governance: true },
  1173→          constraints: {},
  1174→        };
  1175→
  1176→      case 'degraded':
  1177→        return {
  1178→          level: 'degraded',
  1179→          reason: 'Performance degradation detected - reducing throughput',
  1180→          timestamp: new Date().toISOString(),
  1181→          capabilities: { evaluation: true, proofGeneration: true, stateProjection: true, governance: true },
  1182→          constraints: { maxObservationRate: 2 }, // Slow down to 2 ops/sec
  1183→        };
  1184→
  1185→      case 'minimal':
  1186→        return {
  1187→          level: 'minimal',
  1188→          reason: 'Resource constraints - limiting operations',
  1189→          timestamp: new Date().toISOString(),
  1190→          capabilities: { evaluation: true, proofGeneration: false, stateProjection: true, governance: false },
  1191→          constraints: { maxObservationRate: 1, maxStateSize: 1024 * 1024 },
  1192→        };
  1193→
  1194→      case 'failsafe':
  1195→        return {
  1196→          level: 'failsafe',

  1209→### Priority 2: Distributed & Consensus
  1210→
  1211→#### 3.2.1 Distributed Runtime Cluster
  1212→
  1213→**Current State**: Single-node runtime.
  1214→
  1215→**Gap**: Single point of failure for critical infrastructure.
  1216→
  1217→**Solution**: Implement **distributed consensus** for runtime decisions.
  1218→
  1219→```typescript
  1220→// packages/core/runtime/distributed.ts
  1221→
  1222→export interface DistributedRuntime<State, Observation> {
  1223→  /**
  1224→   * Evaluate across a cluster of runtime nodes
  1225→   * - Quorum consensus on verdict
  1226→   * - Byzantine fault tolerance
  1227→   * - Automatic failover
  1228→   */
  1229→  evaluateDistributed(
  1230→    state: State,
  1231→    observation: Observation,
  1232→    ruleSet: CompiledRuleSet,
  1233→    epoch: ConstitutionalEpoch,
  1234→    quorumSize?: number
  1235→  ): Promise<DistributedVerdict<State>>;
  1236→}
  1237→
  1238→export interface DistributedVerdict<State> {
  1239→  accepted: boolean;
  1240→  candidateState: State;
  1241→  proof: ProofRecord;
  1242→  consensus: {
  1243→    totalNodes: number;
  1244→    votesFor: number;
  1245→    votesAgainst: number;
  1246→    quorumReached: boolean;
  1247→  };
  1248→  nodes: {
  1249→    id: string;
  1250→    verdict: Verdict<State>;
  1251→    latency: number;
  1252→  }[];
  1253→}
  1254→
  1255→/**
  1256→ * Implementation: Raft-inspired consensus
  1257→ */
  1258→export class RaftDistributedRuntime<State, Observation>
  1259→  implements DistributedRuntime<State, Observation>
  1260→{
  1261→  private nodes: RuntimeNode[] = [];
  1262→  private currentLeader: string | null = null;
  1263→
  1264→  async evaluateDistributed(
  1265→    state: State,
  1266→    observation: Observation,
  1267→    ruleSet: CompiledRuleSet,
  1268→    epoch: ConstitutionalEpoch,
  1269→    quorumSize: number = Math.floor(this.nodes.length * 2 / 3) + 1
  1270→  ): Promise<DistributedVerdict<State>> {
  1271→    
  1272→    // 1. Find or elect leader
  1273→    const leader = this.getLeader();
  1274→    if (!leader) {
  1275→      throw new Error('No leader available');
  1276→    }
  1277→
  1278→    // 2. Send evaluation request to all nodes
  1279→    const results = await Promise.allSettled(
  1280→      this.nodes.map(node =>
  1281→        node.evaluate(state, observation, ruleSet, epoch)
  1282→      )
  1283→    );
  1284→
  1285→    // 3. Count votes
  1286→    const verdicts = results
  1287→      .filter((r): r is PromiseFulfilledResult<Verdict<State>> => 
  1288→        r.status === 'fulfilled'
  1289→      )
  1290→      .map(r => r.value);
  1291→
  1292→    const votesFor = verdicts.filter(v => v.accepted).length;
  1293→    const votesAgainst = verdicts.filter(v => !v.accepted).length;
  1294→
  1295→    // 4. Check quorum
  1296→    const quorumReached = votesFor >= quorumSize;
  1297→
  1298→    // 5. Return consensus verdict
  1299→    const accepted = quorumReached && votesFor > votesAgainst;
  1300→
  1301→    return {
  1302→      accepted,
  1303→      candidateState: accepted ? this.aggregateState(verdicts) : state,
  1304→      proof: this.aggregateProof(verdicts),
  1305→      consensus: {
  1306→        totalNodes: this.nodes.length,
  1307→        votesFor,
  1308→        votesAgainst,
  1309→        quorumReached,
  1310→      },
  1311→      nodes: verdicts.map((v, i) => ({
  1312→        id: this.nodes[i].id,
  1313→        verdict: v,
  1314→        latency: this.nodes[i].lastLatency,
  1315→      })),
  1316→    };
  1317→  }
  1318→
  1319→  private aggregateState(verdicts: Verdict<State>[]): State {
  1320→    // State aggregation strategy: majority rule
  1321→    // In production: use state reconciliation
  1322→    const stateMap = new Map<string, { count: number; state: State }>();
  1323→    
  1324→    for (const v of verdicts) {
  1325→      const key = hashObject(v.candidateState);
  1326→      const existing = stateMap.get(key);
  1327→      if (existing) {
  1328→        existing.count++;
  1329→      } else {
  1330→        stateMap.set(key, { count: 1, state: v.candidateState });
  1331→      }
  1332→    }
  1333→
  1334→    let maxCount = 0;
  1335→    let maxState: State | null = null;
  1336→    for (const [_, entry] of stateMap) {
  1337→      if (entry.count > maxCount) {
  1338→        maxCount = entry.count;
  1339→        maxState = entry.state;
  1340→      }
  1341→    }
  1342→
  1343→    return maxState || verdicts[0].candidateState;
  1344→  }
  1345→
  1346→  private aggregateProof(verdicts: Verdict<State>[]): ProofRecord {
  1347→    // Combine proofs from multiple nodes
  1348→    // In production: use threshold signatures
  1349→    return {
  1350→      epochId: verdicts[0].proof.epochId,
  1351→      observationHash: verdicts[0].proof.observationHash,
  1352→      inferenceHash: this.combinedInferenceHash(verdicts),
  1353→      previousProofHash: verdicts[0].proof.previousProofHash,
  1354→      signature: this.aggregateSignatures(verdicts),
  1355→      timestamp: new Date().toISOString(),
  1356→    };
  1357→  }
  1358→
  1359→  private combinedInferenceHash(verdicts: Verdict<State>[]): string {
  1360→    // Combine all inference hashes into a single hash
  1361→    return hashObject(verdicts.map(v => v.proof.inferenceHash).sort());
  1362→  }
  1363→
  1364→  private aggregateSignatures(verdicts: Verdict<State>[]): string {
  1365→    // In production: BLS signature aggregation
  1366→    return verdicts.map(v => v.proof.signature).join(':');
  1367→  }
  1368→
  1369→  private getLeader(): RuntimeNode | null {
  1370→    if (this.currentLeader) {
  1371→      const leader = this.nodes.find(n => n.id === this.currentLeader);
  1372→      if (leader && leader.isHealthy()) {
  1373→        return leader;
  1374→      }
  1375→    }
  1376→    // Elect new leader
  1377→    return this.electLeader();
  1378→  }
  1379→
  1380→  private electLeader(): RuntimeNode | null {
  1381→    const healthyNodes = this.nodes.filter(n => n.isHealthy());
  1382→    if (healthyNodes.length === 0) return null;
  1383→    
  1384→    // Simple leader election: oldest healthy node
  1385→    const leader = healthyNodes.reduce((a, b) => 
  1386→      a.startTime < b.startTime ? a : b
  1387→    );
  1388→    this.currentLeader = leader.id;
  1389→    return leader;
  1390→  }
  1391→}

  1396→#### 3.2.2 Byzantine Fault Tolerance (BFT)
  1397→
  1398→**Critical for high-risk systems**: Nodes must tolerate malicious or faulty nodes.
  1399→
  1400→```typescript
  1401→// packages/core/runtime/bft.ts
  1402→
  1403→export interface BFTConfig {
  1404→  f: number; // Maximum Byzantine nodes
  1405→  n: number; // Total nodes (must satisfy n ≥ 3f + 1)
  1406→}
  1407→
  1408→export class BFTDistributedRuntime<State, Observation>
  1409→  extends RaftDistributedRuntime<State, Observation>
  1410→{
  1411→  private config: BFTConfig;
  1412→
  1413→  constructor(nodes: RuntimeNode[], config: BFTConfig) {
  1414→    super(nodes);
  1415→    this.config = config;
  1416→    
  1417→    // Verify BFT condition
  1418→    if (nodes.length < 3 * config.f + 1) {
  1419→      throw new Error(
  1420→        `BFT condition violated: need ${3 * config.f + 1} nodes, have ${nodes.length}`
  1421→      );
  1422→    }
  1423→  }
  1424→
  1425→  async evaluateDistributed(
  1426→    state: State,
  1427→    observation: Observation,
  1428→    ruleSet: CompiledRuleSet,
  1429→    epoch: ConstitutionalEpoch
  1430→  ): Promise<DistributedVerdict<State>> {
  1431→    
  1432→    // 1. Send to all nodes
  1433→    const results = await Promise.allSettled(
  1434→      this.nodes.map(node =>
  1435→        node.evaluate(state, observation, ruleSet, epoch)
  1436→      )
  1437→    );
  1438→
  1439→    // 2. Filter out failed nodes
  1440→    const validVerdicts = results
  1441→      .filter((r): r is PromiseFulfilledResult<Verdict<State>> => 
  1442→        r.status === 'fulfilled' && this.verifyProof(r.value.proof)
  1443→      )
  1444→      .map(r => r.value);
  1445→
  1446→    // 3. BFT quorum: need at least 2f + 1 valid responses
  1447→    const required = 2 * this.config.f + 1;
  1448→    if (validVerdicts.length < required) {
  1449→      throw new Error(
  1450→        `BFT quorum not reached: need ${required}, got ${validVerdicts.length}`
  1451→      );
  1452→    }
  1453→
  1454→    // 4. Byzantine agreement on verdict
  1455→    const verdictCounts = new Map<string, number>();
  1456→    for (const v of validVerdicts) {
  1457→      const key = `${v.accepted}:${hashObject(v.candidateState)}`;
  1458→      verdictCounts.set(key, (verdictCounts.get(key) || 0) + 1);
  1459→    }
  1460→
  1461→    let maxCount = 0;
  1462→    let maxKey = '';
  1463→    for (const [key, count] of verdictCounts) {
  1464→      if (count > maxCount) {
  1465→        maxCount = count;
  1466→        maxKey = key;
  1467→      }
  1468→    }
  1469→
  1470→    // 5. Only accept if agreement > f + 1 (Byzantine majority)
  1471→    const [accepted, stateHash] = maxKey.split(':');
  1472→    const isAgreed = maxCount > this.config.f;
  1473→
  1474→    return {
  1475→      accepted: isAgreed && accepted === 'true',
  1476→      candidateState: isAgreed ? this.findStateByHash(validVerdicts, stateHash) : state,
  1477→      proof: this.aggregateProof(validVerdicts),
  1478→      consensus: {
  1479→        totalNodes: this.nodes.length,
  1480→        votesFor: validVerdicts.filter(v => v.accepted).length,
  1481→        votesAgainst: validVerdicts.filter(v => !v.accepted).length,
  1482→        quorumReached: isAgreed,
  1483→      },
  1484→      nodes: validVerdicts.map((v, i) => ({
  1485→        id: this.nodes[i].id,
  1486→        verdict: v,
  1487→        latency: this.nodes[i].lastLatency,
  1488→      })),
  1489→    };
  1490→  }
  1491→
  1492→  private verifyProof(proof: ProofRecord): boolean {
  1493→    // In production: cryptographic verification
  1494→    return true;
  1495→  }
  1496→
  1497→  private findStateByHash(verdicts: Verdict<State>[], hash: string): State {
  1498→    for (const v of verdicts) {
  1499→      if (hashObject(v.candidateState) === hash) {
  1500→        return v.candidateState;
  1501→      }
  1502→    }
  1503→    return verdicts[0].candidateState;
  1504→  }
  1505→}

  1512→#### 3.3.1 Automated Key Rotation
  1513→
  1514→**Current State**: Static keys in Genesis.
  1515→
  1516→**Gap**: Keys must be rotated periodically to maintain security.
  1517→
  1518→**Solution**: Implement **automated key rotation** with epoch transitions.
  1519→
  1520→```typescript
  1521→// packages/core/governance/key-rotation.ts
  1522→
  1523→export interface KeyRotationPolicy {
  1524→  maxKeyAge: number;        // Seconds before rotation required
  1525→  rotationWindow: number;   // Grace period for rotation
  1526→  quorumRequired: number;   // Number of signers required
  1527→}
  1528→
  1529→export class KeyRotationManager {
  1530→  private currentKey: string;
  1531→  private keyCreatedAt: number;
  1532→  private policy: KeyRotationPolicy;
  1533→
  1534→  constructor(initialKey: string, policy: KeyRotationPolicy) {
  1535→    this.currentKey = initialKey;
  1536→    this.keyCreatedAt = Date.now();
  1537→    this.policy = policy;
  1538→  }
  1539→
  1540→  /**
  1541→   * Check if key rotation is required
  1542→   */
  1543→  public shouldRotate(): boolean {
  1544→    const keyAge = (Date.now() - this.keyCreatedAt) / 1000;
  1545→    return keyAge > this.policy.maxKeyAge;
  1546→  }
  1547→
  1548→  /**
  1549→   * Generate new key and create new epoch
  1550→   */
  1551→  public async rotateKey(
  1552→    currentEpoch: ConstitutionalEpoch,
  1553→    signers: string[]
  1554→  ): Promise<ConstitutionalEpoch> {
  1555→    if (!this.shouldRotate() && !this.isWithinRotationWindow()) {
  1556→      throw new Error('Key rotation not yet required');
  1557→    }
  1558→
  1559→    // 1. Generate new key pair
  1560→    const newKey = this.generateNewKey();
  1561→    
  1562→    // 2. Get quorum signatures
  1563→    const signatures = await this.collectSignatures(newKey, signers);
  1564→    if (signatures.length < this.policy.quorumRequired) {
  1565→      throw new Error(
  1566→        `Insufficient signatures: need ${this.policy.quorumRequired}, got ${signatures.length}`
  1567→      );
  1568→    }
  1569→
  1570→    // 3. Create new epoch
  1571→    return {
  1572→      id: `epoch-${Date.now()}`,
  1573→      version: `${parseInt(currentEpoch.version) + 1}`,
  1574→      parent: currentEpoch.id,
  1575→      effectiveFrom: new Date().toISOString(),
  1576→      specificationHash: currentEpoch.specificationHash,
  1577→      ruleSetHash: currentEpoch.ruleSetHash,
  1578→      signer: newKey.publicKey,
  1579→      multisigThreshold: this.policy.quorumRequired,
  1580→      signatures: signatures.map(s => s.signature),
  1581→    };
  1582→  }
  1583→
  1584→  private generateNewKey(): { publicKey: string; privateKey: string } {
  1585→    // In production: use HSM or secure enclave
  1586→    return {
  1587→      publicKey: `pub-${Date.now()}`,
  1588→      privateKey: `priv-${Date.now()}`,
  1589→    };
  1590→  }
  1591→
  1592→  private async collectSignatures(
  1593→    newKey: { publicKey: string; privateKey: string },
  1594→    signers: string[]
  1595→  ): Promise<{ signer: string; signature: string }[]> {
  1596→    // In production: request signatures from authorized signers
  1597→    return signers.map(s => ({
  1598→      signer: s,
  1599→      signature: `sig-${Date.now()}-${s}`,
  1600→    }));
  1601→  }
  1602→
  1603→  private isWithinRotationWindow(): boolean {
  1604→    const keyAge = (Date.now() - this.keyCreatedAt) / 1000;
  1605→    return keyAge > this.policy.maxKeyAge - this.policy.rotationWindow;
  1606→  }
  1607→}

  1809→      .map((t, i) => ({ time: t, error: this.metrics.errors[i] }))
  1810→      .filter(({ time }) => now - time < 60000)
  1811→      .filter(({ error }) => error === 1)
  1812→      .length;
  1813→    const recentErrorRate = recentErrors / (recent.length || 1);
  1814→
  1815→    // Degradation level
  1816→    let degradationLevel: 'nominal' | 'degraded' | 'minimal' | 'failsafe' = 'nominal';
  1817→    if (recentErrorRate > 0.2) degradationLevel = 'failsafe';
  1818→    else if (recentErrorRate > 0.05) degradationLevel = 'minimal';
  1819→    else if (avgLatency > 5000) degradationLevel = 'degraded';
  1820→
  1821→    // Component health
  1822→    const componentHealth = {
  1823→      engine: avgLatency > 10000 ? 'unhealthy' : avgLatency > 5000 ? 'degraded' : 'healthy',
  1824→      projector: this.checkProjectorHealth(),
  1825→      governance: this.checkGovernanceHealth(),
  1826→      storage: this.checkStorageHealth(),
  1827→    };
  1828→
  1829→    const allHealthy = Object.values(componentHealth).every(h => h === 'healthy');
  1830→    const anyDegraded = Object.values(componentHealth).some(h => h === 'degraded');
  1831→
  1832→    return {
  1833→      status: allHealthy ? 'healthy' : anyDegraded ? 'degraded' : 'unhealthy',
  1834→      timestamp: new Date().toISOString(),

  1845→      healthyNodes: this.getHealthyNodes(),
  1846→      epochId: this.getCurrentEpoch(),
  1847→      keyAge: this.getKeyAge(),
  1848→      certificatesValid: this.getValidCertificates(),
  1849→      certificatesExpiring: this.getExpiringCertificates(),
  1850→      components: componentHealth,
  1851→      degradationLevel,
  1852→    };
  1853→  }

  1874→#### 4.1.2 Incident Reporting & Alerting
  1875→
  1876→**Critical for autonomous operation**: System must report its own failures.
  1877→
  1878→```typescript
  1879→// packages/core/monitoring/incident.ts
  1880→
  1881→export interface Incident {
  1882→  id: string;
  1883→  timestamp: string;
  1884→  severity: 'critical' | 'major' | 'minor' | 'info';
  1885→  category: 'performance' | 'security' | 'governance' | 'consensus' | 'degradation';
  1886→  title: string;
  1887→  description: string;
  1888→  affectedComponents: string[];
  1889→  rootCause?: string;
  1890→  resolution?: string;
  1891→  resolvedAt?: string;
  1892→  proof?: ProofRecord; // Cryptographic proof of incident
  1893→}
  1894→
  1895→export class IncidentReporter {
  1896→  private incidents: Incident[] = [];
  1897→  private alertChannels: AlertChannel[] = [];
  1898→
  1899→  /**
  1900→   * Report an incident
  1901→   */
  1902→  public reportIncident(
  1903→    severity: Incident['severity'],
  1904→    category: Incident['category'],
  1905→    title: string,
  1906→    description: string,
  1907→    affectedComponents: string[],
  1908→    proof?: ProofRecord
  1909→  ): Incident {
  1910→    const incident: Incident = {
  1911→      id: `inc-${Date.now()}`,
  1912→      timestamp: new Date().toISOString(),
  1913→      severity,
  1914→      category,
  1915→      title,
  1916→      description,
  1917→      affectedComponents,
  1918→      proof,
  1919→    };
  1920→
  1921→    this.incidents.push(incident);
  1922→
  1923→    // Send alerts based on severity
  1924→    if (severity === 'critical' || severity === 'major') {
  1925→      this.sendAlerts(incident);
  1926→    }
  1927→
  1928→    // Log to immutable audit log
  1929→    this.logIncident(incident);
  1930→
  1931→    return incident;
  1932→  }
  1933→
  1934→  /**
  1935→   * Resolve an incident
  1936→   */
  1937→  public resolveIncident(incidentId: string, resolution: string): void {
  1938→    const incident = this.incidents.find(i => i.id === incidentId);
  1939→    if (incident) {
  1940→      incident.resolution = resolution;
  1941→      incident.resolvedAt = new Date().toISOString();
  1942→    }
  1943→  }
  1944→
  1945→  /**
  1946→   * Get open incidents
  1947→   */
  1948→  public getOpenIncidents(): Incident[] {
  1949→    return this.incidents.filter(i => !i.resolvedAt);
  1950→  }
  1951→
  1952→  /**
  1953→   * Add an alert channel
  1954→   */
  1955→  public addAlertChannel(channel: AlertChannel): void {
  1956→    this.alertChannels.push(channel);
  1957→  }
  1958→
  1959→  private sendAlerts(incident: Incident): void {
  1960→    for (const channel of this.alertChannels) {
  1961→      try {
  1962→        channel.send(incident);
  1963→      } catch (err) {
  1964→        console.error('Failed to send alert:', err);
  1965→      }
  1966→    }
  1967→  }
  1968→
  1969→  private logIncident(incident: Incident): void {
  1970→    // In production: append to immutable incident log
  1971→    console.log(`[INCIDENT] ${incident.severity}: ${incident.title}`);
  1972→  }
  1973→}
  1974→
  1975→export interface AlertChannel {
  1976→  send(incident: Incident): void;
  1977→}
