  1066→            ingest(t) {
  1067→                let r = this.tracks.get(t.id);
  1068→                if (!r) {
  1069→                    r = { cur: new THREE.Vector3(...t.pos), tgt: new THREE.Vector3(...t.pos), cd: new THREE.Vector3(1, 1,
  1070→                            1), td: new THREE.Vector3(...t.dims), yaw: t.yaw || 0, tYaw: t.yaw || 0, vel: new THREE
  1071→                            .Vector3(...t.vel), tVel: new THREE.Vector3(...t.vel), hasHdg: t.hasHdg || false,

  1073→                        born: performance.now() };
  1074→                    this.tracks.set(t.id, r);
  1075→                } else { r.tgt.set(...t.pos);
  1076→                    r.td.set(...t.dims);
  1077→                    r.tYaw = t.yaw || 0;
  1078→                    r.tVel.set(...t.vel);
  1079→                    r.hasHdg = t.hasHdg || false;
  1080→                    r.conf = t.conf || 0.8;
  1081→                    r.cls = t.cls || 'unknown';
  1082→                    r.seen = performance.now(); }
  1083→            }
  1084→            update(dt, now) {
  1085→                const k = 1 - Math.exp(-dt * LERP_RATE);
  1086→                let i = 0;
  1087→                for (const [id, r] of this.tracks) {
  1088→                    if (now - r.seen > TRACK_TTL) { this.tracks.delete(id); continue; }
  1089→                    r.cur.lerp(r.tgt, k);
  1090→                    r.cd.lerp(r.td, k);
  1091→                    const dy = r.tYaw - r.yaw;
  1092→                    r.yaw += Math.atan2(Math.sin(dy), Math.cos(dy)) * k;

  1094→                    this.dummy.position.copy(r.cur);

  1096→                    this.dummy.scale.set(r.cd.x * s, r.cd.y * s, r.cd.z * s);
  1097→                    this.dummy.updateMatrix();

  1099→                    this.col.setHex(CLASS_COLOR[r.cls] || CLASS_COLOR.unknown);
  1100→                    this.fill.setColorAt(i, this.col);
  1101→                    i++;
  1102→                    if (i >= this.max) break;
  1103→                }
  1104→                this.fill.count = this.edges.count = i;
  1105→                this.fill.instanceMatrix.needsUpdate = true;
  1106→                if (this.fill.instanceColor) this.fill.instanceColor.needsUpdate = true;
  1107→            }

  1114→                }
  1115→                return result;
  1116→            }
  1117→        }
