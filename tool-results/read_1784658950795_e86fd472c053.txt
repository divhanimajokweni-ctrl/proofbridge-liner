   173→/* ─── Rating Color Helper ─── */
   174→function ratingColor(rating: string) {
   175→  switch (rating) {
   176→    case "Excellent": return { text: "text-verified", bg: "bg-verified/10", border: "border-verified/30" };
   177→    case "Good": return { text: "text-repairing", bg: "bg-repairing/10", border: "border-repairing/30" };
   178→    case "Incomplete": return { text: "text-quarantined", bg: "bg-quarantined/10", border: "border-quarantined/30" };
   179→    case "Missing": return { text: "text-violating", bg: "bg-violating/10", border: "border-violating/30" };
   180→    default: return { text: "text-muted-foreground", bg: "bg-muted", border: "border-border/40" };
   181→  }
   182→}
