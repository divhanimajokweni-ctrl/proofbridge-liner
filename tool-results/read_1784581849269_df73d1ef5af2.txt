   120→function fmtRelative(iso: string | null): string {
   121→  if (!iso) return "—";
   122→  const t = new Date(iso).getTime();
   123→  const diff = Date.now() - t;
   124→  if (diff < 0) return "just now";
   125→  const s = Math.floor(diff / 1000);
   126→  if (s < 60) return `${s}s ago`;
   127→  const m = Math.floor(s / 60);
   128→  if (m < 60) return `${m}m ago`;
   129→  const h = Math.floor(m / 60);
   130→  if (h < 24) return `${h}h ago`;
   131→  return `${Math.floor(h / 24)}d ago`;
   132→}

   222→function DriftSparkline({
   223→  events,
   224→  className,
   225→}: {
   226→  events: ShadowEventRow[];
   227→  className?: string;
   228→}) {
   229→  const data = useMemo(() => {

   233→        idx: i,
   234→        drift: Math.abs(Math.sin(i * 0.4) * 0.3 + Math.cos(i * 0.7) * 0.15 + 0.05),
   235→      }));
   236→    }
   237→    return events.slice(0, 20).map((ev, i) => ({
   238→      idx: i,
   239→      drift: ev.divergence,

   243→  const maxDrift = Math.max(...data.map((d) => d.drift), 0.01);

   282→/* ─── Animated gauge needle ─── */
   283→function GaugeNeedle({ value, max, size = 80 }: { value: number; max: number; size?: number }) {
   284→  const clamped = Math.min(Math.max(value / max, 0), 1);
   285→  const angle = -90 + clamped * 180; // -90 to 90 degrees
   286→
   287→  const color =
   288→    clamped < 0.33 ? "oklch(0.78 0.16 160)" : clamped < 0.66 ? "oklch(0.80 0.15 80)" : "oklch(0.64 0.21 25)";
   289→
   290→  return (
   291→    <svg width={size} height={size / 2 + 6} viewBox="0 0 80 46" className="overflow-visible">
   292→      <defs>
   293→        <linearGradient id="gaugeArcGrad" x1="0" y1="0" x2="1" y2="0">
   294→          <stop offset="0%" stopColor="oklch(0.78 0.16 160)" />
   295→          <stop offset="50%" stopColor="oklch(0.80 0.15 80)" />
   296→          <stop offset="100%" stopColor="oklch(0.64 0.21 25)" />
   297→        </linearGradient>
   298→      </defs>
   299→      {/* Background arc */}
   300→      <path
   301→        d="M 8 42 A 34 34 0 0 1 72 42"
   302→        fill="none"
   303→        stroke="oklch(0.32 0.014 165 / 0.4)"
   304→        strokeWidth={5}
   305→        strokeLinecap="round"
   306→      />
   307→      {/* Gradient arc (filled portion) */}
   308→      <motion.path
   309→        d="M 8 42 A 34 34 0 0 1 72 42"
   310→        fill="none"
   311→        stroke="url(#gaugeArcGrad)"
   312→        strokeWidth={5}
   313→        strokeLinecap="round"
   314→        strokeDasharray={`${clamped * 107} 107`}
   315→        initial={{ strokeDasharray: "0 107" }}
   316→        animate={{ strokeDasharray: `${clamped * 107} 107` }}
   317→        transition={{ duration: 1.2, ease: "easeOut" }}
   318→      />
   319→      {/* Needle */}
   320→      <motion.line
   321→        x1="40"
   322→        y1="42"
   323→        x2="40"
   324→        y2="12"
   325→        stroke={color}
   326→        strokeWidth={2}
   327→        strokeLinecap="round"
   328→        initial={{ rotate: -90, transformOrigin: "40px 42px" }}
   329→        animate={{ rotate: angle, transformOrigin: "40px 42px" }}

   331→      />
   332→      {/* Center dot */}
   333→      <circle cx="40" cy="42" r={3} fill={color} />
   334→    </svg>
   335→  );
