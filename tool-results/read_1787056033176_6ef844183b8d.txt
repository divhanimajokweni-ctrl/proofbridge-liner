   108→// ─── Component ──────────────────────────────────────────────────────────────
   109→
   110→export default function Home() {
   111→  const { toast } = useToast();
   112→  const [state, setState] = useState<SystemState | null>(null);
   113→  const [selectedId, setSelectedId] = useState<string | null>(null);
   114→  const [loading, setLoading] = useState(true);
   115→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   116→  const [showCreate, setShowCreate] = useState(false);
   117→
   118→  const refresh = useCallback(async () => {
   119→    try {
   120→      const res = await fetch("/api/state", { cache: "no-store" });
   121→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   122→      const data = (await res.json()) as SystemState;
   123→      setState(data);
   124→      if (!selectedId && data.claims.length > 0) {
   125→        setSelectedId(data.claims[0].id);
   126→      }
   127→    } catch (e) {
   128→      toast({
   129→        title: "Failed to load state",
   130→        description: e instanceof Error ? e.message : String(e),
   131→        variant: "destructive",
   132→      });
   133→    } finally {
   134→      setLoading(false);
   135→    }
   136→  }, [selectedId, toast]);
   137→
   138→  useEffect(() => {
   139→    refresh();
   140→  }, [refresh]);
   141→
   142→  const seed = async () => {
   143→    setActionLoading("seed");
   144→    try {
   145→      const res = await fetch("/api/seed", { method: "POST" });
   146→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   147→      const data = await res.json();
   148→      toast({
   149→        title: "Database seeded",
   150→        description: `${data.count} demonstration claims created with full VVU stack.`,
   151→      });
   152→      setSelectedId(null);
   153→      await refresh();
   154→    } catch (e) {
   155→      toast({
   156→        title: "Seed failed",
   157→        description: e instanceof Error ? e.message : String(e),
   158→        variant: "destructive",
   159→      });
   160→    } finally {
   161→      setActionLoading(null);
   162→    }
   163→  };
   164→
   165→  const ingestEvidence = async (claimId: string) => {
   166→    setActionLoading(`ingest-${claimId}`);
   167→    try {
   168→      const res = await fetch("/api/evidence", {
   169→        method: "POST",
   170→        headers: { "Content-Type": "application/json" },
   171→        body: JSON.stringify({ claimId, seed: Math.floor(Math.random() * 10000) }),
   172→      });
   173→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   174→      const data = await res.json();
   175→      toast({
   176→        title: "Evidence ingested",
   177→        description: `${Array.isArray(data.evidence) ? data.evidence.length : 1} item(s) from Evidence Mesh.`,
   178→      });
   179→      await refresh();
   180→    } catch (e) {
   181→      toast({
   182→        title: "Ingest failed",
   183→        description: e instanceof Error ? e.message : String(e),
   184→        variant: "destructive",
   185→      });
   186→    } finally {
   187→      setActionLoading(null);
   188→    }
   189→  };
   190→
   191→  const verify = async (claimId: string) => {
   192→    setActionLoading(`verify-${claimId}`);
   193→    try {
   194→      const res = await fetch("/api/verify", {
   195→        method: "POST",
   196→        headers: { "Content-Type": "application/json" },
   197→        body: JSON.stringify({ claimId }),
   198→      });
   199→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   200→      const data = await res.json();
   201→      toast({
   202→        title: "Verification complete",
   203→        description: `${data.trace.previousState} → ${data.trace.newState}`,
   204→      });
   205→      await refresh();
   206→    } catch (e) {
   207→      toast({
   208→        title: "Verify failed",
   209→        description: e instanceof Error ? e.message : String(e),
   210→        variant: "destructive",
   211→      });
   212→    } finally {
   213→      setActionLoading(null);
   214→    }
   215→  };
   216→
   217→  const recomputeNInd = async (claimId: string) => {
   218→    setActionLoading(`nind-${claimId}`);
   219→    try {
   220→      const res = await fetch("/api/n-ind", {
   221→        method: "POST",
   222→        headers: { "Content-Type": "application/json" },
   223→        body: JSON.stringify({ claimId }),
   224→      });
   225→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   226→      await refresh();
   227→      toast({
   228→        title: "N_ind recomputed",
   229→        description: "Participation ratio updated from current evidence set.",
   230→      });
   231→    } catch (e) {
   232→      toast({
   233→        title: "N_ind failed",
   234→        description: e instanceof Error ? e.message : String(e),
   235→        variant: "destructive",
   236→      });
   237→    } finally {
   238→      setActionLoading(null);
   239→    }
   240→  };
   241→
   242→  const authorize = async (
   243→    claimId: string,
   244→    safetyOverride: boolean,
   245→    reviewSignedOff: boolean
   246→  ) => {
   247→    setActionLoading(`auth-${claimId}`);
   248→    try {
   249→      const res = await fetch("/api/authorize", {
   250→        method: "POST",
   251→        headers: { "Content-Type": "application/json" },
   252→        body: JSON.stringify({ claimId, safetyOverride, reviewSignedOff }),
   253→      });
   254→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   255→      const data = await res.json();
   256→      toast({
   257→        title: data.authorized ? "Authorized ✓" : "Authorization blocked",
   258→        description: data.authorized
   259→          ? "A = C∧E∧I∧S∧R = true — action permitted."
   260→          : "A = false — fail-closed. See reason for details.",
   261→        variant: data.authorized ? "default" : "destructive",
   262→      });
   263→      await refresh();
   264→    } catch (e) {
   265→      toast({
   266→        title: "Authorize failed",
   267→        description: e instanceof Error ? e.message : String(e),
   268→        variant: "destructive",
   269→      });
   270→    } finally {
   271→      setActionLoading(null);
   272→    }
   273→  };
   274→
   275→  const createClaim = async (form: {
   276→    title: string;
   277→    description: string;
   278→    claimType: ClaimType;
   279→    intendedAction: string;
   280→    safetyCritical: boolean;
   281→  }) => {
   282→    setActionLoading("create");
   283→    try {
   284→      const res = await fetch("/api/claims", {
   285→        method: "POST",
   286→        headers: { "Content-Type": "application/json" },
   287→        body: JSON.stringify(form),
   288→      });
   289→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   290→      const data = await res.json();
   291→      toast({
   292→        title: "Claim created",
   293→        description: `New ${form.claimType} claim: ${form.title}`,
   294→      });
   295→      setShowCreate(false);
   296→      setSelectedId(data.claim.id);
   297→      await refresh();
   298→    } catch (e) {
   299→      toast({
   300→        title: "Create failed",
   301→        description: e instanceof Error ? e.message : String(e),
   302→        variant: "destructive",
   303→      });
   304→    } finally {
   305→      setActionLoading(null);
   306→    }
   307→  };
   308→
   309→  const selected = state?.claims.find((c) => c.id === selectedId) ?? null;
   310→  const latestAuth = selected?.authorizations[0] ?? null;
   311→  const latestNInd = selected?.nIndRecords[0] ?? null;
   312→  const nIndResult: ParticipationRatioResult | null = latestNInd
   313→    ? {
   314→        nInd: latestNInd.nInd,
   315→        numEvidence: latestNInd.numEvidence,
   316→        numSources: latestNInd.numSources,
   317→        gamma: latestNInd.gamma,
   318→        eigenvalues: latestNInd.eigenvalues,
   319→      }
   320→    : null;
   321→
   322→  return (
   323→    <div className="min-h-screen flex flex-col bg-background text-foreground">
   324→      <Toaster />
   325→
   326→      {/* ─── Header ─── */}
   327→      <header className="border-b bg-card/50 backdrop-blur supports-[backdrop-filter]:bg-card/30 sticky top-0 z-40">
   328→        <div className="mx-auto max-w-[1600px] px-4 py-3">
   329→          <div className="flex flex-wrap items-center justify-between gap-3">
   330→            <div className="flex items-center gap-3">
   331→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-foreground text-background">
   332→                <Layers className="h-5 w-5" />
   333→              </div>
   334→              <div>
   335→                <h1 className="text-lg font-bold tracking-tight leading-none">
   336→                  VVU — IVE
   337→                </h1>
   338→                <p className="text-[11px] text-muted-foreground font-mono mt-0.5">
   339→                  Integrated Verification Environment · backend: Evidence Independence Specification
   340→                </p>
   341→              </div>
   342→            </div>
   343→
   344→            <div className="flex items-center gap-2">
   345→              <Button
   346→                variant="outline"
   347→                size="sm"
   348→                onClick={seed}
   349→                disabled={actionLoading === "seed"}
   350→              >
   351→                {actionLoading === "seed" ? (
   352→                  <RefreshCw className="mr-1.5 h-3.5 w-3.5 animate-spin" />
   353→                ) : (
   354→                  <Sparkles className="mr-1.5 h-3.5 w-3.5" />
   355→                )}
   356→                Seed demo
   357→              </Button>
   358→              <Button
   359→                variant="outline"
   360→                size="sm"
   361→                onClick={() => setShowCreate(true)}
   362→              >
   363→                <Plus className="mr-1.5 h-3.5 w-3.5" />
   364→                New claim
   365→              </Button>
   366→              <Button variant="outline" size="sm" onClick={refresh}>
   367→                <RefreshCw className="mr-1.5 h-3.5 w-3.5" />
   368→                Refresh
   369→              </Button>
   370→            </div>
   371→          </div>
   372→
   373→          {/* Evidence bound banner */}
   374→          <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-[11px] font-mono text-muted-foreground">
   375→            <span className="flex items-center gap-1">
   376→              <GitBranch className="h-3 w-3" />
   377→              {state?.summary.evidenceBound ?? "Claim ≤ Evidence ≤ Verification ≤ Authorization ≤ Action"}
   378→            </span>
   379→            <span className="flex items-center gap-1">
   380→              <ShieldCheck className="h-3 w-3" />
   381→              {state?.summary.authorizationFormula ?? "A = C ∧ E ∧ I ∧ S ∧ R"}
   382→            </span>
   383→            <span className="flex items-center gap-1 text-emerald-600 dark:text-emerald-400">
   384→              <Activity className="h-3 w-3" />
   385→              fail-closed
   386→            </span>
   387→          </div>
   388→        </div>
   389→      </header>
   390→
   391→      {/* ─── Summary strip ─── */}
   392→      {state && (
   393→        <div className="border-b bg-muted/20">
   394→          <div className="mx-auto max-w-[1600px] px-4 py-3">
   395→            <div className="grid grid-cols-2 gap-2 sm:grid-cols-3 lg:grid-cols-6">
   396→              <SummaryCard
   397→                icon={<Database className="h-3.5 w-3.5" />}
   398→                label="Claims"
   399→                value={state.summary.totalClaims}
   400→              />
   401→              <SummaryCard
   402→                icon={<ShieldCheck className="h-3.5 w-3.5" />}
   403→                label="Authorized"
   404→                value={state.summary.authorizedClaims}
   405→                tone="emerald"
   406→              />
   407→              <SummaryCard
   408→                icon={<Activity className="h-3.5 w-3.5" />}
   409→                label="Breakers tripped"
   410→                value={state.summary.breakerTripped}
   411→                tone={state.summary.breakerTripped > 0 ? "red" : "emerald"}
   412→              />
   413→              <SummaryCard
   414→                icon={<Terminal className="h-3.5 w-3.5" />}
   415→                label="Evidence items"
   416→                value={state.summary.totalEvidence}
   417→              />
   418→              <SummaryCard
   419→                icon={<Layers className="h-3.5 w-3.5" />}
   420→                label="Claim types"
   421→                value={Object.keys(state.summary.stateCounts).length}
   422→              />
   423→              <SummaryCard
   424→                icon={<GitBranch className="h-3.5 w-3.5" />}
   425→                label="State values"
   426→                value={Object.keys(state.summary.stateCounts).length}
   427→              />
   428→            </div>
   429→          </div>
   430→        </div>
   431→      )}
   432→
   433→      {/* ─── Main ─── */}
   434→      <main className="flex-1 mx-auto max-w-[1600px] w-full px-4 py-4">
   435→        {loading ? (
   436→          <div className="flex flex-col items-center justify-center py-24 gap-3">
   437→            <RefreshCw className="h-8 w-8 animate-spin text-muted-foreground" />
   438→            <p className="text-sm text-muted-foreground font-mono">
   439→              Loading IVE state from EIS backend…
   440→            </p>
   441→          </div>
   442→        ) : !state || state.claims.length === 0 ? (
   443→          <EmptyState onSeed={seed} loading={actionLoading === "seed"} />
   444→        ) : (
   445→          <div className="grid gap-4 lg:grid-cols-[320px_1fr]">
   446→            {/* Claim list */}
   447→            <aside className="space-y-2">
   448→              <div className="flex items-center justify-between px-1 mb-1">
   449→                <h2 className="text-xs font-mono font-semibold uppercase tracking-wider text-muted-foreground">
   450→                  Claims ({state.claims.length})
   451→                </h2>
   452→              </div>
   453→              <div className="max-h-[calc(100vh-280px)] overflow-y-auto pr-1 space-y-1.5">
   454→                {state.claims.map((c) => {
   455→                  const auth = c.authorizations[0];
   456→                  return (
   457→                    <button
   458→                      key={c.id}
   459→                      onClick={() => setSelectedId(c.id)}
   460→                      className={`w-full text-left rounded-md border p-2.5 transition-colors ${
   461→                        selectedId === c.id
   462→                          ? "border-foreground bg-accent"
   463→                          : "border-border bg-card hover:bg-accent/50"
   464→                      }`}
   465→                    >
   466→                      <div className="flex items-start justify-between gap-2 mb-1">
   467→                        <span className="text-xs font-semibold leading-tight line-clamp-2 flex-1">
   468→                          {c.title}
   469→                        </span>
   470→                        <StateBadge state={c.state} size="sm" />
   471→                      </div>
   472→                      <div className="flex items-center justify-between text-[10px] font-mono text-muted-foreground">
   473→                        <span className="uppercase tracking-wider">{c.claimType}</span>
   474→                        <span className="flex items-center gap-1.5">
   475→                          <span>{c.evidence.length} ev</span>
   476→                          {auth && (
   477→                            <span
   478→                              className={
   479→                                auth.authorized
   480→                                  ? "text-emerald-600 dark:text-emerald-400"
   481→                                  : "text-red-600 dark:text-red-400"
   482→                              }
   483→                            >
   484→                              A={auth.authorized ? "✓" : "✗"}
   485→                            </span>
   486→                          )}
   487→                        </span>
   488→                      </div>
   489→                    </button>
   490→                  );
   491→                })}
   492→              </div>
   493→            </aside>
   494→
   495→            {/* Selected claim detail */}
   496→            {selected && (
   497→              <div className="space-y-4">
   498→                {/* Claim header card */}
   499→                <Card>
   500→                  <CardHeader className="pb-3">
   501→                    <div className="flex items-start justify-between gap-3">
   502→                      <div className="flex-1 min-w-0">
   503→                        <div className="flex items-center gap-2 mb-1">
   504→                          <span className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   505→                            {selected.claimType}
   506→                          </span>
   507→                          {selected.safetyCritical && (
   508→                            <span className="text-[10px] font-mono uppercase tracking-wider text-amber-700 dark:text-amber-400 bg-amber-500/10 border border-amber-500/30 px-1.5 py-0.5 rounded">
   509→                              safety-critical
   510→                            </span>
   511→                          )}
   512→                        </div>
   513→                        <CardTitle className="text-base leading-tight">
   514→                          {selected.title}
   515→                        </CardTitle>
   516→                      </div>
   517→                      <StateBadge state={selected.state} size="lg" />
   518→                    </div>
   519→                  </CardHeader>
   520→                  <CardContent className="pt-0">
   521→                    <p className="text-xs text-muted-foreground leading-relaxed mb-3">
   522→                      {selected.description}
   523→                    </p>
   524→                    <div className="flex flex-wrap items-center gap-2">
   525→                      <Button
   526→                        size="sm"
   527→                        variant="outline"
   528→                        onClick={() => ingestEvidence(selected.id)}
   529→                        disabled={actionLoading?.startsWith("ingest-")}
   530→                      >
   531→                        <Database className="mr-1.5 h-3.5 w-3.5" />
   532→                        Ingest evidence
   533→                      </Button>
   534→                      <Button
   535→                        size="sm"
   536→                        variant="outline"
   537→                        onClick={() => verify(selected.id)}
   538→                        disabled={actionLoading?.startsWith("verify-")}
   539→                      >
   540→                        <GitBranch className="mr-1.5 h-3.5 w-3.5" />
   541→                        Verify (IVE)
   542→                      </Button>
   543→                      <Button
   544→                        size="sm"
   545→                        variant="outline"
   546→                        onClick={() => recomputeNInd(selected.id)}
   547→                        disabled={actionLoading?.startsWith("nind-")}
   548→                      >
   549→                        <RefreshCw className="mr-1.5 h-3.5 w-3.5" />
   550→                        N_ind
   551→                      </Button>
   552→                      <Button
   553→                        size="sm"
   554→                        onClick={() => authorize(selected.id, false, false)}
   555→                        disabled={actionLoading?.startsWith("auth-")}
   556→                      >
   557→                        <ShieldCheck className="mr-1.5 h-3.5 w-3.5" />
   558→                        Authorize
   559→                      </Button>
   560→                    </div>
   561→                  </CardContent>
   562→                </Card>
   563→
   564→                {/* Tabs for detailed panels */}
   565→                <Tabs defaultValue="pipeline" className="w-full">
   566→                  <TabsList className="grid w-full grid-cols-2 sm:grid-cols-4 h-auto">
   567→                    <TabsTrigger value="pipeline" className="text-xs">
   568→                      Pipeline
   569→                    </TabsTrigger>
   570→                    <TabsTrigger value="lattice" className="text-xs">
   571→                      State Lattice
   572→                    </TabsTrigger>
   573→                    <TabsTrigger value="diffusion" className="text-xs">
   574→                      Diffusion

   576→                    <TabsTrigger value="raw" className="text-xs">
   577→                      Raw State
   578→                    </TabsTrigger>
   579→                  </TabsList>

   581→                  <TabsContent value="pipeline" className="mt-4 space-y-4">

   586→                        loading={actionLoading?.startsWith("ingest-")}

   592→                        loading={actionLoading?.startsWith("auth-")}
