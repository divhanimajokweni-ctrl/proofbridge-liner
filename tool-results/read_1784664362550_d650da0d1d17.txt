    82→function ThemeToggleBtn() {
    83→  const { resolvedTheme, setTheme } = useTheme();
    84→  const [mounted, setMounted] = useState(false);
    85→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    86→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    87→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
    88→  return (
    89→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
    90→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
    91→    </button>
    92→  );
    93→}

    95→function ScrollToTopButton() {
    96→  const [visible, setVisible] = useState(false);
    97→
    98→  useEffect(() => {
    99→    const onScroll = () => setVisible(window.scrollY > 300);
   100→    window.addEventListener("scroll", onScroll, { passive: true });
   101→    return () => window.removeEventListener("scroll", onScroll);
   102→  }, []);
   103→
   104→  const scrollToTop = useCallback(() => {
   105→    window.scrollTo({ top: 0, behavior: "smooth" });
   106→  }, []);
   107→
   108→  return (
   109→    <AnimatePresence>
   110→      {visible && (
   111→        <motion.button
   112→          type="button"
   113→          onClick={scrollToTop}
   114→          initial={{ opacity: 0, scale: 0.8 }}
   115→          animate={{ opacity: 1, scale: 1 }}
   116→          exit={{ opacity: 0, scale: 0.8 }}
   117→          transition={{ duration: 0.2 }}
   118→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   119→          title="Scroll to top"
   120→        >
   121→          <ArrowUp className="h-4 w-4" />
   122→        </motion.button>

   124→    </AnimatePresence>

   428→/* ─── Quick Stats Bar ─── */
   429→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   430→  const meta = SECTION_META[sectionId];
   431→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   432→  const [isRefreshing, setIsRefreshing] = useState(false);
   433→
   434→  // Auto-refresh countdown (10s cycle, matches health fetch)
   435→  useEffect(() => {
   436→    const t = setInterval(() => {
   437→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   438→    }, 1000);
   439→    return () => clearInterval(t);
   440→  }, []);
   441→
   442→  const handleRefresh = useCallback(() => {
   443→    setIsRefreshing(true);
   444→    fetch("/api/stats").then(() => {
   445→      setIsRefreshing(false);
   446→    }).catch(() => {
   447→      setIsRefreshing(false);
   448→    });
   449→  }, []);
   450→
   451→  return (
   452→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   453→      {meta.stats.map((stat, i) => (
   454→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   455→          {stat}
   456→        </span>
   457→      ))}
   458→      <div className="ml-auto flex items-center gap-2">
   459→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   460→          <Timer className="h-2.5 w-2.5" />
   461→          {refreshCountdown}s
   462→        </span>
   463→        <button
   464→          type="button"
   465→          onClick={handleRefresh}
   466→          disabled={isRefreshing}
   467→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   468→          title="Refresh section data"
   469→        >
   470→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   471→          refresh
   472→        </button>
   473→      </div>
   474→    </div>
   475→  );
   476→}
