   236→function fmtRelativeTime(iso: string): string {
   237→  const diff = Date.now() - new Date(iso).getTime();
   238→  if (diff < 0) return "just now";
   239→  const m = Math.floor(diff / 60000);
   240→  if (m < 60) return `${m}m ago`;
   241→  const h = Math.floor(m / 60);
   242→  if (h < 24) return `${h}h ago`;
   243→  const d = Math.floor(h / 24);
   244→  return `${d}d ago`;
   245→}

   417→/* ─── Radar chart normalized dimensions ─── */
   418→function buildRadarData(policies: ComparisonPolicy[]) {
   419→  if (policies.length === 0) return [];
   420→  const maxHealth = 100;
   421→  const maxMerge = 100;
   422→  const maxInvariant = Math.max(...ALL_POLICIES.map((p) => p.invariantCount));
   423→  const maxZk = Math.max(...ALL_POLICIES.map((p) => p.zkProofs));
   424→
   425→  const axes = [
   426→    { axis: "Health", key: "healthScore" as const, max: maxHealth },
   427→    { axis: "Merge Rate", key: "mergeSuccessRate" as const, max: maxMerge },
   428→    { axis: "Invariant Cov.", key: "invariantCount" as const, max: maxInvariant },
   429→    { axis: "ZK Coverage", key: "zkProofs" as const, max: maxZk },
   430→    { axis: "Violation-Free", key: "violationFree" as const, max: 100 },
   431→  ];
   432→
   433→  return axes.map(({ axis, key, max }) => {
   434→    const entry: Record<string, string | number> = { axis };
   435→    for (const p of policies) {
   436→      let raw: number;
   437→      if (key === "violationFree") {
   438→        raw = Math.max(0, 100 - p.activeViolations * 15);
   439→      } else {
   440→        raw = p[key] as number;
   441→      }
   442→      entry[p.name] = Math.round((raw / max) * 100);
   443→    }
   444→    return entry;
   445→  });
   446→}

   455→/* ─── Difference summary generator ─── */
   456→function generateDiffSummary(policies: ComparisonPolicy[]): string[] {
   457→  if (policies.length < 2) return [];
   458→  const summaries: string[] = [];
   459→
   460→  const best = policies.reduce((a, b) => (a.healthScore > b.healthScore ? a : b));
   461→  const worst = policies.reduce((a, b) => (a.healthScore < b.healthScore ? a : b));
   462→  if (best.id !== worst.id) {
   463→    summaries.push(
   464→      `${best.name} leads in health score at ${best.healthScore}%, while ${worst.name} trails at ${worst.healthScore}% — a ${best.healthScore - worst.healthScore} point gap.`
   465→    );
   466→  }
   467→
   468→  const strictPolicies = policies.filter((p) => p.enforcementMode === "strict");
   469→  const permissivePolicies = policies.filter((p) => p.enforcementMode === "permissive");
   470→  if (strictPolicies.length > 0 && permissivePolicies.length > 0) {
   471→    summaries.push(
   472→      `Enforcement modes differ: ${strictPolicies.map((p) => p.name).join(", ")} run strict, while ${permissivePolicies.map((p) => p.name).join(", ")} are permissive.`
   473→    );
   474→  }
   475→
   476→  const shadowEnabled = policies.filter((p) => p.shadowEnabled);
   477→  const shadowDisabled = policies.filter((p) => !p.shadowEnabled);
   478→  if (shadowEnabled.length > 0 && shadowDisabled.length > 0) {
   479→    summaries.push(
   480→      `${shadowEnabled.length} of ${policies.length} policies have shadow bridges enabled (${shadowEnabled.map((p) => p.name).join(", ")}), while ${shadowDisabled.map((p) => p.name).join(", ")} do not.`
   481→    );
   482→  }
   483→
   484→  const highViolation = policies.filter((p) => p.activeViolations >= 3);
   485→  if (highViolation.length > 0) {
   486→    summaries.push(
   487→      `${highViolation.map((p) => p.name).join(", ")} ${highViolation.length === 1 ? "has" : "have"} elevated violation counts (≥3), suggesting active drift that may require repair intervention.`
   488→    );
   489→  }
   490→
   491→  const bestMerge = policies.reduce((a, b) => (a.mergeSuccessRate > b.mergeSuccessRate ? a : b));
   492→  const worstMerge = policies.reduce((a, b) => (a.mergeSuccessRate < b.mergeSuccessRate ? a : b));
   493→  if (bestMerge.id !== worstMerge.id) {
   494→    summaries.push(
   495→      `Merge success rates span from ${worstMerge.mergeSuccessRate.toFixed(1)}% (${worstMerge.name}) to ${bestMerge.mergeSuccessRate.toFixed(1)}% (${bestMerge.name}).`
   496→    );
   497→  }
   498→
   499→  return summaries;
   500→}
