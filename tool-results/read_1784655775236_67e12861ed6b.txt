    22→/* ─── Helpers ─── */
    23→function fmtTime(iso: string): string {
    24→  const d = Date.now() - new Date(iso).getTime();
    25→  if (d < 60000) return `${Math.floor(d / 1000)}s ago`;
    26→  if (d < 3600000) return `${Math.floor(d / 60000)}m ago`;
    27→  if (d < 86400000) return `${Math.floor(d / 3600000)}h ago`;
    28→  return `${Math.floor(d / 86400000)}d ago`;
    29→}
    30→
    31→function fmtPct(v: number): string {
    32→  return `${(v * 100).toFixed(1)}%`;
    33→}

   263→/* ─── Posterior Distribution Curve (SVG) ─── */
   264→function PosteriorCurve({ data }: { data: Array<{ x: number; y: number }> }) {
   265→  if (data.length < 2) return null;
   266→  const width = 280, height = 60, pad = 4;
   267→  const maxY = Math.max(...data.map(d => d.y), 0.001);
   268→  const pts = data.map(d => ({
   269→    x: pad + d.x * (width - pad * 2),
   270→    y: pad + (1 - d.y / maxY) * (height - pad * 2),
   271→  }));
   272→  const line = pts.map((p, i) => `${i ? "L" : "M"}${p.x.toFixed(1)},${p.y.toFixed(1)}`).join(" ");
   273→  const area = `${line} L${pts.at(-1)!.x.toFixed(1)},${height - pad} L${pts[0].x.toFixed(1)},${height - pad} Z`;
   274→
   275→  return (
   276→    <div className="space-y-1">
   277→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} className="overflow-visible" role="img" aria-label="Posterior distribution curve">
   278→        <path d={area} fill="oklch(0.78 0.16 160 / 0.1)" />
   279→        <path d={line} fill="none" stroke="oklch(0.78 0.16 160 / 0.8)" strokeWidth={1.5} strokeLinecap="round" />
   280→        {/* Peak marker */}
   281→        {pts.length > 0 && (() => {
   282→          const peak = pts.reduce((a, b) => b.y < a.y ? b : a);
   283→          return <circle cx={peak.x} cy={peak.y} r={3} fill="oklch(0.78 0.16 160)" />;
   284→        })()}
   285→      </svg>
   286→      <div className="flex justify-between text-[9px] text-muted-foreground font-mono">
   287→        <span>0.0</span>
   288→        <span>Posterior Distribution</span>
   289→        <span>1.0</span>
   290→      </div>
   291→    </div>
   292→  );
   293→}

   375→/* ─── Main Section Component ─── */
   376→export function TrustRuntimeSection() {
   377→  const [state, setState] = useState<TrustRuntimeState | null>(null);
   378→  const [loading, setLoading] = useState(true);
   379→  const [error, setError] = useState<string | null>(null);
   380→  const [autoRefresh, setAutoRefresh] = useState(true);
   381→  const [countdown, setCountdown] = useState(10);
   382→  const [showDetails, setShowDetails] = useState(false);
   383→
   384→  const fetchData = useCallback(async () => {
   385→    try {
   386→      const res = await fetch("/api/trust-runtime");
   387→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   388→      const data: TrustRuntimeState = await res.json();
   389→      setState(data);
   390→      setError(null);
   391→    } catch (err) {
   392→      setError(err instanceof Error ? err.message : "Failed to load");
   393→    } finally {
   394→      setLoading(false);
   395→    }
   396→  }, []);
   397→
   398→  // Initial fetch + auto-refresh
   399→  useEffect(() => {
   400→    fetchData();
   401→  }, [fetchData]);
   402→
   403→  useEffect(() => {
   404→    if (!autoRefresh) return;
   405→    const t = setInterval(() => {
   406→      setCountdown(c => (c <= 1 ? 10 : c - 1));
   407→      if (countdown <= 1) fetchData();
   408→    }, 1000);
   409→    return () => clearInterval(t);
   410→  }, [autoRefresh, countdown, fetchData]);
   411→
   412→  if (loading) {
   413→    return (
   414→      <div className="flex flex-col items-center justify-center py-20 gap-4">
   415→        <div className="relative flex items-center justify-center">
   416→          <div className="h-10 w-10 rounded-lg bg-verified/10 border border-verified/20 flex items-center justify-center">
   417→            <Shield className="h-5 w-5 text-verified/60" />
   418→          </div>
   419→          <div className="absolute inset-0 rounded-lg animate-ping bg-verified/10 opacity-30" />
   420→        </div>
   421→        <span className="text-sm text-muted-foreground">Loading Trust Runtime…</span>
   422→      </div>
   423→    );
   424→  }
   425→
   426→  if (error || !state) {
