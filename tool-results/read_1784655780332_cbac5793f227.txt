    27→    27→  if (d < 86400000) return `${Math.floor(d / 3600000)}h ago`;
    28→    28→  return `${Math.floor(d / 86400000)}d ago`;
    29→    29→}

   263→   263→/* ─── Posterior Distribution Curve (SVG) ─── */
   264→   264→function PosteriorCurve({ data }: { data: Array<{ x: number; y: number }> }) {
   265→   265→  if (data.length < 2) return null;
   266→   266→  const width = 280, height = 60, pad = 4;
   267→   267→  const maxY = Math.max(...data.map(d => d.y), 0.001);
   268→   268→  const pts = data.map(d => ({
   269→   269→    x: pad + d.x * (width - pad * 2),
   270→   270→    y: pad + (1 - d.y / maxY) * (height - pad * 2),
   271→   271→  }));
   272→   272→  const line = pts.map((p, i) => `${i ? "L" : "M"}${p.x.toFixed(1)},${p.y.toFixed(1)}`).join(" ");
   273→   273→  const area = `${line} L${pts.at(-1)!.x.toFixed(1)},${height - pad} L${pts[0].x.toFixed(1)},${height - pad} Z`;
   274→   274→
   275→   275→  return (
   276→   276→    <div className="space-y-1">
   277→   277→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} className="overflow-visible" role="img" aria-label="Posterior distribution curve">
   278→   278→        <path d={area} fill="oklch(0.78 0.16 160 / 0.1)" />
   279→   279→        <path d={line} fill="none" stroke="oklch(0.78 0.16 160 / 0.8)" strokeWidth={1.5} strokeLinecap="round" />
   280→   280→        {/* Peak marker */}
   281→   281→        {pts.length > 0 && (() => {
   282→   282→          const peak = pts.reduce((a, b) => b.y < a.y ? b : a);
   283→   283→          return <circle cx={peak.x} cy={peak.y} r={3} fill="oklch(0.78 0.16 160)" />;
   284→   284→        })()}
   285→   285→      </svg>
   286→   286→      <div className="flex justify-between text-[9px] text-muted-foreground font-mono">
   287→   287→        <span>0.0</span>
   288→   288→        <span>Posterior Distribution</span>
   289→   289→        <span>1.0</span>
   290→   290→      </div>
   291→   291→    </div>
   292→   292→  );
   293→   293→}
