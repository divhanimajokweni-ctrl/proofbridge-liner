    93→/**
    94→ * Render ANSI-colored text (SGR codes only) as a sequence of styled spans.
    95→ */
    96→function renderAnsi(text: string): React.ReactNode[] {
    97→  if (!text) return [];
    98→  const parts = text.split(/\x1b\[(\d+(?:;\d+)*)m/);
    99→  const nodes: React.ReactNode[] = [];
   100→  let active = new Set<string>();
   101→  let key = 0;
   102→  for (let i = 0; i < parts.length; i++) {
   103→    const part = parts[i];
   104→    if (i % 2 === 1) {
   105→      for (const code of part.split(";")) {
   106→        if (code === "0" || code === "") {
   107→          active = new Set();
   108→        } else {
   109→          const cls = ANSI_CLASS[code];
   110→          if (cls) active.add(cls);
   111→        }
   112→      }
   113→    } else if (part) {
   114→      const cls = Array.from(active).join(" ");
   115→      nodes.push(
   116→        <span key={key++} className={cls || undefined}>
   117→          {part}
   118→        </span>,
   119→      );
   120→    }
   121→  }
   122→  return nodes;
   123→}

   125→/** Syntax highlight .epd keywords for the source editor */
   126→function highlightEpd(text: string): React.ReactNode[] {
   127→  const keywords = [
   128→    "policy", "invariant", "soft", "shard", "by", "expect", "merge",
   129→    "on_violation", "ancestry", "shadowbridge", "shadow_bridge", "export",
   130→    "to", "key", "strategy", "count", "replication", "preserves",
   131→    "locality_preserving", "max_divergence", "proof", "zk", "gossip",
   132→    "anchor", "enabled", "takeover_latency_ms", "whatif_branching",
   133→    "replay", "authoritative", "domain", "version", "description",
   134→    "severity", "predicate", "objective", "max_iters", "notify",
   135→  ];
   136→  const kwSet = new Set(keywords);
   137→  const lines = text.split("\n");
   138→  const nodes: React.ReactNode[] = [];
   139→  let globalKey = 0;
   140→
   141→  for (let li = 0; li < lines.length; li++) {
   142→    const line = lines[li];
   143→    const tokens = line.split(/(\s+|[{}()"[\],;]|\/\/.*$)/);
   144→    for (const token of tokens) {
   145→      if (kwSet.has(token)) {
   146→        nodes.push(
   147→          <span key={globalKey++} className="text-emerald-400 font-medium">{token}</span>
   148→        );
   149→      } else if (/^".*"$/.test(token)) {
   150→        nodes.push(
   151→          <span key={globalKey++} className="text-amber-300">{token}</span>
   152→        );
   153→      } else if (/^\d+(\.\d+)?$/.test(token)) {
   154→        nodes.push(
   155→          <span key={globalKey++} className="text-teal-300">{token}</span>
   156→        );
   157→      } else if (/^\/\/.*$/.test(token)) {
   158→        nodes.push(
   159→          <span key={globalKey++} className="text-zinc-500 italic">{token}</span>
   160→        );
   161→      } else {
   162→        nodes.push(
   163→          <span key={globalKey++}>{token}</span>
   164→        );
   165→      }
   166→    }
   167→    if (li < lines.length - 1) nodes.push(<span key={globalKey++}>{"\n"}</span>);
   168→  }
   169→  return nodes;
   170→}

   237→/** Animated terminal output with typing effect */
   238→function AnimatedOutput({ text, delay = 0 }: { text: string; delay?: number }) {
   239→  const [displayed, setDisplayed] = useState("");
   240→  const [done, setDone] = useState(false);
   241→
   242→  useEffect(() => {
   243→    let timeout: NodeJS.Timeout;
   244→    let i = 0;
   245→    const start = () => {
   246→      const tick = () => {
   247→        if (i < text.length) {
   248→          const chunk = text.length > 200 ? 4 : 1;
   249→          setDisplayed(text.slice(0, i + chunk));
   250→          i += chunk;
   251→          timeout = setTimeout(tick, 8);
   252→        } else {
   253→          setDone(true);
   254→        }
   255→      };
   256→      tick();
   257→    };
   258→    timeout = setTimeout(start, delay);
   259→    return () => clearTimeout(timeout);
   260→  }, [text, delay]);
   261→
   262→  return (
   263→    <span>
   264→      {renderAnsi(displayed)}
   265→      {!done && <BlinkingCursor />}
   266→    </span>
   267→  );
   268→}

   832→function TerminalEntryView({ entry, isNew, isHighlighted }: { entry: TerminalEntry; isNew: boolean; isHighlighted: boolean }) {
   833→  const ok = entry.exitCode === 0;
   834→  const failed = entry.exitCode === -1;
   835→  const [hovered, setHovered] = useState(false);
   836→
   837→  return (
   838→    <motion.div
