   113→function layoutTreemap(domains: DomainInfo[], W: number, H: number): TreemapRect[] {
   114→  const total = domains.reduce((n, d) => n + d.invariantCount, 0) || 1;
   115→  const rects: TreemapRect[] = [];
   116→  let x = 0;
   117→  let y = 0;
   118→  let remainingW = W;
   119→  let remainingH = H;
   120→  let rowHorizontal = true;
   121→
   122→  const sorted = [...domains].sort((a, b) => b.invariantCount - a.invariantCount);
   123→
   124→  for (const d of sorted) {
   125→    const frac = d.invariantCount / total;
   126→    if (rowHorizontal) {
   127→      const w = Math.max(60, frac * remainingW);
   128→      rects.push({ x, y, w, h: remainingH, domain: d });
   129→      x += w;
   130→      remainingW -= w;
   131→    } else {
   132→      const h = Math.max(50, frac * remainingH);
   133→      rects.push({ x, y, w: remainingW, h, domain: d });
   134→      y += h;
   135→      remainingH -= h;
   136→    }
   137→    if (remainingW > 0 && remainingH > 0) {
   138→      const aspect = remainingW / remainingH;
   139→      if ((rowHorizontal && aspect < 1) || (!rowHorizontal && aspect > 1)) {
   140→        rowHorizontal = !rowHorizontal;
   141→      }
   142→    }
   143→  }
   144→  return rects;
   145→}

   147→function buildTreemapData(domains: DomainInfo[], severityFilter: string | null) {
   148→  return domains.map((d, i) => {
   149→    const filteredInvariants = severityFilter
   150→      ? d.invariants.filter((inv) => inv.severity === severityFilter)
   151→      : d.invariants;
   152→    return {
   153→      name: d.domain,
   154→      size: Math.max(1, filteredInvariants.length),
   155→      domain: d,
   156→      colorIdx: i,
   157→    };
   158→  }).filter((item) => item.size > 0);
   159→}

   279→        });
   280→      })()}
   281→    </g>
   282→  );

   560→                      {r.w > 100 && r.h > 90 && (
   561→                        <text
   562→                          x={8}
   563→                          y={52}
   564→                          fontSize={8}
   565→                          fill="oklch(0.60 0.01 165)"
   566→                          fontFamily="var(--font-geist-mono), monospace"
   567→                          style={{ pointerEvents: "none" }}
   568→                        >
   569→                          {dom.policyCount} polic{dom.policyCount === 1 ? "y" : "ies"} · {dom.hardCount} hard · {dom.softCount} soft
   570→                        </text>
   571→                      )}
   572→                      {/* Hover tooltip */}
   573→                      {isHovered && !isSelected && (
   574→                        <g>
   575→                          <rect
   576→                            x={r.w / 2 - 60}
   577→                            y={r.h / 2 - 24}
   578→                            width={120}
   579→                            height={48}
   580→                            rx={4}
   581→                            fill="oklch(0.15 0.02 165)"
   582→                            fillOpacity={0.92}
   583→                            stroke="oklch(0.40 0.02 165)"
   584→                            strokeWidth={0.5}
   585→                          />
   586→                          <text x={r.w / 2} y={r.h / 2 - 8} textAnchor="middle" fontSize={8} fontWeight={600} fill="oklch(0.90 0.01 165)" fontFamily="monospace">
   587→                            {dom.domain.length > 18 ? dom.domain.slice(0, 16) + "…" : dom.domain}
   588→                          </text>
   589→                          <text x={r.w / 2} y={r.h / 2 + 4} textAnchor="middle" fontSize={7} fill="oklch(0.60 0.01 165)">
   590→                            {dom.invariantCount} invariants · {coveragePct}% hard
   591→                          </text>
   592→                          <text x={r.w / 2} y={r.h / 2 + 14} textAnchor="middle" fontSize={7} fill="oklch(0.60 0.01 165)">
   593→                            {dom.criticalCount} critical · {dom.highCount} high
   594→                          </text>
   595→                        </g>
   596→                      )}
   597→                    </g>
   598→                  );
   599→                })}
   600→              </svg>
   601→
   602→              {/* Severity legend */}
   603→              <div className="flex items-center gap-3 mt-2 text-[10px] text-muted-foreground">
   604→                <span className="flex items-center gap-1"><span className="h-2 w-3 rounded-sm" style={{ backgroundColor: "oklch(0.64 0.21 25)" }} /> critical</span>
   605→                <span className="flex items-center gap-1"><span className="h-2 w-3 rounded-sm" style={{ backgroundColor: "oklch(0.80 0.15 80)" }} /> high</span>
   606→                <span className="flex items-center gap-1"><span className="h-2 w-3 rounded-sm" style={{ backgroundColor: "oklch(0.70 0.13 40)" }} /> medium</span>
   607→                <span className="flex items-center gap-1"><span className="h-2 w-3 rounded-sm" style={{ backgroundColor: "oklch(0.55 0.01 165)" }} /> low</span>
   608→                <span className="ml-auto italic">bar = severity mix</span>
   609→              </div>
   610→            </div>
   611→          </Card>
   612→        </GradientBorderCard>
   613→
   614→        {/* Side panel */}
   615→        <div className="space-y-3">
   616→          <AnimatePresence mode="wait">
   617→            {selected ? (
   618→              <motion.div
   619→                key="selected"
   620→                initial={{ opacity: 0, y: 10 }}
   621→                animate={{ opacity: 1, y: 0 }}
   622→                exit={{ opacity: 0, y: -10 }}
   623→              >
   624→                <GradientBorderCard accentBorder="bg-gradient-to-r from-verified/30 via-verified/10 to-transparent">
   625→                  <Card className="bg-card/60 backdrop-blur border-0 p-3 relative overflow-hidden">
   626→                    <div className="bg-grid-fine absolute inset-0 opacity-20" />
   627→                    <div className="relative">
   628→                      <div className="flex items-center gap-2 mb-2">
   629→                        <Layers className="h-3.5 w-3.5 text-verified" />
   630→                        <span className="text-xs font-semibold truncate">{selected.domain}</span>
   631→                        <button onClick={() => setSelectedDomain(null)} className="ml-auto text-muted-foreground hover:text-foreground">
   632→                          <X className="h-3 w-3" />
   633→                        </button>
   634→                      </div>
   635→                      <div className="space-y-1 text-[11px] mb-2">
   636→                        <Row label="policies" value={String(selected.policyCount)} />
   637→                        <Row label="invariants" value={String(selected.invariantCount)} />
   638→                        <Row label="hard / soft" value={`${selected.hardCount} / ${selected.softCount}`} />
   639→                        <Row label="ZK / shadow" value={`${selected.zkCount} / ${selected.shadowCount}`} />
   640→                      </div>
   641→                      {/* Coverage bar */}
   642→                      <div className="mb-2">
   643→                        <div className="flex items-center justify-between text-[9px] mb-0.5">
   644→                          <span className="text-muted-foreground">Hard coverage</span>
   645→                          <span className="font-mono text-verified">{Math.round((selected.hardCount / Math.max(1, selected.invariantCount)) * 100)}%</span>
   646→                        </div>
   647→                        <div className="h-1.5 w-full rounded-full bg-muted/40 overflow-hidden">
   648→                          <motion.div
   649→                            className="h-full rounded-full bg-verified"
   650→                            initial={{ width: 0 }}
   651→                            animate={{ width: `${(selected.hardCount / Math.max(1, selected.invariantCount)) * 100}%` }}
   652→                            transition={{ duration: 0.6, ease: "easeOut" }}
   653→                          />
   654→                        </div>
   655→                      </div>
   656→                      <div className="text-[9px] uppercase tracking-wide text-muted-foreground mb-1">
   657→                        Policies in this domain
   658→                      </div>
   659→                      <div className="space-y-1 max-h-[180px] overflow-y-auto">
   660→                        {selected.policies.map((p) => (
   661→                          <div key={p.id} className="rounded border border-border/40 bg-background/40 px-2 py-1">
   662→                            <div className="flex items-center gap-1.5">
   663→                              <span className="font-mono text-[11px] font-medium truncate flex-1">{p.name}</span>
   664→                              <span className="text-[10px] text-muted-foreground font-mono">{p.invariantCount}</span>
   665→                            </div>
   666→                            <div className="flex items-center gap-1 mt-0.5">
   667→                              {p.zkEnabled && <Lock className="h-2.5 w-2.5 text-verified" />}
   668→                              {p.shadowEnabled && <Cpu className="h-2.5 w-2.5 text-repairing" />}
   669→                              <div className="flex gap-0.5 ml-auto">
   670→                                {p.invariants.map((inv, j) => (
   671→                                  <span
   672→                                    key={j}
   673→                                    className="h-2 w-2 rounded-sm"
   674→                                    style={{ backgroundColor: SEVERITY_ACCENT[inv.severity] ?? "oklch(0.5 0 0)" }}
   675→                                    title={`${inv.name}: ${inv.severity}`}
   676→                                  />
   677→                                ))}
   678→                              </div>
   679→                            </div>
   680→                          </div>
   681→                        ))}
   682→                      </div>
   683→                    </div>
   684→                  </Card>
   685→                </GradientBorderCard>
   686→              </motion.div>
   687→            ) : hovered ? (
   688→              <motion.div
   689→                key="hovered"
   690→                initial={{ opacity: 0, y: 10 }}
   691→                animate={{ opacity: 1, y: 0 }}
   692→                exit={{ opacity: 0, y: -10 }}
   693→              >
   694→                <GradientBorderCard accentBorder="bg-gradient-to-r from-repairing/30 via-repairing/10 to-transparent">
   695→                  <Card className="bg-card/60 backdrop-blur border-0 p-3 relative overflow-hidden">
   696→                    <div className="bg-grid-fine absolute inset-0 opacity-20" />
   697→                    <div className="relative">
   698→                      <div className="flex items-center gap-2 mb-2">
   699→                        <Layers className="h-3.5 w-3.5 text-repairing" />
   700→                        <span className="text-xs font-semibold truncate">{hovered.domain}</span>
   701→                      </div>
   702→                      <div className="space-y-1 text-[11px]">
   703→                        <Row label="policies" value={String(hovered.policyCount)} />
   704→                        <Row label="invariants" value={String(hovered.invariantCount)} />
   705→                        <Row label="critical" value={String(hovered.criticalCount)} />
   706→                        <Row label="high" value={String(hovered.highCount)} />
   707→                        <Row label="medium" value={String(hovered.mediumCount)} />
   708→                        <Row label="low" value={String(hovered.lowCount)} />
   709→                      </div>
   710→                      <p className="text-[10px] text-muted-foreground mt-2 italic">click to inspect policies</p>
   711→                    </div>
   712→                  </Card>
   713→                </GradientBorderCard>
   714→              </motion.div>
   715→            ) : (
   716→              <motion.div key="empty" variants={fadeUp}>
   717→                <Card className="bg-card/60 backdrop-blur border-dashed border-border/60 p-4 text-center">
   718→                  <LayoutGrid className="mx-auto h-6 w-6 text-muted-foreground/30" />
   719→                  <p className="mt-1 text-[11px] text-muted-foreground">Hover or click a tile</p>
   720→                </Card>
   721→              </motion.div>
   722→            )}
   723→          </AnimatePresence>
   724→        </div>
   725→      </motion.div>
   726→    </motion.section>
   727→  );
   728→}

   730→function StatCard({
   731→  label,
   732→  value,
   733→  icon: Icon,
   734→  accent,
   735→}: {
   736→  label: string;
   737→  value: number;
   738→  icon: typeof Layers;
   739→  accent?: "verified" | "violating" | "repairing" | "quarantined";
   740→}) {
   741→  const color =
   742→    accent === "verified" ? "text-verified"
   743→    : accent === "violating" ? "text-violating"
   744→    : accent === "repairing" ? "text-repairing"
   745→    : accent === "quarantined" ? "text-quarantined"
   746→    : "text-foreground";
   747→  const gradBorder = accent === "verified"
   748→    ? "bg-gradient-to-r from-verified/40 via-verified/10 to-transparent"
   749→    : accent === "violating"
   750→      ? "bg-gradient-to-r from-violating/40 via-violating/10 to-transparent"
   751→      : accent === "repairing"
   752→        ? "bg-gradient-to-r from-repairing/40 via-repairing/10 to-transparent"
   753→        : accent === "quarantined"
   754→          ? "bg-gradient-to-r from-quarantined/30 via-quarantined/10 to-transparent"
   755→          : "bg-gradient-to-r from-border/40 via-border/10 to-transparent";
   756→  return (
   757→    <GradientBorderCard accentBorder={gradBorder}>
   758→      <Card className="bg-card/60 backdrop-blur border-0 p-2.5 relative overflow-hidden h-full">
   759→        <div className="bg-grid-fine absolute inset-0 opacity-20" />
   760→        <div className="relative flex items-center gap-2">
   761→          <Icon className={cn("h-3.5 w-3.5", color)} />
   762→          <div>
   763→            <div className={cn("text-base font-semibold tabular-nums leading-none", color)}>{value}</div>
   764→            <div className="text-[9px] uppercase tracking-wide text-muted-foreground mt-0.5">{label}</div>
   765→          </div>
   766→        </div>
   767→      </Card>
   768→    </GradientBorderCard>
   769→  );
   770→}
   771→
   772→function Row({ label, value }: { label: string; value: string }) {
   773→  return (
   774→    <div className="flex items-center justify-between">
   775→      <span className="text-muted-foreground">{label}</span>
   776→      <span className="font-mono">{value}</span>
   777→    </div>
   778→  );
   779→}
