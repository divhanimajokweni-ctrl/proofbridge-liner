   147→function DivergenceMeter({ divergence, authoritative, events }: { divergence: number; authoritative: boolean; events: ShadowEventRow[] }) {
   148→  const pct = Math.min(100, Math.round(Math.sqrt(divergence) * 35));
   149→  const armed = divergence > 0.0001 && authoritative;
   150→  const barClass = divergence < 0.001 ? "[&>div]:bg-verified" : divergence < 1 ? "[&>div]:bg-repairing" : "[&>div]:bg-violating";
   151→  const trend = events.length >= 2 ? events[0].divergence - events[1].divergence : 0;
   152→  const TrendIcon = Math.abs(trend) < 0.0001 ? Minus : trend > 0 ? TrendingUp : TrendingDown;
   153→  const trendColor = Math.abs(trend) < 0.0001 ? "text-muted-foreground" : trend > 0 ? "text-violating" : "text-verified";
   154→
   155→  return (
   156→    <GradientBorderCard gradientFrom={divergence < 0.001 ? "oklch(0.78 0.16 160 / 0.35)" : divergence < 1 ? "oklch(0.80 0.15 80 / 0.35)" : "oklch(0.64 0.21 25 / 0.4)"}
   157→      gradientTo="oklch(0.32 0.014 165 / 0.1)" className={cn(armed && "glow-violating")}>
   158→      <GridOverlay />
   159→      <div className="relative flex flex-col items-center gap-1.5 p-3 w-full">
   160→        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">Twin Divergence</div>
   161→        <motion.div className={cn("font-mono text-2xl font-semibold tabular-nums", divColor(divergence))}
   162→          initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }} transition={{ type: "spring", stiffness: 200, damping: 20 }}>{divergence.toFixed(3)}</motion.div>
   163→        <GaugeNeedle value={pct} max={100} size={100} />
   164→        <Progress value={pct} className={cn("h-1.5 w-full", barClass)} />
   165→        <div className={cn("flex items-center gap-1 text-[10px]", trendColor)}>
   166→          <TrendIcon className="h-3 w-3" /><span className="font-mono">{trend >= 0 ? "+" : ""}{trend.toFixed(4)}</span><span className="text-muted-foreground">trend</span>
   167→        </div>
   168→        <motion.div className={cn("flex items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide",
   169→          armed ? "border-violating/50 bg-violating/15 text-violating animate-epistemic-pulse" : divergence > 0.0001 ? "border-repairing/40 bg-repairing/10 text-repairing" : "border-verified/40 bg-verified/10 text-verified")}
   170→          initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.5 }}>
   171→          <Power className="h-3 w-3" />{armed ? "TAKEOVER ARMED" : divergence > 0.0001 ? "DRIFT" : "ALIGNED"}
   172→        </motion.div>
   173→        <div className="text-[9px] text-muted-foreground text-center leading-tight font-mono mt-0.5">{divergence < 0.001 ? "live ≡ shadow" : divergence < 1 ? "invariant mismatch" : "authoritative breach"}</div>
   174→      </div>
   175→    </GradientBorderCard>
   176→  );
   177→}

   254→export function ShadowBridgeSection() {
   255→  const { toast } = useToast();
   256→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   257→  const [policyId, setPolicyId] = useState("");
   258→  const [bridge, setBridge] = useState<ShadowBridgeRow | null>(null);
   259→  const [loading, setLoading] = useState(true);
   260→  const [takeoverOpen, setTakeoverOpen] = useState(false);
   261→  const [takingOver, setTakingOver] = useState(false);
   262→  const [deltaText, setDeltaText] = useState(DEFAULT_DELTA);
   263→  const [predicate, setPredicate] = useState("");
   264→  const [whatif, setWhatif] = useState<{ ok: boolean; evaluations: InvariantEval[]; mmrRoot: string; divergence: number } | null>(null);
   265→  const [runningWhatif, setRunningWhatif] = useState(false);
   266→
   267→  useEffect(() => {
   268→    let c = false;
   269→    fetch("/api/policies").then((r) => r.json()).then((d: { policies: PolicyRow[] }) => {
   270→      if (c) return; const ps = d.policies ?? []; setPolicies(ps);
   271→      const shadowFirst = ps.find((p) => p.shadowEnabled);
   272→      if (shadowFirst) setPolicyId(shadowFirst.id); else if (ps.length > 0) setPolicyId(ps[0].id);
   273→    }).catch(() => {}); return () => { c = true; };
   274→  }, []);
   275→
   276→  const loadBridge = useCallback(async () => {
   277→    if (!policyId) return;
   278→    try { const r = await fetch(`/api/shadow-bridge?policyId=${encodeURIComponent(policyId)}`); if (!r.ok) throw new Error(`HTTP ${r.status}`); const d: { bridges: ShadowBridgeRow[] } = await r.json(); setBridge((d.bridges ?? [])[0] ?? null); }
   279→    catch (e) { toast({ title: "Failed to load shadow bridge", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" }); }
   280→    finally { setLoading(false); }
   281→  }, [policyId, toast]);
   282→
   283→  useEffect(() => { if (!policyId) return; setLoading(true); loadBridge(); const t = setInterval(loadBridge, POLL_MS); return () => clearInterval(t); }, [policyId, loadBridge]);
   284→
   285→  const doTakeover = useCallback(async () => {
   286→    if (!policyId) return; setTakingOver(true);
   287→    try {
   288→      const r = await fetch("/api/shadow-bridge/takeover", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ policyId }) });
   289→      if (!r.ok) throw new Error(`HTTP ${r.status}`); const d = await r.json();
   290→      toast({ title: "Shadow takeover initiated", description: d.event?.summary ?? "Shadow bridge assumed authority." }); setTakeoverOpen(false); loadBridge();
   291→    } catch (e) { toast({ title: "Takeover failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" }); }
   292→    finally { setTakingOver(false); }
