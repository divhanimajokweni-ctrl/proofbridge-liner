    80→function ThemeToggleBtn() {
    81→  const { resolvedTheme, setTheme } = useTheme();
    82→  const [mounted, setMounted] = useState(false);
    83→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    84→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    85→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
    86→  return (
    87→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
    88→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
    89→    </button>
    90→  );
    91→}

    93→function ScrollToTopButton() {
    94→  const [visible, setVisible] = useState(false);
    95→
    96→  useEffect(() => {
    97→    const onScroll = () => setVisible(window.scrollY > 300);
    98→    window.addEventListener("scroll", onScroll, { passive: true });
    99→    return () => window.removeEventListener("scroll", onScroll);
   100→  }, []);
   101→
   102→  const scrollToTop = useCallback(() => {
   103→    window.scrollTo({ top: 0, behavior: "smooth" });
   104→  }, []);
   105→
   106→  return (
   107→    <AnimatePresence>
   108→      {visible && (
   109→        <motion.button
   110→          type="button"
   111→          onClick={scrollToTop}
   112→          initial={{ opacity: 0, scale: 0.8 }}
   113→          animate={{ opacity: 1, scale: 1 }}
   114→          exit={{ opacity: 0, scale: 0.8 }}
   115→          transition={{ duration: 0.2 }}
   116→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   117→          title="Scroll to top"

   424→/* ─── Quick Stats Bar ─── */
   425→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   426→  const meta = SECTION_META[sectionId];
   427→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   428→  const [isRefreshing, setIsRefreshing] = useState(false);
   429→
   430→  // Auto-refresh countdown (10s cycle, matches health fetch)
   431→  useEffect(() => {
   432→    const t = setInterval(() => {
   433→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   434→    }, 1000);
   435→    return () => clearInterval(t);
   436→  }, []);
   437→
   438→  const handleRefresh = useCallback(() => {
   439→    setIsRefreshing(true);
   440→    fetch("/api/stats").then(() => {
   441→      setIsRefreshing(false);
   442→    }).catch(() => {
   443→      setIsRefreshing(false);
   444→    });
   445→  }, []);
   446→
   447→  return (
   448→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   449→      {meta.stats.map((stat, i) => (
   450→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   451→          {stat}
   452→        </span>
   453→      ))}
   454→      <div className="ml-auto flex items-center gap-2">
   455→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   456→          <Timer className="h-2.5 w-2.5" />
   457→          {refreshCountdown}s
   458→        </span>
   459→        <button
   460→          type="button"
   461→          onClick={handleRefresh}
   462→          disabled={isRefreshing}
   463→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   464→          title="Refresh section data"
   465→        >
   466→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   467→          refresh
   468→        </button>
   469→      </div>
   470→    </div>
   471→  );
   472→}
