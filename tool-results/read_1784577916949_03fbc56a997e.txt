    69→function fmtRelative(iso: string | null): string {
    70→  if (!iso) return "—";
    71→  const t = new Date(iso).getTime();
    72→  const diff = Date.now() - t;
    73→  if (diff < 0) return "just now";
    74→  const s = Math.floor(diff / 1000);
    75→  if (s < 60) return `${s}s ago`;
    76→  const m = Math.floor(s / 60);
    77→  if (m < 60) return `${m}m ago`;
    78→  const h = Math.floor(m / 60);
    79→  if (h < 24) return `${h}h ago`;
    80→  return `${Math.floor(h / 24)}d ago`;
    81→}

   259→/** Central divergence meter — vertical bar + armed indicator. */
   260→function DivergenceMeter({
   261→  divergence,
   262→  authoritative,
   263→}: {
   264→  divergence: number;
   265→  authoritative: boolean;
   266→}) {
   267→  // Map divergence (could be small float) to a 0-100 percentage with a log-ish curve.
   268→  const pct = Math.min(100, Math.round(Math.sqrt(divergence) * 35));
   269→  const armed = divergence > 0.0001 && authoritative;
   270→  const color =
   271→    divergence < 0.001 ? "text-verified" : divergence < 1 ? "text-repairing" : "text-violating";
   272→  const barClass =
   273→    divergence < 0.001
   274→      ? "[&>div]:bg-verified"
   275→      : divergence < 1
   276→        ? "[&>div]:bg-repairing"
   277→        : "[&>div]:bg-violating";
   278→  return (
   279→    <Card
   280→      className={cn(
   281→        "bg-card/60 backdrop-blur border p-3 relative overflow-hidden flex flex-col items-center justify-center gap-2",
   282→        armed ? "border-violating/40 glow-violating" : "border-border/60",
   283→      )}
   284→    >
   285→      <div className="bg-grid-fine absolute inset-0 opacity-20" />
   286→      <div className="relative flex flex-col items-center gap-1.5 w-full">
   287→        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   288→          Twin Divergence
   289→        </div>
   290→        <div className={cn("font-mono text-2xl font-semibold tabular-nums", color)}>
   291→          {divergence.toFixed(3)}
   292→        </div>
   293→        <Progress value={pct} className={cn("h-1.5 w-full", barClass)} />
   294→        <div
   295→          className={cn(
   296→            "mt-1 flex items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide",
   297→            armed
   298→              ? "border-violating/50 bg-violating/15 text-violating animate-epistemic-pulse"
   299→              : divergence > 0.0001
   300→                ? "border-repairing/40 bg-repairing/10 text-repairing"
   301→                : "border-verified/40 bg-verified/10 text-verified",
   302→          )}
   303→        >
   304→          <Power className="h-3 w-3" />
   305→          {armed ? "TAKEOVER ARMED" : divergence > 0.0001 ? "DRIFT" : "ALIGNED"}
   306→        </div>
   307→        <div className="text-[9px] text-muted-foreground text-center leading-tight font-mono mt-0.5">
   308→          {divergence < 0.001
   309→            ? "live ≡ shadow"
   310→            : divergence < 1
   311→              ? "invariant mismatch"
   312→              : "authoritative breach"}
   313→        </div>
   314→      </div>
   315→    </Card>
   316→  );
   317→}

   427→export function ShadowBridgeSection() {
   428→  const { toast } = useToast();
   429→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   430→  const [policyId, setPolicyId] = useState<string>("");
   431→  const [bridge, setBridge] = useState<ShadowBridgeRow | null>(null);
   432→  const [loading, setLoading] = useState(true);
   433→  const [takeoverOpen, setTakeoverOpen] = useState(false);
   434→  const [takingOver, setTakingOver] = useState(false);
   435→
   436→  // What-if composer
   437→  const [deltaText, setDeltaText] = useState(DEFAULT_DELTA);
   438→  const [predicate, setPredicate] = useState("");
   439→  const [whatif, setWhatif] = useState<{
   440→    ok: boolean;
   441→    evaluations: InvariantEval[];
   442→    mmrRoot: string;
   443→    divergence: number;
   444→  } | null>(null);
   445→  const [runningWhatif, setRunningWhatif] = useState(false);
   446→
   447→  // Load policies once, default to first shadow-enabled.
   448→  useEffect(() => {
   449→    let cancelled = false;
   450→    fetch("/api/policies")
   451→      .then((r) => r.json())
   452→      .then((d: { policies: PolicyRow[] }) => {
   453→        if (cancelled) return;
   454→        const ps = d.policies ?? [];
   455→        setPolicies(ps);
   456→        const shadowFirst = ps.find((p) => p.shadowEnabled);
   457→        if (shadowFirst) setPolicyId(shadowFirst.id);
   458→        else if (ps.length > 0) setPolicyId(ps[0].id);
   459→      })
   460→      .catch(() => {});
   461→    return () => {
   462→      cancelled = true;
   463→    };
   464→  }, []);
   465→
   466→  const loadBridge = useCallback(async () => {
   467→    if (!policyId) return;
   468→    try {
   469→      const r = await fetch(`/api/shadow-bridge?policyId=${encodeURIComponent(policyId)}`);
   470→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   471→      const d: { bridges: ShadowBridgeRow[] } = await r.json();
   472→      const b = (d.bridges ?? [])[0] ?? null;
   473→      setBridge(b);
   474→    } catch (e) {
   475→      toast({
   476→        title: "Failed to load shadow bridge",
   477→        description: e instanceof Error ? e.message : "Unknown error",
   478→        variant: "destructive",
   479→      });
   480→    } finally {
   481→      setLoading(false);
   482→    }
   483→  }, [policyId, toast]);
   484→
   485→  useEffect(() => {
   486→    if (!policyId) return;
   487→    setLoading(true);
   488→    loadBridge();
   489→    const t = setInterval(loadBridge, POLL_MS);
   490→    return () => clearInterval(t);
   491→  }, [policyId, loadBridge]);
   492→
   493→  const doTakeover = useCallback(async () => {
   494→    if (!policyId) return;
   495→    setTakingOver(true);
   496→    try {
   497→      const r = await fetch("/api/shadow-bridge/takeover", {
   498→        method: "POST",
   499→        headers: { "Content-Type": "application/json" },
   500→        body: JSON.stringify({ policyId }),
   501→      });
   502→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   503→      const d = await r.json();
   504→      toast({
   505→        title: "Shadow takeover initiated",
   506→        description: d.event?.summary ?? "Shadow bridge assumed authority.",
   507→      });
   508→      setTakeoverOpen(false);
   509→      loadBridge();
   510→    } catch (e) {
   511→      toast({
   512→        title: "Takeover failed",
   513→        description: e instanceof Error ? e.message : "Unknown error",
   514→        variant: "destructive",
   515→      });
   516→    } finally {
   517→      setTakingOver(false);
   518→    }
   519→  }, [policyId, toast, loadBridge]);
   520→
   521→  const runWhatif = useCallback(async () => {
   522→    if (!policyId) return;
   523→    let delta: Record<string, unknown> = {};
   524→    if (deltaText.trim()) {
   525→      try {
   526→        delta = JSON.parse(deltaText);
   527→      } catch (e) {
   528→        toast({
   529→          title: "Invalid state delta JSON",
   530→          description: e instanceof Error ? e.message : "Parse error",
   531→          variant: "destructive",
   532→        });
   533→        return;
   534→      }
   535→    }
   536→    setRunningWhatif(true);
   537→    try {
   538→      const body: Record<string, unknown> = { policyId, stateDelta: delta };
   539→      if (predicate.trim()) body.predicateOverride = predicate.trim();
   540→      const r = await fetch("/api/shadow-bridge/whatif", {
   541→        method: "POST",
   542→        headers: { "Content-Type": "application/json" },
   543→        body: JSON.stringify(body),
   544→      });
   545→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   546→      const d = await r.json();
   547→      setWhatif({
   548→        ok: !!d.ok,
   549→        evaluations: (d.evaluations ?? []).map((e: InvariantEval) => ({
   550→          name: e.name,
   551→          passed: e.passed,
   552→          severity: e.severity,
   553→          soft: e.soft,
   554→          actual: e.actual,
   555→        })),
   556→        mmrRoot: d.mmrRoot ?? "—",
   557→        divergence: d.event?.divergence ?? 0,
   558→      });
   559→      toast({
   560→        title: "What-if replay complete",
   561→        description: `${(d.evaluations ?? []).filter((e: InvariantEval) => !e.passed).length} violation(s) under modified rules`,
   562→      });
   563→      loadBridge();
   564→    } catch (e) {
   565→      toast({
   566→        title: "What-if replay failed",
   567→        description: e instanceof Error ? e.message : "Unknown error",
   568→        variant: "destructive",
   569→      });
   570→    } finally {
   571→      setRunningWhatif(false);
   572→    }
   573→  }, [policyId, deltaText, predicate, toast, loadBridge]);
   574→
   575→  // Build a union of invariant names (live ∪ shadow) preserving order.
   576→  const invariantUnion = useMemo(() => {
   577→    if (!bridge) return [] as { name: string; live?: InvariantEval; shadow?: InvariantEval }[];
   578→    const map = new Map<string, { name: string; live?: InvariantEval; shadow?: InvariantEval }>();
   579→    for (const iv of bridge.liveInvariants) {
