    48→
    49→type EventKind = "merge" | "shadow" | "violation";
    50→
    51→interface TimelineEvent {
    52→  id: string;
    53→  kind: EventKind;
    54→  at: string;
    55→  policyName: string;
    56→  domain: string | null;
    57→  title: string;
    58→  detail: string;
    59→  severity: "info" | "warning" | "critical";
    60→  mergeStatus?: string;
    61→  divergence?: number;
    62→  iterations?: number;
    63→  shadowKind?: string;
    64→  invariant?: string;
    65→  soft?: boolean;
    66→  repaired?: boolean;
    67→  actual?: string | null;
    68→  expected?: string | null;
    69→}
    70→
    71→interface Bucket {
    72→  t: number;
    73→  count: number;
    74→  byKind: Record<string, number>;
    75→}
    76→
    77→interface TimelineData {
    78→  events: TimelineEvent[];
    79→  total: number;
    80→  buckets: Bucket[];
    81→  policyId: string | null;
    82→}
    83→

    85→  EventKind,
    86→  { icon: typeof GitBranch; color: string; text: string; bg: string; label: string; dotBg: string }

    94→    dotBg: "bg-verified",
    95→  },
    96→  shadow: {

    98→    color: "text-repairing",
    99→    text: "text-repairing",
   100→    bg: "bg-repairing",
   101→    label: "Shadow",
   102→    dotBg: "bg-repairing",
   103→  },
   104→  violation: {
   105→    icon: AlertTriangle,
   106→    color: "text-violating",
   107→    text: "text-violating",
   108→    bg: "bg-violating",
   109→    label: "Violation",
   110→    dotBg: "bg-violating",
   111→  },
   112→};
   113→
   114→const SEVERITY_META: Record<string, string> = {
   115→  info: "border-verified/30 bg-verified/5",

   117→  critical: "border-violating/30 bg-violating/10",

   119→
   120→export function TimelineSection() {
   121→  const { toast } = useToast();
   122→  const [data, setData] = useState<TimelineData | null>(null);
   123→  const [loading, setLoading] = useState(true);
   124→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   125→  const [filterPolicy, setFilterPolicy] = useState<string>("all");
   126→  const [filterKind, setFilterKind] = useState<string>("all");
   127→  const [playing, setPlaying] = useState(false);
   128→  const [cursor, setCursor] = useState<number>(0);
   129→  const [hoveredBucket, setHoveredBucket] = useState<number | null>(null);
   130→
   131→  const load = useCallback(() => {
   132→    const qs = filterPolicy !== "all" ? `?policyId=${filterPolicy}` : "";
   133→    fetch(`/api/timeline${qs}`)
   134→      .then((r) => r.json())
   135→      .then((d) => {
   136→        setData(d);
   137→        setLoading(false);
   138→        setCursor(0);
   139→      })
   140→      .catch(() => setLoading(false));
   141→  }, [filterPolicy]);
   142→
   143→  useEffect(() => {
   144→    fetch("/api/policies")
   145→      .then((r) => r.json())
   146→      .then((d) => setPolicies(d.policies ?? []))
   147→      .catch(() => {});
   148→  }, []);
   149→
   150→  useEffect(() => {
   151→    load();
   152→    const t = setInterval(load, 15000);
   153→    return () => clearInterval(t);
   154→  }, [load]);
   155→
   156→  useEffect(() => {
   157→    if (!playing || !data) return;
   158→    const t = setInterval(() => {
   159→      setCursor((c) => {
   160→        if (c >= filtered.length - 1) {
   161→          setPlaying(false);
   162→          return c;
   163→        }
   164→        return c + 1;
   165→      });
   166→    }, 1200);
   167→    return () => clearInterval(t);
   168→  }, [playing, data, filterKind]);
   169→
   170→  const filtered = data ? data.events.filter((e) => filterKind === "all" || e.kind === filterKind) : [];
   171→
   172→  const current = filtered[cursor];
   173→
   174→  const totals = useMemo(() => ({
   175→    merges: data?.events.filter((e) => e.kind === "merge").length ?? 0,
   176→    shadows: data?.events.filter((e) => e.kind === "shadow").length ?? 0,
   177→    violations: data?.events.filter((e) => e.kind === "violation").length ?? 0,
   178→  }), [data]);
   179→
   180→  const histogramData = useMemo(() => {
   181→    if (!data) return [];
   182→    return data.buckets.map((b) => ({
   183→      time: new Date(b.t).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }),
   184→      merge: b.byKind.merge ?? 0,
   185→      shadow: b.byKind.shadow ?? 0,
   186→      violation: b.byKind.violation ?? 0,
   187→      total: b.count,
   188→      ts: b.t,
   189→    }));
   190→  }, [data]);
   191→
   192→  if (loading || !data) {
   193→    return (
   194→      <div className="space-y-4">
   195→        <Skeleton className="h-32 w-full" />
   196→        <Skeleton className="h-64 w-full" />
   197→      </div>
   198→    );
   199→  }
   200→
   201→  const progress = filtered.length > 0 ? ((cursor + 1) / filtered.length) * 100 : 0;
   202→
   203→  return (
   204→    <motion.section
   205→      className="space-y-4"
   206→      variants={containerVariants}
   207→      initial="hidden"
   208→      animate="show"
   209→    >
   210→      {/* Header + filters */}
   211→      <GradientBorderCard gradient="from-verified/40 via-repairing/20 to-violating/20" className="p-4">
   212→        <div className="bg-grid absolute inset-0 opacity-20 pointer-events-none" />
   213→        <div className="relative flex flex-wrap items-center gap-3">
   214→          <motion.div variants={cardVariants} className="flex h-10 w-10 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 shrink-0">
   215→            <Clock className="h-5 w-5 text-verified" />
   216→          </motion.div>
   217→          <motion.div variants={cardVariants} className="min-w-0 flex-1">
   218→            <h2 className="text-base font-semibold flex items-center gap-2">
   219→              Historical Replay Timeline
   220→              <Badge variant="outline" className="border-verified/30 bg-verified/10 text-verified text-[9px] font-mono">
   221→                {data.total} events
   222→              </Badge>
   223→            </h2>
   224→            <p className="text-xs text-muted-foreground mt-0.5">
   225→              Scrub through merge proposals, shadow events, and invariant breaches. Replay any episode with modified invariants.
   226→            </p>
   227→          </motion.div>
   228→          <motion.div variants={cardVariants} className="flex items-center gap-2 flex-wrap">
   229→            {/* Event type filter buttons */}
   230→            <div className="flex items-center gap-1 mr-1">
   231→              <Button
   232→                size="sm"
   233→                variant={filterKind === "all" ? "default" : "outline"}
   234→                className={cn("h-7 px-2 text-[10px]", filterKind === "all" ? "bg-verified/90 hover:bg-verified text-primary-foreground" : "")}
   235→                onClick={() => setFilterKind("all")}
   236→              >
   237→                All <Badge variant="outline" className="ml-1 h-4 px-1 text-[8px] border-current/30 bg-current/10">{data.total}</Badge>
   238→              </Button>
   239→              <Button
   240→                size="sm"
   241→                variant={filterKind === "merge" ? "default" : "outline"}
   242→                className={cn("h-7 px-2 text-[10px]", filterKind === "merge" ? "bg-verified/90 hover:bg-verified text-primary-foreground" : "")}
   243→                onClick={() => setFilterKind("merge")}
   244→              >
   245→                <GitBranch className="h-3 w-3 mr-1" /> Merges <Badge variant="outline" className="ml-1 h-4 px-1 text-[8px] border-current/30 bg-current/10">{totals.merges}</Badge>
   246→              </Button>
   247→              <Button
   248→                size="sm"
   249→                variant={filterKind === "shadow" ? "default" : "outline"}
   250→                className={cn("h-7 px-2 text-[10px]", filterKind === "shadow" ? "bg-repairing/90 hover:bg-repairing text-primary-foreground" : "")}
   251→                onClick={() => setFilterKind("shadow")}
   252→              >
   253→                <Cpu className="h-3 w-3 mr-1" /> Shadow <Badge variant="outline" className="ml-1 h-4 px-1 text-[8px] border-current/30 bg-current/10">{totals.shadows}</Badge>
   254→              </Button>
   255→              <Button
   256→                size="sm"
   257→                variant={filterKind === "violation" ? "default" : "outline"}
   258→                className={cn("h-7 px-2 text-[10px]", filterKind === "violation" ? "bg-violating/90 hover:bg-violating text-primary-foreground" : "")}
   259→                onClick={() => setFilterKind("violation")}
   260→              >
   261→                <AlertTriangle className="h-3 w-3 mr-1" /> Violations <Badge variant="outline" className="ml-1 h-4 px-1 text-[8px] border-current/30 bg-current/10">{totals.violations}</Badge>
   262→              </Button>
   263→            </div>
   264→            <Select value={filterPolicy} onValueChange={setFilterPolicy}>
   265→              <SelectTrigger className="h-8 w-[170px] text-xs">
   266→                <Filter className="h-3 w-3 mr-1" />
   267→                <SelectValue placeholder="Policy" />
   268→              </SelectTrigger>
   269→              <SelectContent>
   270→                <SelectItem value="all">All policies</SelectItem>
   271→                {policies.map((p) => (
   272→                  <SelectItem key={p.id} value={p.id}>
   273→                    {p.name}
   274→                  </SelectItem>
   275→                ))}
   276→              </SelectContent>
   277→            </Select>
   278→          </motion.div>
   279→        </div>
   280→      </GradientBorderCard>
   281→
   282→      {/* Summary + histogram */}
   283→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   284→        <motion.div variants={cardVariants} className="grid grid-cols-3 gap-3 lg:col-span-1">
   285→          <SummaryStat icon={GitBranch} label="Merges" value={totals.merges} accent="verified" />
   286→          <SummaryStat icon={Cpu} label="Shadow" value={totals.shadows} accent="repairing" />
   287→          <SummaryStat icon={AlertTriangle} label="Violations" value={totals.violations} accent="violating" />
   288→        </motion.div>
   289→
   290→        <GradientBorderCard gradient="from-verified/30 via-repairing/15 to-violating/15" className="lg:col-span-2 p-4">
   291→          <div className="relative">
   292→            <div className="flex items-center gap-2 mb-2">
   293→              <BarChart3 className="h-3.5 w-3.5 text-verified" />
   294→              <span className="text-xs font-semibold">Event density · last 24h</span>
   295→              <span className="ml-auto text-[10px] text-muted-foreground font-mono">
   296→                {hoveredBucket !== null && data.buckets[hoveredBucket]
   297→                  ? new Date(data.buckets[hoveredBucket].t).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }) +
   298→                    ` · ${data.buckets[hoveredBucket].count} events`
   299→                  : "hover a bucket"}
   300→              </span>
   301→            </div>
   302→            <RechartsHistogram
   303→              data={histogramData}
   304→              onHover={setHoveredBucket}
   305→              onSelect={(idx) => {
   306→                const bucketTime = data.buckets[idx]?.t;
   307→                if (!bucketTime) return;
   308→                const idx2 = data.events.findIndex(
   309→                  (e) => new Date(e.at).getTime() >= bucketTime,
   310→                );
   311→                if (idx2 >= 0) setCursor(idx2);
   312→              }}
   313→            />
   314→          </div>
   315→        </GradientBorderCard>
   316→      </div>
   317→
   318→      {/* Player + scrubber */}
   319→      <GradientBorderCard gradient="from-verified/30 via-repairing/15 to-verified/20" className="p-4">
   320→        <div className="relative space-y-3">
   321→          <div className="flex items-center gap-2">
   322→            <Gauge className="h-4 w-4 text-verified" />
   323→            <span className="text-sm font-semibold">Episode player</span>
   324→            <span className="ml-auto text-[10px] text-muted-foreground font-mono">
   325→              {filtered.length > 0 ? `${cursor + 1} / ${filtered.length}` : "no events"}
   326→            </span>
   327→          </div>
   328→
   329→          {/* Animated progress bar */}
   330→          <div className="relative h-2 rounded-full bg-muted/30 overflow-hidden">
   331→            <motion.div
   332→              className="absolute inset-y-0 left-0 bg-gradient-to-r from-verified via-repairing to-verified rounded-full"
   333→              initial={{ width: 0 }}
   334→              animate={{ width: `${progress}%` }}
   335→              transition={{ type: "spring", stiffness: 300, damping: 30 }}
   336→            />
   337→          </div>
   338→
   339→          {/* Scrubber */}
   340→          <div className="relative h-10 rounded-md border border-border/60 bg-background/40 overflow-hidden">
   341→            <div className="absolute inset-0 flex">
   342→              {filtered.map((e, i) => {
   343→                const m = KIND_META[e.kind];
   344→                return (
   345→                  <button
   346→                    key={e.id}
   347→                    onClick={() => setCursor(i)}
   348→                    className={cn(
   349→                      "flex-1 min-w-[3px] group relative transition-all",
   350→                      i === cursor ? "bg-opacity-100" : "opacity-50 hover:opacity-100",
   351→                    )}
   352→                    title={`${e.title} · ${timeAgo(e.at)}`}
   353→                  >
   354→                    <div className={cn("h-full w-full", m.bg, i === cursor && "opacity-100")} />
   355→                    {i === cursor && (
   356→                      <motion.div
   357→                        className="absolute -top-0.5 left-0 right-0 h-0.5 bg-foreground"
   358→                        layoutId="scrubber-indicator"
   359→                        transition={{ type: "spring", stiffness: 400, damping: 30 }}
   360→                      />
   361→                    )}
   362→                  </button>
   363→                );
   364→              })}
   365→            </div>
   366→            {filtered.length === 0 && (
   367→              <div className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground italic">
   368→                no events match the filter
   369→              </div>
   370→            )}
   371→          </div>
   372→
   373→          {/* Scrubber legend */}
   374→          <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   375→            <span className="flex items-center gap-1">
   376→              <span className="h-2 w-3 rounded-sm bg-verified" /> merge
   377→            </span>
   378→            <span className="flex items-center gap-1">
   379→              <span className="h-2 w-3 rounded-sm bg-repairing" /> shadow
   380→            </span>
   381→            <span className="flex items-center gap-1">
   382→              <span className="h-2 w-3 rounded-sm bg-violating" /> violation
   383→            </span>
   384→            <span className="ml-auto italic">click a segment to jump</span>
   385→          </div>
   386→
   387→          {/* Transport controls */}
   388→          <div className="flex items-center gap-1.5">
   389→            <Button
   390→              size="sm"
   391→              variant="outline"
   392→              className="h-8 w-8 p-0"
   393→              onClick={() => setCursor(0)}
   394→              disabled={cursor === 0}
   395→              title="Go to start"
   396→            >
   397→              <SkipBack className="h-3.5 w-3.5" />
   398→            </Button>
   399→            <Button
   400→              size="sm"
   401→              variant="outline"
   402→              className="h-8 w-8 p-0"
   403→              onClick={() => setCursor((c) => Math.max(0, c - 1))}
   404→              disabled={cursor === 0}
   405→            >
   406→              <ChevronLeft className="h-4 w-4" />
   407→            </Button>
   408→            <Button
   409→              size="sm"
   410→              className="h-8 px-4 bg-verified/90 hover:bg-verified text-primary-foreground"
   411→              onClick={() => {
   412→                if (cursor >= filtered.length - 1) {
   413→                  setCursor(0);
   414→                  setPlaying(true);
   415→                } else {
   416→                  setPlaying((p) => !p);
   417→                }
   418→              }}
   419→              disabled={filtered.length === 0}
   420→            >
   421→              {playing ? (
   422→                <>
   423→                  <Pause className="h-3.5 w-3.5 mr-1.5" /> Pause
   424→                </>
   425→              ) : (
   426→                <>
   427→                  <Play className="h-3.5 w-3.5 mr-1.5" /> {cursor >= filtered.length - 1 ? "Replay" : "Play"}
   428→                </>
   429→              )}
   430→            </Button>
   431→            <Button
   432→              size="sm"
   433→              variant="outline"
   434→              className="h-8 w-8 p-0"
   435→              onClick={() => setCursor((c) => Math.min(filtered.length - 1, c + 1))}
   436→              disabled={cursor >= filtered.length - 1}
   437→            >
   438→              <ChevronRight className="h-4 w-4" />
   439→            </Button>
   440→            <Button
   441→              size="sm"
   442→              variant="outline"
   443→              className="h-8 w-8 p-0"
   444→              onClick={() => setCursor(filtered.length - 1)}
   445→              disabled={cursor >= filtered.length - 1}
   446→              title="Go to end"
   447→            >
   448→              <SkipForward className="h-3.5 w-3.5" />
   449→            </Button>
   450→            <span className="ml-2 text-[11px] text-muted-foreground font-mono">
   451→              {current ? timeAgo(current.at) : "—"}
   452→            </span>
   453→            <div className="ml-auto flex items-center gap-1.5 text-[10px] text-muted-foreground">
   454→              <Zap className="h-3 w-3 text-repairing" />
   455→              auto-advance 1.2s
   456→            </div>
   457→          </div>
   458→
   459→          {/* Current episode detail */}
   460→          <AnimatePresence mode="wait">
   461→            {current ? (
   462→              <motion.div
   463→                key={current.id}
   464→                initial={{ opacity: 0, y: 8, scale: 0.98 }}
   465→                animate={{ opacity: 1, y: 0, scale: 1 }}
   466→                exit={{ opacity: 0, y: -8, scale: 0.98 }}
   467→                transition={{ type: "spring", stiffness: 280, damping: 24 }}
   468→              >
   469→                <EventDetail event={current} onReplay={() => {
   470→                  toast({
   471→                    title: "Episode queued for replay",
   472→                    description: `${current.title} · ${current.policyName}`,
   473→                  });
   474→                }} />
   475→              </motion.div>
   476→            ) : (
   477→              <div className="rounded-md border border-dashed border-border/60 p-6 text-center text-sm text-muted-foreground">
   478→                No episode selected.
   479→              </div>
   480→            )}
   481→          </AnimatePresence>
   482→        </div>
   483→      </GradientBorderCard>
   484→
   485→      {/* Event list */}
   486→      <GradientBorderCard gradient="from-repairing/30 via-verified/15 to-repairing/20" className="p-4">
   487→        <div className="relative">
   488→          <div className="flex items-center gap-2 mb-2">
   489→            <Activity className="h-4 w-4 text-repairing" />
   490→            <span className="text-sm font-semibold">Event stream</span>
   491→            <span className="ml-auto text-[10px] text-muted-foreground font-mono">
   492→              newest first · {filtered.length} shown
   493→            </span>
   494→          </div>
   495→          <div className="max-h-[320px] overflow-y-auto space-y-1 pr-1">
   496→            {filtered.map((e, i) => {
   497→              const m = KIND_META[e.kind];
   498→              const Icon = m.icon;
   499→              return (
   500→                <motion.button
   501→                  key={e.id}
   502→                  onClick={() => setCursor(i)}
   503→                  variants={itemVariants}
   504→                  custom={i}
   505→                  whileHover={{ scale: 1.01, backgroundColor: "oklch(0.25 0.015 168 / 0.4)" }}
   506→                  whileTap={{ scale: 0.99 }}
   507→                  className={cn(
   508→                    "w-full text-left flex items-start gap-2.5 rounded-md border px-2.5 py-1.5 transition-colors",
   509→                    i === cursor
   510→                      ? "border-verified/40 bg-verified/5"
   511→                      : "border-border/40 bg-background/30",
   512→                  )}
   513→                >
   514→                  {/* Color-coded timeline dot */}
   515→                  <div className="flex flex-col items-center pt-0.5 shrink-0">
   516→                    <div className={cn("h-3 w-3 rounded-full border-2 border-background", m.dotBg)} />
   517→                    {i < filtered.length - 1 && (
   518→                      <div className={cn("w-0.5 h-4 mt-0.5", m.dotBg, "opacity-30")} />
   519→                    )}
   520→                  </div>
   521→                  <div className={cn("mt-0.5 h-5 w-5 shrink-0 rounded flex items-center justify-center", m.bg, "bg-opacity-20")}>
   522→                    <Icon className={cn("h-3 w-3", m.text)} />
   523→                  </div>
   524→                  <div className="min-w-0 flex-1">
   525→                    <div className="flex items-center gap-2">
   526→                      <span className="text-xs font-medium truncate">{e.title}</span>
   527→                      <span className="ml-auto text-[10px] text-muted-foreground font-mono shrink-0">
   528→                        {timeAgo(e.at)}

   586→function Histogram({
   587→  buckets,
   588→  onHover,
   589→  onSelect,
   590→}: {
   591→  buckets: Bucket[];
   592→  onHover: (idx: number | null) => void;
   593→  onSelect: (idx: number) => void;
   594→}) {
   595→  const max = Math.max(1, ...buckets.map((b) => b.count));
   596→  return (
   597→    <div className="flex items-end gap-[2px] h-16">
   598→      {buckets.map((b, i) => {
   599→        const h = (b.count / max) * 100;
   600→        const hasM = b.byKind.merge > 0;
   601→        const hasS = b.byKind.shadow > 0;
   602→        const hasV = b.byKind.violation > 0;
   603→        return (
   604→          <button
   605→            key={i}
   606→            onMouseEnter={() => onHover(i)}
   607→            onMouseLeave={() => onHover(null)}
   608→            onClick={() => onSelect(i)}
   609→            className="flex-1 group relative flex flex-col justify-end h-full"
   610→            title={`${new Date(b.t).toLocaleTimeString([], { hour: "2-digit" })} · ${b.count} events`}
   611→          >
   612→            <div className="w-full overflow-hidden rounded-sm flex flex-col justify-end" style={{ height: `${Math.max(4, h)}%` }}>
   613→              {hasV && <div className="w-full bg-violating/80" style={{ flex: b.byKind.violation }} />}
   614→              {hasS && <div className="w-full bg-repairing/80" style={{ flex: b.byKind.shadow }} />}
   615→              {hasM && <div className="w-full bg-verified/80" style={{ flex: b.byKind.merge }} />}
   616→            </div>
   617→            {b.count === 0 && <div className="w-full h-[3px] bg-muted/30 rounded-sm" />}
   618→          </button>
   619→        );
   620→      })}
   621→    </div>
   622→  );
   623→}
