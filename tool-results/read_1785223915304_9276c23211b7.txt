    94→function ThemeToggleBtn() {
    95→  const { resolvedTheme, setTheme } = useTheme();
    96→  const [mounted, setMounted] = useState(false);
    97→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    98→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    99→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
   100→  return (
   101→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>

   107→function ScrollToTopButton() {
   108→  const [visible, setVisible] = useState(false);
   109→
   110→  useEffect(() => {
   111→    const onScroll = () => setVisible(window.scrollY > 300);
   112→    window.addEventListener("scroll", onScroll, { passive: true });
   113→    return () => window.removeEventListener("scroll", onScroll);
   114→  }, []);
   115→
   116→  const scrollToTop = useCallback(() => {
   117→    window.scrollTo({ top: 0, behavior: "smooth" });
   118→  }, []);
   119→
   120→  return (
   121→    <AnimatePresence>
   122→      {visible && (
   123→        <motion.button
   124→          type="button"
   125→          onClick={scrollToTop}
   126→          initial={{ opacity: 0, scale: 0.8 }}
   127→          animate={{ opacity: 1, scale: 1 }}
   128→          exit={{ opacity: 0, scale: 0.8 }}
   129→          transition={{ duration: 0.2 }}
   130→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   131→          title="Scroll to top"
   132→        >
   133→          <ArrowUp className="h-4 w-4" />
   134→        </motion.button>
   135→      )}
   136→    </AnimatePresence>

   452→/* ─── Quick Stats Bar ─── */
   453→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   454→  const meta = SECTION_META[sectionId];
   455→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   456→  const [isRefreshing, setIsRefreshing] = useState(false);
   457→
   458→  // Auto-refresh countdown (10s cycle, matches health fetch)
   459→  useEffect(() => {
   460→    const t = setInterval(() => {
   461→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   462→    }, 1000);
   463→    return () => clearInterval(t);
   464→  }, []);
   465→
   466→  const handleRefresh = useCallback(() => {
   467→    setIsRefreshing(true);
   468→    fetch("/api/stats").then(() => {
   469→      setIsRefreshing(false);
   470→    }).catch(() => {
   471→      setIsRefreshing(false);
   472→    });
   473→  }, []);
   474→
   475→  return (
   476→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   477→      {meta.stats.map((stat, i) => (
   478→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   479→          {stat}
   480→        </span>
   481→      ))}
   482→      <div className="ml-auto flex items-center gap-2">
   483→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   484→          <Timer className="h-2.5 w-2.5" />
   485→          {refreshCountdown}s
   486→        </span>
   487→        <button
   488→          type="button"
   489→          onClick={handleRefresh}
   490→          disabled={isRefreshing}
   491→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   492→          title="Refresh section data"
   493→        >
   494→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   495→          refresh
   496→        </button>
   497→      </div>
   498→    </div>
   499→  );
   500→}
