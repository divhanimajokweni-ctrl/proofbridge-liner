    87→/* ─── Color helpers ─── */
    88→function coverageColor(pct: number): string {
    89→  if (pct >= 80) return "oklch(0.78 0.16 160)"; // green
    90→  if (pct >= 60) return "oklch(0.70 0.13 120)"; // light green
    91→  if (pct >= 40) return "oklch(0.75 0.15 80)";  // amber
    92→  if (pct >= 20) return "oklch(0.72 0.13 50)";  // orange
    93→  return "oklch(0.65 0.2 25)";                   // red
    94→}
    95→
    96→function coverageTextClass(pct: number): string {
    97→  if (pct >= 80) return "text-verified";
    98→  if (pct >= 60) return "text-emerald-400";
    99→  if (pct >= 40) return "text-repairing";
   100→  if (pct >= 20) return "text-orange-400";
   101→  return "text-violating";
   102→}
   103→
   104→function coverageBgClass(pct: number): string {
   105→  if (pct >= 80) return "bg-verified/10 border-verified/30";
   106→  if (pct >= 60) return "bg-emerald-500/10 border-emerald-500/30";
   107→  if (pct >= 40) return "bg-repairing/10 border-repairing/30";
   108→  if (pct >= 20) return "bg-orange-500/10 border-orange-500/30";
   109→  return "bg-violating/10 border-violating/30";
   110→}
   111→
   112→function gapSeverityClass(severity: "critical" | "high" | "medium"): string {
   113→  if (severity === "critical") return "border-violating/40 bg-violating/10 text-violating";
   114→  if (severity === "high") return "border-repairing/40 bg-repairing/10 text-repairing";
   115→  return "border-quarantined/40 bg-quarantined/10 text-quarantined";
   116→}

   118→/* ─── Mock data generation ─── */
   119→function generateCoverageData(): CoverageData {
   120→  const shardNames = [
   121→    "us-east-1-alpha", "us-east-1-beta", "us-west-2-gamma",
   122→    "eu-west-1-delta", "eu-central-1-epsilon", "ap-south-1-zeta",
   123→    "ap-northeast-1-eta", "us-east-1-theta", "sa-east-1-iota",
   124→    "eu-north-1-kappa", "us-west-1-lambda", "ap-southeast-1-mu",
   125→  ];
   126→
   127→  const statuses: ShardStatus[] = ["healthy", "repairing", "violating"];
   128→
   129→  const shards: ShardCoverage[] = shardNames.map((name, i) => {
   130→    const coveragePct = Math.round(
   131→      i < 3 ? 85 + Math.random() * 15 :
   132→      i < 6 ? 50 + Math.random() * 30 :
   133→      i < 9 ? 20 + Math.random() * 30 :
   134→      Math.random() * 20
   135→    );
   136→    return {
   137→      id: `shard-${i}`,
   138→      name,
   139→      hash: `${name.slice(0, 3)}${Math.random().toString(36).slice(2, 8)}`,
   140→      invariantCount: Math.floor(8 + Math.random() * 50),
   141→      coveragePct: Math.min(100, coveragePct),
   142→      status: coveragePct >= 70 ? "healthy" : coveragePct >= 40 ? "repairing" : "violating",
   143→      lastChecked: new Date(Date.now() - Math.floor(Math.random() * 3600000)).toISOString(),
   144→      trend: coveragePct >= 70 ? (Math.random() > 0.3 ? "up" : "stable") :
   145→             coveragePct >= 40 ? (Math.random() > 0.5 ? "stable" : "down") :
   146→             (Math.random() > 0.4 ? "down" : "stable"),
   147→    };
   148→  });
   149→
   150→  const totalInvariants = shards.reduce((s, sh) => s + sh.invariantCount, 0);
   151→  const avgCoverage = Math.round(shards.reduce((s, sh) => s + sh.coveragePct, 0) / shards.length);
   152→  const uncoveredCount = shards.reduce((s, sh) => s + Math.round(sh.invariantCount * (1 - sh.coveragePct / 100)), 0);
   153→  const criticalGaps = shards.filter((sh) => sh.coveragePct === 0).reduce((s, sh) => s + sh.invariantCount, 0);
   154→
   155→  // Distribution buckets
   156→  const buckets = [
   157→    { label: "0-20%", min: 0, max: 20, fill: "oklch(0.65 0.2 25)" },
   158→    { label: "21-40%", min: 21, max: 40, fill: "oklch(0.72 0.13 50)" },
   159→    { label: "41-60%", min: 41, max: 60, fill: "oklch(0.75 0.15 80)" },
   160→    { label: "61-80%", min: 61, max: 80, fill: "oklch(0.70 0.13 120)" },
   161→    { label: "81-100%", min: 81, max: 100, fill: "oklch(0.78 0.16 160)" },
   162→  ];
   163→
   164→  const distribution = buckets.map((b) => ({
   165→    bucket: b.label,
   166→    count: shards.filter((sh) => sh.coveragePct >= b.min && sh.coveragePct <= b.max)
   167→      .reduce((s, sh) => s + sh.invariantCount, 0),
   168→    fill: b.fill,
   169→  }));
   170→
   171→  // Coverage gaps
   172→  const gaps: CoverageGap[] = [];
   173→  for (const shard of shards) {
   174→    if (shard.coveragePct < 40) {
   175→      const uncoveredCountForShard = Math.ceil(shard.invariantCount * (1 - shard.coveragePct / 100));
   176→      const count = Math.min(uncoveredCountForShard, 3);
   177→      for (let j = 0; j < count; j++) {
   178→        gaps.push({
   179→          id: `gap-${shard.id}-${j}`,
   180→          invariantName: `inv_${shard.name.replace(/-/g, "_")}_${["bounds_check", "state_integrity", "prop_constraint", "drift_limit", "hash_valid"][j % 5]}`,
   181→          shardName: shard.name,
   182→          shardHash: shard.hash,
   183→          coveragePct: Math.max(0, shard.coveragePct - Math.floor(Math.random() * 20)),
   184→          severity: shard.coveragePct < 10 ? "critical" : shard.coveragePct < 25 ? "high" : "medium",
   185→        });
   186→      }
   187→    }
   188→  }
   189→
   190→  return {
   191→    shards,
   192→    totalInvariants,
   193→    avgCoverage,
   194→    uncoveredCount,
   195→    criticalGaps,
   196→    distribution,
   197→    generatedAt: Date.now(),
   198→  };
   199→}

   201→/* ─── Custom Treemap Content ─── */
   202→function CoverageTreemapContent(props: {
   203→  x?: number; y?: number; width?: number; height?: number;
   204→  name?: string; size?: number; coverage?: number; depth?: number;
   205→}) {
   206→  const { x = 0, y = 0, width = 0, height = 0, name = "", size = 0, coverage = 0, depth = 0 } = props;
   207→  if (depth !== 1 || width < 20 || height < 16) return null;
   208→
   209→  const fillColor = coverageColor(coverage);
   210→  const opacity = 0.15 + (coverage / 100) * 0.35;
   211→
   212→  return (
   213→    <g>
   214→      <rect
   215→        x={x} y={y} width={width} height={height}
   216→        fill={fillColor} opacity={opacity} rx={3}
   217→        stroke={fillColor} strokeWidth={1} strokeOpacity={0.5}
   218→      />
   219→      {width > 50 && height > 22 && (
   220→        <text x={x + 6} y={y + 13} fill="var(--foreground)" fontSize={8} fontFamily="var(--font-geist-mono)" opacity={0.9}>
   221→          {name.length > 14 ? name.slice(0, 12) + "…" : name}
   222→        </text>
   223→      )}
   224→      {width > 50 && height > 36 && (
   225→        <text x={x + 6} y={y + 25} fill={fillColor} fontSize={9} fontFamily="var(--font-geist-mono)" fontWeight={600}>
   226→          {coverage}% · {size}
   227→        </text>
   228→      )}
   229→    </g>
   230→  );
   231→}

   252→/* ─── Main Component ─── */
   253→export function CoverageTreemapSection() {
   254→  const [data, setData] = useState<CoverageData | null>(null);
   255→  const [loading, setLoading] = useState(true);
   256→  const [autoRefresh, setAutoRefresh] = useState(false);
   257→  const [sortKey, setSortKey] = useState<SortKey>("coveragePct");
   258→  const [sortDir, setSortDir] = useState<SortDir>("asc");
   259→  const [lastRefresh, setLastRefresh] = useState<number>(Date.now());
   260→
   261→  const load = useCallback(() => {
   262→    const d = generateCoverageData();
   263→    setData(d);
   264→    setLastRefresh(Date.now());
   265→    setLoading(false);
   266→  }, []);
   267→
   268→  useEffect(() => { load(); }, [load]);
   269→
   270→  useEffect(() => {
   271→    if (!autoRefresh) return;
   272→    const t = setInterval(load, 30000);
   273→    return () => clearInterval(t);
   274→  }, [autoRefresh, load]);
   275→
   276→  /* ─── Derived data ─── */
   277→  const treemapData = useMemo(() => {
   278→    if (!data) return [];
   279→    return data.shards.map((sh) => ({
   280→      name: sh.name,
   281→      size: sh.invariantCount,
   282→      coverage: sh.coveragePct,
   283→    }));
   284→  }, [data]);
   285→
   286→  const sortedShards = useMemo(() => {
   287→    if (!data) return [];
   288→    const arr = [...data.shards];
   289→    return arr.sort((a, b) => {
   290→      let cmp = 0;
   291→      if (sortKey === "name") cmp = a.name.localeCompare(b.name);
   292→      else if (sortKey === "invariantCount") cmp = a.invariantCount - b.invariantCount;
   293→      else if (sortKey === "coveragePct") cmp = a.coveragePct - b.coveragePct;
   294→      else if (sortKey === "lastChecked") cmp = new Date(a.lastChecked).getTime() - new Date(b.lastChecked).getTime();
   295→      return sortDir === "asc" ? cmp : -cmp;
   296→    }).slice(0, 10);
   297→  }, [data, sortKey, sortDir]);
   298→
   299→  const coverageGaps = useMemo(() => {
   300→    if (!data) return [];
   301→    const gaps: CoverageGap[] = [];
   302→    for (const shard of data.shards) {
   303→      if (shard.coveragePct < 40) {
   304→        const uncovered = Math.ceil(shard.invariantCount * (1 - shard.coveragePct / 100));
   305→        const count = Math.min(uncovered, 3);
   306→        for (let j = 0; j < count; j++) {
   307→          gaps.push({
   308→            id: `gap-${shard.id}-${j}`,
   309→            invariantName: `inv_${shard.name.replace(/-/g, "_")}_${["bounds_check", "state_integrity", "prop_constraint", "drift_limit", "hash_valid"][j % 5]}`,
   310→            shardName: shard.name,
   311→            shardHash: shard.hash,
   312→            coveragePct: Math.max(0, shard.coveragePct - Math.floor(j * 8)),
   313→            severity: shard.coveragePct < 10 ? "critical" : shard.coveragePct < 25 ? "high" : "medium",
   314→          });
   315→        }
   316→      }
   317→    }
   318→    return gaps;
   319→  }, [data]);
   320→
   321→  const trendIcon = useMemo(() => {
   322→    if (!data) return Minus;
   323→    return data.avgCoverage >= 70 ? TrendingUp : data.avgCoverage >= 40 ? Minus : TrendingDown;
   324→  }, [data]);
   325→
   326→  const handleSort = useCallback((key: SortKey) => {
   327→    setSortKey((prev) => {
   328→      if (prev === key) {
   329→        setSortDir((d) => d === "asc" ? "desc" : "asc");
   330→      } else {
   331→        setSortDir("asc");
   332→      }
   333→      return key;
   334→    });
   335→  }, []);
   336→
   337→  const fmtTime = (iso: string) => {
   338→    const diff = Date.now() - new Date(iso).getTime();
   339→    const s = Math.floor(diff / 1000);
   340→    if (s < 60) return `${s}s ago`;
   341→    const m = Math.floor(s / 60);
   342→    if (m < 60) return `${m}m ago`;
   343→    const h = Math.floor(m / 60);
   344→    return `${h}h ago`;
   345→  };
   346→
   347→  if (loading) {
   348→    return (
   349→      <div className="flex items-center justify-center py-20">
   350→        <div className="flex items-center gap-3 text-muted-foreground">
   351→          <div className="h-5 w-5 animate-spin rounded-full border-2 border-muted-foreground/30 border-t-verified" />
   352→          <span className="text-sm">Loading coverage data…</span>
   353→        </div>
   354→      </div>
   355→    );
   356→  }
   357→
   358→  if (!data) return null;
   359→
   360→  const TrendIcon = trendIcon;
   361→
   362→  return (
   363→    <motion.div
   364→      className="space-y-5"
   365→      initial="hidden"
   366→      animate="visible"
   367→      variants={containerVariants}
   368→    >
   369→      {/* ── Header Card ── */}
   370→      <motion.div variants={cardVariants}>
   371→        <Card className="relative overflow-hidden bg-card/60 backdrop-blur border-border/60 p-5">
   372→          <TopAccentBar color="oklch(0.78 0.16 160)" />
   373→          <GridOverlay opacity="opacity-10" />
   374→          <div className="relative flex flex-col sm:flex-row items-start sm:items-center gap-3">
   375→            <SectionHeader
   376→              icon={GitGraph}
   377→              title="Coverage Treemap"
   378→              subtitle="Invariant coverage visualization across shards & policies"
   379→              iconClass="border-verified/30 bg-verified/10 text-verified"
   380→            />
   381→            <div className="sm:ml-auto flex flex-wrap items-center gap-2">
   382→              <Button
   383→                variant="outline"
   384→                size="sm"
   385→                className="h-7 gap-1.5 border-border/60 bg-muted/30 text-xs"
   386→                onClick={load}
   387→              >
   388→                <RefreshCw className={cn("h-3 w-3", loading && "animate-spin")} />
   389→                Refresh
   390→              </Button>
   391→              <Button
   392→                variant="outline"
   393→                size="sm"
   394→                className="h-7 gap-1.5 border-verified/30 bg-verified/10 text-verified text-xs hover:bg-verified/20"
   395→              >
   396→                <Download className="h-3 w-3" />
   397→                Export Report
   398→              </Button>
   399→              <div
   400→                className={cn(
   401→                  "flex items-center gap-2 rounded-full border px-3 py-1 text-[10px] cursor-pointer select-none transition-colors",
   402→                  autoRefresh
   403→                    ? "border-verified/30 bg-verified/10 text-verified"
   404→                    : "border-border/60 bg-muted/30 text-muted-foreground"

   406→                onClick={() => setAutoRefresh((v) => !v)}

   421→      {/* ── 2. KPI Strip ── */}

   428→          <Card className="relative overflow-hidden bg-card/60 backdrop-blur border-border/60 p-4 h-full">
   429→            <div className="absolute top-0 left-0 right-0 h-0.5 opacity-60" style={{ background: "linear-gradient(90deg, oklch(0.78 0.16 160), transparent)" }} />
   430→            <div className="bg-grid-fine absolute inset-0 opacity-15" />
