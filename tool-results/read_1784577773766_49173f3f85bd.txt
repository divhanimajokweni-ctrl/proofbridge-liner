    53→// ─── Seeded sparkline generator ───────────────────────────────────────────────
    54→function generateSparkline(base: number, variance: number, length = 12): number[] {
    55→  let seed = base * 7 + 31;
    56→  const rand = () => {
    57→    seed = (seed * 1103515245 + 12345) & 0x7fffffff;
    58→    return seed / 0x7fffffff;
    59→  };
    60→  return Array.from({ length }, () =>
    61→    Math.max(0, Math.round((base + (rand() - 0.5) * 2 * variance) * 10) / 10)
    62→  );
    63→}
