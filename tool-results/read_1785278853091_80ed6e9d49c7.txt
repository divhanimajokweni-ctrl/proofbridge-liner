   195→# ─── Custom Page Template ───
   196→class DarkDocTemplate(BaseDocTemplate):
   197→    def __init__(self, filename, **kw):
   198→        BaseDocTemplate.__init__(self, filename, **kw)
   199→        page_width, page_height = A4
   200→        frame = Frame(
   201→            20*mm, 25*mm, page_width - 40*mm, page_height - 50*mm,
   202→            id='normal'
   203→        )
   204→        template = PageTemplate(id='dark', frames=[frame], onPage=self._dark_page)
   205→        self.addPageTemplates([template])
   206→
   207→    def _dark_page(self, canvas, doc):
   208→        canvas.saveState()
   209→        # Full dark background
   210→        canvas.setFillColor(VVU_BG)
   211→        canvas.rect(0, 0, A4[0], A4[1], fill=1, stroke=0)
   212→        # Top accent line
   213→        canvas.setStrokeColor(VVU_GREEN)
   214→        canvas.setLineWidth(2)
   215→        canvas.line(20*mm, A4[1] - 20*mm, A4[0] - 20*mm, A4[1] - 20*mm)
   216→        # Footer
   217→        canvas.setFont('Helvetica', 8)
   218→        canvas.setFillColor(VVU_GRAY)
   219→        canvas.drawCentredString(A4[0]/2, 15*mm,
   220→            f'VVU EARTH TECH — Outreach Distribution Document — {datetime.now().strftime("%Y-%m")}')
   221→        # Bottom accent line
   222→        canvas.setStrokeColor(VVU_GREEN)
   223→        canvas.setLineWidth(1)
   224→        canvas.line(20*mm, 20*mm, A4[0] - 20*mm, 20*mm)
   225→        canvas.restoreState()

   288→def make_table(headers, rows, col_widths=None):
   289→    """Create a styled table."""
   290→    header_paras = [Paragraph(h, ParagraphStyle('TH', fontName='Helvetica-Bold',
   291→                                                  fontSize=9, textColor=VVU_PURE_WHITE,
   292→                                                  alignment=TA_CENTER)) for h in headers]
   293→    data = [header_paras]
   294→    for row in rows:
   295→        row_paras = [Paragraph(str(cell), ParagraphStyle('TC', fontName='Helvetica',
   296→                                                          fontSize=9, textColor=VVU_WHITE)) for cell in row]
   297→        data.append(row_paras)
   298→
   299→    page_width = A4[0] - 40*mm
   300→    if col_widths is None:
   301→        n_cols = len(headers)
   302→        col_widths = [page_width / n_cols] * n_cols
   303→
   304→    t = Table(data, colWidths=col_widths)
   305→    t.setStyle(TableStyle([
   306→        ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_BG),
   307→        ('BACKGROUND', (0, 1), (-1, -1), TABLE_ROW_BG),
   308→        ('ROWBACKGROUNDS', (0, 1), (-1, -1), [TABLE_ROW_BG, TABLE_ALT_ROW]),
   309→        ('TEXTCOLOR', (0, 0), (-1, 0), VVU_PURE_WHITE),
   310→        ('TEXTCOLOR', (0, 1), (-1, -1), VVU_WHITE),
   311→        ('GRID', (0, 0), (-1, -1), 0.5, HexColor('#334155')),
   312→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   313→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   314→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   315→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   316→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   317→    ]))
   318→    return t
