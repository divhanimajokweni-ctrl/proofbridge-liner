     1→"""VVU Earth Tech Ledger — Complete exception hierarchy.
     2→
     3→Every exception carries a machine-readable ``code`` attribute and an optional
     4→``detail`` dictionary so that callers can programmatically inspect failures
     5→without parsing message strings.
     6→
     7→Hierarchy::
     8→
     9→    LedgerError
    10→    ├── SerializationError
    11→    │   ├── DepthExceededError
    12→    │   ├── SizeExceededError
    13→    │   └── InvalidTypeError
    14→    ├── HashError
    15→    │   ├── DomainViolationError
    16→    │   └── HashMismatchError
    17→    ├── CryptoError
    18→    │   ├── SignatureError
    19→    │   │   ├── InvalidSignatureError
    20→    │   │   ├── KeyNotFoundError
    21→    │   │   └── KeyExpiredError
    22→    │   └── KeyRotationError
    23→    │       ├── RotationInProgressError
    24→    │       └── InvalidEpochError
    25→    ├── MMRError
    26→    │   ├── InvalidIndexError
    27→    │   ├── InvalidProofError
    28→    │   └── RootMismatchError
    29→    ├── StorageError
    30→    │   ├── DatabaseError
    31→    │   │   ├── DBConnectionFailedError
    32→    │   │   ├── DatabaseBusyError
    33→    │   │   ├── DatabaseCorruptError
    34→    │   │   └── MigrationFailedError
    35→    │   └── SnapshotError
    36→    │       ├── SnapshotCreationError
    37→    │       ├── SnapshotRestorationError
    38→    │       └── SnapshotIntegrityError
    39→    ├── ReplayError
    40→    │   ├── SequenceViolationError
    41→    │   ├── ParentChainBrokenError
    42→    │   ├── ReplayHashMismatchError
    43→    │   └── ValidatorHistoryError
    44→    ├── ValidatorError
    45→    │   ├── ValidatorNotFoundError
    46→    │   ├── DuplicateValidatorError
    47→    │   ├── ValidatorExpiredError
    48→    │   ├── WeightInvalidError
    49→    │   └── QuorumFailedError
    50→    ├── EnvelopeError
    51→    │   ├── InvalidEnvelopeFormatError
    52→    │   ├── MissingFieldError
    53→    │   └── EnvelopeSignatureError
    54→    ├── ConfigError
    55→    │   ├── InvalidValueError
    56→    │   ├── MissingConfigFieldError
    57→    │   └── ImmutableAfterStartupError
    58→    └── NetworkError
    59→        ├── NetworkConnectionFailedError
    60→        ├── TLSError
    61→        ├── AuthFailedError
    62→        └── RateLimitedError
    63→"""
    64→
    65→from __future__ import annotations
    66→
    67→from typing import Any
    68→
    69→
    70→# ---------------------------------------------------------------------------
    71→# Base
    72→# ---------------------------------------------------------------------------
    73→
    74→class LedgerError(Exception):
    75→    """Base exception for all ledger errors.
    76→
    77→    Attributes:
    78→        code: Machine-readable error code (e.g. ``"LEDGER_UNKNOWN"``).
    79→        detail: Optional dictionary with additional context.
    80→    """
    81→
    82→    def __init__(
    83→        self,
    84→        message: str = "",
    85→        *,
    86→        code: str = "LEDGER_UNKNOWN",
    87→        detail: dict[str, Any] | None = None,
    88→    ) -> None:
    89→        super().__init__(message)
    90→        self.code: str = code
    91→        self.detail: dict[str, Any] | None = detail
    92→
    93→    def __repr__(self) -> str:  # pragma: no cover
    94→        return f"{type(self).__name__}(code={self.code!r}, message={str(self)!r})"
    95→
