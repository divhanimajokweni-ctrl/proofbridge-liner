    85→    85→function generateGitActions(count: number): GitAction[] {
    86→    86→  const actions: GitAction["action"][] = ["commit", "push", "merge", "deploy", "test", "validate"];
    87→    87→  const branches = ["main", "hbk/mk-ii", "hbk/bayesian-engine", "hbk/sensor-cal", "hbk/cad-layout", "hbk/partners"];
    88→    88→  const messages = [
    89→    89→    "feat: add AMD Ryzen AI compute module bounding box",
    90→    90→    "fix: sensor isolation distance verified at X=20, Y=180",
    91→    91→    "refactor: BMS power distribution routing updated",
    92→    92→    "feat: NVMe storage bay vibration dampening mounts",
    93→    93→    "chore: IP67 transit shell envelope clearance check",
    94→    94→    "test: Bayesian inference engine MCMC convergence",
    95→    95→    "feat: Comms routing node Cellular/GNSS integration",
    96→    96→    "validate: 72h validation phase V3 metrics passing",
    97→    97→    "deploy: HBK Mk-II chassis layout to FreeCAD workspace",
    98→    98→    "merge: hbk/sensor-cal → main (acoustic filtering verified)",
    99→    99→    "feat: equity split 70/20/5 embedded in metadata",
   100→   100→    "fix: analog isolation clearance zone expanded",
   101→   101→    "test: Kria SoM edge-compute integration",
   102→   102→    "validate: Brier Score ≤ 0.02 threshold check",
   103→   103→    "feat: Founding Partners campaign framework",
   104→   104→    "chore: resource register initial data migration",
   105→   105→  ];
   106→   106→  const authors = ["eng-lead", "cad-operator", "bayesian-eng", "sensor-tech", "field-ops", "devops"];
   107→   107→  const result: GitAction[] = [];
   108→   108→  const now = Date.now();
   109→   109→  for (let i = 0; i < count; i++) {
   110→   110→    const action = actions[Math.floor(Math.random() * actions.length)];
   111→   111→    const hash = Math.random().toString(16).slice(2, 9);
   112→   112→    result.push({
   113→   113→      id: `ga-${i}`,
   114→   114→      timestamp: new Date(now - (count - i) * 180000 + Math.random() * 60000).toISOString(),
   115→   115→      action,
   116→   116→      branch: branches[Math.floor(Math.random() * branches.length)],
   117→   117→      message: messages[Math.floor(Math.random() * messages.length)],
   118→   118→      author: authors[Math.floor(Math.random() * authors.length)],
   119→   119→      status: Math.random() > 0.08 ? "success" : Math.random() > 0.5 ? "running" : "failed",
   120→   120→      hash,
   121→   121→    });
   122→   122→  }
   123→   123→  return result;
   124→   124→}
