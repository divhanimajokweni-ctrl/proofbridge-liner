    86→function ThemeToggleBtn() {
    87→  const { resolvedTheme, setTheme } = useTheme();
    88→  const [mounted, setMounted] = useState(false);
    89→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
    90→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
    91→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
    92→  return (
    93→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
    94→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
    95→    </button>
    96→  );
    97→}

    99→function ScrollToTopButton() {
   100→  const [visible, setVisible] = useState(false);
   101→
   102→  useEffect(() => {
   103→    const onScroll = () => setVisible(window.scrollY > 300);
   104→    window.addEventListener("scroll", onScroll, { passive: true });
   105→    return () => window.removeEventListener("scroll", onScroll);
   106→  }, []);
   107→
   108→  const scrollToTop = useCallback(() => {
   109→    window.scrollTo({ top: 0, behavior: "smooth" });
   110→  }, []);
   111→
   112→  return (
   113→    <AnimatePresence>
   114→      {visible && (
   115→        <motion.button
   116→          type="button"
   117→          onClick={scrollToTop}
   118→          initial={{ opacity: 0, scale: 0.8 }}
   119→          animate={{ opacity: 1, scale: 1 }}
   120→          exit={{ opacity: 0, scale: 0.8 }}
   121→          transition={{ duration: 0.2 }}
   122→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   123→          title="Scroll to top"
   124→        >
   125→          <ArrowUp className="h-4 w-4" />

   128→    </AnimatePresence>

   436→/* ─── Quick Stats Bar ─── */
   437→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   438→  const meta = SECTION_META[sectionId];
   439→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   440→  const [isRefreshing, setIsRefreshing] = useState(false);
   441→
   442→  // Auto-refresh countdown (10s cycle, matches health fetch)
   443→  useEffect(() => {
   444→    const t = setInterval(() => {
   445→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   446→    }, 1000);
   447→    return () => clearInterval(t);
   448→  }, []);
   449→
   450→  const handleRefresh = useCallback(() => {
   451→    setIsRefreshing(true);
   452→    fetch("/api/stats").then(() => {
   453→      setIsRefreshing(false);
   454→    }).catch(() => {
   455→      setIsRefreshing(false);
   456→    });
   457→  }, []);
   458→
   459→  return (
   460→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   461→      {meta.stats.map((stat, i) => (
   462→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   463→          {stat}
   464→        </span>
   465→      ))}
   466→      <div className="ml-auto flex items-center gap-2">
   467→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   468→          <Timer className="h-2.5 w-2.5" />
   469→          {refreshCountdown}s
   470→        </span>
   471→        <button
   472→          type="button"
   473→          onClick={handleRefresh}
   474→          disabled={isRefreshing}
   475→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   476→          title="Refresh section data"
   477→        >
   478→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   479→          refresh
   480→        </button>
   481→      </div>
   482→    </div>
   483→  );
   484→}
