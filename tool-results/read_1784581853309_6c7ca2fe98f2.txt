    74→function fmtRelative(iso: string | null | undefined): string {
    75→  if (!iso) return "—";
    76→  const t = new Date(iso).getTime();
    77→  const diff = Date.now() - t;
    78→  if (diff < 0) return "just now";
    79→  const s = Math.floor(diff / 1000);
    80→  if (s < 60) return `${s}s ago`;
    81→  const m = Math.floor(s / 60);
    82→  if (m < 60) return `${m}m ago`;
    83→  const h = Math.floor(m / 60);
    84→  if (h < 24) return `${h}h ago`;
    85→  return `${Math.floor(h / 24)}d ago`;
    86→}

    93→function severityBorderColor(severity: Severity, accepted: boolean): string {
    94→  if (accepted) return "border-verified/50";
    95→  switch (severity) {
    96→    case "critical": return "border-violating/50";
    97→    case "high": return "border-repairing/50";
    98→    case "medium": return "border-quarantined/50";
    99→    case "low": return "border-border/60";
   100→  }

   104→function severityGlow(severity: Severity, accepted: boolean): string {
   105→  if (accepted) return "glow-verified";

   113→/** Confidence gauge — radial progress ring */
   114→function ConfidenceGauge({
   115→  confidence,
   116→  accepted,
   117→  size = 52,
   118→}: {
   119→  confidence: number;
   120→  accepted: boolean;
   121→  size?: number;
   122→}) {
   123→  const pct = Math.round(confidence * 100);
   124→  const band =
   125→    confidence >= 0.8 ? "verified" : confidence >= 0.6 ? "repairing" : "muted";
   126→  const colorMap: Record<string, string> = {
   127→    verified: "var(--verified)",
   128→    repairing: "var(--repairing)",
   129→    muted: "var(--muted-foreground)",
   130→  };
   131→  const color = accepted ? "var(--verified)" : colorMap[band];
   132→  const textColorMap: Record<string, string> = {
   133→    verified: "text-verified",
   134→    repairing: "text-repairing",
   135→    muted: "text-muted-foreground",
   136→  };
   137→  const textColor = accepted ? "text-verified" : textColorMap[band];
   138→
   139→  const strokeWidth = 4;
   140→  const radius = (size - strokeWidth) / 2;
   141→  const circumference = 2 * Math.PI * radius;
   142→  const offset = circumference - (pct / 100) * circumference;
   143→
   144→  return (
   145→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   146→      <svg width={size} height={size} className="-rotate-90">
   147→        <circle
   148→          cx={size / 2}
   149→          cy={size / 2}
   150→          r={radius}
   151→          fill="none"
   152→          stroke="var(--muted)"
   153→          strokeWidth={strokeWidth}
   154→          opacity={0.3}
   155→        />
   156→        <motion.circle
   157→          cx={size / 2}
   158→          cy={size / 2}
   159→          r={radius}
   160→          fill="none"
   161→          stroke={color}
   162→          strokeWidth={strokeWidth}
   163→          strokeDasharray={circumference}
   164→          initial={{ strokeDashoffset: circumference }}
   165→          animate={{ strokeDashoffset: offset }}
   166→          transition={{ duration: 0.8, ease: "easeOut" }}
   167→          strokeLinecap="round"
   168→        />
   169→      </svg>
   170→      <div className="absolute inset-0 flex items-center justify-center">
   171→        <span className={cn("font-mono text-[11px] font-bold tabular-nums", textColor)}>
   172→          {pct}
   173→        </span>
   174→      </div>
   175→    </div>
   176→  );
   177→}

   425→export function InvariantMinerSection() {
   426→  const { toast } = useToast();
   427→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   428→  const [mined, setMined] = useState<MinedInvariantRow[]>([]);
   429→  const [loading, setLoading] = useState(true);
   430→  const [fetchError, setFetchError] = useState(false);
   431→  const [filterPolicyId, setFilterPolicyId] = useState<string>("all");
   432→  const [mining, setMining] = useState(false);
   433→  const [miningProgress, setMiningProgress] = useState(0);
   434→  const [patchingId, setPatchingId] = useState<string | null>(null);
   435→  const [driftSummary, setDriftSummary] = useState<DriftSummary | null>(null);
   436→  const [llmUsed, setLlmUsed] = useState<boolean | null>(null);
   437→  const [stats, setStats] = useState<StatsResponse | null>(null);
   438→
   439→  const policyMap = useMemo(() => {
   440→    const m: Record<string, PolicyRow> = {};
   441→    for (const p of policies) m[p.id] = p;
   442→    return m;
   443→  }, [policies]);
   444→
   445→  const policyName = (id: string | null, fallback?: string | null) =>
   446→    fallback ?? (id ? policyMap[id]?.name ?? "global" : "global");
   447→
   448→  // Load policies once.
   449→  useEffect(() => {
   450→    let cancelled = false;
   451→    fetch("/api/policies")
   452→      .then((r) => r.json())
   453→      .then((d: { policies: PolicyRow[] }) => {
   454→        if (cancelled) return;
   455→        setPolicies(d.policies ?? []);
   456→      })
   457→      .catch(() => {});
   458→    return () => {
   459→      cancelled = true;
   460→    };
   461→  }, []);
   462→
   463→  // Load stats for drift telemetry panel.
   464→  useEffect(() => {
   465→    let cancelled = false;
   466→    const load = () =>
   467→      fetch("/api/stats")
   468→        .then((r) => r.json())
   469→        .then((d: StatsResponse) => {
   470→          if (cancelled) return;
   471→          setStats(d);
   472→        })
   473→        .catch(() => {});
   474→    load();
   475→    const t = setInterval(load, POLL_MS);
   476→    return () => {
   477→      cancelled = true;
   478→      clearInterval(t);
   479→    };
   480→  }, []);
   481→
   482→  const loadMined = useCallback(async () => {
   483→    try {
   484→      const url =
   485→        filterPolicyId && filterPolicyId !== "all"
   486→          ? `/api/miner?policyId=${encodeURIComponent(filterPolicyId)}`
   487→          : "/api/miner";
   488→      const r = await fetch(url);
   489→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   490→      const d: { mined: MinedInvariantRow[] } = await r.json();
   491→      setMined(d.mined ?? []);
   492→      setFetchError(false);
   493→    } catch {
   494→      setFetchError(true);
   495→    } finally {
   496→      setLoading(false);
   497→    }
   498→  }, [filterPolicyId]);
   499→
   500→  useEffect(() => {
   501→    setLoading(true);
   502→    loadMined();
   503→    const t = setInterval(loadMined, POLL_MS);
   504→    return () => clearInterval(t);
   505→  }, [loadMined]);
   506→
   507→  const mine = useCallback(async () => {
   508→    setMining(true);
   509→    setMiningProgress(0);
   510→    // Simulate progress while mining
   511→    const progressInterval = setInterval(() => {
   512→      setMiningProgress((p) => Math.min(p + Math.random() * 15, 90));
   513→    }, 400);
   514→
   515→    try {
   516→      const body: { policyId?: string; useLLM: boolean } = { useLLM: true };
   517→      if (filterPolicyId && filterPolicyId !== "all") body.policyId = filterPolicyId;
   518→      const r = await fetch("/api/miner/mine", {
   519→        method: "POST",
   520→        headers: { "Content-Type": "application/json" },
   521→        body: JSON.stringify(body),
   522→      });
   523→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   524→      const d: { mined: MinedInvariantRow[]; llmUsed: boolean; driftSummary: DriftSummary } =
   525→        await r.json();
   526→      const fresh = d.mined ?? [];
   527→      setLlmUsed(d.llmUsed);
   528→      setDriftSummary(d.driftSummary ?? null);
   529→      setMined((prev) => {
   530→        const seen = new Set(prev.map((m) => m.id));
   531→        const merged = [...fresh.filter((m) => !seen.has(m.id)), ...prev];
   532→        return merged.sort((a, b) => b.confidence - a.confidence);
   533→      });
   534→      setFetchError(false);
   535→      setMiningProgress(100);
   536→      toast({
   537→        title: `Mined ${fresh.length} candidate${fresh.length === 1 ? "" : "s"}`,
   538→        description: d.llmUsed
   539→          ? "LLM augmented the statistical drift baseline."
   540→          : "Statistical fallback only (LLM unavailable).",
   541→      });
   542→    } catch (e) {
   543→      toast({
   544→        title: "Mining failed",
   545→        description: e instanceof Error ? e.message : "Unknown error",
   546→        variant: "destructive",
   547→      });
   548→    } finally {
   549→      clearInterval(progressInterval);
   550→      setTimeout(() => {
   551→        setMining(false);
   552→        setMiningProgress(0);
   553→      }, 600);
   554→    }
   555→  }, [filterPolicyId, toast]);
   556→
   557→  const patch = useCallback(
   558→    async (id: string, accepted: boolean) => {
   559→      setPatchingId(id);
   560→      try {
   561→        const r = await fetch("/api/miner", {
   562→          method: "PATCH",
   563→          headers: { "Content-Type": "application/json" },
   564→          body: JSON.stringify({ id, accepted }),
   565→        });
   566→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   567→        const d: { mined: MinedInvariantRow } = await r.json();
   568→        setMined((prev) =>
   569→          prev
   570→            .map((m) => (m.id === id ? { ...m, ...d.mined } : m))
   571→            .sort((a, b) => b.confidence - a.confidence),
   572→        );
   573→        toast({
   574→          title: accepted ? "Invariant hardened" : "Candidate rejected",
   575→          description: accepted
   576→            ? "Accepted invariants compile into verified enforcers."
   577→            : "Rejected — will not be promoted to an enforcer.",
   578→        });
   579→      } catch (e) {
   580→        toast({
