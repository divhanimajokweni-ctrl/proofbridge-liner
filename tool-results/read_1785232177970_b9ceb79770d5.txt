   302→function createGitEntry(workflow: typeof GIT_WORKFLOWS[0], hour: number, trigger: string, phase: string): GitActionLogEntry {
   303→  const statusRoll = Math.random();
   304→  let status: GitActionLogEntry['status'];
   305→  // P1: mostly success, P6: some failures (security testing), P5: some cancelled (node failure)
   306→  if (phase === 'P6' && statusRoll < 0.15) status = 'failure';

   308→  else if (statusRoll < 0.05) status = 'failure';
   309→  else status = 'success';
   310→
   311→  const duration = 30000 + Math.floor(Math.random() * 120000); // 30-150 seconds
   312→  const commitHash = `sha256:${Math.random().toString(16).slice(2, 10).padEnd(7, '0')}`;

   315→  const stepLogs = workflow.steps.map((step, i) => {

'✅ All steps passed' : status === 'failure' ? '❌ Pipeline failed — see step logs above' : '⏸️ Run cancelled (node failure recovery)'}`,

   438→      const packetLoss = hour < 27 ? 5 : hour < 30 ? 15 : hour < 33 ? 25 : 40;
   439→      m.cpu_pct = 40 + packetLoss * 0.3 + Math.random() * 10;
   440→      m.latency_p99_ms = 200 + packetLoss * 20 + Math.random() * 100;
   441→      m.circuit_breaker = packetLoss > 25 ? 'DEGRADED' : 'NORMAL';
   442→      m.air_state = packetLoss > 25 ? 'WARNING' : 'NORMAL';
   443→      m.facts_rejected += Math.floor(Math.random() * 3);
   444→      m.risk_score = 0.1 + packetLoss * 0.008 + Math.random() * 0.05;
   445→      break;
   446→
   447→    case 'P4':
   448→      const diskFill = hour < 39 ? 70 : hour < 42 ? 80 : hour < 45 ? 90 : 95;
   449→      m.cpu_pct = diskFill > 90 ? 85 : 50 + diskFill * 0.3 + Math.random() * 5;
   450→      m.ram_pct = diskFill > 90 ? 78 : 55 + diskFill * 0.2;
   451→      m.latency_p99_ms = diskFill > 90 ? 500 : 100 + diskFill * 5 + Math.random() * 30;
   452→      m.circuit_breaker = diskFill > 90 ? 'DEGRADED' : 'NORMAL';
   453→      m.air_state = diskFill > 90 ? 'WARNING' : 'NORMAL';
   454→      m.facts_rejected += Math.floor(Math.random() * 2);
   455→      m.risk_score = 0.1 + diskFill * 0.005 + Math.random() * 0.03;
   456→      break;
   457→
   458→    case 'P5':
   459→      m.cpu_pct = 45 + Math.random() * 15;
   460→      m.ram_pct = 48 + Math.random() * 10;
   461→      m.circuit_breaker = Math.random() < 0.3 ? 'RECOVERING' : 'NORMAL';
   462→      m.air_state = Math.random() < 0.3 ? 'RECOVERY' : 'NORMAL';
   463→      m.latency_p99_ms = 80 + Math.random() * 40;
   464→      m.facts_rejected += Math.floor(Math.random() * 2);
   465→      m.risk_score = 0.08 + Math.random() * 0.12;
   466→      break;
   467→
   468→    case 'P6':
   469→      m.cpu_pct = 40 + Math.random() * 15;
   470→      m.ram_pct = 42 + Math.random() * 8;
   471→      m.circuit_breaker = 'NORMAL'; // Spoofed payloads are quarantined, not degrading
   472→      m.air_state = 'NORMAL';
   473→      m.latency_p99_ms = 50 + Math.random() * 20;
   474→      const spoofedCount = 20 + Math.floor(Math.random() * 15);
   475→      m.spoofed_payloads_injected += spoofedCount;
   476→      m.spoofed_payloads_quarantined += spoofedCount; // ALL quarantined
   477→      m.facts_rejected += spoofedCount; // Rejected at Pass 2
   478→      m.risk_score = 0.05 + Math.random() * 0.03;
   479→      break;
   480→
   481→    case 'P7':
   482→      if (hour >= 66 && hour < 71) {
   483→        // Partition phase
   484→        m.circuit_breaker = 'DEGRADED';
   485→        m.air_state = 'WARNING';
   486→        m.facts_queued += 20;
   487→        m.risk_score = 0.2 + Math.random() * 0.1;
   488→      } else {
   489→        // Merge phase
   490→        m.circuit_breaker = 'NORMAL';
   491→        m.air_state = 'RECOVERY';
   492→        m.merge_count += 1;
   493→        m.merge_conflicts_observed = 0; // Zero conflicts — the claim
   494→        m.facts_merged += Math.floor(m.facts_queued * 0.3);
   495→        m.facts_queued = Math.floor(m.facts_queued * 0.7);
   496→        m.risk_score = 0.08 + Math.random() * 0.03;
   497→      }
   498→      m.cpu_pct = 40 + Math.random() * 10;
   499→      m.latency_p99_ms = 60 + Math.random() * 30;
   500→      break;
   501→  }
   502→
   503→  // Running counters
   504→  m.policy_violations += Math.floor(Math.random() * (phase === 'P6' ? 5 : phase === 'P3' ? 3 : 1));
   505→  m.policy_violations_handled = m.policy_violations; // All handled
   506→  m.proof_count += Math.floor(Math.random() * 10) + 5;
   507→  m.evidence_bundles_total += 1;
   508→  m.evidence_bundles_verified = m.evidence_bundles_total; // All verified
   509→  m.fact_log_checksum = `sha256:${m.facts_accepted.toString(16).padStart(8, '0')}${m.phase_id.toLowerCase().padEnd(8, '0')}`;
   510→  m.replay_checksum = m.fact_log_checksum; // MUST match — replay determinism
   511→  m.mmr_root = `sha256:${m.facts_accepted.toString(16).padStart(8, '0')}mmr${Math.floor(m.risk_score * 100).toString(16).padStart(4, '0')}`;
   512→
   513→  // AIR pipeline (separate from municipal breaker)
   514→  m.risk_score_smoothed = m.risk_score_smoothed * 0.8 + m.risk_score * 0.2;
   515→
   516→  // Validation Index computation
   517→  let index = 100;
   518→  if (m.facts_rejected > 0 && phase !== 'P6') index -= m.facts_rejected * 0.5;
   519→  if (m.merge_conflicts_observed > 0) index -= 10;
   520→  if (m.replay_status === 'DIVERGENT') index -= 20;
   521→  if (m.circuit_breaker === 'FAIL-CLOSED') index -= 15;
   522→  m.validation_index = Math.max(0, Math.min(100, Math.round(index * 100) / 100));
   523→
   524→  // P6 specific: spoofed payloads MUST all be quarantined
   525→  if (phase === 'P6') {
   526→    m.tee_status = 'ATTESTED'; // No mock boolean — real TEE attestation simulation
   527→    m.spoofed_payloads_quarantined = m.spoofed_payloads_injected; // 100% quarantine rate
   528→  }
   529→
   530→  return m;
   531→}

   534→// SIMULATION TICK LOOP
   535→// ════════════════════════════════════════════════════════════════════════
   536→
   537→function tick() {
   538→  if (!state.running || state.paused) return;
   539→
   540→  // Advance simulated time
   541→  const prevPhase = state.currentPhase;
   542→  state.currentHour += state.speedMultiplier / 3600; // speedMultiplier = simulated seconds per real second
   543→
   544→  // Check phase transition
   545→  const newPhase = computePhaseForHour(Math.floor(state.currentHour));
   546→  if (newPhase !== prevPhase) {
   547→    state.phaseHistory.push({ phase: prevPhase, enteredAt: state.currentHour, exitAt: state.currentHour });
   548→    state.currentPhase = newPhase;
   549→    console.log(`[SIM] Phase transition: ${prevPhase} → ${newPhase} at hour ${state.currentHour.toFixed(1)}`);
   550→  }
   551→
   552→  // Check milestones
   553→  for (const milestone of state.milestones) {
   554→    if (!milestone.triggered && Math.floor(state.currentHour) >= milestone.hour) {
   555→      milestone.triggered = true;
   556→      milestone.triggeredAt = Date.now();
   557→      console.log(`[SIM] Milestone ${milestone.id}: ${milestone.name} triggered`);
   558→    }
   559→  }
   560→
   561→  // Compute metrics
   562→  state.metrics = computeMetricsForTick(Math.floor(state.currentHour), state.currentPhase, state.metrics);
   563→
   564→  // Generate HBK telemetry
   565→  state.hbkTelemetry = generateHBKTelemetry(Math.floor(state.currentHour), state.currentPhase);
   566→
   567→  // Generate Git Actions
   568→  const newGitActions = generateGitActions(Math.floor(state.currentHour), state.currentPhase);
   569→  state.gitActions = [...state.gitActions.slice(-200), ...newGitActions]; // Keep last 200 entries
   570→
   571→  // Check end condition
   572→  if (state.currentHour >= 72) {
   573→    state.running = false;
   574→    state.currentHour = 72;
   575→    console.log('[SIM] 72-hour simulation complete');
   576→    io.emit('sim:complete', { finalMetrics: state.metrics, milestones: state.milestones });
   577→  }
   578→
   579→  // Broadcast to all connected clients
   580→  io.emit('sim:tick', {
   581→    hour: Math.floor(state.currentHour),
   582→    phase: state.currentPhase,
   583→    metrics: state.metrics,
   584→    hbkTelemetry: state.hbkTelemetry,
   585→    gitActions: state.gitActions.slice(-20), // Send last 20
   586→    milestones: state.milestones,
   587→    phaseHistory: state.phaseHistory,
   588→    running: state.running,
   589→    speedMultiplier: state.speedMultiplier,
   590→  });
   591→}
