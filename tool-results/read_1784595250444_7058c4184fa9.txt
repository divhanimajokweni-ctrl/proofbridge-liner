    43→function DriftSparkline({ events }: { events: ShadowEventRow[] }) {

   154→function DivergenceMeter({ divergence, authoritative, events }: { divergence: number; authoritative: boolean; events: ShadowEventRow[] }) {
   155→  const pct = Math.min(100, Math.round(Math.sqrt(divergence) * 35));
   156→  const armed = divergence > 0.0001 && authoritative;
   157→  const barClass = divergence < 0.001 ? "[&>div]:bg-verified" : divergence < 1 ? "[&>div]:bg-repairing" : "[&>div]:bg-violating";
   158→  const trend = events.length >= 2 ? events[0].divergence - events[1].divergence : 0;
   159→  const TrendIcon = Math.abs(trend) < 0.0001 ? Minus : trend > 0 ? TrendingUp : TrendingDown;
   160→  const trendColor = Math.abs(trend) < 0.0001 ? "text-muted-foreground" : trend > 0 ? "text-violating" : "text-verified";
   161→
   162→  return (
   163→    <GradientBorderCard gradientFrom={divergence < 0.001 ? "oklch(0.78 0.16 160 / 0.35)" : divergence < 1 ? "oklch(0.80 0.15 80 / 0.35)" : "oklch(0.64 0.21 25 / 0.4)"}
   164→      gradientTo="oklch(0.32 0.014 165 / 0.1)" className={cn(armed && "glow-violating")}>
   165→      <GridOverlay />
   166→      <div className="relative flex flex-col items-center gap-1.5 p-3 w-full">
   167→        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">Twin Divergence</div>
   168→        <motion.div className={cn("font-mono text-2xl font-semibold tabular-nums", divColor(divergence))}
   169→          initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }} transition={{ type: "spring", stiffness: 200, damping: 20 }}>{divergence.toFixed(3)}</motion.div>
   170→        <GaugeNeedle value={pct} max={100} size={100} />
   171→        <Progress value={pct} className={cn("h-1.5 w-full", barClass)} />
   172→        <div className={cn("flex items-center gap-1 text-[10px]", trendColor)}>
   173→          <TrendIcon className="h-3 w-3" /><span className="font-mono">{trend >= 0 ? "+" : ""}{trend.toFixed(4)}</span><span className="text-muted-foreground">trend</span>
   174→        </div>
   175→        <motion.div className={cn("flex items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide",
   176→          armed ? "border-violating/50 bg-violating/15 text-violating animate-epistemic-pulse" : divergence > 0.0001 ? "border-repairing/40 bg-repairing/10 text-repairing" : "border-verified/40 bg-verified/10 text-verified")}
   177→          initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.5 }}>
   178→          <Power className="h-3 w-3" />{armed ? "TAKEOVER ARMED" : divergence > 0.0001 ? "DRIFT" : "ALIGNED"}
   179→        </motion.div>
   180→        <div className="text-[9px] text-muted-foreground text-center leading-tight font-mono mt-0.5">{divergence < 0.001 ? "live ≡ shadow" : divergence < 1 ? "invariant mismatch" : "authoritative breach"}</div>
   181→      </div>
   182→    </GradientBorderCard>
   183→  );
   184→}
