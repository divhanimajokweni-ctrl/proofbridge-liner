    21→function synthesizeCircuit(policy: PolicyNode): Circuit {
    22→  const gates: CircuitGate[] = [], wires: CircuitWire[] = [];
    23→  const fields = new Set<string>();
    24→  for (const inv of policy.invariants) { if (inv.predicate) extractIdents(inv.predicate, fields); }
    25→  const privateInputs = Array.from(fields), publicInputs = ["mmr_root", "policy_hash", "nonce"];
    26→  const inputX = 40, constraintX = 220, aggregateX = 420, outputX = 560;
    27→  privateInputs.forEach((f, i) => {
    28→    const y = 50 + i * 42;
    29→    gates.push({ id: `in-${f}`, type: "input", label: f, sublabel: "private", x: inputX, y });
    30→    policy.invariants.forEach((inv) => { const invFields = new Set<string>(); if (inv.predicate) extractIdents(inv.predicate, invFields); if (invFields.has(f)) wires.push({ from: `in-${f}`, to: `c-${inv.name}`, kind: "private" }); });
    31→  });
    32→  policy.invariants.forEach((inv, j) => {
    33→    const y = 50 + j * 52;
    34→    gates.push({ id: `c-${inv.name}`, type: "constraint", label: inv.name, sublabel: inv.severity + (inv.soft ? " · soft" : ""), x: constraintX, y, invariant: inv.name });
    35→    wires.push({ from: `c-${inv.name}`, to: "agg-all", kind: "private" });
    36→  });
    37→  const aggY = 50 + (policy.invariants.length - 1) * 26;
    38→  gates.push({ id: "agg-all", type: "aggregate", label: "ALL(invariants)", sublabel: "conjunction", x: aggregateX, y: aggY });
    39→  gates.push({ id: "out-pass", type: "output", label: "pass", sublabel: "public", x: outputX, y: aggY - 20 });
    40→  gates.push({ id: "out-root", type: "output", label: "new_mmr_root", sublabel: "public", x: outputX, y: aggY + 20 });
    41→  wires.push({ from: "agg-all", to: "out-pass", kind: "public" }, { from: "agg-all", to: "out-root", kind: "public" });
    42→  publicInputs.forEach((p, i) => {
    43→    gates.push({ id: `pub-${p}`, type: "input", label: p, sublabel: "public", x: inputX, y: 50 + (privateInputs.length + i) * 42 + 20 });
    44→    wires.push({ from: `pub-${p}`, to: "agg-all", kind: "public" });
    45→  });
    46→  const constraintCount = policy.invariants.length + privateInputs.length * 2 + 4;
    47→  const estimatedRows = constraintCount * 8 + privateInputs.length * 3;
    48→  return { gates, wires, publicInputs, privateInputs, constraintCount, estimatedRows, provingKeySize: `${(estimatedRows * 0.12).toFixed(1)} MB`, verificationKeySize: `${(constraintCount * 0.04 + 0.8).toFixed(1)} KB` };
    49→}

    51→function extractIdents(expr: unknown, out: Set<string>) {
    52→  if (!expr || typeof expr !== "object") return;
    53→  const e = expr as Record<string, unknown>;
    54→  if (e.kind === "ident") out.add(e.name as string);
    55→  else if (e.kind === "unary") extractIdents(e.operand, out);
    56→  else if (e.kind === "binary" || e.kind === "logic" || e.kind === "compare") { extractIdents(e.left, out); extractIdents(e.right, out); }
    57→  else if (e.kind === "in") { extractIdents(e.value, out); extractIdents((e.range as unknown[])[0], out); extractIdents((e.range as unknown[])[1], out); }
    58→  else if (e.kind === "call") (e.args as unknown[]).forEach((a) => extractIdents(a, out));
    59→}

    61→function generateProofHistory(circuit: Circuit): ProofHistoryEntry[] {
    62→  const entries: ProofHistoryEntry[] = [], now = Date.now();
    63→  for (let i = 0; i < 8; i++) {
    64→    const status = i === 2 ? "failed" : i === 5 ? "pending" : "verified";
    65→    entries.push({ id: `proof-${i + 1}`, timestamp: new Date(now - (8 - i) * 180000).toISOString(), status, provingTimeMs: Math.round(800 + Math.random() * 2400), constraintCount: circuit.constraintCount + Math.round(Math.random() * 4 - 2), shardKey: `shard-${String.fromCharCode(65 + (i % 3))}` });
    66→  }
    67→  return entries;
    68→}

   545→function GateShape({ gate, revealPrivate, isHovered, onHover }: { gate: CircuitGate; revealPrivate: boolean; isHovered: boolean; onHover: (id: string | null) => void }) {
   546→  const w = 90, h = 30, x = gate.x, y = gate.y - h / 2;
   547→  const isPrivate = gate.sublabel === "private", hidden = isPrivate && !revealPrivate;
   548→  const config: Record<CircuitGate["type"], { fill: string; stroke: string; text: string }> = {
   549→    input: { fill: "oklch(0.22 0.014 168)", stroke: "oklch(0.45 0.01 165)", text: "oklch(0.85 0.01 165)" },
   550→    constraint: { fill: "oklch(0.25 0.03 80 / 0.6)", stroke: "oklch(0.80 0.15 80 / 0.7)", text: "oklch(0.90 0.10 80)" },
   551→    aggregate: { fill: "oklch(0.25 0.03 40 / 0.6)", stroke: "oklch(0.70 0.13 40 / 0.8)", text: "oklch(0.90 0.10 40)" },
   552→    output: { fill: "oklch(0.22 0.05 160 / 0.7)", stroke: "oklch(0.78 0.16 160)", text: "oklch(0.90 0.12 160)" },
   553→  };
   554→  const c = config[gate.type];
   555→  return (
   556→    <g onMouseEnter={() => onHover(gate.id)} onMouseLeave={() => onHover(null)} style={{ cursor: "pointer" }}>
   557→      <rect x={x} y={y} width={w} height={h} rx={4} fill={hidden ? "oklch(0.20 0.01 165)" : c.fill} stroke={hidden ? "oklch(0.35 0.02 165)" : c.stroke}
   558→        strokeWidth={isHovered ? 2 : 1.2} strokeDasharray={hidden ? "3 2" : "0"} filter={isHovered ? "url(#gate-glow)" : "url(#gate-shadow)"} />
   559→      {isHovered && <rect x={x - 2} y={y - 2} width={w + 4} height={h + 4} rx={5} fill="none" stroke={c.stroke} strokeWidth={0.5} opacity={0.5}><animate attributeName="opacity" values="0.5;0.1;0.5" dur="1.5s" repeatCount="indefinite" /></rect>}
   560→      <text x={x + w / 2} y={y + 13} textAnchor="middle" fontSize={9.5} fontFamily="var(--font-geist-mono), monospace" fill={hidden ? "oklch(0.50 0.02 165)" : c.text} fontWeight={gate.type === "output" || gate.type === "aggregate" ? 600 : 400}>{hidden ? "•••••" : gate.label.length > 14 ? gate.label.slice(0, 12) + "…" : gate.label}</text>
   561→      <text x={x + w / 2} y={y + 24} textAnchor="middle" fontSize={7} fill="oklch(0.55 0.01 165)">{gate.sublabel}</text>
   562→    </g>
   563→  );
