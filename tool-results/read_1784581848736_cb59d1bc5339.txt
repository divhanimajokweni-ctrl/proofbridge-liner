   151→function fmtRelative(iso: string | null): string {
   152→  if (!iso) return "—";
   153→  const t = new Date(iso).getTime();
   154→  const diff = Date.now() - t;
   155→  if (diff < 0) return "just now";
   156→  const s = Math.floor(diff / 1000);
   157→  if (s < 60) return `${s}s ago`;
   158→  const m = Math.floor(s / 60);
   159→  if (m < 60) return `${m}m ago`;
   160→  const h = Math.floor(m / 60);
   161→  if (h < 24) return `${h}h ago`;
   162→  return `${Math.floor(h / 24)}d ago`;
   163→}
   164→
   165→function fmtVal(v: unknown): string {
   166→  if (v === null) return "null";
   167→  if (typeof v === "object") return JSON.stringify(v);
   168→  return String(v);
   169→}

   171→/** Visualize a single applied repair: field: from → to with enhanced delta indicators */
   172→function AppliedDiff({ repair }: { repair: AppliedRepair }) {
   173→  const delta =
   174→    typeof repair.from === "number" && typeof repair.to === "number"
   175→      ? repair.to - repair.from
   176→      : null;
   177→  const isPositive = delta !== null && delta > 0;
   178→  const isNegative = delta !== null && delta < 0;
   179→  const isViolation = isNegative; // Decreases often represent violations being repaired
   180→
   181→  return (
   182→    <motion.div
   183→      variants={itemVariants}
   184→      className={cn(
   185→        "flex flex-col gap-1.5 rounded-md border p-2.5 transition-all hover:shadow-md hover:scale-[1.01]",
   186→        isViolation
   187→          ? "border-violating/40 bg-violating/5 hover:border-violating/60"
   188→          : isPositive
   189→            ? "border-verified/40 bg-verified/5 hover:border-verified/60"
   190→            : "border-border/60 bg-muted/20 hover:border-border",
   191→      )}
   192→    >
   193→      <div className="flex items-center justify-between gap-2">
   194→        <span className="font-mono text-xs text-foreground font-medium">{repair.field}</span>
   195→        <div className="flex items-center gap-1.5">
   196→          {delta !== null && (
   197→            <>
   198→              <motion.span
   199→                initial={{ scale: 0 }}
   200→                animate={{ scale: 1 }}
   201→                transition={{ type: "spring", stiffness: 400, damping: 20 }}
   202→                className={cn(
   203→                  "inline-flex items-center gap-0.5 rounded-sm px-1.5 py-0.5 text-[10px] font-mono font-semibold",
   204→                  isPositive
   205→                    ? "bg-verified/15 text-verified"
   206→                    : isNegative
   207→                      ? "bg-violating/15 text-violating"
   208→                      : "bg-muted text-muted-foreground",
   209→                )}
   210→              >
   211→                {isPositive ? <TrendingUp className="h-2.5 w-2.5" /> : isNegative ? <AlertTriangle className="h-2.5 w-2.5" /> : null}
   212→                {delta > 0 ? "+" : ""}
   213→                {delta.toFixed(3)}
   214→              </motion.span>
   215→            </>
   216→          )}
   217→        </div>
   218→      </div>
   219→      <div className="flex items-center gap-2 text-xs">
   220→        <span className={cn(
   221→          "font-mono px-1.5 py-0.5 rounded bg-violating/10 text-violating line-through decoration-violating/50",
   222→        )}>
   223→          {fmtVal(repair.from)}
   224→        </span>
   225→        <motion.div
   226→          initial={{ x: -4, opacity: 0 }}
   227→          animate={{ x: 0, opacity: 1 }}
   228→          transition={{ delay: 0.15 }}
   229→        >
   230→          <ArrowRight className={cn(
   231→            "h-3 w-3 shrink-0",
   232→            isViolation ? "text-violating" : "text-verified",
   233→          )} />
   234→        </motion.div>
   235→        <span className={cn(
   236→          "font-mono px-1.5 py-0.5 rounded bg-verified/10 text-verified",
   237→        )}>
   238→          {fmtVal(repair.to)}
   239→        </span>
   240→      </div>
   241→    </motion.div>
   242→  );
   243→}

   378→/** Repair cost bar chart showing adjustment costs */
   379→function RepairCostChart({ repair }: { repair: RepairResult }) {
   380→  const data = useMemo(() => {
   381→    if (!repair.applied || repair.applied.length === 0) return [];
   382→    return repair.applied.map((a, i) => {
   383→      const from = typeof a.from === "number" ? a.from : 0;
   384→      const to = typeof a.to === "number" ? a.to : 0;
   385→      const cost = Math.abs(to - from);
   386→      return {
   387→        name: a.field.length > 12 ? a.field.slice(0, 10) + "…" : a.field,
   388→        cost,
   389→        direction: to - from,
   390→      };
   391→    });
   392→  }, [repair]);
   393→
   394→  if (data.length === 0) return null;
   395→
   396→  return (
   397→    <motion.div
   398→      initial={{ opacity: 0, y: 8 }}
   399→      animate={{ opacity: 1, y: 0 }}
   400→      transition={{ type: "spring", stiffness: 260, damping: 24 }}
   401→      className="space-y-2"
   402→    >

   404→        <Activity className="h-3 w-3" />

   406→      </div>
   407→      <div className="h-28 w-full">
   408→        <ResponsiveContainer width="100%" height="100%">
   409→          <BarChart data={data} margin={{ top: 4, right: 4, bottom: 4, left: 4 }}>
   410→            <XAxis
   411→              dataKey="name"
   412→              tick={{ fontSize: 9, fill: "oklch(0.68 0.015 160)" }}

   783→/** Inline color-coded diff between proposed and repaired states */
   784→function InlineStateDiff({
   785→  proposed,
   786→  repaired,
   787→}: {
   788→  proposed: Record<string, unknown>;
   789→  repaired: Record<string, unknown>;
   790→}) {
   791→  const allKeys = useMemo(() => {
   792→    const keys = new Set<string>();
   793→    Object.keys(proposed ?? {}).forEach((k) => keys.add(k));
   794→    Object.keys(repaired ?? {}).forEach((k) => keys.add(k));
   795→    return Array.from(keys).sort();
   796→  }, [proposed, repaired]);
   797→
   798→  const diffs = useMemo(() => {
   799→    return allKeys
   800→      .map((key) => {
   801→        const fromVal = proposed?.[key];
   802→        const toVal = repaired?.[key];
   803→        const changed = JSON.stringify(fromVal) !== JSON.stringify(toVal);
   804→        const delta =
   805→          typeof fromVal === "number" && typeof toVal === "number" ? toVal - fromVal : null;
   806→        return { key, from: fromVal, to: toVal, changed, delta };
   807→      })
   808→      .filter((d) => d.changed);
   809→  }, [allKeys, proposed, repaired]);
   810→
   811→  if (diffs.length === 0) return null;
   812→
   813→  return (
   814→    <div>
   815→      <div className="mb-1.5 text-[10px] uppercase tracking-wide text-foreground/70 flex items-center gap-1.5">
   816→        <Eye className="h-3 w-3" />
   817→        State delta ({diffs.length} change{diffs.length === 1 ? "" : "s"})
   818→      </div>
   819→      <div className="grid grid-cols-1 gap-1 sm:grid-cols-2">
   820→        {diffs.map((d) => (
   821→          <div
   822→            key={d.key}
   823→            className={cn(
   824→              "flex items-center justify-between gap-2 rounded border px-2 py-1 text-[11px]",
   825→              d.delta !== null
   826→                ? d.delta > 0
   827→                  ? "border-verified/30 bg-verified/5"
   828→                  : "border-violating/30 bg-violating/5"
   829→                : "border-repairing/30 bg-repairing/5",
   830→            )}
   831→          >
   832→            <div className="flex items-center gap-1.5 min-w-0">
   833→              <span className="font-mono text-muted-foreground truncate">{d.key}</span>
   834→              <ArrowRight className="h-2.5 w-2.5 shrink-0 text-muted-foreground" />
   835→            </div>
   836→            <div className="flex items-center gap-1 shrink-0">
   837→              <span className={cn(
   838→                "font-mono line-through decoration-violating/50",
   839→                d.delta !== null && d.delta < 0 ? "text-violating" : "text-muted-foreground",
   840→              )}>
   841→                {fmtVal(d.from)}
   842→              </span>
   843→              <span className={cn(
   844→                "font-mono",
   845→                d.delta !== null
   846→                  ? d.delta > 0 ? "text-verified" : "text-violating"
   847→                  : "text-verified",
   848→              )}>
   849→                {fmtVal(d.to)}
   850→              </span>
   851→              {d.delta !== null && (
   852→                <Badge variant="outline" className={cn(
   853→                  "text-[8px] px-1 py-0 h-4",
   854→                  d.delta > 0
   855→                    ? "border-verified/30 text-verified"
   856→                    : "border-violating/30 text-violating",
   857→                )}>
   858→                  {d.delta > 0 ? "+" : ""}{d.delta.toFixed(3)}
   859→                </Badge>
   860→              )}
   861→            </div>
   862→          </div>
   863→        ))}
   864→      </div>
   865→    </div>
   866→  );
   867→}

   869→/** Summary statistics for the merge list */
   870→function MergeSummary({ merges }: { merges: MergeRow[] }) {
   871→  const stats = useMemo(() => {
   872→    const clean = merges.filter((m) => getMergeType(m) === "clean").length;
   873→    const repaired = merges.filter((m) => getMergeType(m) === "repaired").length;
   874→    const rejected = merges.filter((m) => getMergeType(m) === "rejected").length;
   875→    const avgDiv = merges.length > 0
   876→      ? merges.reduce((sum, m) => sum + m.divergence, 0) / merges.length
   877→      : 0;
   878→    const totalIters = merges.reduce((sum, m) => sum + m.iterations, 0);
   879→    return { clean, repaired, rejected, avgDiv, totalIters, total: merges.length };
   880→  }, [merges]);
   881→
   882→  return (
   883→    <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   884→      {[
   885→        { label: "Clean", value: stats.clean, color: "text-verified", bg: "bg-verified/10", border: "border-verified/30" },
   886→        { label: "Repaired", value: stats.repaired, color: "text-repairing", bg: "bg-repairing/10", border: "border-repairing/30" },
   887→        { label: "Rejected", value: stats.rejected, color: "text-violating", bg: "bg-violating/10", border: "border-violating/30" },
   888→        { label: "Avg div", value: stats.avgDiv.toFixed(3), color: "text-foreground", bg: "bg-muted/20", border: "border-border/60" },
   889→      ].map((item) => (
   890→        <div
   891→          key={item.label}
   892→          className={cn(
   893→            "rounded-md border px-2.5 py-2 text-center",
   894→            item.bg,
   895→            item.border,
   896→          )}
   897→        >
   898→          <div className="text-[9px] uppercase tracking-wide text-muted-foreground">{item.label}</div>
   899→          <div className={cn("mt-0.5 font-mono text-sm font-semibold", item.color)}>{item.value}</div>
   900→        </div>
   901→      ))}
   902→    </div>
   903→  );
