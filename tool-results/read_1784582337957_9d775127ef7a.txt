    92→    92→function fmtRelative(iso: string | null | undefined): string {
    93→    93→  if (!iso) return "—";
    94→    94→  const t = new Date(iso).getTime();
    95→    95→  const diff = Date.now() - t;
    96→    96→  if (diff < 0) return "just now";
    97→    97→  const s = Math.floor(diff / 1000);
    98→    98→  if (s < 60) return `${s}s ago`;
    99→    99→  const m = Math.floor(s / 60);
   100→   100→  if (m < 60) return `${m}m ago`;
   101→   101→  const h = Math.floor(m / 60);
   102→   102→  if (h < 24) return `${h}h ago`;
   103→   103→  return `${Math.floor(h / 24)}d ago`;
   104→   104→}
