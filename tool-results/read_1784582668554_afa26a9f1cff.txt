   192→    a.download = `${report.reportId}.json`;
   193→    a.click();

   195→    toast({ title: "Report downloaded", description: `${report.reportId}.json` });

   212→    ];
   213→    for (const p of report.policies) {
   214→      rows.push([
   215→        p.name,
   216→        p.domain ?? "",

   236→    a.download = `${report.reportId}.csv`;
   237→    a.click();
   238→    URL.revokeObjectURL(url);

   331→      {/* Header */}

   716→function SummaryStat({
   717→  icon: Icon,
   718→  label,
   719→  value,
   720→  accent,
   721→}: {
   722→  icon: typeof GitBranch;
   723→  label: string;
   724→  value: number;
   725→  accent?: "verified" | "repairing" | "violating";
   726→}) {
   727→  const color =
   728→    accent === "verified"
   729→      ? "text-verified"
   730→      : accent === "repairing"
   731→        ? "text-repairing"
   732→        : accent === "violating"
   733→          ? "text-violating"
   734→          : "text-foreground";
   735→  const gradient =
   736→    accent === "verified"
   737→      ? "from-verified/30 to-verified/10"
   738→      : accent === "repairing"
   739→        ? "from-repairing/30 to-repairing/10"
   740→        : accent === "violating"
   741→          ? "from-violating/30 to-violating/10"
   742→          : "from-border/30 to-border/10";
   743→
   744→  return (
   745→    <GradientBorderCard gradient={gradient} className="p-2.5">
   746→      <motion.div
   747→        className="relative"
   748→        whileHover={{ scale: 1.05 }}
   749→        whileTap={{ scale: 0.95 }}
   750→      >
   751→        <Icon className={cn("h-3.5 w-3.5", color)} />
   752→        <div className={cn("text-lg font-semibold tabular-nums leading-none mt-1", color)}>{value}</div>
   753→        <div className="text-[9px] uppercase tracking-wide text-muted-foreground mt-0.5">{label}</div>
   754→      </motion.div>
   755→    </GradientBorderCard>
   756→  );
   757→}

   781→function csvEscape(s: string): string {
   782→  if (s.includes(",") || s.includes('"') || s.includes("\n")) {
   783→    return `"${s.replace(/"/g, '""')}"`;
   784→  }
   785→  return s;
