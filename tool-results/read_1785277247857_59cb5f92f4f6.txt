   165→// SIMULATED GIT ACTIONS GENERATOR
   166→// ════════════════════════════════════════════════════════════════════════
   167→
   168→function generateGitActions(count: number): GitAction[] {
   169→  const actions: GitAction["action"][] = ["commit", "push", "merge", "deploy", "test", "validate"];
   170→  const branches = ["main", "hbk/mk-ii", "hbk/bayesian-engine", "hbk/sensor-cal", "hbk/cad-layout", "hbk/consortium"];
   171→  const messages = [
   172→    "feat: add AMD Ryzen AI compute module bounding box",
   173→    "fix: sensor isolation distance verified at X=20, Y=180",
   174→    "refactor: BMS power distribution routing updated",
   175→    "feat: NVMe storage bay vibration dampening mounts",
   176→    "chore: IP67 transit shell envelope clearance check",
   177→    "test: Bayesian inference engine MCMC convergence",
   178→    "feat: Comms routing node Cellular/GNSS integration",
   179→    "validate: 72h validation phase V3 metrics passing",
   180→    "deploy: HBK Mk-II chassis layout to FreeCAD workspace",
   181→    "merge: hbk/sensor-cal → main (acoustic filtering verified)",
   182→    "feat: VVU 100% ownership model embedded in metadata",
   183→    "fix: analog isolation clearance zone expanded",
   184→    "test: Kria SoM edge-compute integration",
   185→    "validate: Brier Score ≤ 0.02 threshold check",
   186→    "feat: consortium architecture hierarchy definition",
   187→    "chore: resource register initial data migration",
   188→  ];
   189→  const authors = ["eng-lead", "cad-operator", "bayesian-eng", "sensor-tech", "field-ops", "devops"];
   190→  const result: GitAction[] = [];
   191→  const now = Date.now();
   192→  for (let i = 0; i < count; i++) {
   193→    const action = actions[Math.floor(Math.random() * actions.length)];
   194→    const hash = Math.random().toString(16).slice(2, 9);
   195→    result.push({
   196→      id: `ga-${i}`,
   197→      timestamp: new Date(now - (count - i) * 180000 + Math.random() * 60000).toISOString(),
   198→      action,
   199→      branch: branches[Math.floor(Math.random() * branches.length)],
   200→      message: messages[Math.floor(Math.random() * messages.length)],
   201→      author: authors[Math.floor(Math.random() * authors.length)],
   202→      status: Math.random() > 0.08 ? "success" : Math.random() > 0.5 ? "running" : "failed",
   203→      hash,
   204→    });
   205→  }
   206→  return result;
   207→}

   218→  const [simElapsed, setSimElapsed] = useState(0);
   219→  const [hoveredModule, setHoveredModule] = useState<string | null>(null);
   220→  const simIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   221→
   222→  // Simulation timer
   223→  useEffect(() => {
   224→    if (simRunning) {
   225→      simIntervalRef.current = setInterval(() => {
   226→        setSimElapsed(prev => prev + 1);
   227→      }, 1000);
   228→    } else if (simIntervalRef.current) {
   229→      clearInterval(simIntervalRef.current);
   230→    }
   231→    return () => { if (simIntervalRef.current) clearInterval(simIntervalRef.current); };
   232→  }, [simRunning]);
   233→
