   233→# ─── TocDocTemplate (for TOC support) ───
   234→class TocDocTemplate(BaseDocTemplate):
   235→    """Custom DocTemplate that supports Table of Contents via afterFlowable.
   236→    Inherits from BaseDocTemplate so we can control page templates directly."""
   237→    def afterFlowable(self, flowable):
   238→        if hasattr(flowable, 'bookmark_name'):
   239→            level = getattr(flowable, 'bookmark_level', 0)
   240→            text = getattr(flowable, 'bookmark_text', '')
   241→            key = getattr(flowable, 'bookmark_key', '')
   242→            self.notify('TOCEntry', (level, text, self.page, key))
   243→
   244→
   245→# ─── Heading helper (with bookmark for TOC) ───
   246→def add_heading(text, style, level=0):
   247→    key = f'h_{hashlib.md5(text.encode()).hexdigest()[:8]}'
   248→    p = Paragraph(f'<a name="{key}"/>{text}', style)
   249→    p.bookmark_name = key
   250→    p.bookmark_level = level
   251→    p.bookmark_text = text
   252→    p.bookmark_key = key
   253→    return p

   270→# ─── Custom Flowables ───
   271→class GoldRule(Flowable):
   272→    """A thin gold horizontal rule for section emphasis."""
   273→    def __init__(self, width, thickness=1.5, color=VVU_GOLD):
   274→        Flowable.__init__(self)
   275→        self.width = width
   276→        self.thickness = thickness
   277→        self.color = color
   278→        self.height = thickness + 4
   279→
   280→    def draw(self):
   281→        self.canv.setStrokeColor(self.color)
   282→        self.canv.setLineWidth(self.thickness)
   283→        self.canv.line(0, 2, self.width, 2)

   288→    def __init__(self, width, label=''):
   289→        Flowable.__init__(self)
   290→        self.width = width

   299→            self.canv.setFont('Helvetica', 7)
