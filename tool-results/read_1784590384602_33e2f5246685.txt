   392→   392→export function TemplateLibrarySection({ onCreate }: { onCreate?: (source: string, filename: string) => void }) {
   393→   393→  const { toast } = useToast();
   394→   394→  const [selected, setSelected] = useState<Template | null>(null);
   395→   395→  const [editedSource, setEditedSource] = useState("");
   396→   396→  const [policyName, setPolicyName] = useState("");
   397→   397→  const [validation, setValidation] = useState<{ ok: boolean; invCount: number; errors: number } | null>(null);
   398→   398→  const [copied, setCopied] = useState(false);
   399→   399→  const [searchQuery, setSearchQuery] = useState("");
   400→   400→  const [activeCategory, setActiveCategory] = useState("all");
   401→   401→  const [deploying, setDeploying] = useState(false);
   402→   402→
   403→   403→  const openTemplate = (t: Template) => {
   404→   404→    setSelected(t);
   405→   405→    setEditedSource(t.source);
   406→   406→    const m = t.source.match(/policy "([^"]+)"/);
   407→   407→    setPolicyName(m ? m[1] : t.name);
   408→   408→    setValidation(null);
   409→   409→  };
   410→   410→
   411→   411→  const validate = () => {
   412→   412→    const result = validateEpd(editedSource);
   413→   413→    setValidation({
   414→   414→      ok: result.ok,
   415→   415→      invCount: result.invariantCount,
   416→   416→      errors: result.diagnostics.filter((d) => d.level === "error").length,
   417→   417→    });
   418→   418→    if (result.ok) {
   419→   419→      toast({ title: "Template valid", description: `${result.invariantCount} invariants compiled` });
   420→   420→    } else {
   421→   421→      toast({ title: "Validation failed", description: `${result.diagnostics.filter((d) => d.level === "error").length} error(s)`, variant: "destructive" });
   422→   422→    }
   423→   423→  };
   424→   424→
   425→   425→  const create = async () => {
   426→   426→    const result = validateEpd(editedSource);
   427→   427→    if (!result.ok) {
   428→   428→      toast({ title: "Cannot create — fix errors first", variant: "destructive" });
   429→   429→      return;
   430→   430→    }
   431→   431→    setDeploying(true);
   432→   432→    const filename = `${policyName.replace(/[^a-zA-Z0-9]+/g, "-").toLowerCase()}.epd`;
   433→   433→    try {
   434→   434→      const res = await fetch("/api/policies", {
   435→   435→        method: "POST",
   436→   436→        headers: { "Content-Type": "application/json" },
   437→   437→        body: JSON.stringify({ source: editedSource, filename }),
   438→   438→      });
   439→   439→      const d = await res.json();
   440→   440→      if (!res.ok) {
   441→   441→        toast({ title: "Create failed", description: d.error ?? "invalid", variant: "destructive" });
   442→   442→      } else {
   443→   443→        toast({ title: "Policy created", description: `${d.policy.name} saved & compiled` });
   444→   444→        setSelected(null);
   445→   445→        onCreate?.(editedSource, filename);
   446→   446→      }
   447→   447→    } catch (e) {
   448→   448→      toast({ title: "Create failed", description: String(e), variant: "destructive" });
   449→   449→    } finally {
   450→   450→      setDeploying(false);
   451→   451→    }
   452→   452→  };
   453→   453→
   454→   454→  const copy = async () => {
   455→   455→    try {
   456→   456→      await navigator.clipboard.writeText(editedSource);
   457→   457→      setCopied(true);
   458→   458→      setTimeout(() => setCopied(false), 1400);
   459→   459→      toast({ title: "Source copied" });
   460→   460→    } catch {
   461→   461→      toast({ title: "Copy failed", variant: "destructive" });
   462→   462→    }
   463→   463→  };
   464→   464→
   465→   465→  const filteredTemplates = useMemo(() => {
   466→   466→    return TEMPLATES.filter((t) => {
   467→   467→      const matchesCategory = activeCategory === "all" || t.category === activeCategory;
   468→   468→      const matchesSearch = !searchQuery ||
   469→   469→        t.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   470→   470→        t.domain.toLowerCase().includes(searchQuery.toLowerCase()) ||
   471→   471→        t.description.toLowerCase().includes(searchQuery.toLowerCase());
   472→   472→      return matchesCategory && matchesSearch;
   473→   473→    });
   474→   474→  }, [activeCategory, searchQuery]);
   475→   475→
   476→   476→  const totalInvariants = TEMPLATES.reduce((sum, t) => sum + t.invariants, 0);
   477→   477→  const domains = new Set(TEMPLATES.map((t) => t.domain)).size;
   478→   478→
   479→   479→  return (
   480→   480→    <section className="space-y-4">
   481→   481→      {/* Header */}
   482→   482→      <motion.div
   483→   483→        initial={{ opacity: 0, y: -8 }}
   484→   484→        animate={{ opacity: 1, y: 0 }}
   485→   485→        transition={{ duration: 0.4 }}
   486→   486→      >
   487→   487→        <Card className="bg-card/60 backdrop-blur border-border/60 p-4 relative overflow-hidden">
   488→   488→          <div className="absolute top-0 left-0 right-0 h-[2px] bg-gradient-to-r from-verified/0 via-verified/50 to-verified/0" />
   489→   489→          <div className="bg-grid absolute inset-0 opacity-20" />
   490→   490→          <div className="relative flex flex-wrap items-center gap-3">
   491→   491→            <div className="flex h-10 w-10 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 shrink-0">
   492→   492→              <Library className="h-5 w-5 text-verified" />
   493→   493→            </div>
   494→   494→            <div className="min-w-0 flex-1">
   495→   495→              <h2 className="text-base font-semibold flex items-center gap-2">
   496→   496→                Policy Template Library
   497→   497→                <Badge variant="outline" className="border-verified/30 bg-verified/10 text-verified text-[9px] font-mono">
   498→   498→                  {TEMPLATES.length} templates
   499→   499→                </Badge>
   500→   500→              </h2>
   501→   501→              <p className="text-xs text-muted-foreground mt-0.5">
   502→   502→                Start from a domain template — edit, validate, and compile a new `.epd` policy in one click.
   503→   503→              </p>
   504→   504→            </div>
   505→   505→            {/* Stats */}
   506→   506→            <div className="flex items-center gap-3 shrink-0">
   507→   507→              <div className="text-center">
   508→   508→                <div className="font-mono text-lg font-bold text-verified">{totalInvariants}</div>
