    76→    76→const ANSI_CLASS: Record<string, string> = {
    77→    77→  "0": "", // reset
    78→    78→  "1": "font-bold",
    79→    79→  "2": "opacity-60",
    80→    80→  "31": "text-violating",
    81→    81→  "32": "text-verified",
    82→    82→  "33": "text-repairing",
    83→    83→  "34": "text-chart-2", // blue → teal (design system forbids blue)
    84→    84→  "35": "text-chart-5", // magenta → warm violet
    85→    85→  "36": "text-chart-2", // cyan → teal
    86→    86→  "90": "text-muted-foreground", // bright black / gray
    87→    87→};

    89→    89→/**
    90→    90→ * Render ANSI-colored text (SGR codes only) as a sequence of styled spans.
    91→    91→ * Splits on `\x1b[...m` escape sequences and applies the accumulated class
    92→    92→ * set to each text segment. Handles combined codes like `\x1b[1;36m`.
    93→    93→ */
    94→    94→function renderAnsi(text: string): React.ReactNode[] {
    95→    95→  if (!text) return [];
    96→    96→  const parts = text.split(/\x1b\[(\d+(?:;\d+)*)m/);
    97→    97→  const nodes: React.ReactNode[] = [];
    98→    98→  let active = new Set<string>();
    99→    99→  let key = 0;
   100→   100→  for (let i = 0; i < parts.length; i++) {
   101→   101→    const part = parts[i];
   102→   102→    if (i % 2 === 1) {
   103→   103→      // escape code segment (one or more ;-separated codes)
   104→   104→      for (const code of part.split(";")) {
   105→   105→        if (code === "0" || code === "") {
   106→   106→          active = new Set();
   107→   107→        } else {
   108→   108→          const cls = ANSI_CLASS[code];
   109→   109→          if (cls) active.add(cls);
   110→   110→        }
   111→   111→      }
   112→   112→    } else if (part) {
   113→   113→      const cls = Array.from(active).join(" ");
   114→   114→      // Preserve whitespace by wrapping in a span; split newlines so React
   115→   115→      // renders them within the pre-formatted container.
   116→   116→      nodes.push(
   117→   117→        <span key={key++} className={cls || undefined}>
   118→   118→          {part}
   119→   119→        </span>,
   120→   120→      );
   121→   121→    }
   122→   122→  }
   123→   123→  return nodes;
   124→   124→}
