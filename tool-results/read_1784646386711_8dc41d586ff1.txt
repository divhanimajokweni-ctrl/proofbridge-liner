    72→    72→function ThemeToggleBtn() {

    78→    78→  return (
    79→    79→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ?     85→    85→function ScrollToTopButton() {
    86→    86→  const [visible, setVisible] = useState(false);
    87→    87→
    88→    88→  useEffect(() => {
    89→    89→    const onScroll = () => setVisible(window.scrollY > 300);
    90→    90→    window.addEventListener("scroll", onScroll, { passive: true });
    91→    91→    return () => window.removeEventListener("scroll", onScroll);
    92→    92→  }, []);
    93→    93→
    94→    94→  const scrollToTop = useCallback(() => {
    95→    95→    window.scrollTo({ top: 0, behavior: "smooth" });
    96→    96→  }, []);
    97→    97→
    98→    98→  return (
    99→    99→    <AnimatePresence>
   100→   100→      {visible && (
   101→   101→        <motion.button
   102→   102→          type="button"
   103→   103→          onClick={scrollToTop}
   104→   104→          initial={{ opacity: 0, scale: 0.8 }}

   406→   406→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   407→   407→  const meta = SECTION_META[sectionId];

   412→   412→  useEffect(() => {

   421→   421→    fetch("/api/stats").then(() => {
   422→   422→      setIsRefreshing(false);
   423→   423→    }).catch(() => {
   424→   424→      setIsRefreshing(false);
   425→   425→    });
   426→   426→  }, []);

   428→   428→  return (
   429→   429→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   430→   430→      {meta.stats.map((stat, i) => (
