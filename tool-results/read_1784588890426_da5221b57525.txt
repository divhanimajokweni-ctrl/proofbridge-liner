    64→function generateSparkline(base: number, variance: number, length = 12): number[] {
    65→  let seed = base * 7 + 31;
    66→  const rand = () => {
    67→    seed = (seed * 1103515245 + 12345) & 0x7fffffff;
    68→    return seed / 0x7fffffff;
    69→  };
    70→  return Array.from({ length }, () =>
    71→    Math.max(0, Math.round((base + (rand() - 0.5) * 2 * variance) * 10) / 10)
    72→  );
    73→}
