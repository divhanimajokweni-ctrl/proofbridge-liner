    70→function GaugeNeedle({ value, max, size = 80 }: { value: number; max: number; size?: number }) {

    74→  return (
    75→    <svg width={size} height={size / 2 + 6} viewBox="0 0 80 46" className="overflow-visible">

   189→function DivergenceMeter({ divergence, authoritative, events }: { divergence: number; authoritative: boolean; events: ShadowEventRow[] }) {
   190→  const pct = Math.min(100, Math.round(Math.sqrt(divergence) * 35));
   191→  const armed = divergence > 0.0001 && authoritative;
   192→  const barClass = divergence < 0.001 ? "[&>div]:bg-verified" : divergence < 1 ? "[&>div]:bg-repairing" : "[&>div]:bg-violating";
   193→  const trend = events.length >= 2 ? events[0].divergence - events[1].divergence : 0;
   194→  const TrendIcon = Math.abs(trend) < 0.0001 ? Minus : trend > 0 ? TrendingUp : TrendingDown;
   195→  const trendColor = Math.abs(trend) < 0.0001 ? "text-muted-foreground" : trend > 0 ? "text-violating" : "text-verified";
   196→
   197→  return (
   198→    <GradientBorderCard gradientFrom={divergence < 0.001 ? "oklch(0.78 0.16 160 / 0.35)" : divergence < 1 ? "oklch(0.80 0.15 80 / 0.35)" : "oklch(0.64 0.21 25 / 0.4)"}
   199→      gradientTo="oklch(0.32 0.014 165 / 0.1)" className={cn(armed && "glow-violating")}>
   200→      <div className="bg-grid-fine absolute inset-0 opacity-20 rounded-lg" />
   201→      <div className="relative flex flex-col items-center gap-1.5 p-3 w-full">
   202→        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">Twin Divergence</div>
   203→        <motion.div className={cn("font-mono text-2xl font-semibold tabular-nums", divColor(divergence))}
   204→          initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }}
   205→          transition={{ type: "spring", stiffness: 200, damping: 20 }}>
   206→          {divergence.toFixed(3)}
   207→        </motion.div>
   208→        <GaugeNeedle value={pct} max={100} size={100} />
   209→        <Progress value={pct} className={cn("h-1.5 w-full", barClass)} />
   210→        <div className={cn("flex items-center gap-1 text-[10px]", trendColor)}>
   211→          <TrendIcon className="h-3 w-3" />
   212→          <span className="font-mono">{trend >= 0 ? "+" : ""}{trend.toFixed(4)}</span>
   213→          <span className="text-muted-foreground">trend</span>
   214→        </div>
   215→        <motion.div className={cn("flex items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide",
   216→          armed ? "border-violating/50 bg-violating/15 text-violating animate-epistemic-pulse" : divergence > 0.0001 ? "border-repairing/40 bg-repairing/10 text-repairing" : "border-verified/40 bg-verified/10 text-verified")}
   217→          initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.5 }}>
   218→          <Power className="h-3 w-3" />{armed ? "TAKEOVER ARMED" : divergence > 0.0001 ? "DRIFT" : "ALIGNED"}
   219→        </motion.div>
   220→        <div className="text-[9px] text-muted-foreground text-center leading-tight font-mono mt-0.5">
   221→          {divergence < 0.001 ? "live ≡ shadow" : divergence < 1 ? "invariant mismatch" : "authoritative breach"}
   222→        </div>
   223→      </div>
   224→    </GradientBorderCard>
   225→  );
   226→}

   333→
   334→  useEffect(() => {

   358→
   359→  useEffect(() => {
   360→    if (!policyId) return;
   361→    setLoading(true);

   364→    return () => clearInterval(t);

   366→
   367→  const doTakeover = useCallback(async () => {

   369→    setTakingOver(true);

   371→      const r = await fetch("/api/shadow-bridge/takeover", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ policyId }) });
   372→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   373→      const d = await r.json();
   374→      toast({ title: "Shadow takeover initiated", description: d.event?.summary ?? "Shadow bridge assumed authority." });
   375→      setTakeoverOpen(false);
   376→      loadBridge();

   378→      toast({ title: "Takeover failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" });
   379→    } finally { setTakingOver(false); }
   380→  }, [policyId, toast, loadBridge]);
   381→
   382→  const runWhatif = useCallback(async () => {
   383→    if (!policyId) return;
   384→    let delta: Record<string, unknown> = {};

   386→      try { delta = JSON.parse(deltaText); } catch (e) {
   387→        toast({ title: "Invalid state delta JSON", description: e instanceof Error ? e.message : "Parse error", variant: "destructive" });
   388→        return;
   389→      }
   390→    }
   391→    setRunningWhatif(true);
   392→    try {
   393→      const body: Record<string, unknown> = { policyId, stateDelta: delta };
   394→      if (predicate.trim()) body.predicateOverride = predicate.trim();
   395→      const r = await fetch("/api/shadow-bridge/whatif", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   396→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   397→      const d = await r.json();
   398→      setWhatif({
   399→        ok: !!d.ok,
   400→        evaluations: (d.evaluations ?? []).map((e: InvariantEval) => ({ name: e.name, passed: e.passed, severity: e.severity, soft: e.soft, actual: e.actual })),
   401→        mmrRoot: d.mmrRoot ?? "—",
   402→        divergence: d.event?.divergence ?? 0,
   403→      });
   404→      toast({ title: "What-if replay complete", description: `${(d.evaluations ?? []).filter((e: InvariantEval) => !e.passed).length} violation(s) under modified rules` });
   405→      loadBridge();
   406→    } catch (e) {
   407→      toast({ title: "What-if replay failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" });
   408→    } finally { setRunningWhatif(false); }
   409→  }, [policyId, deltaText, predicate, toast, loadBridge]);
   410→
   411→  const invariantUnion = useMemo(() => {
   412→    if (!bridge) return [] as { name: string; live?: InvariantEval; shadow?: InvariantEval }[];
   413→    const map = new Map<string, { name: string; live?: InvariantEval; shadow?: InvariantEval }>();
   414→    for (const iv of bridge.liveInvariants) map.set(iv.name, { name: iv.name, live: iv });
   415→    for (const iv of bridge.shadowInvariants) {
   416→      const ex = map.get(iv.name);
   417→      if (ex) ex.shadow = iv; else map.set(iv.name, { name: iv.name, shadow: iv });
   418→    }
   419→    return Array.from(map.values());
   420→  }, [bridge]);
   421→
   422→  const events = bridge?.events ?? [];
   423→  const selectedPolicy = policies.find((p) => p.id === policyId);
   424→  const shadowEnabled = !!selectedPolicy?.shadowEnabled;
   425→  const takeoverLatency = bridge?.policy.takeoverLatencyMs ?? selectedPolicy?.takeoverLatencyMs ?? null;
   426→  const authoritative = bridge?.policy.authoritative ?? false;
   427→  const divergence = bridge?.divergence ?? 0;
   428→
   429→  return (
   430→    <motion.section variants={containerVariants} initial="hidden" animate="visible" className="space-y-4">
   431→      <motion.div variants={cardVariants} className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   432→        <SectionHeader icon={Cpu} title="Shadow Bridge" subtitle="Digital-twin control plane · live vs shadow invariants · takeover-ready" />
   433→        <div className="flex items-center gap-2 flex-wrap">
   434→          <BridgeStatusIndicator shadowEnabled={shadowEnabled} authoritative={authoritative} divergence={divergence} />
   435→          <Select value={policyId} onValueChange={setPolicyId}>
   436→            <SelectTrigger className="w-[220px] sm:w-[260px] bg-card/60"><SelectValue placeholder="Select policy" /></SelectTrigger>
   437→            <SelectContent>
   438→              {policies.map((p) => (
   439→                <SelectItem key={p.id} value={p.id}>
   440→                  <span className="flex items-center gap-2">{p.name}{p.shadowEnabled && <span className="text-[10px] text-quarantined">●shadow</span>}</span>
   441→                </SelectItem>
   442→              ))}
   443→            </SelectContent>
   444→          </Select>
   445→        </div>
   446→      </motion.div>
   447→
   448→      <motion.div variants={cardVariants} className="flex flex-wrap items-center gap-2">
   449→        <Badge variant="outline" className={cn("border-border/60 transition-colors hover:border-quarantined/40", shadowEnabled ? "text-quarantined border-quarantined/40 bg-quarantined/10" : "text-muted-foreground")}>
   450→          <Moon className="h-3 w-3" />{shadowEnabled ? "shadow enabled" : "shadow disabled"}
   451→        </Badge>
   452→        <Badge variant="outline" className={cn("border-border/60 transition-colors hover:border-violating/40", authoritative ? "text-violating border-violating/40 bg-violating/10" : "text-muted-foreground")}>
   453→          <ShieldAlert className="h-3 w-3" />{authoritative ? "authoritative" : "non-authoritative"}
   454→        </Badge>
   455→        <Badge variant="outline" className="border-border/60 text-muted-foreground"><Gauge className="h-3 w-3" />takeover {takeoverLatency ?? "—"}ms</Badge>
   456→        <Badge variant="outline" className="border-border/60 text-muted-foreground"><Activity className="h-3 w-3" />{events.length} event{events.length === 1 ? "" : "s"}</Badge>
   457→      </motion.div>
   458→
   459→      {!shadowEnabled && (
   460→        <motion.div variants={cardVariants}>
   461→          <Card className="bg-repairing/5 border-repairing/30 p-3 text-xs text-repairing flex items-center gap-2">
   462→            <AlertTriangle className="h-4 w-4 shrink-0" />
   463→            <span>Selected policy does not have a shadow bridge enabled. Pick a shadow-enabled policy for full functionality.</span>
   464→          </Card>
   465→        </motion.div>
   466→      )}
   467→
   468→      <motion.div variants={cardVariants}>
   469→        <GradientBorderCard gradientFrom="oklch(0.80 0.15 80 / 0.3)" gradientTo="oklch(0.32 0.014 165 / 0.1)">
   470→          <div className="p-3">
   471→            <div className="flex items-center justify-between mb-2">
   472→              <div className="flex items-center gap-2"><BarChart3 className="h-4 w-4 text-repairing" /><span className="text-sm font-semibold text-foreground">Drift Monitoring</span></div>
   473→              <Badge variant="outline" className="border-repairing/30 bg-repairing/10 text-repairing text-[10px]">live</Badge>
   474→            </div>
   475→            <DriftSparkline events={events} />
   476→            <div className="flex items-center justify-between mt-1.5 text-[10px] text-muted-foreground">
   477→              <span className="font-mono">last 20 events</span><span className="font-mono">divergence over time</span>
   478→            </div>
   479→          </div>
   480→        </GradientBorderCard>
   481→      </motion.div>
   482→
   483→      {loading ? (
   484→        <div className="grid grid-cols-1 gap-3 md:grid-cols-[1fr_auto_1fr]">
   485→          <Skeleton className="h-80 w-full rounded-lg bg-muted/40" />
   486→          <Skeleton className="h-80 w-32 rounded-lg bg-muted/40" />
   487→          <Skeleton className="h-80 w-full rounded-lg bg-muted/40" />
   488→        </div>
   489→      ) : !bridge ? (
   490→        <motion.div variants={cardVariants}>
   491→          <Card className="bg-card/60 border-border/60 p-8 text-center">
   492→            <Moon className="mx-auto h-8 w-8 text-muted-foreground/50" />
   493→            <p className="mt-2 text-sm text-muted-foreground">No shadow bridge data for this policy.</p>
   494→          </Card>
   495→        </motion.div>
   496→      ) : (
   497→        <div className="grid grid-cols-1 gap-3 md:grid-cols-[1fr_180px_1fr] md:items-stretch">
   498→          <motion.div variants={cardVariants}><TwinCard side="live" state={bridge.liveState} invariants={bridge.liveInvariants} /></motion.div>
   499→          <motion.div variants={cardVariants} className="flex md:flex-col gap-3">
   500→            <DivergenceMeter divergence={bridge.divergence} authoritative={authoritative} events={events} />
   501→            <Card className="bg-card/40 border-border/60 p-2.5 relative overflow-hidden flex-1">
   502→              <div className="bg-grid-fine absolute inset-0 opacity-20" />
   503→              <div className="relative">
   504→                <div className="mb-1.5 text-[10px] uppercase tracking-wide text-muted-foreground">L vs S matrix</div>
   505→                <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-0.5">
   506→                  {invariantUnion.length === 0 && <div className="text-[11px] text-muted-foreground italic">no invariants</div>}
   507→                  {invariantUnion.map((row) => <InvariantRow key={row.name} live={row.live} shadow={row.shadow} />)}
   508→                </motion.div>
   509→              </div>
   510→            </Card>
   511→          </motion.div>
   512→          <motion.div variants={cardVariants}><TwinCard side="shadow" state={bridge.shadowState} invariants={bridge.shadowInvariants} authoritative={authoritative} /></motion.div>
   513→        </div>
   514→      )}
   515→
   516→      <motion.div variants={containerVariants} className="grid grid-cols-1 gap-3 lg:grid-cols-3">
   517→        <motion.div variants={cardVariants}>
   518→          <GradientBorderCard gradientFrom="oklch(0.64 0.21 25 / 0.3)" gradientTo="oklch(0.32 0.014 165 / 0.1)">
   519→            <div className="bg-grid-fine absolute inset-0 opacity-20 rounded-lg" />
   520→            <div className="relative space-y-3 p-4">
   521→              <div className="flex items-center gap-2"><Power className="h-4 w-4 text-violating" /><span className="text-sm font-semibold text-foreground">Authority transfer</span></div>
   522→              <p className="text-[11px] text-muted-foreground">
   523→                Promote shadow twin to authoritative control within <span className="font-mono text-foreground">{takeoverLatency ?? "—"}ms</span>.

   537→                      <Eye className="h-3 w-3" />monitoring

   545→                    <ShieldAlert className="h-4 w-4" />Initiate takeover

   570→            <div className="bg-grid-fine absolute inset-0 opacity-20 rounded-lg" />
   571→            <div className="relative space-y-3 p-4">
   572→              <div className="flex items-center justify-between">
   573→                <div className="flex items-center gap-2"><FlaskConical className="h-4 w-4 text-repairing" /><span className="text-sm font-semibold text-foreground">What-if replay</span></div>

   587→                    {runningWhatif ? <RotateCw className="h-4 w-4 animate-spin" /> : <Play className="h-4 w-4" />}
   588→                    {runningWhatif ? "Replaying…" : "Run what-if"}
   589→                  </Button>
   590→                </div>
   591→              </div>
   592→              <AnimatePresence>{whatif && <WhatIfDiff whatif={whatif} />}</AnimatePresence>
   593→            </div>
   594→          </GradientBorderCard>
   595→        </motion.div>
   596→      </motion.div>
   597→
   598→      <motion.div variants={cardVariants}>
   599→        <GradientBorderCard gradientFrom="oklch(0.70 0.13 40 / 0.25)" gradientTo="oklch(0.32 0.014 165 / 0.1)">
   600→          <div className="bg-grid-fine absolute inset-0 opacity-15 rounded-lg" />
   601→          <div className="relative p-4">
   602→            <div className="mb-3 flex items-center justify-between">
   603→              <div className="flex items-center gap-2"><Activity className="h-4 w-4 text-quarantined" /><span className="text-sm font-semibold text-foreground">Event timeline</span>
   604→                <Badge variant="outline" className="border-border/60 text-[10px]">{events.length}</Badge>
   605→              </div>
   606→              <div className="hidden sm:flex items-center gap-2 text-[10px] text-muted-foreground flex-wrap">

   610→                })}
   611→              </div>
   612→            </div>
   613→            {events.length === 0 ? (
   614→              <div className="flex flex-col items-center justify-center py-8 text-center">
   615→                <Clock className="h-6 w-6 text-muted-foreground/50" /><p className="mt-2 text-xs text-muted-foreground">No shadow events recorded.</p>
   616→              </div>
   617→            ) : (
   618→              <motion.div variants={containerVariants} initial="hidden" animate="visible" className="max-h-80 overflow-y-auto epistemic-scroll space-y-2 pr-1">
   619→                {events.map((ev) => <EventTimelineItem key={ev.id} ev={ev} />)}
   620→              </motion.div>
   621→            )}
   622→          </div>
   623→        </GradientBorderCard>
   624→      </motion.div>
   625→    </motion.section>
   626→  );
