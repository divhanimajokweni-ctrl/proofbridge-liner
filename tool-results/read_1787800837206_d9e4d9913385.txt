    61→
    62→interface ComputeResponse {
    63→  verdict: Omit<EISVerdict, 'observations'>;
    64→  pipeline: PipelinePass[];
    65→  observations: CorrelatedObservation[];
    66→  auditHash: string;
    67→  auditShortHash: string;
    68→  generatedAtUtc: string;

    76→// ─── Page ────────────────────────────────────────────────────────────────
    77→
    78→export default function Home() {
    79→  const [view, setView] = useState<View>('eis');
    80→  const [calibration, setCalibration] = useState<DMACalibration>(DEFAULT_CALIBRATION);
    81→  const [stepIdx, setStepIdx] = useState(0); // start at BASELINE (index 0)
    82→  const [playing, setPlaying] = useState(false);
    83→  const [pumpSim, setPumpSim] = useState(false); // false-positive toggle
    84→  const [resp, setResp] = useState<ComputeResponse | null>(null);
    85→  const [loading, setLoading] = useState(false);
    86→  const [error, setError] = useState<string | null>(null);
    87→
    88→  const currentStep: ReplayStep = REPLAY_STEPS[stepIdx] ?? REPLAY_STEPS[1];
    89→
    90→  // Compute EIS via API whenever calibration, replay step, or pump sim changes.
    91→  const compute = useCallback(async () => {
    92→    setLoading(true);
    93→    setError(null);
    94→    try {
    95→      const res = await fetch('/api/evidence/compute', {
    96→        method: 'POST',
    97→        headers: { 'Content-Type': 'application/json' },
    98→        body: JSON.stringify({
    99→          calibration,
   100→          includeAnomaly: currentStep.includeAnomaly,
   101→          includeField: currentStep.includeField,
   102→          includeAcoustic: currentStep.includeAcoustic,
   103→          includeContext: currentStep.includeContext,
   104→          pumpStateChanged: pumpSim,
   105→        }),
   106→      });
   107→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   108→      const data = (await res.json()) as ComputeResponse;
   109→      setResp(data);
   110→    } catch (e) {
   111→      setError(e instanceof Error ? e.message : 'Compute failed');
   112→    } finally {
   113→      setLoading(false);
   114→    }
   115→  }, [calibration, currentStep, pumpSim]);
   116→
   117→  useEffect(() => {
   118→    void compute();
   119→  }, [compute]);
   120→
   121→  // Auto-play the 10-step replay
   122→  useEffect(() => {
   123→    if (!playing) return;
   124→    const id = setInterval(() => {
   125→      setStepIdx((prev) => {
   126→        if (prev >= REPLAY_STEPS.length - 1) {
   127→          setPlaying(false);
   128→          return prev;
   129→        }
   130→        return prev + 1;
   131→      });
   132→    }, 1800);
   133→    return () => clearInterval(id);
   134→  }, [playing]);
   135→
   136→  const step = (delta: number) => {
   137→    setStepIdx((prev) =>
   138→      Math.max(0, Math.min(REPLAY_STEPS.length - 1, prev + delta)),
   139→    );
   140→  };
   141→
   142→  const reset = () => {
   143→    setPlaying(false);
   144→    setStepIdx(0);
   145→    setPumpSim(false);
   146→    setCalibration(DEFAULT_CALIBRATION);
   147→  };
   148→
   149→  const exportAudit = async () => {
   150→    const res = await fetch('/api/evidence/audit', {
   151→      method: 'POST',
   152→      headers: { 'Content-Type': 'application/json' },
   153→      body: JSON.stringify({
   154→        calibration,
   155→        includeAnomaly: currentStep.includeAnomaly,
   156→        includeField: currentStep.includeField,
   157→        includeAcoustic: currentStep.includeAcoustic,
   158→        includeContext: currentStep.includeContext,
   159→        pumpStateChanged: pumpSim,
   160→      }),
   161→    });
   162→    const blob = await res.blob();
   163→    const url = URL.createObjectURL(blob);
   164→    const a = document.createElement('a');
   165→    a.href = url;
   166→    a.download = `leak_candidate_audit_${DMA_ID}.json`;
   167→    document.body.appendChild(a);
   168→    a.click();
   169→    a.remove();
   170→    URL.revokeObjectURL(url);
   171→  };
   172→
   173→  return (
   174→    <div className="kernel-theme min-h-screen flex flex-col k-grid-bg">
   175→      <HeaderBar
   176→        view={view}
   177→        onViewChange={setView}
   178→        stepIdx={stepIdx}
   179→        totalSteps={REPLAY_STEPS.length}
   180→        auditShortHash={resp?.auditShortHash ?? '——————'}
   181→        loading={loading}
   182→      />
   183→
   184→      <main className="flex-1 px-3 sm:px-6 pb-8 max-w-[1600px] w-full mx-auto">
   185→        {view === 'hbk' ? (
   186→          <HBKPanel />
   187→        ) : (
   188→          <>
   189→            {/* Replay timeline */}
   190→            <ReplayTimeline
   191→              steps={REPLAY_STEPS}
   192→              currentIdx={stepIdx}
   193→              onSelect={(i) => setStepIdx(i)}
   194→            />
   195→
   196→            {/* Control bar */}
   197→            <ControlBar
   198→              playing={playing}
   199→              onPlay={() => setPlaying(true)}
   200→              onPause={() => setPlaying(false)}
   201→              onStepFwd={() => step(1)}
   202→              onStepBack={() => step(-1)}
   203→              onReset={reset}
   204→              pumpSim={pumpSim}
   205→              onPumpSimChange={setPumpSim}
   206→              onExport={exportAudit}
   207→              currentStep={currentStep}
   208→            />
   209→
   210→            {error && (
   211→              <div className="k-card k-glow-red mt-4 text-sm">
   212→                <span className="k-danger">[ERROR]</span>{' '}
   213→                <span className="k-fg">{error}</span>
   214→              </div>
   215→            )}
   216→
   217→            {/* Main grid: 2 columns on lg, 1 col on mobile */}
   218→            <div className="grid grid-cols-1 lg:grid-cols-12 gap-4 mt-4">
   219→              {/* Left column: pipeline + calibration */}
   220→              <div className="lg:col-span-4 flex flex-col gap-4">
   221→                <PipelinePanel pipeline={resp?.pipeline ?? []} />
   222→                <CalibrationPanel
   223→                  calibration={calibration}
   224→                  onChange={setCalibration}
   225→                />
   226→              </div>
   227→
   228→              {/* Middle column: trust gauge + evidence chain */}
   229→              <div className="lg:col-span-4 flex flex-col gap-4">
   230→                <TrustGaugePanel verdict={resp?.verdict ?? null} loading={loading} />
   231→                <EvidenceChainPanel
   232→                  observations={resp?.observations ?? []}
   233→                  pumpSim={pumpSim}
   234→                />
   235→              </div>
   236→
   237→              {/* Right column: SCADA telemetry + audit */}
   238→              <div className="lg:col-span-4 flex flex-col gap-4">
   239→                <ScadaPanel stepIdx={stepIdx} />
   240→                <AuditPanel
   241→                  resp={resp}
   242→                  onExport={exportAudit}
   243→                />
   244→              </div>
   245→            </div>
   246→
   247→            {/* Provenance + classification section */}
   248→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 mt-4">
   249→              <ProvenancePanel
   250→                observations={resp?.observations ?? []}
   251→                auditHash={resp?.auditHash ?? '——————'}
   252→                generatedAt={resp?.generatedAtUtc ?? '——————'}
   253→              />
   254→              <ClassificationPanel />
   255→            </div>
   256→          </>
   257→        )}
   258→      </main>
   259→
   260→      <FooterBar
   261→        auditShortHash={resp?.auditShortHash ?? '——————'}
   262→        generatedAt={resp?.generatedAtUtc ?? '——————'}
   263→      />
   264→    </div>
   265→  );
   266→}

   365→function ReplayTimeline({
   366→  steps,
   367→  currentIdx,
   368→  onSelect,
   369→}: {
   370→  steps: ReplayStep[];
   371→  currentIdx: number;
   372→  onSelect: (i: number) => void;
   373→}) {
   374→  return (
   375→    <Card className="k-card border-[var(--k-line)] bg-[var(--k-panel)]">
   376→      <CardHeader className="pb-2">
   377→        <CardTitle className="k-card-title">
   378→          <Layers className="w-4 h-4" />
   379→          Hydraulic Incident Replay · 10-Step Sequence
   380→        </CardTitle>
   381→      </CardHeader>
   382→      <CardContent>
   383→        <div className="flex gap-1 overflow-x-auto pb-2">
   384→          {steps.map((s, arrayIdx) => {
   385→            const active = arrayIdx === currentIdx;
   386→            const passed = arrayIdx < currentIdx;
   387→            return (
   388→              <button
   389→                key={s.id}
   390→                onClick={() => onSelect(arrayIdx)}
   391→                className={`group flex-shrink-0 flex flex-col items-center gap-1 px-2 py-2 rounded-md border transition-all min-w-[64px] ${
   392→                  active
   393→                    ? 'border-[var(--k-cyan-bright)] k-glow-cyan bg-[rgba(0,212,255,0.06)]'
   394→                    : passed
   395→                      ? 'border-[var(--k-green-bright)]/40 bg-[rgba(0,255,136,0.04)] hover:border-[var(--k-green-bright)]'
   396→                      : 'border-[var(--k-line)] bg-[var(--k-bg-elevated)] hover:border-[var(--k-cyan)]'
   397→                }`}
   398→                title={s.title}
   399→              >
   400→                <span
   401→                  className={`text-[10px] font-bold ${
   402→                    active
   403→                      ? 'text-[var(--k-cyan-bright)]'
   404→                      : passed
   405→                        ? 'text-[var(--k-green-bright)]'
   406→                        : 'text-[var(--k-dim)]'
   407→                  }`}
   408→                >
   409→                  {String(s.id).padStart(2, '0')}
   410→                </span>
   411→                <span
   412→                  className={`text-[9px] uppercase tracking-wider text-center ${
   413→                    active
   414→                      ? 'text-[var(--k-fg-bright)]'
   415→                      : passed
   416→                        ? 'text-[var(--k-green)]'
   417→                        : 'text-[var(--k-dim)]'
   418→                  }`}
   419→                >
   420→                  {s.key}
   421→                </span>
   422→              </button>
   423→            );
   424→          })}
   425→        </div>
   426→      </CardContent>
   427→    </Card>
   428→  );
   429→}
