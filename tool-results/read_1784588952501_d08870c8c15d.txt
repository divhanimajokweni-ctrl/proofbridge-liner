    51→/* ─── Types ─── */
    52→interface MetricsData {
    53→  throughput: {
    54→    totalMerges: number;
    55→    appliedMerges: number;
    56→    rejectedMerges: number;
    57→    successRate: number;
    58→    avgDivergence: number;
    59→    avgIterations: number;
    60→  };
    61→  timeSeries: { t: number; merges: number; violations: number; repairs: number }[];
    62→  nodeLoad: { node: string; shards: number; healthy: number; repairing: number; violating: number; load: number }[];
    63→  severityBreakdown: { critical: number; high: number; medium: number; low: number };
    64→  latency: { p50: number; p95: number; p99: number };
    65→  shardCount: number;
    66→  timestamp: number;
    67→}
    68→
    69→/* ─── Color helpers ─── */
    70→function successRateColor(rate: number): string {
    71→  if (rate >= 80) return "text-verified";
    72→  if (rate >= 50) return "text-repairing";
    73→  return "text-violating";
    74→}
    75→
    76→function successRateBg(rate: number): string {
    77→  if (rate >= 80) return "bg-verified/10 border-verified/30";
    78→  if (rate >= 50) return "bg-repairing/10 border-repairing/30";
    79→  return "bg-violating/10 border-violating/30";
    80→}
    81→
    82→function divergenceColor(v: number): string {
    83→  if (v < 0.001) return "text-verified";
    84→  if (v < 1) return "text-repairing";
    85→  return "text-violating";
    86→}
    87→
    88→function loadColor(load: number): string {
    89→  if (load < 40) return "oklch(0.78 0.16 160)"; // green
    90→  if (load < 70) return "oklch(0.75 0.15 80)";  // amber
    91→  return "oklch(0.65 0.2 25)";                   // red
    92→}
    93→
    94→function loadBgClass(load: number): string {
    95→  if (load < 40) return "bg-verified/20 border-verified/30";
    96→  if (load < 70) return "bg-repairing/20 border-repairing/30";
    97→  return "bg-violating/20 border-violating/30";
    98→}

   195→    <motion.div
   196→      className="space-y-5"
   197→      initial="hidden"

   204→          <TopAccentBar color="oklch(0.78 0.16 160)" />
