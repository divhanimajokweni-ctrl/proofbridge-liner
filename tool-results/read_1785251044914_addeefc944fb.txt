   126→export function SimulationDashboard() {
   127→  const socketRef = useRef<Socket | null>(null);
   128→  const [connected, setConnected] = useState(false);
   129→  const [simData, setSimData] = useState<SimTickData | null>(null);
   130→  const [simRunning, setSimRunning] = useState(false);
   131→  const [simPaused, setSimPaused] = useState(false);
   132→  const [speed, setSpeed] = useState(60);
   133→  const [activeTab, setActiveTab] = useState<"overview" | "hbk" | "git" | "metrics">("overview");
   134→  const [metricsHistory, setMetricsHistory] = useState<{ cpu: number[]; ram: number[]; queue: number[]; latency: number[]; risk: number[]; vi: number[] }>({ cpu: [], ram: [], queue: [], latency: [], risk: [], vi: [] });
   135→
   136→  useEffect(() => {
   137→    const socket = io("/?XTransformPort=3003", { transports: ["websocket"], reconnection: true, reconnectionAttempts: 10, reconnectionDelay: 2000 });
   138→    socket.on("connect", () => { setConnected(true); });
   139→    socket.on("disconnect", () => { setConnected(false); });
   140→    socket.on("sim:state", (data: any) => { setSimData(data); setSimRunning(data.running); });
   141→    socket.on("sim:tick", (data: SimTickData) => {
   142→      setSimData(data);
   143→      setMetricsHistory(prev => {
   144→        const h = { ...prev, cpu: [...prev.cpu, data.metrics.cpu_pct], ram: [...prev.ram, data.metrics.ram_pct], queue: [...prev.queue, data.metrics.queue_depth], latency: [...prev.latency, data.metrics.latency_p99_ms], risk: [...prev.risk, data.metrics.risk_score], vi: [...prev.vi, data.metrics.validation_index] };
   145→        for (const k of Object.keys(h) as (keyof typeof h)[]) { if (h[k].length > 60) h[k] = h[k].slice(-60); }
   146→        return h;
   147→      });
   148→    });
   149→    socket.on("sim:started", () => { setSimRunning(true); setSimPaused(false); });
   150→    socket.on("sim:paused", (d: { paused: boolean }) => { setSimPaused(d.paused); });
   151→    socket.on("sim:stopped", () => { setSimRunning(false); setSimPaused(false); });
   152→    socket.on("sim:reset", () => { setSimRunning(false); setSimPaused(false); setSimData(null); });
   153→    socketRef.current = socket;
   154→    return () => { socket.disconnect(); };
   155→  }, []);
   156→
   157→  const startSim = useCallback(() => { socketRef.current?.emit("sim:start", { speedMultiplier: speed }); }, [speed]);
   158→  const pauseSim = useCallback(() => { socketRef.current?.emit("sim:pause"); }, []);
   159→  const stopSim = useCallback(() => { socketRef.current?.emit("sim:stop"); }, []);
   160→  const resetSim = useCallback(() => { socketRef.current?.emit("sim:reset"); }, []);
   161→  const changeSpeed = useCallback((newSpeed: number) => { setSpeed(newSpeed); socketRef.current?.emit("sim:speed", { speedMultiplier: newSpeed }); }, []);
   162→
   163→  const metrics = simData?.metrics;
   164→  const hour = simData?.hour ?? 0;
   165→  const phase = simData?.phase ?? "P1";
   166→  const currentPhaseInfo = PHASES.find(p => p.id === phase) ?? PHASES[0];
   167→  const hbkData = simData?.hbkTelemetry ?? [];
   168→  const gitData = simData?.gitActions ?? [];
   169→  const milestones = simData?.milestones ?? [];
   170→  const phaseProgress = currentPhaseInfo ? ((hour - currentPhaseInfo.startHour) / (currentPhaseInfo.endHour - currentPhaseInfo.startHour)) * 100 : 0;
   171→  const overallProgress = (hour / 72) * 100;
   172→
   173→  return (
   174→    <div className="flex h-full flex-col overflow-hidden">
   175→      {/* SIMULATION CONTROLLER BAR */}
   176→      <div className="flex flex-wrap items-center gap-3 border-b border-white/[0.06] px-4 py-2.5 backdrop-blur-xl" style={{ background: "rgba(15,15,24,0.7)" }}>
   177→        <div className="flex items-center gap-2">
   178→          <span className={`h-2 w-2 rounded-full ${connected ? "bg-green-400" : "bg-red-400"}`} style={{ boxShadow: connected ? "0 0 6px rgba(52,211,153,0.5)" : "none" }} />
   179→          <span className="font-mono text-[10px] text-muted-foreground">{connected ? "SIM ENGINE LIVE" : "SIM ENGINE OFFLINE"}</span>
   180→        </div>
   181→        <div className="flex items-center gap-1.5">
   182→          {!simRunning ? (
   183→            <button onClick={startSim} className="inline-flex items-center gap-1.5 rounded-md border border-emerald-500/30 bg-emerald-500/10 px-3 py-1.5 text-xs font-medium text-emerald-400 hover:bg-emerald-500/20">
   184→              <Play className="h-3 w-3" />Start 72h Loop
   185→            </button>
   186→          ) : (
   187→            <>
   188→              <button onClick={pauseSim} className={`inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs font-medium ${simPaused ? "border-emerald-500/30 bg-emerald-500/10 text-emerald-400" : "border-yellow-500/30 bg-yellow-500/10 text-yellow-400"}`}>
   189→                {simPaused ? <Play className="h-3 w-3" /> : <Pause className="h-3 w-3" />}{simPaused ? "Resume" : "Pause"}
   190→              </button>
   191→              <button onClick={stopSim} className="inline-flex items-center gap-1.5 rounded-md border border-red-500/30 bg-red-500/10 px-2.5 py-1.5 text-xs font-medium text-red-400 hover:bg-red-500/20">
   192→                <Square className="h-3 w-3" />Stop
   193→              </button>
   194→            </>
   195→          )}
   196→          <button onClick={resetSim} className="inline-flex items-center gap-1.5 rounded-md border border-white/[0.08] bg-white/[0.03] px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground">
   197→            <RotateCcw className="h-3 w-3" />Reset
   198→          </button>
   199→        </div>
   200→        <div className="flex items-center gap-2">
   201→          <Gauge className="h-3 w-3 text-muted-foreground" />
   202→          <span className="font-mono text-[10px] text-muted-foreground">Speed:</span>
   203→          {[{ label: "1x", value: 1 }, { label: "60x", value: 60 }, { label: "360x", value: 360 }, { label: "3600x", value: 3600 }].map(s => (
   204→            <button key={s.value} onClick={() => changeSpeed(s.value)} className={`rounded border px-2 py-1 font-mono text-[10px] transition-all ${speed === s.value ? "border-[#C9A84C]/30 bg-[#C9A84C]/10 text-[#C9A84C]" : "border-white/[0.08] bg-white/[0.03] text-muted-foreground hover:text-foreground"}`}>
   205→              {s.label}
   206→            </button>
   207→          ))}
   208→        </div>
   209→        <div className="flex items-center gap-4 ml-auto">
   210→          <div className="flex items-center gap-2">
   211→            <Clock className="h-3.5 w-3.5" style={{ color: currentPhaseInfo.color }} />
   212→            <span className="font-mono text-sm font-bold" style={{ color: currentPhaseInfo.color }}>H{hour.toString().padStart(2,"0")}:00</span>
   213→            <span className="font-mono text-[10px] text-muted-foreground">/ H72</span>
   214→          </div>
   215→          <div className="flex items-center gap-1.5">
   216→            <span className="font-mono text-[10px] uppercase tracking-wider" style={{ color: currentPhaseInfo.color }}>{currentPhaseInfo.id}</span>
   217→            <span className="text-xs font-medium" style={{ color: currentPhaseInfo.color }}>{currentPhaseInfo.name}</span>
   218→          </div>
   219→          {metrics && (
   220→            <>
   221→              <div className="flex items-center gap-1.5">
   222→                <span className="h-1.5 w-1.5 rounded-full" style={{ background: CB_COLORS[metrics.circuit_breaker]||"#6b7280", boxShadow: `0 0 4px ${CB_COLORS[metrics.circuit_breaker]||"#6b7280"}80` }} />
   223→                <span className="font-mono text-[10px] text-muted-foreground">CB:</span>
   224→                <span className="font-mono text-[10px] font-medium" style={{ color: CB_COLORS[metrics.circuit_breaker] }}>{metrics.circuit_breaker}</span>
   225→              </div>
   226→              <div className="flex items-center gap-1.5">
   227→                <Shield className="h-3 w-3" style={{ color: AIR_COLORS[metrics.air_state] }} />
   228→                <span className="font-mono text-[10px] text-muted-foreground">AIR:</span>
   229→                <span className="font-mono text-[10px] font-medium" style={{ color: AIR_COLORS[metrics.air_state] }}>{metrics.air_state}</span>
   230→              </div>
   231→            </>
   232→          )}
   233→        </div>
   234→      </div>
   235→
   236→      {/* PROGRESS BAR */}
   237→      <div className="px-4 pt-1 pb-2">
   238→        <div className="relative h-1.5 rounded-full bg-white/[0.06] overflow-hidden">
   239→          <div className="absolute inset-0 rounded-full" style={{ width: `${overallProgress}%`, background: `linear-gradient(90deg, ${PHASES.map(p=>p.color).join(",")})`, transition: "width 0.3s" }} />
   240→          {PHASES.map(p => <div key={p.id} className="absolute top-0 h-full w-[2px]" style={{ left: `${(p.startHour/72)*100}%`, background: `${p.color}60` }} />)}
   241→        </div>
   242→        <div className="flex justify-between mt-1 font-mono text-[9px] text-muted-foreground/60">
   243→          {PHASES.map(p => <span key={p.id} style={{ color: phase===p.id ? p.color : undefined }}>{p.id}</span>)}
   244→        </div>
   245→      </div>
   246→
   247→      {/* TAB BAR */}
   248→      <div className="flex gap-1 px-4 pb-2">
   249→        {([["overview","Overview",Activity],["hbk","HBK Digital Twin",Droplets],["git","Git Actions Log",GitBranch],["metrics","Real-Time Metrics",Cpu]] as const).map(([id,label,Icon])=>(
   250→          <button key={id} onClick={()=>setActiveTab(id as any)} className={`inline-flex items-center gap-1.5 rounded-md border px-3 py-1.5 text-xs font-medium transition-all ${activeTab===id?"border-white/10 bg-white/[0.05] text-foreground":"border-transparent text-muted-foreground hover:bg-white/[0.03] hover:text-foreground"}`}>
   251→            <Icon className="h-3 w-3" />{label}
   252→          </button>
   253→        ))}
   254→      </div>
   255→
   256→      {/* MAIN CONTENT */}
   257→      <div className="flex-1 overflow-y-auto px-4 pb-4" style={{ scrollbarWidth:"thin" }}>
   258→        <AnimatePresence mode="wait">
   259→          <motion.div key={activeTab} initial={{ opacity:0, y:8 }} animate={{ opacity:1, y:0 }} exit={{ opacity:0, y:-8 }} transition={{ duration:0.15 }}>
   260→            {/* OVERVIEW TAB */}
   261→            {activeTab==="overview" && (
   262→              <div className="space-y-4">
   263→                {metrics && (
   264→                  <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-6">
   265→                    {([
   266→                      ["Validation Index", metrics.validation_index.toFixed(1), Eye, metrics.validation_index>=90?"#10b981":"#ef4444", "/ 100"],
   267→                      ["Facts Accepted", metrics.facts_accepted.toLocaleString(), CheckCircle2, "#3dffb0", ""],
   268→                      ["Facts Rejected", metrics.facts_rejected.toLocaleString(), XCircle, metrics.facts_rejected>0?"#ef4444":"#6b7280", ""],
   269→                      ["Queue Depth", metrics.queue_depth.toLocaleString(), Database, metrics.queue_depth>10000?"#ef4444":"#3dffb0", ""],
   270→                      ["Risk Score", metrics.risk_score_smoothed.toFixed(3), AlertTriangle, AIR_COLORS[metrics.air_state], "[0,1]"],
   271→                      ["Proofs", metrics.proof_count.toLocaleString(), Shield, "#c9a84c", ""],
   272→                      ["TEE Status", metrics.tee_status, Cpu, metrics.tee_status==="ATTESTED"?"#3dffb0":"#ef4444", ""],
   273→                      ["Replay", metrics.replay_status, FileText, metrics.replay_status==="VERIFIED"?"#3dffb0":"#ef4444", ""],
   274→                      ["Spoofed Quarantined", `${metrics.spoofed_payloads_quarantined}/${metrics.spoofed_payloads_injected}`, Shield, metrics.spoofed_payloads_injected>0&&metrics.spoofed_payloads_quarantined===metrics.spoofed_payloads_injected?"#3dffb0":"#ef4444", ""],
   275→                      ["Merge Conflicts", metrics.merge_conflicts_observed.toString(), GitBranch, metrics.merge_conflicts_observed===0?"#3dffb0":"#ef4444", "observed"],
   276→                      ["Evidence Bundles", `${metrics.evidence_bundles_verified}/${metrics.evidence_bundles_total}`, HardDrive, metrics.evidence_bundles_verified===metrics.evidence_bundles_total?"#3dffb0":"#ef4444", "verified"],
   277→                      ["Fail-Closed Dur.", `${metrics.fail_closed_s.toFixed(0)}s`, Zap, metrics.fail_closed_s>0?"#ef4444":"#6b7280", ""],

   279→                      <div key={label} className="rounded-lg border border-white/[0.06] p-3 backdrop-blur-sm" style={{ background:"rgba(15,15,24,0.6)" }}>
   280→                        <div className="flex items-center gap-1.5 mb-1"><Icon className="h-3 w-3" style={{ color }} strokeWidth={1.8} /><span className="font-mono text-[9px] uppercase tracking-wider text-muted-foreground/70">{label}</span></div>

   295→                          <div className="text-xs font-medium truncate" style={{ color:isActive||isComplete?p.color:"#6b7280" }}>{p.name}</div>

   318→              </div>
