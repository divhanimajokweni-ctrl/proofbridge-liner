   250→function InlineStateDiff({ proposed, repaired }: { proposed: Record<string, unknown>; repaired: Record<string, unknown> }) {
   251→  const diffs = useMemo(() => {
   252→    const keys = new Set([...Object.keys(proposed ?? {}), ...Object.keys(repaired ?? {})]);
   253→    return Array.from(keys).sort().map((key) => {
   254→      const fromVal = proposed?.[key]; const toVal = repaired?.[key];
   255→      const changed = JSON.stringify(fromVal) !== JSON.stringify(toVal);
   256→      const delta = typeof fromVal === "number" && typeof toVal === "number" ? toVal - fromVal : null;
   257→      return { key, from: fromVal, to: toVal, changed, delta };
   258→    }).filter((d) => d.changed);
   259→  }, [proposed, repaired]);
   260→  if (!diffs.length) return null;
   261→  return (
   262→    <div>
   263→      <div className="mb-1.5 text-[10px] uppercase tracking-wide text-foreground/70 flex items-center gap-1.5"><Eye className="h-3 w-3" />State delta ({diffs.length})</div>
   264→      <div className="grid grid-cols-1 gap-1 sm:grid-cols-2">
   265→        {diffs.map((d) => (
   266→          <div key={d.key} className={cn("flex items-center justify-between gap-2 rounded border px-2 py-1 text-[11px]",
   267→            d.delta !== null ? d.delta > 0 ? "border-verified/30 bg-verified/5" : "border-violating/30 bg-violating/5" : "border-repairing/30 bg-repairing/5")}>
   268→            <div className="flex items-center gap-1.5 min-w-0"><span className="font-mono text-muted-foreground truncate">{d.key}</span><ArrowRight className="h-2.5 w-2.5 shrink-0 text-muted-foreground" /></div>
   269→            <div className="flex items-center gap-1 shrink-0">
   270→              <span className={cn("font-mono line-through", d.delta !== null && d.delta < 0 ? "text-violating" : "text-muted-foreground")}>{fmtVal(d.from)}</span>
   271→              <span className={cn("font-mono", d.delta !== null ? d.delta > 0 ? "text-verified" : "text-violating" : "text-verified")}>{fmtVal(d.to)}</span>
   272→              {d.delta !== null && <Badge variant="outline" className={cn("text-[8px] px-1 py-0 h-4", d.delta > 0 ? "border-verified/30 text-verified" : "border-violating/30 text-violating")}>{d.delta > 0 ? "+" : ""}{d.delta.toFixed(3)}</Badge>}
   273→            </div>
   274→          </div>
   275→        ))}
   276→      </div>
   277→    </div>
   278→  );
   279→}

   281→function MergeSummary({ merges }: { merges: MergeRow[] }) {
   282→  const stats = useMemo(() => {
   283→    const clean = merges.filter((m) => getMergeType(m) === "clean").length;
   284→    const repaired = merges.filter((m) => getMergeType(m) === "repaired").length;
   285→    const rejected = merges.filter((m) => getMergeType(m) === "rejected").length;
   286→    const avgDiv = merges.length > 0 ? merges.reduce((s, m) => s + m.divergence, 0) / merges.length : 0;
   287→    return { clean, repaired, rejected, avgDiv };
   288→  }, [merges]);
   289→  return (
   290→    <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   291→      <StatCard label="Clean" value={stats.clean} color="text-verified" bg="bg-verified/10" border="border-verified/30" />
   292→      <StatCard label="Repaired" value={stats.repaired} color="text-repairing" bg="bg-repairing/10" border="border-repairing/30" />
   293→      <StatCard label="Rejected" value={stats.rejected} color="text-violating" bg="bg-violating/10" border="border-violating/30" />
   294→      <StatCard label="Avg div" value={stats.avgDiv.toFixed(3)} bg="bg-muted/20" border="border-border/60" />
   295→    </div>
   296→  );
   297→}

   331→    } catch { toast({ title: "Repair request failed", variant: "destructive" }); } finally { setSubmitting(false); }

   339→      if (!r.ok) throw new Error();

   373→              : <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-2 max-h-[640px] overflow-y-auto pr-1">{recentMerges.map((m) => <MergeItem key={m.id} merge={m} />)}</motion.div>

   402→        </div>
   403→      </div>
   404→
   405→      <MergeSimulator policies={policies} />
   406→    </motion.section>
