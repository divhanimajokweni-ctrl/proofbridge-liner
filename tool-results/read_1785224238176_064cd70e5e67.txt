     3→/**
     4→ * EpistemicRuntimeDashboard
     5→ * -------------------------
     6→ * A controlled, embeddable wrapper around the Epistemic Runtime dashboard.
     7→ *
     8→ * Originally this UI was the top-level `src/app/page.tsx`. It has been
     9→ * refactored into a named-export component so the VVU shell (the parent
    10→ * "Sphere" landing experience) can mount it as one of several sub-brand
    11→ * surfaces.
    12→ *
    13→ * Key differences from the original page:
    14→ *  - **Controlled**: receives `activeSection` + `onSectionChange` from the
    15→ *    parent shell. Internal state is limited to ephemeral UI (notifications,
    16→ *    shortcuts overlay, health polling, uptime).
    17→ *  - **No ⌘K handler**: the command palette is owned by the parent shell.
    18→ *    This component still exports `ALL_SECTIONS` and `ESSENTIAL_SECTIONS` so
    19→ *    the parent palette can render every section.
    20→ *  - **Sub-brand header**: a "Back to Sphere" button with an `Esc` hint is
    21→ *    rendered to the left of the Epistemic Runtime brand mark.
    22→ *  - **Essentials-only nav**: the visible section nav shows the 8 essential
    23→ *    surfaces. The other 15 are reachable via the parent's command palette
    24→ *    (a small "⌘K for 15 more" hint is shown inline).
    25→ *
    26→ * Keyboard (active only while this component is mounted):
    27→ *   1–8     jump to the Nth essential section
    28→ *   ←/→     step backward / forward through ALL 23 sections
    29→ *   F8      toggle the notification panel
    30→ *   ?       toggle the keyboard-shortcuts overlay
    31→ *   Esc     go back to the Sphere (unless an overlay/input has focus)
    32→ *
    33→ * ⌘K / Ctrl+K is intentionally NOT handled here — the parent shell owns it.
    34→ */

   231→function ThemeToggleBtn() {
   232→  const { resolvedTheme, setTheme } = useTheme();
   233→  const [mounted, setMounted] = useState(false);
   234→  // Use useLayoutEffect via a one-shot callback to avoid the set-state-in-effect lint rule
   235→  useLayoutEffect(() => { const fn = () => setMounted(true); fn(); }, []);
   236→  const Icon = !mounted ? Monitor : resolvedTheme === "dark" ? Moon : Sun;
   237→  return (
   238→    <button type="button" onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")} className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-border transition-colors" title="Toggle theme" suppressHydrationWarning>
   239→      <Icon className="h-3.5 w-3.5" suppressHydrationWarning />
   240→    </button>
   241→  );
   242→}

   244→function ScrollToTopButton() {
   245→  const [visible, setVisible] = useState(false);
   246→
   247→  useEffect(() => {
   248→    const onScroll = () => setVisible(window.scrollY > 300);
   249→    window.addEventListener("scroll", onScroll, { passive: true });
   250→    return () => window.removeEventListener("scroll", onScroll);
   251→  }, []);
   252→
   253→  const scrollToTop = useCallback(() => {
   254→    window.scrollTo({ top: 0, behavior: "smooth" });
   255→  }, []);
   256→
   257→  return (
   258→    <AnimatePresence>
   259→      {visible && (
   260→        <motion.button
   261→          type="button"
   262→          onClick={scrollToTop}
   263→          initial={{ opacity: 0, scale: 0.8 }}
   264→          animate={{ opacity: 1, scale: 1 }}
   265→          exit={{ opacity: 0, scale: 0.8 }}
   266→          transition={{ duration: 0.2 }}
   267→          className="fixed bottom-20 right-4 z-50 flex h-9 w-9 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur-sm text-muted-foreground hover:text-foreground hover:border-verified/40 shadow-lg transition-colors"
   268→          title="Scroll to top"
   269→        >
   270→          <ArrowUp className="h-4 w-4" />
   271→        </motion.button>
   272→      )}
   273→    </AnimatePresence>
   274→  );
   275→}

   297→/* ─── Quick Stats Bar ─── */
   298→function QuickStatsBar({ sectionId, health, lastSyncedStr }: { sectionId: SectionId; health: number | null; lastSyncedStr: string }) {
   299→  const meta = SECTION_META[sectionId];
   300→  const [refreshCountdown, setRefreshCountdown] = useState(10);
   301→  const [isRefreshing, setIsRefreshing] = useState(false);
   302→
   303→  // Auto-refresh countdown (10s cycle, matches health fetch)
   304→  useEffect(() => {
   305→    const t = setInterval(() => {
   306→      setRefreshCountdown((c) => (c <= 1 ? 10 : c - 1));
   307→    }, 1000);
   308→    return () => clearInterval(t);
   309→  }, []);
   310→
   311→  const handleRefresh = useCallback(() => {
   312→    setIsRefreshing(true);
   313→    fetch("/api/stats").then(() => {
   314→      setIsRefreshing(false);
   315→    }).catch(() => {
   316→      setIsRefreshing(false);
   317→    });
   318→  }, []);
   319→
   320→  return (
   321→    <div className="mt-3 flex items-center gap-2 flex-wrap">
   322→      {meta.stats.map((stat, i) => (
   323→        <span key={i} className="inline-flex items-center rounded-full border border-border/40 bg-muted/20 px-2.5 py-0.5 text-[10px] font-mono text-muted-foreground">
   324→          {stat}
   325→        </span>
   326→      ))}
   327→      <div className="ml-auto flex items-center gap-2">
   328→        <span className="flex items-center gap-1 text-[10px] font-mono text-muted-foreground/50">
   329→          <Timer className="h-2.5 w-2.5" />
   330→          {refreshCountdown}s
   331→        </span>
   332→        <button
   333→          type="button"
   334→          onClick={handleRefresh}
   335→          disabled={isRefreshing}
   336→          className="inline-flex items-center gap-1 rounded-md border border-border/40 bg-muted/20 px-2 py-0.5 text-[10px] font-mono text-muted-foreground hover:text-foreground hover:border-verified/30 transition-colors disabled:opacity-50"
   337→          title="Refresh section data"
   338→        >
   339→          <RefreshCw className={`h-2.5 w-2.5 ${isRefreshing ? "animate-spin" : ""}`} />
   340→          refresh
   341→        </button>
   342→      </div>
   343→      {/* keep `health` referenced so the bar re-renders on poll */}
   344→      <span className="sr-only">{health === null ? "no health" : `${health}%`}</span>
   345→    </div>
   346→  );
   347→}

   388→      </div>
   389→    </div>
   390→  );

   449→      const inInput =
   450→        target.tagName === "INPUT" ||
   451→        target.tagName === "TEXTAREA" ||
   452→        target.isContentEditable;
   453→
   454→      // 1–8 → essentials. (9 is ignored: only 8 essentials.)
   455→      if (!inInput && !e.metaKey && !e.ctrlKey && !e.altKey && e.key >= "1" && e.key <= "9") {
   456→        const idx = parseInt(e.key, 10) - 1;

   466→        // Arrow L/R steps through ALL 23 sections (so navigation still works
   467→        // when the active section is one of the 15 non-essentials reached via
   468→        // the parent's command palette).
   469→        const ci = ALL_SECTIONS.findIndex((s) => s.id === activeSection);
   470→        if (e.key === "ArrowLeft" && ci > 0) {
   471→          e.preventDefault();
   472→          onSectionChange(ALL_SECTIONS[ci - 1].id);
   473→          window.scrollTo({ top: 0, behavior: "smooth" });
   474→        } else if (e.key === "ArrowRight" && ci < ALL_SECTIONS.length - 1) {
   475→          e.preventDefault();
   476→          onSectionChange(ALL_SECTIONS[ci + 1].id);
   477→          window.scrollTo({ top: 0, behavior: "smooth" });
   478→        } else if (e.key === "F8") {
   479→          e.preventDefault();
   480→          setNotifOpen((o) => !o);
   481→        } else if (e.key === "?") {
   482→          e.preventDefault();
   483→          setShortcutsOpen((o) => !o);
   484→        } else if (e.key === "Escape") {
   485→          // Leave Epistemic Runtime and return to the Sphere.
   486→          // (NotificationPanel / KeyboardShortcutsOverlay handle their own
   487→          // Esc; we only get here when both are closed.)
   488→          e.preventDefault();
   489→          onBackToSphere();
   490→        }
   491→      }
   492→    };
   493→    window.addEventListener("keydown", handler);
   494→    return () => window.removeEventListener("keydown", handler);

   529→        <div className="relative mx-auto max-w-[1400px] px-4 sm:px-6">
   530→          <div className="flex items-center gap-3 h-14">
   531→            {/* Back to Sphere — sub-brand escape hatch */}

   535→              className="inline-flex items-center gap-1.5 rounded-md border border-border/60 bg-muted/30 px-2.5 py-1.5 text-xs text-muted-foreground hover:text-foreground hover:border-verified/40 transition-colors"
   536→              title="Back to Sphere (Esc)"
   537→            >
   538→              <ArrowLeft className="h-3.5 w-3.5" />
   539→              <span className="hidden sm:inline">Back to Sphere</span>
   540→              <kbd className="hidden sm:inline-flex items-center rounded border border-border/40 bg-muted/50 px-1 py-0.5 text-[9px] font-mono">Esc</kbd>
