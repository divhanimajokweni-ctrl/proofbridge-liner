    92→function fmtRelative(iso: string | null | undefined): string {
    93→  if (!iso) return "—";
    94→  const t = new Date(iso).getTime();
    95→  const diff = Date.now() - t;
    96→  if (diff < 0) return "just now";
    97→  const s = Math.floor(diff / 1000);
    98→  if (s < 60) return `${s}s ago`;
    99→  const m = Math.floor(s / 60);
   100→  if (m < 60) return `${m}m ago`;
   101→  const h = Math.floor(m / 60);
   102→  if (h < 24) return `${h}h ago`;
   103→  return `${Math.floor(h / 24)}d ago`;
   104→}
   105→
   106→function fmtTimestamp(iso: string | null | undefined): string {
   107→  if (!iso) return "—";
   108→  const d = new Date(iso);
   109→  return d.toLocaleString(undefined, {
   110→    year: "numeric",
   111→    month: "short",
   112→    day: "2-digit",
   113→    hour: "2-digit",
   114→    minute: "2-digit",
   115→    second: "2-digit",
   116→  });
   117→}

   535→  const { toast } = useToast();
   536→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   537→  const [proofs, setProofs] = useState<ProofRow[]>([]);
   538→  const [loading, setLoading] = useState(true);
   539→  const [filterPolicyId, setFilterPolicyId] = useState<string>("all");
   540→  const [searchQuery, setSearchQuery] = useState("");
   541→  const [mintPolicyId, setMintPolicyId] = useState<string>("");
   542→  const [minting, setMinting] = useState(false);
   543→
   544→  const policyMap = useMemo(() => {
   545→    const m: Record<string, PolicyRow> = {};
   546→    for (const p of policies) m[p.id] = p;
   547→    return m;
   548→  }, [policies]);
   549→
   550→  // Load policies once.
   551→  useEffect(() => {
   552→    let cancelled = false;
   553→    fetch("/api/policies")
   554→      .then((r) => r.json())
   555→      .then((d: { policies: PolicyRow[] }) => {
   556→        if (cancelled) return;
   557→        const ps = d.policies ?? [];
   558→        setPolicies(ps);
   559→        if (ps.length > 0) setMintPolicyId(ps[0].id);
   560→      })
   561→      .catch(() => {});
   562→    return () => {
   563→      cancelled = true;
   564→    };
   565→  }, []);
   566→
   567→  const loadProofs = useCallback(async () => {
   568→    try {
   569→      const url =
   570→        filterPolicyId && filterPolicyId !== "all"
   571→          ? `/api/proofs?policyId=${encodeURIComponent(filterPolicyId)}`
   572→          : "/api/proofs";
   573→      const r = await fetch(url);
   574→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   575→      const d: { proofs: ProofRow[] } = await r.json();
   576→      setProofs(d.proofs ?? []);
   577→    } catch (e) {
   578→      toast({
   579→        title: "Failed to load proofs",
   580→        description: e instanceof Error ? e.message : "Unknown error",
   581→        variant: "destructive",
   582→      });
   583→    } finally {
   584→      setLoading(false);
   585→    }
   586→  }, [filterPolicyId, toast]);
   587→
   588→  useEffect(() => {
   589→    setLoading(true);
   590→    loadProofs();
   591→    const t = setInterval(loadProofs, POLL_MS);
   592→    return () => clearInterval(t);
   593→  }, [loadProofs]);
   594→
   595→  const mintProof = useCallback(async () => {
   596→    if (!mintPolicyId) {
   597→      toast({ title: "Select a policy first", variant: "destructive" });
   598→      return;
   599→    }
   600→    setMinting(true);
   601→    try {
   602→      const r = await fetch("/api/proofs", {
   603→        method: "POST",
   604→        headers: { "Content-Type": "application/json" },
   605→        body: JSON.stringify({ policyId: mintPolicyId }),
   606→      });
   607→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   608→      const d = await r.json();
   609→      const created: ProofRow = {
   610→        ...d.proof,
   611→        verified: true,
   612→        policy: d.proof.policy ?? {
   613→          name: policyMap[mintPolicyId]?.name ?? "unknown",
   614→          zkEnabled: policyMap[mintPolicyId]?.zkEnabled ?? false,
   615→          proofKind: policyMap[mintPolicyId]?.proofKind ?? "mmr",
   616→        },
   617→      };
   618→      setProofs((prev) => [created, ...prev]);
   619→      toast({
   620→        title: "MMR proof minted",
   621→        description: `root ${created.mmrRoot} · ${created.proofPath.length} sibling(s)`,
   622→      });
   623→    } catch (e) {
   624→      toast({
   625→        title: "Mint failed",
   626→        description: e instanceof Error ? e.message : "Unknown error",
   627→        variant: "destructive",
   628→      });
   629→    } finally {
   630→      setMinting(false);
   631→    }
   632→  }, [mintPolicyId, policyMap, toast]);
   633→
   634→  // Filter proofs by search query
   635→  const filteredProofs = useMemo(() => {
   636→    if (!searchQuery.trim()) return proofs;
   637→    const q = searchQuery.toLowerCase();
   638→    return proofs.filter((p) => {
   639→      const name = p.policy?.name ?? policyMap[p.policyId]?.name ?? "";
   640→      return (
   641→        name.toLowerCase().includes(q) ||
   642→        p.shardKey.toLowerCase().includes(q) ||
   643→        p.mmrRoot.toLowerCase().includes(q) ||
   644→        (p.anchor ?? "").toLowerCase().includes(q)
   645→      );
   646→    });
   647→  }, [proofs, searchQuery, policyMap]);
   648→
   649→  // Summary metrics.
   650→  const total = proofs.length;
   651→  const zkCount = proofs.filter((p) => p.zkProof).length;
   652→  const anchoredCount = proofs.filter((p) => p.anchored).length;
   653→  const anchoredRate = total > 0 ? Math.round((anchoredCount / total) * 100) : 0;
   654→  const allVerified = total > 0 && proofs.every((p) => p.verified);
   655→  const healthy = total > 0 && allVerified;
   656→
   657→  return (
   658→    <TooltipProvider delayDuration={150}>
   659→      <motion.section
   660→        className="space-y-4"
   661→        variants={containerVariants}
   662→        initial="hidden"
   663→        animate="show"
   664→      >
   665→        {/* Header */}
   666→        <motion.div variants={itemVariants} className="flex items-center gap-2.5">
   667→          <div className="flex h-9 w-9 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 glow-verified">
   668→            <KeyRound className="h-4.5 w-4.5 text-verified" />
   669→          </div>
   670→          <div>
   671→            <h2 className="text-base font-semibold text-foreground">MMR Ancestry Proof Inspector</h2>
   672→            <p className="text-xs text-muted-foreground">
   673→              Merkle-mountain-range ancestry · zero-knowledge merge proofs · transparency anchoring
   674→            </p>
   675→          </div>
   676→        </motion.div>
   677→
   678→        {/* Summary bar */}
   679→        <motion.div variants={cardVariants}>
   680→          <GradientBorderCard gradient="from-verified/20 via-verified/10 to-repairing/20">
   681→            <div className="p-4 space-y-3">
   682→              <div className="bg-grid-fine absolute inset-0 opacity-30" />
   683→              <div className="relative grid grid-cols-2 gap-3 sm:grid-cols-4">
   684→                <SummaryStat label="Total proofs" value={total} icon={<Mountain className="h-3.5 w-3.5" />} />
   685→                <SummaryStat
   686→                  label="ZK proofs"
   687→                  value={zkCount}
   688→                  icon={<KeyRound className="h-3.5 w-3.5" />}
   689→                  accent="verified"
   690→                />
   691→                <SummaryStat
   692→                  label="Anchored"
   693→                  value={`${anchoredRate}%`}
   694→                  icon={<Anchor className="h-3.5 w-3.5" />}
   695→                  accent="verified"
   696→                />
   697→                <SummaryStat
   698→                  label="Verification"
   699→                  value={total === 0 ? "—" : healthy ? "healthy" : "degraded"}
   700→                  icon={<ShieldCheck className="h-3.5 w-3.5" />}
   701→                  accent={total === 0 ? "muted" : healthy ? "verified" : "violating"}
   702→                  pulse={!healthy && total > 0}
   703→                />
   704→              </div>
   705→
   706→              {/* Proof chain integrity */}
   707→              {total > 0 && (
   708→                <div className="relative border-t border-border/40 pt-3">
   709→                  <ProofChainIntegrity proofs={proofs} />
   710→                </div>
   711→              )}
   712→            </div>
   713→          </GradientBorderCard>
   714→        </motion.div>
   715→
   716→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   717→          {/* LEFT — proof list (2/3) */}
   718→          <div className="space-y-3 lg:col-span-2">
   719→            <motion.div variants={cardVariants}>
   720→              <GradientBorderCard gradient="from-verified/20 via-border/10 to-verified/20">
   721→                <div className="p-3 space-y-3">
   722→                  <div className="bg-grid-fine absolute inset-0 opacity-20" />
   723→                  <div className="relative flex flex-wrap items-center justify-between gap-2">
   724→                    <div className="flex items-center gap-2">
   725→                      <span className="text-sm font-medium text-foreground">Issued proofs</span>
   726→                      <Badge variant="outline" className="border-border/60 text-[10px]">{total}</Badge>
   727→                    </div>
   728→                    <div className="flex items-center gap-2">
   729→                      <div className="relative">
   730→                        <Search className="absolute left-2 top-1/2 h-3 w-3 -translate-y-1/2 text-muted-foreground" />
   731→                        <Input
   732→                          value={searchQuery}
   733→                          onChange={(e) => setSearchQuery(e.target.value)}
   734→                          placeholder="Search proofs…"
   735→                          className="h-8 w-[180px] pl-7 text-xs bg-card/60"
   736→                        />
   737→                      </div>
   738→                      <Select value={filterPolicyId} onValueChange={setFilterPolicyId}>
   739→                        <SelectTrigger className="h-8 w-[200px] bg-card/60 text-xs">
   740→                          <SelectValue placeholder="All policies" />
   741→                        </SelectTrigger>
   742→                        <SelectContent>
   743→                          <SelectItem value="all">All policies</SelectItem>
   744→                          {policies.map((p) => (
   745→                            <SelectItem key={p.id} value={p.id}>
   746→                              {p.name}
   747→                            </SelectItem>
   748→                          ))}
   749→                        </SelectContent>
   750→                      </Select>
   751→                    </div>
   752→                  </div>
   753→                </div>
   754→              </GradientBorderCard>
   755→            </motion.div>
   756→
   757→            {loading ? (
   758→              <div className="space-y-2">
   759→                {Array.from({ length: 4 }).map((_, i) => (
   760→                  <Skeleton key={i} className="h-20 w-full rounded-lg bg-muted/40" />
   761→                ))}
   762→              </div>
   763→            ) : filteredProofs.length === 0 ? (
   764→              <Card className="bg-card/60 border-dashed border-border/60 p-8 text-center">
   765→                <KeyRound className="mx-auto h-8 w-8 text-muted-foreground/40" />
   766→                <p className="mt-2 text-sm text-muted-foreground">
   767→                  {searchQuery ? "No proofs match your search." : "No ancestry proofs issued."}
   768→                </p>
   769→                <p className="text-xs text-muted-foreground/70">
   770→                  {searchQuery ? "Try a different query." : "Mint one with the composer →"}
   771→                </p>
   772→              </Card>
   773→            ) : (
   774→              <motion.div
   775→                className="space-y-2 max-h-[640px] overflow-y-auto pr-1"
   776→                variants={containerVariants}
   777→                initial="hidden"
   778→                animate="show"
   779→              >
   780→                {filteredProofs.map((p, i) => (
   781→                  <ProofCard
   782→                    key={p.id}
   783→                    proof={p}
   784→                    policyName={p.policy?.name ?? policyMap[p.policyId]?.name ?? "unknown"}
   785→                    defaultOpen={i === 0}
   786→                  />
   787→                ))}
   788→              </motion.div>
   789→            )}
   790→          </div>
   791→
   792→          {/* RIGHT — mint composer + educational callout (1/3) */}
   793→          <div className="space-y-3">
   794→            <motion.div variants={cardVariants}>
   795→              <GradientBorderCard gradient="from-verified/30 via-verified/10 to-verified/20">
   796→                <div className="p-4 space-y-3">
   797→                  <div className="bg-grid-fine absolute inset-0 opacity-20" />
   798→                  <div className="relative space-y-3">
   799→                    <div className="flex items-center justify-between">
   800→                      <div className="flex items-center gap-2">
   801→                        <Plus className="h-4 w-4 text-verified" />
   802→                        <span className="text-sm font-semibold text-foreground">Mint MMR proof</span>
   803→                      </div>
   804→                      <Badge variant="outline" className="border-verified/30 bg-verified/10 text-verified text-[10px]">
   805→                        ancestry
   806→                      </Badge>
   807→                    </div>
   808→
   809→                    <div className="space-y-1.5">
   810→                      <label className="text-[11px] uppercase tracking-wide text-muted-foreground">
   811→                        Source policy
   812→                      </label>
   813→                      <Select value={mintPolicyId} onValueChange={setMintPolicyId}>
   814→                        <SelectTrigger className="bg-card/60">
   815→                          <SelectValue placeholder="Select policy" />
   816→                        </SelectTrigger>
   817→                        <SelectContent>
   818→                          {policies.map((p) => (
   819→                            <SelectItem key={p.id} value={p.id}>
   820→                              {p.name}
   821→                              {p.zkEnabled && <span className="ml-1 text-verified">●zk</span>}
   822→                            </SelectItem>
   823→                          ))}
   824→                        </SelectContent>
   825→                      </Select>
   826→                    </div>
   827→
   828→                    <Button
   829→                      onClick={mintProof}
   830→                      disabled={minting || !mintPolicyId}
   831→                      className="w-full bg-verified/15 border border-verified/40 text-verified hover:bg-verified/25"
   832→                      variant="outline"
   833→                    >
   834→                      {minting ? (
   835→                        <RotateCw className="h-4 w-4 animate-spin" />
   836→                      ) : (
   837→                        <Sparkles className="h-4 w-4" />
   838→                      )}
   839→                      {minting ? "Minting…" : "Mint MMR proof"}
   840→                    </Button>
   841→
   842→                    {mintPolicyId && policyMap[mintPolicyId]?.zkEnabled && (
   843→                      <p className="text-[10px] text-verified/80 flex items-center gap-1">
   844→                        <KeyRound className="h-2.5 w-2.5" />
   845→                        ZK enabled — a SNARK over the state delta will be generated.
   846→                      </p>
   847→                    )}
   848→                  </div>
   849→                </div>
   850→              </GradientBorderCard>
   851→            </motion.div>
   852→
   853→            {/* Educational callout */}
   854→            <motion.div variants={cardVariants}>
   855→              <GradientBorderCard gradient="from-quarantined/20 via-quarantined/10 to-quarantined/20">
   856→                <div className="p-4 space-y-2">
   857→                  <div className="bg-grid-fine absolute inset-0 opacity-20" />
   858→                  <div className="relative space-y-2">
   859→                    <div className="flex items-center gap-2">
   860→                      <div className="flex h-7 w-7 items-center justify-center rounded-md border border-quarantined/30 bg-quarantined/10">
   861→                        <Info className="h-3.5 w-3.5 text-quarantined" />
   862→                      </div>
   863→                      <span className="text-sm font-semibold text-foreground">How MMR ancestry works</span>
   864→                    </div>
   865→                    <p className="text-[11px] leading-relaxed text-muted-foreground">
   866→                      <span className="text-foreground">MMR (Merkle Mountain Range)</span> ancestry
   867→                      proofs let any node verify a merge without seeing full history — only the leaf
   868→                      and its sibling path are needed to recompute the root.
   869→                    </p>
   870→                    <p className="text-[11px] leading-relaxed text-muted-foreground">
   871→                      With <span className="text-verified">ZK enabled</span>, the proof becomes a
   872→                      SNARK/STARK over the state delta — enabling{" "}
   873→                      <span className="text-foreground">trustless edge→cloud sync</span> where the
   874→                      edge proves compliance without revealing raw state.
   875→                    </p>
   876→                    <div className="flex flex-wrap gap-1.5 pt-1">
   877→                      <Badge variant="outline" className="border-border/60 text-[9px] font-mono">append-only</Badge>
   878→                      <Badge variant="outline" className="border-border/60 text-[9px] font-mono">O(log n) path</Badge>
   879→                      <Badge variant="outline" className="border-border/60 text-[9px] font-mono">transparent</Badge>
   880→                      <Badge variant="outline" className="border-border/60 text-[9px] font-mono">CRDT-safe</Badge>
   881→                    </div>
   882→                  </div>
   883→                </div>
   884→              </GradientBorderCard>
   885→            </motion.div>
   886→          </div>
   887→        </div>
   888→      </motion.section>
   889→    </TooltipProvider>
   890→  );
   891→}

   893→function SummaryStat({
   894→  label,
   895→  value,
   896→  icon,
   897→  accent = "muted",
   898→  pulse,
   899→}: {
   900→  label: string;
   901→  value: string | number;
   902→  icon?: React.ReactNode;
   903→  accent?: "muted" | "verified" | "repairing" | "violating" | "quarantined";
   904→  pulse?: boolean;
   905→}) {
   906→  const colorMap: Record<string, string> = {
   907→    muted: "text-foreground",
   908→    verified: "text-verified",
   909→    repairing: "text-repairing",
   910→    violating: "text-violating",
   911→    quarantined: "text-quarantined",
   912→  };
   913→  return (
   914→    <div className="rounded-md border border-border/60 bg-background/40 p-2.5 hover:bg-background/60 transition-colors">
   915→      <div className="flex items-center gap-1.5 text-[10px] uppercase tracking-wide text-muted-foreground">
   916→        {icon}
   917→        {label}
   918→      </div>
   919→      <div className={cn("mt-1 font-mono text-lg tabular-nums", colorMap[accent], pulse && "animate-epistemic-pulse")}>
   920→        {value}
   921→      </div>
   922→    </div>
   923→  );
   924→}
