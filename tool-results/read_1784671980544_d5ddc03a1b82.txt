    97→    97→function ScrollToTopButton() {
    98→    98→  const [visible, setVisible] = useState(false);
    99→    99→
   100→   100→  useEffect(() => {
   101→   101→    const onScroll = () => setVisible(window.scrollY > 300);
   102→   102→    window.addEventListener("scroll", onScroll, { passive: true });
   103→   103→    return () => window.removeEventListener("scroll", onScroll);
   104→   104→  }, []);
   105→   105→

   107→   107→    window.scrollTo({ top: 0, behavior: "smooth" });

   110→   110→  return (
   111→   111→    <AnimatePresence>
   112→   112→      {visible && (

   114→   114→          type="button"
   115→   115→          onClick={scrollToTop}
   116→   116→          initial={{ opacity: 0, scale: 0.8 }}
