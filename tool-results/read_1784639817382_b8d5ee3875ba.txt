    17→function generateSparkline(base: number, variance: number, length = 12): number[] {
    18→  let seed = base * 7 + 31;
    19→  const rand = () => { seed = (seed * 1103515245 + 12345) & 0x7fffffff; return seed / 0x7fffffff; };

    21→}
