    62→function generateSparkline(base: number, variance: number, length = 12): number[] {
    63→  let seed = base * 7 + 31;
    64→  const rand = () => {
    65→    seed = (seed * 1103515245 + 12345) & 0x7fffffff;
    66→    return seed / 0x7fffffff;
    67→  };
    68→  return Array.from({ length }, () =>
    69→    Math.max(0, Math.round((base + (rand() - 0.5) * 2 * variance) * 10) / 10)
    70→  );
    71→}
