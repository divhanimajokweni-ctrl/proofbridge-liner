    69→    69→function fmtRelative(iso: string | null): string {
    70→    70→  if (!iso) return "—";
    71→    71→  const t = new Date(iso).getTime();
    72→    72→  const diff = Date.now() - t;
    73→    73→  if (diff < 0) return "just now";
    74→    74→  const s = Math.floor(diff / 1000);
    75→    75→  if (s < 60) return `${s}s ago`;
    76→    76→  const m = Math.floor(s / 60);
    77→    77→  if (m < 60) return `${m}m ago`;
    78→    78→  const h = Math.floor(m / 60);
    79→    79→  if (h < 24) return `${h}h ago`;
    80→    80→  return `${Math.floor(h / 24)}d ago`;
    81→    81→}

   264→   264→  divergence: number;

   270→   270→  const color =

   274→   274→      ? "[&>div]:bg-verified"
   275→   275→      : divergence < 1
   276→   276→        ? "[&>div]:bg-repairing"
   277→   277→        : "[&>div]:bg-violating";

   312→   312→              : "authoritative breach"}
