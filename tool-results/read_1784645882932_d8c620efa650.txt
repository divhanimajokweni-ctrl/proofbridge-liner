   212→export function MmrProofsSection() {
   213→  const { toast } = useToast();
   214→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   215→  const [proofs, setProofs] = useState<ProofRow[]>([]);
   216→  const [loading, setLoading] = useState(true);
   217→  const [filterPolicyId, setFilterPolicyId] = useState<string>("all");
   218→  const [searchQuery, setSearchQuery] = useState("");
   219→  const [mintPolicyId, setMintPolicyId] = useState<string>("");
   220→  const [minting, setMinting] = useState(false);
   221→
   222→  const policyMap = useMemo(() => { const m: Record<string, PolicyRow> = {}; for (const p of policies) m[p.id] = p; return m; }, [policies]);
   223→
   224→  useEffect(() => { let c = false; fetch("/api/policies").then((r) => r.json()).then((d: { policies: PolicyRow[] }) => { if (!c) { const ps = d.policies ?? []; setPolicies(ps); if (ps.length > 0) setMintPolicyId(ps[0].id); } }).catch(() => {}); return () => { c = true; }; }, []);
   225→
   226→  const loadProofs = useCallback(async () => {
   227→    try {
   228→      const url = filterPolicyId && filterPolicyId !== "all" ? `/api/proofs?policyId=${encodeURIComponent(filterPolicyId)}` : "/api/proofs";
   229→      const r = await fetch(url); if (!r.ok) throw new Error(`HTTP ${r.status}`);
   230→      const d: { proofs: ProofRow[] } = await r.json(); setProofs(d.proofs ?? []);
   231→    } catch (e) { toast({ title: "Failed to load proofs", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" }); }
   232→    finally { setLoading(false); }
   233→  }, [filterPolicyId, toast]);
   234→
   235→  useEffect(() => { setLoading(true); loadProofs(); const t = setInterval(loadProofs, POLL_MS); return () => clearInterval(t); }, [loadProofs]);
   236→
   237→  const mintProof = useCallback(async () => {
   238→    if (!mintPolicyId) { toast({ title: "Select a policy first", variant: "destructive" }); return; }
   239→    setMinting(true);
   240→    try {
   241→      const r = await fetch("/api/proofs", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ policyId: mintPolicyId }) });
   242→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   243→      const d = await r.json();
   244→      const created: ProofRow = { ...d.proof, verified: true, policy: d.proof.policy ?? { name: policyMap[mintPolicyId]?.name ?? "unknown", zkEnabled: policyMap[mintPolicyId]?.zkEnabled ?? false, proofKind: policyMap[mintPolicyId]?.proofKind ?? "mmr" } };
   245→      setProofs((prev) => [created, ...prev]);

   247→    } catch (e) { toast({ title: "Mint failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" }); }
   248→    finally { setMinting(false); }
   249→  }, [mintPolicyId, policyMap, toast]);
   250→
   251→  const filteredProofs = useMemo(() => {
   252→    if (!searchQuery.trim()) return proofs;
   253→    const q = searchQuery.toLowerCase();
   254→    return proofs.filter((p) => { const n = p.policy?.name ?? policyMap[p.policyId]?.name ?? ""; return n.toLowerCase().includes(q) || p.shardKey.toLowerCase().includes(q) || p.mmrRoot.toLowerCase().includes(q) || (p.anchor ?? "").toLowerCase().includes(q); });
   255→  }, [proofs, searchQuery, policyMap]);
   256→
   257→  const total = proofs.length;
   258→  const zkCount = proofs.filter((p) => p.zkProof).length;
   259→  const anchoredCount = proofs.filter((p) => p.anchored).length;
   260→  const anchoredRate = total > 0 ? Math.round((anchoredCount / total) * 100) : 0;
   261→  const allVerified = total > 0 && proofs.every((p) => p.verified);
   262→  const healthy = total > 0 && allVerified;
   263→
   264→  return (
   265→    <TooltipProvider delayDuration={150}>
   266→      <motion.section className="space-y-4" variants={containerVariants} initial="hidden" animate="visible">
   267→        <motion.div variants={itemVariants}>
   268→          <div className="flex items-center gap-2.5">
   269→            <div className="flex h-9 w-9 items-center justify-center rounded-lg border border-verified/30 bg-verified/10 glow-verified"><KeyRound className="h-4.5 w-4.5 text-verified" /></div>
   270→            <div><h2 className="text-base font-semibold text-foreground">MMR Ancestry Proof Inspector</h2><p className="text-xs text-muted-foreground">Merkle-mountain-range ancestry · zero-knowledge merge proofs · transparency anchoring</p></div>
   271→          </div>
   272→        </motion.div>
   273→
   274→        <motion.div variants={cardVariants}>
   275→          <MmrGradientCard gradient="from-verified/20 via-verified/10 to-repairing/20">
   276→            <div className="p-4 space-y-3"><div className="bg-grid-fine absolute inset-0 opacity-30" />
   277→              <div className="relative grid grid-cols-2 gap-3 sm:grid-cols-4">
   278→                <SummaryStat label="Total proofs" value={total} icon={<Mountain className="h-3.5 w-3.5" />} />
   279→                <SummaryStat label="ZK proofs" value={zkCount} icon={<KeyRound className="h-3.5 w-3.5" />} accent="verified" />
   280→                <SummaryStat label="Anchored" value={`${anchoredRate}%`} icon={<Anchor className="h-3.5 w-3.5" />} accent="verified" />
   281→                <SummaryStat label="Verification" value={total === 0 ? "—" : healthy ? "healthy" : "degraded"} icon={<ShieldCheck className="h-3.5 w-3.5" />} accent={total === 0 ? "muted" : healthy ? "verified" : "violating"} pulse={!healthy && total > 0} />
   282→              </div>
   283→              {total > 0 && <div className="relative border-t border-border/40 pt-3"><ProofChainIntegrity proofs={proofs} /></div>}

   289→          <div className="space-y-3 lg:col-span-2">
   290→            <motion.div variants={cardVariants}>
