    35→function AppliedDiff({ repair }: { repair: AppliedRepair }) {
    36→  const delta = typeof repair.from === "number" && typeof repair.to === "number" ? repair.to - repair.from : null;
    37→  const isPos = delta !== null && delta > 0;
    38→  const isNeg = delta !== null && delta < 0;
    39→  const isViolation = isNeg;
    40→  return (
    41→    <motion.div variants={itemVariants}
    42→      className={cn("flex flex-col gap-1.5 rounded-md border p-2.5 transition-all hover:shadow-md hover:scale-[1.01]",
    43→        isViolation ? "border-violating/40 bg-violating/5" : isPos ? "border-verified/40 bg-verified/5" : "border-border/60 bg-muted/20")}>
    44→      <div className="flex items-center justify-between gap-2">
    45→        <span className="font-mono text-xs text-foreground font-medium">{repair.field}</span>
    46→        {delta !== null && (
    47→          <motion.span initial={{ scale: 0 }} animate={{ scale: 1 }} transition={{ type: "spring", stiffness: 400, damping: 20 }}
    48→            className={cn("inline-flex items-center gap-0.5 rounded-sm px-1.5 py-0.5 text-[10px] font-mono font-semibold",
    49→              isPos ? "bg-verified/15 text-verified" : isNeg ? "bg-violating/15 text-violating" : "bg-muted text-muted-foreground")}>
    50→            {isPos ? <TrendingUp className="h-2.5 w-2.5" /> : isNeg ? <AlertTriangle className="h-2.5 w-2.5" /> : null}
    51→            {delta > 0 ? "+" : ""}{delta.toFixed(3)}
    52→          </motion.span>
    53→        )}
    54→      </div>
    55→      <div className="flex items-center gap-2 text-xs">
    56→        <span className="font-mono px-1.5 py-0.5 rounded bg-violating/10 text-violating line-through decoration-violating/50">{fmtVal(repair.from)}</span>
    57→        <motion.div initial={{ x: -4, opacity: 0 }} animate={{ x: 0, opacity: 1 }} transition={{ delay: 0.15 }}>
    58→          <ArrowRight className={cn("h-3 w-3 shrink-0", isViolation ? "text-violating" : "text-verified")} />
    59→        </motion.div>
    60→        <span className="font-mono px-1.5 py-0.5 rounded bg-verified/10 text-verified">{fmtVal(repair.to)}</span>
    61→      </div>
    62→    </motion.div>
    63→  );
    64→}

   297→function InlineStateDiff({ proposed, repaired }: { proposed: Record<string, unknown>; repaired: Record<string, unknown> }) {
   298→  const diffs = useMemo(() => {
   299→    const keys = new Set([...Object.keys(proposed ?? {}), ...Object.keys(repaired ?? {})]);
   300→    return Array.from(keys).sort().map((key) => {
   301→      const fromVal = proposed?.[key]; const toVal = repaired?.[key];
   302→      const changed = JSON.stringify(fromVal) !== JSON.stringify(toVal);
   303→      const delta = typeof fromVal === "number" && typeof toVal === "number" ? toVal - fromVal : null;
   304→      return { key, from: fromVal, to: toVal, changed, delta };
   305→    }).filter((d) => d.changed);
   306→  }, [proposed, repaired]);
   307→  if (!diffs.length) return null;
   308→  return (
   309→    <div>
   310→      <div className="mb-1.5 text-[10px] uppercase tracking-wide text-foreground/70 flex items-center gap-1.5"><Eye className="h-3 w-3" />State delta ({diffs.length} change{diffs.length === 1 ? "" : "s"})</div>
   311→      <div className="grid grid-cols-1 gap-1 sm:grid-cols-2">
   312→        {diffs.map((d) => (
   313→          <div key={d.key} className={cn("flex items-center justify-between gap-2 rounded border px-2 py-1 text-[11px]",
   314→            d.delta !== null ? d.delta > 0 ? "border-verified/30 bg-verified/5" : "border-violating/30 bg-violating/5" : "border-repairing/30 bg-repairing/5")}>
   315→            <div className="flex items-center gap-1.5 min-w-0"><span className="font-mono text-muted-foreground truncate">{d.key}</span><ArrowRight className="h-2.5 w-2.5 shrink-0 text-muted-foreground" /></div>
   316→            <div className="flex items-center gap-1 shrink-0">
   317→              <span className={cn("font-mono line-through decoration-violating/50", d.delta !== null && d.delta < 0 ? "text-violating" : "text-muted-foreground")}>{fmtVal(d.from)}</span>
   318→              <span className={cn("font-mono", d.delta !== null ? d.delta > 0 ? "text-verified" : "text-violating" : "text-verified")}>{fmtVal(d.to)}</span>
   319→              {d.delta !== null && <Badge variant="outline" className={cn("text-[8px] px-1 py-0 h-4", d.delta > 0 ? "border-verified/30 text-verified" : "border-violating/30 text-violating")}>{d.delta > 0 ? "+" : ""}{d.delta.toFixed(3)}</Badge>}
   320→            </div>
   321→          </div>
   322→        ))}
   323→      </div>
   324→    </div>
   325→  );
   326→}

   328→function MergeSummary({ merges }: { merges: MergeRow[] }) {
   329→  const stats = useMemo(() => {
   330→    const clean = merges.filter((m) => getMergeType(m) === "clean").length;
   331→    const repaired = merges.filter((m) => getMergeType(m) === "repaired").length;
   332→    const rejected = merges.filter((m) => getMergeType(m) === "rejected").length;
   333→    const avgDiv = merges.length > 0 ? merges.reduce((s, m) => s + m.divergence, 0) / merges.length : 0;
   334→    return { clean, repaired, rejected, avgDiv, total: merges.length };
   335→  }, [merges]);
   336→  return (
   337→    <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   338→      <StatCard label="Clean" value={stats.clean} color="text-verified" bg="bg-verified/10" border="border-verified/30" />
   339→      <StatCard label="Repaired" value={stats.repaired} color="text-repairing" bg="bg-repairing/10" border="border-repairing/30" />
   340→      <StatCard label="Rejected" value={stats.rejected} color="text-violating" bg="bg-violating/10" border="border-violating/30" />
   341→      <StatCard label="Avg div" value={stats.avgDiv.toFixed(3)} bg="bg-muted/20" border="border-border/60" />
   342→    </div>
   343→  );
   344→}

   346→export function MergeReconciliationSection() {
   347→  const { toast } = useToast();
   348→  const [policies, setPolicies] = useState<PolicyRow[]>([]);
   349→  const [merges, setMerges] = useState<MergeRow[]>([]);
   350→  const [loadingMerges, setLoadingMerges] = useState(true);
   351→  const [policyId, setPolicyId] = useState<string>("");
   352→  const [proposedText, setProposedText] = useState(DEFAULT_PROPOSED);
   353→  const [submitting, setSubmitting] = useState(false);
   354→  const [rerunning, setRerunning] = useState(false);
   355→  const [lastResult, setLastResult] = useState<{ mergeId: string; repair: RepairResult } | null>(null);
   356→
   357→  useEffect(() => {
   358→    let cancelled = false;
   359→    fetch("/api/policies").then((r) => r.json()).then((d: { policies: PolicyRow[] }) => {
   360→      if (cancelled) return;
   361→      const ps = d.policies ?? [];
   362→      setPolicies(ps);
   363→      if (ps.length > 0 && !policyId) setPolicyId(ps[0].id);
   364→    }).catch(() => {});
   365→    return () => { cancelled = true; };
   366→  }, []);
   367→
   368→  const loadMerges = useCallback(async () => {
   369→    try {
   370→      const r = await fetch("/api/merges");
   371→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   372→      const d: { merges: MergeRow[] } = await r.json();
   373→      setMerges(d.merges ?? []);
   374→    } catch (e) {
   375→      toast({ title: "Failed to load merges", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" });
   376→    } finally { setLoadingMerges(false); }
   377→  }, [toast]);
   378→
   379→  useEffect(() => { loadMerges(); const t = setInterval(loadMerges, POLL_MS); return () => clearInterval(t); }, [loadMerges]);
   380→
   381→  const runRepair = useCallback(async () => {
   382→    if (!policyId) { toast({ title: "Select a policy first", variant: "destructive" }); return; }
   383→    let proposed: Record<string, unknown>;
   384→    try { proposed = JSON.parse(proposedText); } catch (e) {
   385→      toast({ title: "Invalid JSON", description: e instanceof Error ? e.message : "Parse error", variant: "destructive" }); return;
   386→    }
   387→    setSubmitting(true);
   388→    try {
   389→      const r = await fetch("/api/merges", { method: "POST", headers: { "Content-Type": "application/json" },
   390→        body: JSON.stringify({ policyId, sourceShardName: "manual", targetShard: "default", proposedState: proposed }) });
   391→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   392→      const d = await r.json();
   393→      const mergeId: string = d.merge?.id;
   394→      const repair: RepairResult = d.repair;
   395→      setLastResult({ mergeId, repair });
   396→      toast({ title: repair.ok ? "Self-repair succeeded" : "Self-repair unresolved",
   397→        description: repair.ok ? `Applied ${repair.applied.length} fix(es) · div ${repair.divergence.toFixed(3)}` : `${repair.violations.length} still violating`,
   398→        variant: repair.ok ? "default" : "destructive" });
   399→      loadMerges();
   400→    } catch (e) {
   401→      toast({ title: "Repair request failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" });
   402→    } finally { setSubmitting(false); }
   403→  }, [policyId, proposedText, toast, loadMerges]);
   404→
   405→  const rerun = useCallback(async () => {
   406→    if (!lastResult?.mergeId) return;
   407→    setRerunning(true);
   408→    try {
   409→      const r = await fetch("/api/merges/repair", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ mergeId: lastResult.mergeId }) });
   410→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   411→      const d = await r.json();
   412→      const repair: RepairResult = d.repair;
   413→      setLastResult((prev) => (prev ? { ...prev, repair } : prev));
   414→      toast({ title: repair.ok ? "Re-run succeeded" : "Re-run unresolved",
   415→        description: `div ${repair.divergence.toFixed(3)} · ${repair.applied.length} applied`, variant: repair.ok ? "default" : "destructive" });
   416→      loadMerges();
   417→    } catch (e) {
   418→      toast({ title: "Re-run failed", description: e instanceof Error ? e.message : "Unknown error", variant: "destructive" });
   419→    } finally { setRerunning(false); }
   420→  }, [lastResult, toast, loadMerges]);

   428→          iconClass="border-repairing/30 bg-repairing/10 text-repairing" />

   431→      <motion.div variants={cardVariants}>

   433→          <div className="p-3 space-y-3">
   434→            <div className="flex items-center justify-between">
   435→              <div className="flex items-center gap-2"><Activity className="h-4 w-4 text-repairing" /><span className="text-sm font-semibold text-foreground">Merge Analytics</span></div>
   436→              <Badge variant="outline" className="border-border/60 text-[10px]">{merges.length} total</Badge>
   437→            </div>
   438→            <MergeSummary merges={merges} />
   439→            <DivergenceChart merges={merges} />
