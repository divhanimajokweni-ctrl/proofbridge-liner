   658→        //  Sequential Bayesian state estimation over a discrete location grid:

   688→        // Gaussian noise via Box-Muller
   689→        function gaussianNoise(mean, sigma) {
   690→            const u = 1 - Math.random(), v = Math.random();
   691→            return mean + sigma * Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
   692→        }

   694→        // Marsaglia-Tsang gamma sampler (for Beta-Binomial decision layer)
   695→        function sampleGamma(shape) {
   696→            if (shape < 1) {
   697→                const u = Math.random();
   698→                return sampleGamma(1 + shape) * Math.pow(u, 1 / shape);
   699→            }
   700→            const d = shape - 1 / 3;
   701→            const c = 1 / Math.sqrt(9 * d);
   702→            while (true) {
   703→                let x, v;
   704→                do {
   705→                    x = gaussianNoise(0, 1);
   706→                    v = 1 + c * x;
   707→                } while (v <= 0);
   708→                v = v * v * v;
   709→                const u = Math.random();
   710→                if (u < 1 - 0.0331 * x * x * x * x) return d * v;
   711→                if (Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
   712→            }
   713→        }
   714→        function sampleBeta(a, b) {
   715→            const x = sampleGamma(a), y = sampleGamma(b);
   716→            return x / (x + y);
   717→        }

   719→        // Physical forward model: predicted signal amplitude at a listening
   720→        // tap given a candidate leak location & magnitude (inverse-square decay)
   721→        function predictedAmplitude(cx, cz, q, sensor) {
   722→            const dx = cx - sensor.x, dz = cz - sensor.z;
   723→            const d2 = dx * dx + dz * dz;
   724→            const decay = 2.2;
   725→            return q * 3.2 * Math.exp(-d2 / (2 * decay * decay));
   726→        }

   744→        // Sequential Bayesian update: posterior(x) ∝ prior(x) · L(O_t | x)
   745→        // Magnitude q is profiled from the live flow surplus each tick.
   746→        function bayesianUpdate(obs) {
   747→            const qHat = Math.max(0.5, state.currentFlow - state.baselineFlow);
   748→            // Mixture-noise robustness: widen (down-weight) the likelihood
   749→            // during a blast instead of hard-excluding samples — this is the
   750→            // FPR-suppression mixture from the paper.
   751→            const sigma = state.blastActive ? 0.9 : 0.18;
   752→            const newPosterior = new Float64Array(GRID_N * GRID_N);
   753→            let total = 0;
   754→
   755→            for (let i = 0; i < GRID_N; i++) {
   756→                for (let j = 0; j < GRID_N; j++) {
   757→                    const idx = i * GRID_N + j;
   758→                    const { x, z } = cellCenter(i, j);
   759→                    let logL = 0;
   760→                    sensorNodes.forEach(s => {
   761→                        const pred = predictedAmplitude(x, z, qHat, s);
   762→                        const diff = obs[s.name] - pred;
   763→                        logL += -(diff * diff) / (2 * sigma * sigma);
   764→                    });
   765→                    const val = posterior[idx] * Math.exp(logL);
   766→                    newPosterior[idx] = val;
   767→                    total += val;
   768→                }
   769→            }
   770→            if (total > 0) {
   771→                for (let k = 0; k < newPosterior.length; k++) newPosterior[k] /= total;
   772→            }
   773→            posterior = newPosterior;
   774→        }

   776→        // MAP estimate + 95% credible radius (smallest radius around the MAP

   778→        function computeMapAndRadius() {
   779→            let maxP = -1, mi = 0, mj = 0;
   780→            for (let i = 0; i < GRID_N; i++) {
   781→                for (let j = 0; j < GRID_N; j++) {
   782→                    const idx = i * GRID_N + j;
   783→                    if (posterior[idx] > maxP) { maxP = posterior[idx]; mi = i; mj = j; }
   784→                }
   785→            }
   786→            const mapCenter = cellCenter(mi, mj);
   787→            state.map = mapCenter;
   788→            state.posteriorPeak = maxP;
   789→
   790→            const cells = [];
   791→            for (let i = 0; i < GRID_N; i++) {
   792→                for (let j = 0; j < GRID_N; j++) {
   793→                    const idx = i * GRID_N + j;
   794→                    const c = cellCenter(i, j);
   795→                    cells.push({ p: posterior[idx], d: Math.hypot(c.x - mapCenter.x, c.z - mapCenter.z) });
   796→                }
   797→            }
   798→            cells.sort((a, b) => b.p - a.p);

   803→                if (cum >= 0.95) break;
   804→            }

  1287→        function bayesTick() {
  1288→            state.bayesTicks++;
  1289→
  1290→            // Live hydraulic telemetry converges toward the true leak signature
  1291→            const targetFlow = state.baselineFlow + state.trueLeak.q;
  1292→            const targetPressure = state.baselinePressure - state.trueLeak.q * 0.18;
  1293→            state.currentFlow += (targetFlow - state.currentFlow) * 0.35 + gaussianNoise(0, 0.4);
  1294→            state.currentPressure += (targetPressure - state.currentPressure) * 0.35 + gaussianNoise(0, 0.08);
  1295→            updateHUD();
  1296→
  1297→            // Sequential Bayesian update: P(S_t|O_1:t) ∝ P(O_t|S_t)·P(S_t|S_t-1)
  1298→            const obs = simulateObservations();
  1299→            bayesianUpdate(obs);
  1300→            const { mapCenter } = computeMapAndRadius();
  1301→            updatePosteriorVisual();
  1302→
  1303→            document.getElementById('bayes-peak').textContent = (state.posteriorPeak * 100).toFixed(1) + '%';
  1304→            document.getElementById('bayes-radius').textContent = Math.round(state.credibleRadiusM).toLocaleString() + 'm';
  1305→
  1306→            if (state.blastActive) {
  1307→                state.blastFiltered++;
  1308→                document.getElementById('bayes-filtered').textContent = state.blastFiltered;
  1309→                if (state.bayesTicks % 3 === 0) {
  1310→                    logEvent('BAYES', 'Blast-corrupted samples down-weighted (mixture likelihood)');
  1311→                }
  1312→            }
  1313→
  1314→            updateEIS();
  1315→
  1316→            if (!state.verifiedLogged && state.credibleRadiusM <= 500 && state.posteriorPeak > 0.04) {
  1317→                state.verifiedLogged = true;
  1318→                clearInterval(state.bayesInterval);
  1319→
  1320→                const errM = Math.hypot(mapCenter.x - state.trueLeak.x, mapCenter.z - state.trueLeak.z) * METERS_PER_UNIT;
  1321→                logEvent('SUCCESS', `HBK: VERIFIED_CANDIDATE. 95% credible radius ${Math.round(state.credibleRadiusM)}m ≤ 500m target`);
  1322→                logEvent('BAYES', `Localization error vs ground truth: ${Math.round(errM)}m`);
  1323→                revealLeakMarker(mapCenter);
  1324→
  1325→                addFieldEvidence('SCADA_FLOW', 'FM01', `Flow surge +${((state.trueLeak.q / state.baselineFlow) * 100).toFixed(1)}%`);
  1326→                addFieldEvidence('SCADA_PRESS', 'PT14', `Pressure drop ${(state.baselinePressure - state.currentPressure).toFixed(1)}m`);
  1327→                addFieldEvidence('ACOUSTIC', 'AC-07', `Posterior concentrated within 500m (95% CI)`);
  1328→
  1329→                document.getElementById('simulate-leak-btn').disabled = false;
  1330→            } else if (state.bayesTicks > 150) {

  1378→        // ---------- Decision Support Layer (Beta-Binomial) ----------
  1379→        function runRiskAnalysis() {
  1380→            const mu = 0.12, rho = 0.15; // baseline segment failure rate & intra-cluster correlation
  1381→            const a = mu * (1 - rho) / rho;
  1382→            const b = (1 - mu) * (1 - rho) / rho;
  1383→
  1384→            function simulateDMA(nSegments, trials, riskMultiplier) {
  1385→                let fails = 0, total = 0;
  1386→                for (let s = 0; s < nSegments; s++) {
  1387→                    const p = Math.min(0.95, sampleBeta(a, b) * riskMultiplier);
  1388→                    let f = 0;
  1389→                    for (let t = 0; t < trials; t++) if (Math.random() < p) f++;
  1390→                    fails += f; total += trials;
  1391→                }
  1392→                return fails / total;
  1393→            }
  1394→
  1395→            const treatmentRate = simulateDMA(18, 40, 1.4); // near the detected leak zone
  1396→            const controlRate = simulateDMA(18, 40, 1.0);
  1397→            const rr = treatmentRate / controlRate;
  1398→            const rd = (treatmentRate - controlRate) * 100;
  1399→            const infoGained = state.posteriorPeak > 0 ? Math.min(1, state.posteriorPeak * GRID_N * GRID_N / 40) : 0;
  1400→            const id = Math.max(0, Math.min(100, 55 + rr * 12 + infoGained * 20));
  1401→
  1402→            document.getElementById('risk-output').innerHTML =
  1403→                `Relative Risk: <b style="color:var(--amber)">${rr.toFixed(2)}x</b><br>` +
  1404→                `Risk Difference: <b style="color:${rd >= 0 ? 'var(--red)' : 'var(--green)'}">${rd >= 0 ? '+' : ''}${rd.toFixed(1)}%</b><br>` +
  1405→                `Information Density: <b style="color:var(--cyan)">${id.toFixed(1)}/100</b>`;
  1406→            logEvent('BAYES', `Risk analysis: RR=${rr.toFixed(2)}x, RD=${rd.toFixed(1)}%`);
  1407→        }
