   330→/** Health gauge - donut chart showing federation health */
   331→function FederationHealthGauge({ channels }: { channels: FedChannel[] }) {
   332→  const verified = channels.filter((c) => c.status === "verified").length;
   333→  const negotiating = channels.filter((c) => c.status === "negotiating").length;
   334→  const drift = channels.filter((c) => c.status === "drift").length;
   335→
   336→  const data = [
   337→    { name: "verified", value: verified, color: "oklch(0.78 0.16 160)" },
   338→    { name: "negotiating", value: negotiating, color: "oklch(0.80 0.15 80)" },
   339→    { name: "drift", value: drift, color: "oklch(0.64 0.21 25)" },
   340→  ].filter((d) => d.value > 0);
   341→
   342→  const total = verified + negotiating + drift;
   343→  const healthPct = total > 0 ? Math.round((verified / total) * 100) : 0;
   344→
   345→  if (data.length === 0) {
   346→    return (
   347→      <div className="flex items-center justify-center h-[120px] text-xs text-muted-foreground italic">
   348→        No channels
   349→      </div>
   350→    );
   351→  }
   352→
   353→  return (
   354→    <div className="relative h-[120px]">
   355→      <ResponsiveContainer width="100%" height="100%">
   356→        <PieChart>
   357→          <Pie
   358→            data={data}
   359→            cx="50%"
   360→            cy="50%"
   361→            innerRadius={35}
   362→            outerRadius={48}
   363→            paddingAngle={2}
   364→            dataKey="value"
   365→            strokeWidth={0}
   366→          >
   367→            {data.map((entry, i) => (
   368→              <Cell key={i} fill={entry.color} fillOpacity={0.8} />
   369→            ))}
   370→          </Pie>
   371→        </PieChart>
   372→      </ResponsiveContainer>
   373→      <div className="absolute inset-0 flex items-center justify-center">
   374→        <div className="text-center">
   375→          <motion.div
   376→            className={cn(
   377→              "text-lg font-bold font-mono",
   378→              healthPct >= 80 ? "text-verified" : healthPct >= 50 ? "text-repairing" : "text-violating",
   379→            )}
   380→            initial={{ opacity: 0, scale: 0.8 }}
   381→            animate={{ opacity: 1, scale: 1 }}
   382→            transition={{ type: "spring", stiffness: 300 }}
   383→          >
   384→            {healthPct}%
   385→          </motion.div>
   386→          <div className="text-[8px] uppercase tracking-wide text-muted-foreground">health</div>
   387→        </div>
   388→      </div>
   389→    </div>
   390→  );
   391→}

   401→    fetch("/api/federation")
   402→      .then((r) => r.json())
   403→      .then((d) => {
   404→        setData(d);
   405→        setLoading(false);
   406→      })
   407→      .catch(() => setLoading(false));
   408→  }, []);
   409→
   410→  useEffect(() => {
   411→    load();

   420→        method: "POST",
   421→        headers: { "Content-Type": "application/json" },

   426→        toast({ title: "Reconciliation failed", description: d.error, variant: "destructive" });
