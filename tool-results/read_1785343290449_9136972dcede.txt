     1→"""VVU Earth Tech Ledger — Complete exception hierarchy.
     2→
     3→Every exception carries a machine-readable ``code`` attribute and an optional
     4→``detail`` dictionary so that callers can programmatically inspect failures
     5→without parsing message strings.
     6→
     7→Hierarchy::
     8→
     9→    LedgerError
    10→    ├── SerializationError
    11→    ├── HashError
    12→    ├── CryptoError
    13→    │   ├── SignatureError
    14→    │   └── KeyRotationError
    15→    ├── MMRError
    16→    ├── StorageError
    17→    │   ├── DatabaseError
    18→    │   └── SnapshotError
    19→    ├── ReplayError
    20→    ├── ValidatorError
    21→    ├── EnvelopeError
    22→    ├── ConfigError
    23→    └── NetworkError
    24→"""
    25→
    26→from __future__ import annotations
    27→
    28→from typing import Any
    29→
    30→
    31→# ---------------------------------------------------------------------------
    32→# Base
    33→# ---------------------------------------------------------------------------
    34→
    35→class LedgerError(Exception):
    36→    """Base exception for all ledger errors.
    37→
    38→    Attributes:
    39→        code: Machine-readable error code (e.g. ``"LEDGER_UNKNOWN"``).
    40→        detail: Optional dictionary with additional context.
    41→    """
    42→
    43→    def __init__(
    44→        self,
    45→        message: str = "",
    46→        *,
    47→        code: str = "LEDGER_UNKNOWN",
    48→        detail: dict[str, Any] | None = None,
    49→    ) -> None:
    50→        super().__init__(message)
    51→        self.code: str = code
    52→        self.detail: dict[str, Any] | None = detail
    53→
    54→    def __repr__(self) -> str:  # pragma: no cover
    55→        return f"{type(self).__name__}(code={self.code!r}, message={str(self)!r})"
    56→
    57→
