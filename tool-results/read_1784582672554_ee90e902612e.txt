   516→  useEffect(() => {
   517→    let cancelled = false;
   518→    fetch("/api/policies")
   519→      .then((r) => r.json())
   520→      .then((d: { policies: PolicyRow[] }) => {
   521→        if (!cancelled) setPolicies(d.policies ?? []);
   522→      })
   523→      .catch(() => {
   524→
   525→      });
   526→    return () => {
   527→      cancelled = true;
   528→    };
   529→  }, []);
   530→
   531→  const loadShards = useCallback(async () => {
   532→    const url =
   533→      policyId === "all" ? "/api/shards" : `/api/shards?policyId=${encodeURIComponent(policyId)}`;
   534→    try {
   535→      const r = await fetch(url);
   536→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   537→      const d: { shards: ShardRow[] } = await r.json();
   538→      setShards(d.shards ?? []);
   539→    } catch (e) {
   540→      toast({
   541→        title: "Failed to load shards",
   542→        description: e instanceof Error ? e.message : "Unknown error",
   543→        variant: "destructive",
   544→      });
   545→    } finally {
   546→      setLoading(false);
   547→      setRefreshing(false);
   548→    }
   549→  }, [policyId, toast]);
   550→
   551→  useEffect(() => {
   552→    setLoading(true);
   553→    loadShards();
   554→    const t = setInterval(loadShards, POLL_MS);
   555→    return () => clearInterval(t);
   556→  }, [loadShards]);
   557→
   558→  const sorted = useMemo(
   559→    () =>
   560→      [...shards].sort((a, b) => {
   561→        const rank: Record<ShardStatus, number> = { violating: 0, repairing: 1, healthy: 2 };
   562→        const r = rank[a.invariantStatus] - rank[b.invariantStatus];
   563→        if (r !== 0) return r;
   564→        return a.region.localeCompare(b.region);
   565→      }),
   566→    [shards],
   567→  );
   568→
   569→  return (
   570→    <motion.section
   571→      className="space-y-4"
   572→      variants={containerVariants}
   573→      initial="hidden"
   574→      animate="visible"
   575→    >
   576→      {/* Header / filter */}
   577→      <motion.div
   578→        variants={cardVariants}
   579→        className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between"
   580→      >
   581→        <div className="flex items-center gap-2.5">
   582→          <motion.div
   583→            className="flex h-9 w-9 items-center justify-center rounded-lg border border-verified/30 bg-verified/10"
   584→            whileHover={{ scale: 1.1, rotate: 5 }}
   585→            whileTap={{ scale: 0.95 }}
   586→          >
   587→            <Network className="h-4.5 w-4.5 text-verified" />
   588→          </motion.div>
   589→          <div>
   590→            <h2 className="text-base font-semibold text-foreground">DAG Shard Topology</h2>
   591→            <p className="text-xs text-muted-foreground">
   592→              Live sharded state across regions · MMR-anchored · peer-gossiped
   593→            </p>
   594→          </div>
   595→        </div>
   596→        <div className="flex items-center gap-2">
   597→          <Select value={policyId} onValueChange={setPolicyId}>
   598→            <SelectTrigger className="w-[220px] bg-card/60">
   599→              <SelectValue placeholder="Filter by policy" />
   600→            </SelectTrigger>
   601→            <SelectContent>
   602→              <SelectItem value="all">All policies</SelectItem>
   603→              {policies.map((p) => (
   604→                <SelectItem key={p.id} value={p.id}>
   605→                  {p.name}
   606→                </SelectItem>
   607→              ))}
   608→            </SelectContent>
   609→          </Select>
   610→          <motion.button
   611→            type="button"
   612→            onClick={() => {
   613→              setRefreshing(true);
   614→              loadShards();
   615→            }}
   616→            className="flex h-9 w-9 items-center justify-center rounded-md border border-border/60 bg-card/60 text-muted-foreground hover:text-foreground hover:bg-muted/40 transition-colors"
   617→            aria-label="Refresh topology"
   618→            whileHover={{ scale: 1.08, rotate: 90 }}
   619→            whileTap={{ scale: 0.92 }}
   620→          >
   621→            <RefreshCw className={cn("h-4 w-4", refreshing && "animate-spin")} />
   622→          </motion.button>
   623→        </div>
   624→      </motion.div>
   625→
   626→      <SummaryBar shards={sorted} />
   627→
   628→      {/* Legend */}
   629→      <motion.div
   630→        variants={containerVariants}
   631→        className="flex flex-wrap items-center gap-4 px-1"
   632→      >
   633→        <LegendItem
   634→          icon={CheckCircle2}
   635→          label="Healthy — all invariants satisfied"
   636→          color="bg-verified/10"
   637→          border="border-verified/30"
   638→          text="text-verified"
   639→        />
   640→        <LegendItem
   641→          icon={AlertTriangle}
   642→          label="Repairing — self-heal in progress"
   643→          color="bg-repairing/10"
   644→          border="border-repairing/30"
   645→          text="text-repairing"
   646→        />
   647→        <LegendItem
   648→          icon={XCircle}
   649→          label="Violating — invariant breach"
   650→          color="bg-violating/10"
   651→          border="border-violating/30"
   652→          text="text-violating"
   653→        />
   654→      </motion.div>
   655→
   656→      {/* Topology grid */}
   657→      <motion.div variants={cardVariants}>
   658→        <Card className="bg-card/40 backdrop-blur border-border/60 p-4 relative overflow-hidden">
   659→          {/* Gradient top border */}
   660→          <div className="absolute inset-x-0 top-0 h-0.5 bg-gradient-to-r from-verified/50 via-repairing/30 to-violating/30" />
   661→          <div className="bg-grid absolute inset-0 opacity-20 pointer-events-none" />
   662→          <div className="relative">
   663→            <div className="mb-3 flex items-center justify-between">
   664→              <div className="flex items-center gap-2">
   665→                <Activity className="h-4 w-4 text-verified" />
   666→                <span className="text-sm font-medium text-foreground">Mesh view</span>
   667→                <span className="text-xs text-muted-foreground">
   668→                  {sorted.length} node{sorted.length === 1 ? "" : "s"} · dotted links = peer gossip
   669→                </span>
   670→              </div>
   671→            </div>
   672→
   673→            {loading ? (
   674→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
   675→                {Array.from({ length: 6 }).map((_, i) => (
   676→                  <Skeleton key={i} className="h-36 w-full rounded-lg bg-muted/40" />
   677→                ))}
   678→              </div>
   679→            ) : sorted.length === 0 ? (
   680→              <div className="flex flex-col items-center justify-center py-12 text-center">
   681→                <Boxes className="h-8 w-8 text-muted-foreground/50" />
   682→                <p className="mt-2 text-sm text-muted-foreground">No shards found for this policy.</p>
   683→              </div>
   684→            ) : (
   685→              <div className="space-y-3">
   686→                <motion.div
   687→                  variants={containerVariants}
   688→                  className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4"
   689→                >
   690→                  {sorted.map((s, idx) => (
   691→                    <ShardNodeCard key={s.id} shard={s} index={idx} />
   692→                  ))}
   693→                </motion.div>
   694→                <MeshHint />
   695→                <motion.div
   696→                  className="flex items-center justify-center gap-1.5 text-[10px] text-muted-foreground"
   697→                  initial={{ opacity: 0 }}
   698→                  animate={{ opacity: 1 }}
   699→                  transition={{ delay: 0.5 }}
   700→                >
   701→                  <span className="font-mono">DAG</span>
   702→                  <ArrowRight className="h-3 w-3" />
   703→                  <span>locality-preserving merges</span>
   704→                  <ArrowRight className="h-3 w-3" />
   705→                  <span className="font-mono">MMR root</span>
   706→                </motion.div>
   707→              </div>
   708→            )}
   709→          </div>
   710→        </Card>
   711→      </motion.div>
   712→
   713→      {/* Interactive force-directed graph */}
   714→      {!loading && sorted.length > 0 && <InteractiveDagGraph shards={sorted} />}
   715→
   716→      {/* Rebalance + ledger in a two-column layout on large screens */}
   717→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   718→        <motion.div variants={cardVariants} className="lg:col-span-1">
   719→          <ShardRebalancePanel onMigrated={loadShards} />
   720→        </motion.div>
   721→        <div className="lg:col-span-2">
   722→          <motion.div variants={itemVariants} className="mb-2 flex items-center gap-2 px-1">
   723→            <span className="text-sm font-medium text-foreground">Shard ledger</span>
   724→            <span className="text-xs text-muted-foreground">· sorted by status severity</span>
   725→            <Info className="h-3 w-3 text-muted-foreground ml-1" />
   726→          </motion.div>
   727→          {loading ? (
   728→            <Skeleton className="h-64 w-full rounded-lg bg-muted/40" />
   729→          ) : sorted.length === 0 ? (
   730→            <Card className="bg-card/60 border-border/60 p-8 text-center text-sm text-muted-foreground">
   731→              No shards to display.
   732→            </Card>
   733→          ) : (
   734→            <ShardTable shards={sorted} />
   735→          )}
   736→        </div>
   737→      </div>
   738→    </motion.section>
   739→  );
   740→}
