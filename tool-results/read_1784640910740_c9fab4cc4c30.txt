    18→function generateSparkline(base: number, variance: number, length = 12): number[] {
    19→  let seed = base * 7 + 31;
    20→  const rand = () => { seed = (seed * 1103515245 + 12345) & 0x7fffffff; return seed / 0x7fffffff; };
    21→  return Array.from({ length }, () => Math.max(0, Math.round((base + (rand() - 0.5) * 2 * variance) * 10) / 10));
    22→}
